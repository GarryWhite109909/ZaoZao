  91| 	 * returns which standard is used, CII or UBL

  95| 	private EStandard findOutStandardFromRootNode(InputStream fis)
  96| 		throws ParserConfigurationException {
  97| 
  98| 		String zf1Signature = "CrossIndustryDocument";
  99| 		String zf2Signature = "CrossIndustryInvoice";
 100| 		String ublSignature = "Invoice";
 101| 		String ublCreditNoteSignature = "CreditNote";
 102| 		String cioSignature = "SCRDMCCBDACIOMessageStructure";
 103| 
 104| 		DocumentBuilderFactory dbf = DocumentBuilderFactory.newInstance();
 105| 		dbf.setNamespaceAware(true);
 106| 		dbf.setExpandEntityReferences(false);
 107| 		dbf.setFeature(XMLConstants.FEATURE_SECURE_PROCESSING, true);
 108| 		dbf.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
 109| 		dbf.setFeature("http://xml.org/sax/features/external-general-entities", false);
 110| 		dbf.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
 111| 		try {
 112| 			DocumentBuilder db = dbf.newDocumentBuilder();
 113| 			Document doc = db.parse(new InputSource(fis));
 114| 			Element root = doc.getDocumentElement();
 115| 			if (root.getLocalName().equals(zf1Signature)) {
 116| 				return EStandard.zugferd;
 117| 			} else if (root.getLocalName().equals(zf2Signature)) {
 118| 				return EStandard.facturx;
 119| 			} else if (root.getLocalName().equals(ublSignature)) {
 120| 				return EStandard.ubl;
 121| 			} else if (root.getLocalName().equals(ublCreditNoteSignature)) {
 122| 				return EStandard.ubl_creditnote;
 123| 			} else if (root.getLocalName().equals(cioSignature)) {
 124| 				return EStandard.orderx;
 125| 			}
 126| 		} catch (Exception e) {
 127| 			LOGGER.error("Failed to recognize standard", e);
 128| 		}
 129| 		return null;
 130| 	}

 139| 		throws IOException, TransformerException, ParserConfigurationException {
 140| 		initTemplates(lang);

 142| 		String fileContent = new String(IOUtils.toByteArray(inputXml), StandardCharsets.UTF_8);

 145| 
 146|

 150| 			applyZF1XSLT(xmlContentStream, htmlOutput);
 151| 			return htmlOutput.toString(StandardCharsets.UTF_8);
 152| 		} else if (thestandard == EStandard.facturx) {
 153| 			//zf2 or fx
 154| 			applyZF2XSLT(xmlContentStream, htmlOutput);
 155| 		} else if (thestandard == EStandard.ubl) {
 156| 			//zf2 or fx
 157| 			applyUBL2XSLT(xmlContentStream, htmlOutput);
 158| 		} else if (thestandard == EStandard.ubl_creditnote) {
 159| 			//zf2 or fx
 160| 			applyUBLCreditNote2XSLT(xmlContentStream, htmlOutput);
 161| 		} else if (thestandard == EStandard.orderx) {
 162| 			//zf2 or fx
 163| 			applyCIO2XSLT(xmlContentStream, htmlOutput);
 164| 		} else {
 165| 			throw new IllegalArgumentException("File does not look like CII or UBL");
 166| 		}
 167| 		Optional<InputStream> in = copyStream(htmlOutput);
 168| 		ByteArrayOutputStream htmlOutStream = new ByteArrayOutputStream();
 169| 		if (in.isPresent()) {

 174| 	}

 179| 	private Optional<InputStream> copyStream(ByteArrayOutputStream byteArrayOutputStream) {
 180| 		// take the copy of the stream and re-write it to an InputStream
 181| 		PipedInputStream in = new PipedInputStream();
 182| 		try {
 183| 			PipedOutputStream out = new PipedOutputStream(in);
 184| 			new Thread(() -> {
 185| 				try {
 186| 					// write the original OutputStream to the PipedOutputStream
 187| 					// note that in order for the below method to work, you need
 188| 					// to ensure that the data has finished writing to the
 189| 					// ByteArrayOutputStream
 190| 					byteArrayOutputStream.writeTo(out);
 191| 				} catch (IOException e) {
 192| 					LOGGER.error("Failed to write to stream", e);
 193| 				} finally {
 194| 					// close the PipedOutputStream here because we're done writing data
 195| 					// once this thread has completed its run
 196| 					StreamHelper.close(out);
 197| 				}
 198| 			}).start();
 199| 		} catch (IOException e1) {
 200| 			LOGGER.error("Failed to create HTML", e1);
 201| 			return Optional.empty();
 202| 		}
 203| 		return Optional.of(in);
 204| 	}

 315| 				LOGGER.error("Failed to create PDF", e);
 316| 			}
 317| 			byteHolder.set(os.toByteArray());
 318| 		});
 319| 		
 320| 		return byteHolder.get();

 323| 	private void toPDFfromFOP(String fopInput, Supplier<OutputStream> outputStreamDelegate, Consumer<OutputStream> consumerDelegate) {
 324| 
 325| 		DefaultConfigurationBuilder cfgBuilder = new DefaultConfigurationBuilder();
 326| 
 327| 		Configuration cfg = null;
 328| 		try {
 329| 			cfg = cfgBuilder.build(CLASS_LOADER.getResourceAsStream("fop-config.xconf"));
 330| 		} catch (ConfigurationException e) {
 331| 			throw new RuntimeException(e);
 332| 		}
 333| 
 334| 		FopFactoryBuilder builder = new FopFactoryBuilder(new File(".").toURI(), new ClasspathResolverURIAdapter()).setConfiguration(cfg);
 335| // Step 1: Construct a FopFactory by specifying a reference to the configuration file
 336| // (reuse if you plan to render multiple documents!)
 337| 
 338| 		FopFactory fopFactory = builder.build();//FopFactory.newInstance(new File("c:\\Users\\jstaerk\\temp\\fop-config.xconf"));
 339| 
 340| 		fopFactory.getFontManager().setResourceResolver(
 341| 			ResourceResolverFactory.createInternalResourceResolver(
 342| 				new File(".").toURI(),
 343| 				new ClasspathResolverURIAdapter()));
 344| 
 345| 		FOUserAgent userAgent = fopFactory.newFOUserAgent();
 346| 
 347| 		userAgent.getRendererOptions().put("pdf-a-mode", "PDF/A-3b");
 348| 
 349| // Step 2: Set up output stream.
 350| // Note: Using BufferedOutputStream for performance reasons (helpful with FileOutputStreams).
 351| 
 352| 		try (OutputStream out = new BufferedOutputStream(outputStreamDelegate.get())) {
 353| 
 354| 			// Step 3: Construct fop with desired output format
 355| 			Fop fop = fopFactory.newFop(MimeConstants.MIME_PDF, userAgent, out);
 356| 
 357| 			// Step 4: Setup JAXP using identity transformer
 358| 			TransformerFactory factory = TransformerFactory.newInstance();
 359| 			Transformer transformer = factory.newTransformer(); // identity transformer
 360| 
 361| 			// Step 5: Setup input and output for XSLT transformation
 362| 			// Setup input stream
 363| 			Source src = new StreamSource(new ByteArrayInputStream(fopInput.getBytes(StandardCharsets.UTF_8)));
 364| 
 365| 			// Resulting SAX events (the generated FO) must be piped through to FOP
 366| 			Result res = new SAXResult(fop.getDefaultHandler());
 367| 
 368| 			// Step 6: Start XSLT transformation and FOP processing
 369| 			transformer.transform(src, res);
 370| 			
 371| 			consumerDelegate.accept(out);
 372| 
 373| 		} catch (FOPException | IOException | TransformerException e) {
 374| 			LOGGER.error("Failed to create PDF", e);
 375| 		}
 376| 	}
