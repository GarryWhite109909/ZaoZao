 237| // sendAPIAuthMiddleware handles authentication specifically for the send API endpoint.
 238| // It can use dedicated send API authentication or accept any credentials based on configuration.
 239| // It communicates skip-UI-auth intent via request context rather than mutating the global
 240| // auth.UICredentials pointer, which would be a data race under concurrent load.
 241| func sendAPIAuthMiddleware(fn http.HandlerFunc) http.HandlerFunc {
 242| 	return func(w http.ResponseWriter, r *http.Request) {
 243| 		// Override the default 5 MB body cap with the send-specific limit so that
 244| 		// middleWareFunc applies config.MaxMessageSize (0 = unlimited) instead.
 245| 		r = r.WithContext(context.WithValue(r.Context(), bodyLimitKey, int64(config.MaxMessageSize)*1024*1024))
 246| 
 247| 		// If send API auth accept any is enabled, bypass all authentication.
 248| 		if config.SendAPIAuthAcceptAny {
 249| 			ctx := context.WithValue(r.Context(), skipUIAuthKey, true)
 250| 			middleWareFunc(fn)(w, r.WithContext(ctx))
 251| 			return
 252| 		}
 253| 
 254| 		// If Send API credentials are configured, only accept those credentials.
 255| 		if auth.SendAPICredentials != nil {
 256| 			user, pass, ok := r.BasicAuth()
 257| 
 258| 			if !ok {
 259| 				basicAuthResponse(w)
 260| 				return
 261| 			}
 262| 
 263| 			if !auth.SendAPICredentials.Match(user, pass) {
 264| 				basicAuthResponse(w)
 265| 				return
 266| 			}
 267| 
 268| 			// Valid Send API credentials — bypass UI auth via context flag.
 269| 			ctx := context.WithValue(r.Context(), skipUIAuthKey, true)
 270| 			middleWareFunc(fn)(w, r.WithContext(ctx))
 271| 			return
 272| 		}
 273| 
 274| 		// No Send API credentials configured — fall back to UI auth.
 275| 		middleWareFunc(fn)(w, r)
 276| 	}
 277| }

 288| // MiddleWareFunc http middleware adds optional basic authentication
 289| // and gzip compression.
 290| func middleWareFunc(fn http.HandlerFunc) http.HandlerFunc {
 291| 	return func(w http.ResponseWriter, r *http.Request) {
 292| 		// Limit request body size to 5 MB to prevent memory-exhaustion DoS via large
 293| 		// JSON bodies. sendAPIAuthMiddleware sets bodyLimitKey in the context to signal
 294| 		// that the handler manages its own limit (send.go uses config.MaxMessageSize),
 295| 		// so we skip the cap here for that route only.
 296| 		if _, ok := r.Context().Value(bodyLimitKey).(int64); !ok {
 297| 			r.Body = http.MaxBytesReader(w, r.Body, 5*1024*1024)
 298| 		}
 299| 
 300| 		w.Header().Set("Referrer-Policy", "no-referrer")
 301| 
 302| 		// generate a new random nonce on every request
 303| 		randomNonce := shortuuid.New()
 304| 		// header used to pass nonce through to function
 305| 		r.Header.Set("mp-nonce", randomNonce)
 306| 
 307| 		// Prevent JavaScript XSS by adding a nonce for script-src
 308| 		cspHeader := strings.Replace(
 309| 			config.ContentSecurityPolicy,
 310| 			"script-src 'self';",
 311| 			fmt.Sprintf("script-src 'nonce-%s';", randomNonce),
 312| 			1,
 313| 		)
 314| 
 315| 		w.Header().Set("Content-Security-Policy", cspHeader)
 316| 
 317| 		if htmlPreviewRouteRe == nil {
 318| 			htmlPreviewRouteRe = regexp.MustCompile(`^` + regexp.QuoteMeta(config.Webroot) + `view/[a-zA-Z0-9]+\.html$`)
 319| 		}
 320| 
 321| 		if strings.HasPrefix(r.RequestURI, config.Webroot+"api/") || htmlPreviewRouteRe.MatchString(r.RequestURI) {
 322| 			if allowed := corsOriginAccessControl(r); !allowed {
 323| 				http.Error(w, "Blocked due to CORS violation", http.StatusForbidden)
 324| 				return
 325| 			}
 326| 			w.Header().Set("Access-Control-Allow-Origin", r.Header.Get("Origin"))
 327| 			w.Header().Set("Access-Control-Allow-Methods", "GET, POST, DELETE, PUT, OPTIONS")
 328| 			w.Header().Set("Access-Control-Allow-Headers", "*")
 329| 		}
 330| 
 331| 		// Check basic authentication headers if configured.
 332| 		// OPTIONS requests are skipped if CORS is enabled, since browsers omit credentials for preflight checks.
 333| 		// skipUIAuthKey in the request context allows sendAPIAuthMiddleware to bypass UI auth
 334| 		// for a specific request without touching the global auth.UICredentials pointer.
 335| 		skipUIAuth, _ := r.Context().Value(skipUIAuthKey).(bool)
 336| 		isCORSOptionsRequest := AccessControlAllowOrigin != "" && r.Method == http.MethodOptions
 337| 		if !skipUIAuth && !isCORSOptionsRequest && auth.UICredentials != nil {
 338| 			user, pass, ok := r.BasicAuth()
 339| 
 340| 			if !ok {
 341| 				basicAuthResponse(w)
 342| 				return
 343| 			}
 344| 
 345| 			if !auth.UICredentials.Match(user, pass) {
 346| 				basicAuthResponse(w)
 347| 				return
 348| 			}
 349| 		}
 350| 
 351| 		// WebSocket upgrade requests must not be wrapped in a gzip writer:
 352| 		// gzipResponseWriter does not implement http.Hijacker, which the
 353| 		// WebSocket library requires to take over the raw TCP connection.
 354| 		isWebSocketUpgrade := strings.EqualFold(r.Header.Get("Upgrade"), "websocket")
 355| 		if isWebSocketUpgrade || config.DisableHTTPCompression || !strings.Contains(r.Header.Get("Accept-Encoding"), "gzip") {
 356| 			fn(w, r)
 357| 			return
 358| 		}
 359| 
 360| 		w.Header().Set("Content-Encoding", "gzip")
 361| 		gz := gzip.NewWriter(w)
 362| 		defer func() { _ = gz.Close() }()
 363| 		gzr := gzipResponseWriter{Writer: gz, ResponseWriter: w}
 364| 		fn(gzr, r)
 365| 	}
 366| }

 235| // sendAPIAuthMiddleware handles authentication specifically for the send API endpoint.
 236| // It can use dedicated send API authentication or accept any credentials based on configuration.
 237| // It communicates skip-UI-auth intent via request context rather than mutating the global
 238| // auth.UICredentials pointer, which would be a data race under concurrent load.
 239| func sendAPIAuthMiddleware(fn http.HandlerFunc) http.HandlerFunc {
 240| 	return func(w http.ResponseWriter, r *http.Request) {
 241| 		// Override the default 5 MB body cap with the send-specific limit so that
 242| 		// middleWareFunc applies config.MaxMessageSize (0 = unlimited) instead.
 243| 		r = r.WithContext(context.WithValue(r.Context(), bodyLimitKey, int64(config.MaxMessageSize)*1024*1024))
 244| 
 245| 		// If send API auth accept any is enabled, bypass all authentication.
 246| 		if config.SendAPIAuthAcceptAny {
 247| 			ctx := context.WithValue(r.Context(), skipUIAuthKey, true)
 248| 			middleWareFunc(fn)(w, r.WithContext(ctx))
 249| 			return
 250| 		}
 251| 
 252| 		// If Send API credentials are configured, only accept those credentials.
 253| 		if auth.SendAPICredentials != nil {
 254| 			user, pass, ok := r.BasicAuth()
 255| 
 256| 			if !ok {
 257| 				basicAuthResponse(w)
 258| 				return
 259| 			}
 260| 
 261| 			if !auth.SendAPICredentials.Match(user, pass) {
 262| 				basicAuthResponse(w)
 263| 				return
 264| 			}
 265| 
 266| 			// Valid Send API credentials — bypass UI auth via context flag.
 267| 			ctx := context.WithValue(r.Context(), skipUIAuthKey, true)
 268| 			middleWareFunc(fn)(w, r.WithContext(ctx))
 269| 			return
 270| 		}
 271| 
 272| 		// No Send API credentials configured — fall back to UI auth.
 273| 		middleWareFunc(fn)(w, r)
 274| 	}

 286| // MiddleWareFunc http middleware adds optional basic authentication
 287| // and gzip compression.
 288| func middleWareFunc(fn http.HandlerFunc) http.HandlerFunc {
 289| 	return func(w http.ResponseWriter, r *http.Request) {
 290| 		// Limit request body size to 5 MB to prevent memory-exhaustion DoS via large
 291| 		// JSON bodies. sendAPIAuthMiddleware sets bodyLimitKey in the context to signal
 292| 		// that the handler manages its own limit (send.go uses config.MaxMessageSize),
 293| 		// so we skip the cap here for that route only.
 294| 		if _, ok := r.Context().Value(bodyLimitKey).(int64); !ok {
 295| 			r.Body = http.MaxBytesReader(w, r.Body, 5*1024*1024)
 296| 		}
 297| 
 298| 		w.Header().Set("Referrer-Policy", "no-referrer")
 299| 
 300| 		// generate a new random nonce on every request
 301| 		randomNonce := shortuuid.New()
 302| 		// header used to pass nonce through to function
 303| 		r.Header.Set("mp-nonce", randomNonce)
 304| 
 305| 		// Prevent JavaScript XSS by adding a nonce for script-src
 306| 		cspHeader := strings.Replace(
 307| 			config.ContentSecurityPolicy,
 308| 			"script-src 'self';",
 309| 			fmt.Sprintf("script-src 'nonce-%s';", randomNonce),
 310| 			1,
 311| 		)
 312| 
 313| 		w.Header().Set("Content-Security-Policy", cspHeader)
 314| 
 315| 		if htmlPreviewRouteRe == nil {
 316| 			htmlPreviewRouteRe = regexp.MustCompile(`^` + regexp.QuoteMeta(config.Webroot) + `view/[a-zA-Z0-9]+\.html$`)
 317| 		}
 318| 
 319| 		if strings.HasPrefix(r.URL.Path, config.Webroot+"api/") || htmlPreviewRouteRe.MatchString(r.URL.Path) {
 320| 			if allowed := corsOriginAccessControl(r); !allowed {
 321| 				http.Error(w, "Blocked due to CORS violation", http.StatusForbidden)
 322| 				return
 323| 			}
 324| 			w.Header().Set("Access-Control-Allow-Origin", r.Header.Get("Origin"))
 325| 			w.Header().Set("Access-Control-Allow-Methods", "GET, POST, DELETE, PUT, OPTIONS")
 326| 			w.Header().Set("Access-Control-Allow-Headers", "*")
 327| 		}
 328| 
 329| 		// Check basic authentication headers if configured.
 330| 		// OPTIONS requests are skipped if CORS is enabled, since browsers omit credentials for preflight checks.
 331| 		// skipUIAuthKey in the request context allows sendAPIAuthMiddleware to bypass UI auth
 332| 		// for a specific request without touching the global auth.UICredentials pointer.
 333| 		skipUIAuth, _ := r.Context().Value(skipUIAuthKey).(bool)
 334| 		isCORSOptionsRequest := AccessControlAllowOrigin != "" && r.Method == http.MethodOptions
 335| 		if !skipUIAuth && !isCORSOptionsRequest && auth.UICredentials != nil {
 336| 			user, pass, ok := r.BasicAuth()
 337| 
 338| 			if !ok {
 339| 				basicAuthResponse(w)
 340| 				return
 341| 			}
 342| 
 343| 			if !auth.UICredentials.Match(user, pass) {
 344| 				basicAuthResponse(w)
 345| 				return
 346| 			}
 347| 		}
 348| 
 349| 		// WebSocket upgrade requests must not be wrapped in a gzip writer:
 350| 		// gzipResponseWriter does not implement http.Hijacker, which the
 351| 		// WebSocket library requires to take over the raw TCP connection.
 352| 		isWebSocketUpgrade := strings.EqualFold(r.Header.Get("Upgrade"), "websocket")
 353| 		if isWebSocketUpgrade || config.DisableHTTPCompression || !strings.Contains(r.Header.Get("Accept-Encoding"), "gzip") {
 354| 			fn(w, r)

 356| 		}
 357| 
 358| 		w.Header().Set("Content-Encoding", "gzip")
 359| 		gz := gzip.NewWriter(w)
 360| 		defer func() { _ = gz.Close() }()

 362| 		fn(gzr, r)
 363| 	}
 364| }
