 105| func VerifySignatureWithKeys(publicKeys []PublicKeyInfo, messageBytes []byte, signature []byte) error {
 106| 	var lastErr error
 107| 	allMismatch := true
 108| 	for _, publicKeyInfo := range publicKeys {
 109| 		err := publicKeyInfo.VerifySignature(messageBytes, signature)
 110| 		if err == nil {
 111| 			return nil
 112| 		}
 113| 		lastErr = err
 114| 		if !errors.Is(err, ErrSignatureMismatch) {
 115| 			allMismatch = false
 116| 		}
 117| 	}
 118| 
 119| 	// If at least one key failed for a structural reason (wrong size, unsupported algorithm),
 120| 	// surface that error directly — it's more actionable than a generic "didn't match" message.
 121| 	if !allMismatch {
 122| 		return lastErr
 123| 	}
 124| 
 125| 	// Every key was tried and produced a clean cryptographic mismatch. Include short
 126| 	// fingerprints of every key that was tried so users can tell which published keys the
 127| 	// registry actually saw — the most common cause of this error is a stale record left
 128| 	// behind after a key rotation, which is otherwise indistinguishable from a generic
 129| 	// crypto failure.
 130| 	fingerprints := make([]string, 0, len(publicKeys))
 131| 	for _, publicKeyInfo := range publicKeys {
 132| 		fingerprints = append(fingerprints, publicKeyInfo.Fingerprint())
 133| 	}
 134| 	if len(publicKeys) == 1 {
 135| 		return fmt.Errorf(
 136| 			"signature verification failed (tried published key %s); "+
 137| 				"if this is not the key you are signing with, the published record may be stale",
 138| 			fingerprints[0],
 139| 		)
 140| 	}
 141| 	return fmt.Errorf(
 142| 		"signature verification failed against all %d published keys (tried: %s); "+
 143| 			"if you recently rotated keys, remove any stale records from the apex domain",
 144| 		len(publicKeys), strings.Join(fingerprints, ", "),

 148| // Fingerprint returns a short, human-readable identifier for the public key.
 149| // Format: "<algorithm>:<first 8 base64 chars of the raw key>". Public keys are not secret,
 150| // but truncating keeps error messages readable.
 151| func (pki *PublicKeyInfo) Fingerprint() string {
 152| 	const prefixLen = 8
 153| 	var raw []byte
 154| 	switch pki.Algorithm {
 155| 	case AlgorithmEd25519:
 156| 		if k, ok := pki.Key.(ed25519.PublicKey); ok {
 157| 			raw = k
 158| 		}
 159| 	case AlgorithmECDSAP384:
 160| 		if k, ok := pki.Key.(ecdsa.PublicKey); ok {
 161| 			raw = elliptic.MarshalCompressed(k.Curve, k.X, k.Y) //nolint:staticcheck // SA1019: matches the encoding used in DNS records
 162| 		}
 163| 	}
 164| 	if len(raw) == 0 {
 165| 		return fmt.Sprintf("%s:unknown", pki.Algorithm)
 166| 	}
 167| 	encoded := base64.StdEncoding.EncodeToString(raw)
 168| 	if len(encoded) > prefixLen {
 169| 		encoded = encoded[:prefixLen]
 170| 	}
 171| 	return fmt.Sprintf("%s:%s", pki.Algorithm, encoded)

 174| // VerifySignature verifies a signature using the appropriate algorithm
 175| func (pki *PublicKeyInfo) VerifySignature(message, signature []byte) error {
 176| 	switch pki.Algorithm {
 177| 	case AlgorithmEd25519:
 178| 		if ed25519Key, ok := pki.Key.(ed25519.PublicKey); ok {
 179| 			if len(signature) != ed25519.SignatureSize {
 180| 				return fmt.Errorf("invalid signature size for Ed25519")
 181| 			}
 182| 			if !ed25519.Verify(ed25519Key, message, signature) {
 183| 				return fmt.Errorf("Ed25519: %w", ErrSignatureMismatch)
 184| 			}
 185| 			return nil
 186| 		}
 187| 	case AlgorithmECDSAP384:
 188| 		if ecdsaKey, ok := pki.Key.(ecdsa.PublicKey); ok {
 189| 			if len(signature) != 96 {
 190| 				return fmt.Errorf("invalid signature size for ECDSA P-384")
 191| 			}
 192| 			r := new(big.Int).SetBytes(signature[:48])
 193| 			s := new(big.Int).SetBytes(signature[48:])
 194| 			digest := sha512.Sum384(message)
 195| 			if !ecdsa.Verify(&ecdsaKey, digest[:], r, s) {
 196| 				return fmt.Errorf("ECDSA P-384: %w", ErrSignatureMismatch)
 197| 			}
 198| 			return nil
 199| 		}
 200| 	}
 201| 	return fmt.Errorf("unsupported public key algorithm")
 202| }

 329| func ParsePublicKey(algorithm, publicKey string) (*PublicKeyInfo, error) {
 330| 	publicKeyBytes, err := base64.StdEncoding.DecodeString(publicKey)
 331| 	if err != nil {
 332| 		return nil, fmt.Errorf("failed to decode public key: %w", err)
 333| 	}
 334| 
 335| 	// match to a supported crypto algorithm
 336| 	switch algorithm {
 337| 	case string(AlgorithmEd25519):
 338| 		if len(publicKeyBytes) != ed25519.PublicKeySize {
 339| 			return nil, fmt.Errorf("invalid Ed25519 public key size")
 340| 		}
 341| 		return &PublicKeyInfo{
 342| 			Algorithm: AlgorithmEd25519,
 343| 			Key:       ed25519.PublicKey(publicKeyBytes),
 344| 		}, nil
 345| 	case string(AlgorithmECDSAP384):
 346| 		if len(publicKeyBytes) != 49 {
 347| 			return nil, fmt.Errorf("invalid ECDSA P-384 public key size")
 348| 		}
 349| 		if publicKeyBytes[0] != 0x02 && publicKeyBytes[0] != 0x03 {
 350| 			return nil, fmt.Errorf("invalid ECDSA P-384 public key format (must be compressed, with a leading 0x02 or 0x03 byte)")
 351| 		}
 352| 		curve := elliptic.P384()
 353| 		x, y := elliptic.UnmarshalCompressed(curve, publicKeyBytes)
 354| 		if x == nil || y == nil {
 355| 			return nil, fmt.Errorf("failed to decompress ECDSA P-384 public key")
 356| 		}
 357| 		return &PublicKeyInfo{
 358| 			Algorithm: AlgorithmECDSAP384,
 359| 			Key:       ecdsa.PublicKey{Curve: curve, X: x, Y: y},
 360| 		}, nil
 361| 	}
 362| 
 363| 	return nil, fmt.Errorf("unsupported public key algorithm: %s", algorithm)
 364| }

 366| // ReverseString reverses a domain string (example.com -> com.example)
 367| func ReverseString(domain string) string {
 368| 	parts := strings.Split(domain, ".")
 369| 	for i, j := 0, len(parts)-1; i < j; i, j = i+1, j-1 {
 370| 		parts[i], parts[j] = parts[j], parts[i]
 371| 	}
 372| 	return strings.Join(parts, ".")
 373| }

 375| func IsValidDomain(domain string) bool {
 376| 	if len(domain) == 0 || len(domain) > 253 {
 377| 		return false
 378| 	}
 379| 
 380| 	// Reject IP literals — this auth method proves domain ownership, not IP
 381| 	// ownership, and IP literals are an SSRF vector into internal networks.
 382| 	if net.ParseIP(domain) != nil {
 383| 		return false
 384| 	}
 385| 
 386| 	// Require at least one dot — rejects single-label names like "localhost"
 387| 	// or "kubernetes" that resolve only inside private networks.
 388| 	if !strings.Contains(domain, ".") {
 389| 		return false
 390| 	}
 391| 
 392| 	// Check for valid characters and structure
 393| 	domainPattern := regexp.MustCompile(`^[a-zA-Z0-9]([a-zA-Z0-9-]*[a-zA-Z0-9])?(\.[a-zA-Z0-9]([a-zA-Z0-9-]*[a-zA-Z0-9])?)*$`)
 394| 	return domainPattern.MatchString(domain)
 395| }
