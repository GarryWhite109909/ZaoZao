  64| function launchEditor (file, specifiedEditor, onErrorCallback) {
  65|   const parsed = parseFile(file)
  66|   let { fileName } = parsed
  67|   const { lineNumber, columnNumber } = parsed
  68| 
  69|   if (!fs.existsSync(fileName)) {
  70|     return
  71|   }
  72| 
  73|   if (typeof specifiedEditor === 'function') {
  74|     onErrorCallback = specifiedEditor
  75|     specifiedEditor = undefined
  76|   }
  77| 
  78|   onErrorCallback = wrapErrorCallback(onErrorCallback)
  79| 
  80|   const [editor, ...args] = guessEditor(specifiedEditor)
  81|   if (!editor) {
  82|     onErrorCallback(fileName, null)
  83|     return
  84|   }
  85| 
  86|   if (
  87|     process.platform === 'linux' &&
  88|     fileName.startsWith('/mnt/') &&
  89|     /Microsoft/i.test(os.release())
  90|   ) {
  91|     // Assume WSL / "Bash on Ubuntu on Windows" is being used, and
  92|     // that the file exists on the Windows file system.
  93|     // `os.release()` is "4.4.0-43-Microsoft" in the current release
  94|     // build of WSL, see: https://github.com/Microsoft/BashOnWindows/issues/423#issuecomment-221627364
  95|     // When a Windows editor is specified, interop functionality can
  96|     // handle the path translation, but only if a relative path is used.
  97|     fileName = path.relative('', fileName)
  98|   }
  99| 
 100|   if (lineNumber) {
 101|     const extraArgs = getArgumentsForPosition(editor, fileName, lineNumber, columnNumber)
 102|     args.push.apply(args, extraArgs)
 103|   } else {
 104|     args.push(fileName)
 105|   }
 106| 
 107|   if (_childProcess && isTerminalEditor(editor)) {
 108|     // There's an existing editor process already and it's attached
 109|     // to the terminal, so go kill it. Otherwise two separate editor
 110|     // instances attach to the stdin/stdout which gets confusing.
 111|     _childProcess.kill('SIGKILL')
 112|   }
 113| 
 114|   if (process.platform === 'win32') {
 115|     // On Windows, we need to use `exec` with the `shell: true` option,
 116|     // and some more sanitization is required.
 117| 
 118|     // However, CMD.exe on Windows is vulnerable to RCE attacks given a file name of the
 119|     // form "C:\Users\myusername\Downloads\& curl 172.21.93.52".
 120|     // `create-react-app` used a safe file name pattern to validate user-provided file names:
 121|     // - https://github.com/facebook/create-react-app/pull/4866
 122|     // - https://github.com/facebook/create-react-app/pull/5431
 123|     // But that's not a viable solution for this package because
 124|     // it's depended on by so many meta frameworks that heavily rely on
 125|     // special characters in file names for filesystem-based routing.
 126|     // We need to at least:
 127|     // - Support `+` because it's used in SvelteKit and Vike
 128|     // - Support `$` because it's used in Remix
 129|     // - Support `(` and `)` because they are used in Analog, SolidStart, and Vike
 130|     // - Support `@` because it's used in Vike
 131|     // - Support `[` and `]` because they are widely used for [slug]
 132|     // So here we choose to use `^` to escape special characters instead.
 133| 
 134|     // According to https://ss64.com/nt/syntax-esc.html,
 135|     // we can use `^` to escape `&`, `<`, `>`, `|`, `%`, and `^`
 136|     // I'm not sure if we have to escape all of these, but let's do it anyway
 137|     function escapeCmdArgs (cmdArgs) {
 138|       return cmdArgs.replace(/([&|<>,;=^])/g, '^$1')
 139|     }
 140| 
 141|     // Need to double quote the editor path in case it contains spaces;
 142|     // If the fileName contains spaces, we also need to double quote it in the arguments
 143|     // However, there's a case that it's concatenated with line number and column number
 144|     // which is separated by `:`. We need to double quote the whole string in this case.
 145|     // Also, if the string contains the escape character `^`, it needs to be quoted, too.
 146|     function doubleQuoteIfNeeded(str) {

 152|         return `"${str}"`
 153|       } 
 154|       return str
 155|     }
 156|     const launchCommand = [editor, ...args.map(escapeCmdArgs)]

 158|       .join(' ')
 159| 
 160|     _childProcess = childProcess.exec(launchCommand, {
 161|       stdio: 'inherit',
 162|       shell: true
 163|     })
 164|   } else {
 165|     _childProcess = childProcess.spawn(editor, args, { stdio: 'inherit' })
 166|   }
 167|   _childProcess.on('exit', function (errorCode) {
 168|     _childProcess = null

 170|     if (errorCode) {

 172|     }
 173|   })

 179|     }

 181|   })
