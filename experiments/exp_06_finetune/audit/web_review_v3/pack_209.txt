 258| // Run is a blocking function (that should be invoked as a goroutine)
 259| // that scans the data source at regular intervals for pending campaigns,
 260| // and queues them for processing. The process queue fetches batches of
 261| // subscribers and pushes messages to them for each queued campaign
 262| // until all subscribers are exhausted, at which point, a campaign is marked
 263| // as "finished".
 264| func (m *Manager) Run() {
 265| 	if m.cfg.ScanCampaigns {
 266| 		// Periodically scan campaigns and push running campaigns to nextPipes
 267| 		// to fetch subscribers from the campaign.
 268| 		go m.scanCampaigns(m.cfg.ScanInterval)
 269| 	}
 270| 
 271| 	// Spawn N message workers.
 272| 	for i := 0; i < m.cfg.Concurrency; i++ {
 273| 		go m.worker()
 274| 	}
 275| 
 276| 	// Indefinitely wait on the pipe queue to fetch the next set of subscribers
 277| 	// for any active campaigns.
 278| 	for p := range m.nextPipes {
 279| 		has, err := p.NextSubscribers()
 280| 		if err != nil {
 281| 			m.log.Printf("error processing campaign batch (%s): %v", p.camp.Name, err)
 282| 			continue
 283| 		}
 284| 
 285| 		if has {
 286| 			// There are more subscribers to fetch. Queue again.
 287| 			select {
 288| 			case m.nextPipes <- p:
 289| 			default:
 290| 			}
 291| 		} else {
 292| 			// The pipe is created with a +1 on the waitgroup pseudo counter
 293| 			// so that it immediately waits. Subsequently, every message created
 294| 			// is incremented in the counter in pipe.newMessage(), and when it's'
 295| 			// processed (or ignored when a campaign is paused or cancelled),
 296| 			// the count is's reduced in worker().
 297| 			//
 298| 			// This marks down the original non-message +1, causing the waitgroup
 299| 			// to be released and the pipe to end, triggering the pg.Wait()
 300| 			// in newPipe() that calls pipe.cleanup().
 301| 			p.wg.Done()
 302| 		}
 303| 	}
 304| }
