 273|     ``categories()``, which returns a list of the categories for the

 287|     def __init__(self, kwargs):

 305|         more than one argument is specified, an exception will be
 306|         raised.

 309|         self._c2f = None  #: category-to-file mapping

 311|         self._pattern = None  #: regexp specifying the mapping
 312|         self._map = None  #: dict specifying the mapping
 313|         self._file = None  #: fileid of file containing the mapping
 314|         self._delimiter = None  #: delimiter for ``self._file``
 315| 
 316|         if "cat_pattern" in kwargs:
 317|             self._pattern = kwargs["cat_pattern"]
 318|             del kwargs["cat_pattern"]
 319|         elif "cat_map" in kwargs:
 320|             self._map = kwargs["cat_map"]
 321|             del kwargs["cat_map"]
 322|         elif "cat_file" in kwargs:
 323|             self._file = kwargs["cat_file"]
 324|             del kwargs["cat_file"]

 338|     def _init(self):

 340|         self._c2f = defaultdict(set)

 344|                 category = re.match(self._pattern, file_id).group(1)
 345|                 self._add(file_id, category)

 369|     def categories(self, fileids=None):
