  86|     public KeyPair generateKeyPair(String algorithm, String keyId, Object parameterSpec) throws Exception {
  87|         LOG.info("Generating key pair for algorithm: {}, keyId: {}", algorithm, keyId);
  88| 
  89|         KeyPairGenerator generator;
  90|         String provider = determineProvider(algorithm);
  91| 
  92|         if (provider != null) {
  93|             generator = KeyPairGenerator.getInstance(getAlgorithmName(algorithm), provider);
  94|         } else {
  95|             generator = KeyPairGenerator.getInstance(getAlgorithmName(algorithm));
  96|         }
  97| 
  98|         // Initialize with parameter spec if provided
  99|         if (parameterSpec != null) {
 100|             if (parameterSpec instanceof AlgorithmParameterSpec algorithmParamSpec) {
 101|                 generator.initialize(algorithmParamSpec, new SecureRandom());
 102|             } else if (parameterSpec instanceof Integer keySize) {
 103|                 generator.initialize(keySize, new SecureRandom());
 104|             }
 105|         } else {
 106|             // Use default parameter spec for the algorithm
 107|             AlgorithmParameterSpec defaultSpec = getDefaultParameterSpec(algorithm);
 108|             if (defaultSpec != null) {
 109|                 generator.initialize(defaultSpec, new SecureRandom());
 110|             } else {
 111|                 generator.initialize(getDefaultKeySize(algorithm), new SecureRandom());
 112|             }
 113|         }
 114| 
 115|         KeyPair keyPair = generator.generateKeyPair();
 116| 
 117|         // Create metadata
 118|         KeyMetadata metadata = new KeyMetadata(keyId, algorithm);
 119|         metadata.setDescription("Generated on " + new Date());
 120| 
 121|         // Store the key
 122|         storeKey(keyId, keyPair, metadata);
 123| 
 124|         LOG.info("Generated key pair: {}", metadata);
 125|         return keyPair;

 332|     private KeyPair readStandardKeyPair(String keyId, Path privateKeyFile, Path publicKeyFile) throws Exception {
 333|         String privateJson = Files.readString(privateKeyFile, StandardCharsets.UTF_8);
 334|         KeyFileData privateData = objectMapper.readValue(privateJson, KeyFileData.class);
 335|         byte[] privateKeyBytes = Base64.getDecoder().decode(privateData.key());
 336|         PKCS8EncodedKeySpec privateSpec = new PKCS8EncodedKeySpec(privateKeyBytes);
 337| 
 338|         String publicJson = Files.readString(publicKeyFile, StandardCharsets.UTF_8);
 339|         KeyFileData publicData = objectMapper.readValue(publicJson, KeyFileData.class);
 340|         byte[] publicKeyBytes = Base64.getDecoder().decode(publicData.key());
 341|         X509EncodedKeySpec publicSpec = new X509EncodedKeySpec(publicKeyBytes);
 342| 
 343|         String algorithm = privateData.algorithm();
 344|         String algorithmName = getAlgorithmName(algorithm);
 345|         String provider = determineProvider(algorithm);
 346| 
 347|         KeyFactory keyFactory;
 348|         if (provider != null) {
 349|             keyFactory = KeyFactory.getInstance(algorithmName, provider);
 350|         } else {
 351|             keyFactory = KeyFactory.getInstance(algorithmName);
 352|         }
 353| 
 354|         PrivateKey privateKey = keyFactory.generatePrivate(privateSpec);
 355|         PublicKey publicKey = keyFactory.generatePublic(publicSpec);
 356|         return new KeyPair(publicKey, privateKey);
 357|     }
