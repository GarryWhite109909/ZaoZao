 200|     """Analyze images using LLM vision capabilities."""

 203|     image_url = params.get('image_url', '')
 204|     prompt = params.get('prompt', 'Describe this image in detail')
 205|     provider = params.get('provider', 'openai')
 206|     model = params.get('model', 'gpt-4o')
 207|     api_key = params.get('api_key')
 208|     max_tokens = params.get('max_tokens', 1000)
 209|     detail = params.get('detail', 'auto')
 210| 
 211|     if not image_path and not image_url:
 212|         raise ValidationError(

 214|             field="image_path",
 215|         )
 216| 
 217|     # Resolve API key from environment if not provided
 218|     if not api_key:
 219|         env_vars = {
 220|             'openai': 'OPENAI_API_KEY',
 221|             'anthropic': 'ANTHROPIC_API_KEY',
 222|         }
 223|         api_key = os.getenv(env_vars.get(provider, ''))
 224|

 226|         raise ValidationError(
 227|             f"API key not provided for {provider}",
 228|             field="api_key",
 229|             hint=f"Set {provider.upper()}_API_KEY environment variable or provide api_key parameter",
 230|         )
 231| 
 232|     # Read image as base64 if path is provided
 233|     image_b64 = None
 234|     media_type = 'image/png'
 235| 
 236|     if image_path:
 237|         if not os.path.exists(image_path):
 238|             raise ValidationError(
 239|                 f"Image file not found: {image_path}",

 243|         if mime_type:
 244|             media_type = mime_type

 270| async def _call_openai_vision(
 271|     session: aiohttp.ClientSession,
 272|     api_key: str,
 273|     model: str,
 274|     prompt: str,
 275|     max_tokens: int,
 276|     detail: str,
 277|     image_b64: Optional[str],
 278|     image_url: str,
 279|     media_type: str,
 280| ) -> Dict[str, Any]:
 281|     """Call OpenAI Vision API."""
 282|     # Build image content
 283|     if image_b64:
 284|         image_content = {
 285|             "type": "image_url",
 286|             "image_url": {
 287|                 "url": f"data:{media_type};base64,{image_b64}",
 288|                 "detail": detail,
 289|             },
 290|         }
 291|     else:
 292|         image_content = {
 293|             "type": "image_url",
 294|             "image_url": {
 295|                 "url": image_url,
 296|                 "detail": detail,
 297|             },
 298|         }
 299| 
 300|     payload = {
 301|         "model": model,
 302|         "messages": [
 303|             {
 304|                 "role": "user",
 305|                 "content": [
 306|                     {"type": "text", "text": prompt},
 307|                     image_content,
 308|                 ],
 309|             }
 310|         ],
 311|         "max_tokens": max_tokens,
 312|     }
 313| 
 314|     headers = {
 315|         "Authorization": f"Bearer {api_key}",
 316|         "Content-Type": "application/json",
 317|     }
 318| 
 319|     async with session.post(
 320|         "https://api.openai.com/v1/chat/completions",
 321|         json=payload,

 329|

 332|

 344| async def _call_anthropic_vision(
 345|     session: aiohttp.ClientSession,
 346|     api_key: str,
 347|     model: str,
 348|     prompt: str,
 349|     max_tokens: int,
 350|     image_b64: Optional[str],
 351|     image_url: str,
 352|     media_type: str,
 353| ) -> Dict[str, Any]:
 354|     """Call Anthropic Vision API."""
 355|     # Build image source
 356|     if image_b64:
 357|         image_source = {
 358|             "type": "base64",
 359|             "media_type": media_type,
 360|             "data": image_b64,
 361|         }
 362|     else:
 363|         # For URL, we need to download and encode
 364|         async with session.get(image_url) as img_resp:
 365|             if img_resp.status != 200:
 366|                 raise ModuleError(f"Failed to download image from URL: {image_url}")
 367|             img_data = await img_resp.read()
 368|             content_type = img_resp.headers.get('Content-Type', media_type)
 369|             image_source = {
 370|                 "type": "base64",
 371|                 "media_type": content_type.split(';')[0],
 372|                 "data": base64.b64encode(img_data).decode('utf-8'),
 373|             }
 374| 
 375|     payload = {
 376|         "model": model,
 377|         "max_tokens": max_tokens,
 378|         "messages": [
 379|             {
 380|                 "role": "user",
 381|                 "content": [
 382|                     {
 383|                         "type": "image",
 384|                         "source": image_source,
 385|                     },
 386|                     {
 387|                         "type": "text",
 388|                         "text": prompt,
 389|                     },
 390|                 ],
 391|             }
 392|         ],
 393|     }
 394| 
 395|     headers = {
 396|         "x-api-key": api_key,
 397|         "anthropic-version": "2023-06-01",
 398|         "Content-Type": "application/json",
 399|     }
 400| 
 401|     async with session.post(
 402|         "https://api.anthropic.com/v1/messages",
 403|         json=payload,
 404|         headers=headers,
 405|     ) as resp:
 406|         data = await resp.json()
 407| 
 408|         if resp.status != 200:
 409|             error_msg = data.get('error', {}).get('message', str(data))
 410|             raise ModuleError(f"Anthropic API error: {error_msg}")
 411| 
 412|         content_blocks = data.get('content', [])
 413|         analysis = ''
 414|         for block in content_blocks:
 415|             if block.get('type') == 'text':
 416|                 analysis += block.get('text', '')
 417| 
 418|         usage = data.get('usage', {})
 419|         tokens_used = usage.get('input_tokens', 0) + usage.get('output_tokens', 0)
 420| 
 421|         return {
 422|             'ok': True,
 423|             'data': {
 424|                 'analysis': analysis,
 425|                 'model': data.get('model', model),
 426|                 'provider': 'anthropic',
 427|                 'tokens_used': tokens_used,
 428|             },
 429|         }

# 逐条分析

## 入口点枚举（全文）

1. **`ai_vision_analyze` 主入口**：`params` 中 `image_path`(L~200)、`image_url`、`prompt`、`provider`、`model`、`api_key`、`max_tokens`、`detail`。

4. **网络出站**：OpenAI/Anthropic 固定 URL POST；Anthropic 分支中 `session.get(image_url)`（L~345）下载任意 URL。

## 数据流验证

**流 A：`image_path` → `open()`**
- source: params 的 `image_path`；sink: L~243 `open(image_path, 'rb')`。
- 防御检查：仅有 `os.path.exists` 检查，**无路径规范化/白名单限制**。但该模块声明 `required_permissions=['filesystem.read']`，属于工作流引擎的受控模块，工作流作者本身可指定任意路径。若工作流参数来自不可信输入（如 Web 表单），则构成**任意文件读取**：读取 `/etc/passwd` 等后 base64 编码发送给外部 LLM API，数据外泄到第三方。mimetypes 仅决定 media_type，不拦截任何扩展名。**判定为真实漏洞（条件性）**：SSRF/任意文件读取决于平台是否允许不可信用户提交 params。作为开源 workflow 引擎模块，这是常见的高危模式。

**流 B：`image_url` → `session.get(image_url)`（仅 Anthropic 分支）**
- source: params 的 `image_url`；sink: L~345 GET 请求。
- 无任何 scheme/host 校验，无内网 IP 黑名单。**SSRF**：可访问 `http://169.254.169.254/latest/meta-data/`、内网服务，响应体被 base64 后进入 payload 并可能通过错误信息或分析结果回显（响应内容经 LLM 分析输出到 `analysis` 字段——间接回显通道）。OpenAI 分支虽不主动 fetch，但把 URL 交给 OpenAI 服务端 fetch，同样 SSRF（由 OpenAI 出口发起，风险较低）。

**流 C：`api_key` → headers**
- 凭据仅发送至固定官方域名，无泄漏面。

**流 D：`prompt`/`model`/`detail` → JSON payload**
- 经 aiohttp json 序列化，无注入面；`model` 可任意字符串但只影响 API 调用。`detail` 未校验取值，仅传给 API，无害。

**其他**：
- 错误处理将 API 响应 message 拼入异常，信息泄露轻微。
- `resp.json()` 对非 JSON 响应会抛异常，被 except ClientError 捕获不到（JSONDecodeError 不是 ClientError）——健壮性问题，非安全漏洞。
- 无重定向大小限制/超时设置于图片下载——DoS 面（次要）。

## 结论

主要漏洞：**SSRF（CWE-918）**，通过 `image_url` 参数在 Anthropic 分支直接请求任意 URL（含云元数据端点）；次要：**任意文件读取（CWE-22）**，`image_path` 无路径约束，文件内容外发至第三方 API。
