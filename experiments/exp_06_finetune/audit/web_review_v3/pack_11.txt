 269|     def _do_expiration(self) -> None:
 270|         """Remove expired cookies."""
 271|         if not (expire_heap_len := len(self._expire_heap)):
 272|             return
 273| 
 274|         # If the expiration heap grows larger than the number expirations
 275|         # times two, we clean it up to avoid keeping expired entries in
 276|         # the heap and consuming memory. We guard this with a minimum
 277|         # threshold to avoid cleaning up the heap too often when there are
 278|         # only a few scheduled expirations.
 279|         if (
 280|             expire_heap_len > _MIN_SCHEDULED_COOKIE_EXPIRATION
 281|             and expire_heap_len > len(self._expirations) * 2
 282|         ):
 283|             # Remove any expired entries from the expiration heap
 284|             # that do not match the expiration time in the expirations
 285|             # as it means the cookie has been re-added to the heap
 286|             # with a different expiration time.
 287|             self._expire_heap = [
 288|                 entry
 289|                 for entry in self._expire_heap
 290|                 if self._expirations.get(entry[1]) == entry[0]
 291|             ]
 292|             heapq.heapify(self._expire_heap)
 293| 
 294|         now = time.time()
 295|         to_del: list[tuple[str, str, str]] = []
 296|         # Find any expired cookies and add them to the to-delete list
 297|         while self._expire_heap:
 298|             when, cookie_key = self._expire_heap[0]
 299|             if when > now:
 300|                 break
 301|             heapq.heappop(self._expire_heap)
 302|             # Check if the cookie hasn't been re-added to the heap
 303|             # with a different expiration time as it will be removed
 304|             # later when it reaches the top of the heap and its
 305|             # expiration time is met.
 306|             if self._expirations.get(cookie_key) == when:
 307|                 to_del.append(cookie_key)
 308| 
 309|         if to_del:
 310|             self._delete_cookies(to_del)

 312|     def _delete_cookies(self, to_del: list[tuple[str, str, str]]) -> None:
 313|         for domain, path, name in to_del:
 314|             self._host_only_cookies.discard((domain, name))
 315|             self._cookies[(domain, path)].pop(name, None)
 316|             self._morsel_cache[(domain, path)].pop(name, None)
 317|             self._expirations.pop((domain, path, name), None)

 319|     def _expire_cookie(self, when: float, domain: str, path: str, name: str) -> None:
 320|         cookie_key = (domain, path, name)
 321|         if self._expirations.get(cookie_key) == when:

 489|     def _build_morsel(self, cookie: Morsel[str]) -> Morsel[str]:
 490|         """Build a morsel for sending, respecting quote_cookie setting."""
 491|         if self._quote_cookie and cookie.coded_value and cookie.coded_value[0] == '"':
 492|             return preserve_morsel_with_coded_value(cookie)
 493|         morsel: Morsel[str] = Morsel()

 495|             value, coded_value = _SIMPLE_COOKIE.value_encode(cookie.value)
 496|         else:

 498|         # We use __setstate__ instead of the public set() API because it allows us to
 499|         # bypass validation and set already validated state. This is more stable than
 500|         # setting protected attributes directly and unlikely to change since it would

 502|         morsel.__setstate__({"key": cookie.key, "value": value, "coded_value": coded_value})  # type: ignore[attr-defined]

 521|     @classmethod
 522|     def _parse_date(cls, date_str: str) -> int | None:
 523|         """Implements date string parsing adhering to RFC 6265."""
 524|         if not date_str:
 525|             return None
 526| 
 527|         found_time = False
 528|         found_day = False
 529|         found_month = False
 530|         found_year = False
 531| 
 532|         hour = minute = second = 0
 533|         day = 0
 534|         month = 0
 535|         year = 0
 536| 
 537|         for token_match in cls.DATE_TOKENS_RE.finditer(date_str):
 538| 
 539|             token = token_match.group("token")
 540| 
 541|             if not found_time:
 542|                 time_match = cls.DATE_HMS_TIME_RE.match(token)
 543|                 if time_match:
 544|                     found_time = True
 545|                     hour, minute, second = (int(s) for s in time_match.groups())
 546|                     continue
 547| 
 548|             if not found_day:
 549|                 day_match = cls.DATE_DAY_OF_MONTH_RE.match(token)
 550|                 if day_match:
 551|                     found_day = True
 552|                     day = int(day_match.group())
 553|                     continue
 554| 
 555|             if not found_month:
 556|                 month_match = cls.DATE_MONTH_RE.match(token)
 557|                 if month_match:
 558|                     found_month = True
 559|                     assert month_match.lastindex is not None
 560|                     month = month_match.lastindex
 561|                     continue
 562| 
 563|             if not found_year:
 564|                 year_match = cls.DATE_YEAR_RE.match(token)
 565|                 if year_match:
 566|                     found_year = True
 567|                     year = int(year_match.group())
 568|
