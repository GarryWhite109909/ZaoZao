 316| def _number_captions(md_body: str) -> str:
 317|     """Incrementally number tables and image captions."""
 318|     images = {}
 319|     tables = {}
 320|     output = md_body.splitlines()
 321| 
 322|     for index, line in enumerate(output):
 323|         if re.match(r'!\[.+\]\(.+\)', line):
 324|             images[index] = line
 325|         if output[index].lower().startswith('table: '):
 326|             tables[index] = line
 327| 
 328|     for index, row in enumerate(tables):
 329|         output[row] = f'Table: Table {index + 1} - {tables[row].split(": ")[1]}'
 330| 
 331|     for index, row in enumerate(images):
 332|         output[row] = images[row].replace('![', f'![Figure {index + 1} - ')
 333| 
 334|     return '\n'.join(output)
