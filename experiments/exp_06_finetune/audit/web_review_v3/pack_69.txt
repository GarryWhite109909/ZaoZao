 165|             -1 foo -> index = -1, value = foo
 166|             0 foo -> index = 0, value = foo
 167|             0 foo bar -> index = 0, value = foo bar
 168|             0 42 -> index = 0, value = 42

 222| def _get_formatter(value, **context):
 223|     """
 224|     Invokes settings.get from the annotation in value.
 225| 
 226|     value can be one of the following:
 227| 
 228|     @get KEY
 229|     @get KEY @int
 230|     @get KEY default_value
 231|     @get KEY @int default_value
 232| 
 233|     @marker KEY_TO_LOOKUP @OPTIONAL_CAST OPTIONAL_DEFAULT_VALUE
 234| 
 235|     key group will match the key
 236|     cast group will match anything provided after @
 237|     the default group will match anything between single or double quotes
 238|     """
 239|     tokens = value.strip().split()
 240|     if not tokens or len(tokens) > 3:
 241|         raise DynaconfFormatError(f"Error parsing {value} malformed syntax.")
 242| 
 243|     params = {
 244|         "key": tokens[0],
 245|         "cast": None,
 246|         "default": None,
 247|     }
 248|     for token in tokens[1:]:
 249|         if token.startswith("@"):
 250|             params["cast"] = token
 251|         else:
 252|             params["default"] = token
 253| 
 254|     if not params["default"] and params["key"] not in context["this"]:
 255|         raise DynaconfParseError(
 256|             f"Key {params['key']} not found in settings and no default value provided."
 257|         )

 431| def _parse_conf_data(data, tomlfy=False, box_settings=None):
 432|     """
 433|     @int @bool @float @json (for lists and dicts)
 434|     strings does not need converters
 435| 
 436|     export DYNACONF_DEFAULT_THEME='material'
 437|     export DYNACONF_DEBUG='@bool True'
 438|     export DYNACONF_DEBUG_TOOLBAR_ENABLED='@bool False'
 439|     export DYNACONF_PAGINATION_PER_PAGE='@int 20'
 440|     export DYNACONF_MONGODB_SETTINGS='@json {"DB": "quokka_db"}'
 441|     export DYNACONF_ALLOWED_EXTENSIONS='@json ["jpg", "png"]'
 442|     """
 443|     # not enforced to not break backwards compatibility with custom loaders
 444|     box_settings = box_settings or {}
 445| 
 446|     castenabled = box_settings.get("AUTO_CAST_FOR_DYNACONF", empty)
 447|     if castenabled is empty:
 448|         castenabled = (
 449|             os.environ.get("AUTO_CAST_FOR_DYNACONF", "true").lower()
 450|             not in false_values
 451|         )
 452| 
 453|     if (
 454|         castenabled
 455|         and data
 456|         and isinstance(data, str)
 457|         and data.startswith(tuple(converters.keys()))
 458|     ):
 459|         # Check combination token is used
 460|         comb_token = re.match(
 461|             f"^({'|'.join(converters.keys())}) @(jinja|format)",
 462|             data,
 463|         )
 464|         if comb_token:
 465|             tokens = comb_token.group(0)
 466|             converter_key_list = tokens.split(" ")
 467|             value = data.replace(tokens, "").strip()
 468|         else:
 469|             parts = data.partition(" ")
 470|             converter_key_list = [parts[0]]
 471|             value = parts[-1]
 472| 
 473|         # Parse the converters iteratively
 474|         for converter_key in converter_key_list[::-1]:
 475|             value = apply_converter(converter_key, value, box_settings)
 476|     else:
 477|         value = parse_with_toml(data) if tomlfy else data
 478| 
 479|     if isinstance(value, dict):
 480|         value = DynaBox(value, box_settings=box_settings)
 481| 
 482|     return value

 485| def parse_conf_data(data, tomlfy=False, box_settings=None):
 486|     """
 487|     Apply parsing tokens recursively and return transformed data.
 488| 
 489|     Strings with lazy parser (e.g, @format) will become Lazy objects.
 490|     """
 491| 
 492|     # fix for https://github.com/dynaconf/dynaconf/issues/595
 493|     if isnamedtupleinstance(data):
 494|         return data
 495| 
 496|     # not enforced to not break backwards compatibility with custom loaders
 497|     box_settings = box_settings or {}
 498| 
 499|     if isinstance(data, (tuple, list)):
 500|         # recursively parse each sequence item
 501|         return [
 502|             parse_conf_data(item, tomlfy=tomlfy, box_settings=box_settings)
 503|             for item in data
 504|         ]
 505| 
 506|     if isinstance(data, DynaBox):
 507|         # recursively parse inner dict items
 508|         # It is important to keep the same object id because
 509|         # of mutability
 510|         for k, v in data.items(bypass_eval=True):
 511|             data[k] = parse_conf_data(
 512|                 v, tomlfy=tomlfy, box_settings=box_settings
 513|             )
 514|         return data
 515| 
 516|     if isinstance(data, dict):
 517|         # recursively parse inner dict items
 518|         for k, v in data.items():
 519|             data[k] = parse_conf_data(
 520|                 v, tomlfy=tomlfy, box_settings=box_settings
 521|             )
 522|         return data
 523| 
 524|     # return parsed string value
 525|     return _parse_conf_data(data, tomlfy=tomlfy, box_settings=box_settings)
