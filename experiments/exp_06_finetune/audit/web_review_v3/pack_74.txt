 104| // symbolOffsets extracts all symbols offsets from an ELF and indexes them by
 105| // section and variable name.
 106| //
 107| // References to variables in BTF data sections carry unsigned 32-bit offsets.
 108| // Some ELF symbols (e.g. in vmlinux) may point to virtual memory that is well
 109| // beyond this range. Since these symbols cannot be described by BTF info,
 110| // ignore them here.
 111| func symbolOffsets(file *internal.SafeELFFile) (map[elfSymbol]uint32, error) {
 112| 	symbols, err := file.Symbols()
 113| 	if err != nil {
 114| 		return nil, fmt.Errorf("can't read symbols: %v", err)
 115| 	}
 116| 
 117| 	offsets := make(map[elfSymbol]uint32)
 118| 	for _, sym := range symbols {
 119| 		if idx := sym.Section; idx >= elf.SHN_LORESERVE && idx <= elf.SHN_HIRESERVE {
 120| 			// Ignore things like SHN_ABS
 121| 			continue
 122| 		}
 123| 
 124| 		if sym.Value > math.MaxUint32 {
 125| 			// VarSecinfo offset is u32, cannot reference symbols in higher regions.
 126| 			continue
 127| 		}
 128| 
 129| 		if int(sym.Section) >= len(file.Sections) {
 130| 			return nil, fmt.Errorf("symbol %s: invalid section %d", sym.Name, sym.Section)
 131| 		}
 132| 
 133| 		secName := file.Sections[sym.Section].Name
 134| 		offsets[elfSymbol{secName, sym.Name}] = uint32(sym.Value)
 135| 	}
 136| 
 137| 	return offsets, nil
 138| }

 195| func loadRawSpec(btf []byte, base *Spec) (*Spec, error) {
 196| 	var (
 197| 		baseDecoder *decoder
 198| 		baseStrings *stringTable
 199| 		err         error
 200| 	)
 201| 
 202| 	if base != nil {
 203| 		baseDecoder = base.decoder
 204| 		baseStrings = base.strings

 207| 	header, bo, err := parseBTFHeader(btf)
 208| 	if err != nil {

 210| 	}

 240| // fixupDatasec attempts to patch up missing info in Datasecs and its members by
 241| // supplementing them with information from the ELF headers and symbol table.
 242| func (elf *elfData) fixupDatasec(typ Type) error {
 243| 	if elf == nil {
 244| 		return nil
 245| 	}
 246| 
 247| 	if ds, ok := typ.(*Datasec); ok {
 248| 		if elf.fixups[ds] {
 249| 			return nil
 250| 		}
 251| 		elf.fixups[ds] = true
 252| 
 253| 		name := ds.Name
 254| 
 255| 		// Some Datasecs are virtual and don't have corresponding ELF sections.
 256| 		switch name {
 257| 		case ".ksyms":
 258| 			// .ksyms describes forward declarations of kfunc signatures, as well as
 259| 			// references to kernel symbols.
 260| 			// Nothing to fix up, all sizes and offsets are 0.
 261| 			for _, vsi := range ds.Vars {
 262| 				switch t := vsi.Type.(type) {
 263| 				case *Func:
 264| 					continue
 265| 				case *Var:
 266| 					if _, ok := t.Type.(*Void); !ok {
 267| 						return fmt.Errorf("data section %s: expected %s to be *Void, not %T: %w", name, vsi.Type.TypeName(), vsi.Type, ErrNotSupported)
 268| 					}
 269| 				default:
 270| 					return fmt.Errorf("data section %s: expected to be either *btf.Func or *btf.Var, not %T: %w", name, vsi.Type, ErrNotSupported)
 271| 				}
 272| 			}
 273| 
 274| 			return nil
 275| 		case ".kconfig":
 276| 			// .kconfig has a size of 0 and has all members' offsets set to 0.
 277| 			// Fix up all offsets and set the Datasec's size.
 278| 			if err := fixupDatasecLayout(ds); err != nil {
 279| 				return err
 280| 			}
 281| 
 282| 			// Fix up extern to global linkage to avoid a BTF verifier error.
 283| 			for _, vsi := range ds.Vars {
 284| 				vsi.Type.(*Var).Linkage = GlobalVar
 285| 			}
 286| 
 287| 			return nil
 288| 		}
 289| 
 290| 		if ds.Size != 0 {
 291| 			return nil
 292| 		}
 293| 
 294| 		ds.Size, ok = elf.sectionSizes[name]
 295| 		if !ok {
 296| 			return fmt.Errorf("data section %s: missing size", name)
 297| 		}
 298| 
 299| 		for i := range ds.Vars {
 300| 			symName := ds.Vars[i].Type.TypeName()
 301| 			ds.Vars[i].Offset, ok = elf.symbolOffsets[elfSymbol{name, symName}]
 302| 			if !ok {
 303| 				return fmt.Errorf("data section %s: missing offset for symbol %s", name, symName)
 304| 			}
 305| 		}
 306| 	}
 307| 
 308| 	return nil
 309| }

 311| // fixupDatasecLayout populates ds.Vars[].Offset according to var sizes and
 312| // alignment. Calculate and set ds.Size.
 313| func fixupDatasecLayout(ds *Datasec) error {
 314| 	var off uint32
 315| 
 316| 	for i, vsi := range ds.Vars {
 317| 		v, ok := vsi.Type.(*Var)
 318| 		if !ok {
 319| 			return fmt.Errorf("member %d: unsupported type %T", i, vsi.Type)
 320| 		}
 321| 
 322| 		size, err := Sizeof(v.Type)
 323| 		if err != nil {
 324| 			return fmt.Errorf("variable %s: getting size: %w", v.Name, err)
 325| 		}
 326| 		align, err := alignof(v.Type)
 327| 		if err != nil {
 328| 			return fmt.Errorf("variable %s: getting alignment: %w", v.Name, err)
 329| 		}
 330| 
 331| 		// Align the current member based on the offset of the end of the previous
 332| 		// member and the alignment of the current member.
 333| 		off = internal.Align(off, uint32(align))
 334| 
 335| 		ds.Vars[i].Offset = off
 336| 
 337| 		off += uint32(size)
 338| 	}
 339| 
 340| 	ds.Size = off
 341| 
 342| 	return nil
 343| }

 439| // TypeByName searches for a Type with a specific name. Since multiple Types
 440| // with the same name can exist, the parameter typ is taken to narrow down the
 441| // search in case of a clash.
 442| //
 443| // typ must be a non-nil pointer to an implementation of a Type. On success, the
 444| // address of the found Type will be copied to typ.
 445| //
 446| // Returns an error wrapping ErrNotFound if no matching Type exists in the Spec.
 447| // Returns an error wrapping ErrMultipleTypes if multiple candidates are found.
 448| func (s *Spec) TypeByName(name string, typ any) error {
 449| 	if err := internal.IsNilPointer(typ); err != nil {
 450| 		return fmt.Errorf("type argument: %w", err)
 451| 	}
 452| 
 453| 	typeInterface := reflect.TypeFor[Type]()
 454| 
 455| 	// typ may be **T or *Type
 456| 	typValue := reflect.ValueOf(typ)
 457| 	if typValue.Kind() != reflect.Pointer {
 458| 		return fmt.Errorf("%T is not a pointer", typ)
 459| 	}
 460| 	typPtr := typValue.Elem()
 461| 	if !typPtr.CanSet() {
 462| 		return fmt.Errorf("%T cannot be set", typ)
 463| 	}
 464| 
 465| 	wanted := typPtr.Type()
 466| 	if wanted == typeInterface {
 467| 		// This is *Type. Unwrap the value's type.
 468| 		if typPtr.IsNil() {
 469| 			return fmt.Errorf("%T points to a nil Type", typ)
 470| 		}
 471| 		wanted = typPtr.Elem().Type()
 472| 	}
 473| 
 474| 	if !wanted.AssignableTo(typeInterface) {
 475| 		return fmt.Errorf("%T does not satisfy Type interface", typ)
 476| 	}
 477| 
 478| 	types, err := s.AnyTypesByName(name)
 479| 	if err != nil {
 480| 		return err
 481| 	}
 482| 
 483| 	var candidate Type
 484| 	for _, typ := range types {
 485| 		if reflect.TypeOf(typ) != wanted {
 486| 			continue
 487| 		}
 488| 
 489| 		if candidate != nil {
 490| 			return fmt.Errorf("type %s(%T): %w", name, typ, ErrMultipleMatches)
 491| 		}
 492| 
 493| 		candidate = typ
 494| 	}
 495| 
 496| 	if candidate == nil {
 497| 		return fmt.Errorf("%s %s: %w", wanted, name, ErrNotFound)
 498| 	}
 499| 
 500| 	typPtr.Set(reflect.ValueOf(candidate))
 501| 
 502| 	return nil
 503| }

 532| // All iterates over all types.
 533| func (s *Spec) All() iter.Seq2[Type, error] {
 534| 	return func(yield func(Type, error) bool) {
 535| 		for id := s.firstTypeID; ; id++ {
 536| 			typ, err := s.TypeByID(id)
 537| 			if errors.Is(err, ErrNotFound) {
 538| 				return
 539| 			} else if err != nil {
 540| 				yield(nil, err)
 541| 				return
 542| 			}
 543| 
 544| 			// Skip declTags, during unmarshaling declTags become `Tags` fields of other types.
 545| 			// We keep them in the spec to avoid holes in the ID space, but for the purposes of
 546| 			// iteration, they are not useful to the user.
 547| 			if _, ok := typ.(*declTag); ok {
 548| 				continue
 549| 			}
 550| 
 551| 			if !yield(typ, nil) {
 552| 				return
 553| 			}
 554| 		}
 555| 	}
