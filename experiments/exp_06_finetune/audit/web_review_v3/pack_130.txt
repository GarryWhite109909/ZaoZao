 156| 		}

 158| 		count, err := db.CountByBean(ctx, &LFSMetaObject{Pointer: lfs.Pointer{Oid: oid}})
 159| 		if err != nil {
 160| 			return count, err

 163| 		if fn != nil {

 209| // LFSAutoAssociate auto associates accessible LFSMetaObjects
 210| func LFSAutoAssociate(ctx context.Context, metas []*LFSMetaObject, user *user_model.User, repoID int64) error {
 211| 	return db.WithTx(ctx, func(ctx context.Context) error {
 212| 		oids := make([]any, len(metas))
 213| 		oidMap := make(map[string]*LFSMetaObject, len(metas))
 214| 		for i, meta := range metas {
 215| 			oids[i] = meta.Oid
 216| 			oidMap[meta.Oid] = meta
 217| 		}
 218| 
 219| 		if !user.IsAdmin {
 220| 			newMetas := make([]*LFSMetaObject, 0, len(metas))
 221| 			cond := builder.In(
 222| 				"`lfs_meta_object`.repository_id",
 223| 				builder.Select("`repository`.id").From("repository").Where(repo_model.AccessibleRepositoryCondition(user, unit.TypeInvalid)),
 224| 			)
 225| 			if err := db.GetEngine(ctx).Cols("oid").Where(cond).In("oid", oids...).GroupBy("oid").Find(&newMetas); err != nil {
 226| 				return err
 227| 			}
 228| 			if len(newMetas) != len(oidMap) {
 229| 				return fmt.Errorf("unable collect all LFS objects from database, expected %d, actually %d", len(oidMap), len(newMetas))
 230| 			}
 231| 			for i := range newMetas {
 232| 				newMetas[i].Size = oidMap[newMetas[i].Oid].Size
 233| 				newMetas[i].RepositoryID = repoID
 234| 			}
 235| 			return db.Insert(ctx, newMetas)
 236| 		}
 237| 
 238| 		// admin can associate any LFS object to any repository, and we do not care about errors (eg: duplicated unique key),
 239| 		// even if error occurs, it won't hurt users and won't make things worse
 240| 		for i := range metas {
 241| 			p := lfs.Pointer{Oid: metas[i].Oid, Size: metas[i].Size}
 242| 			if err := db.Insert(ctx, &LFSMetaObject{
 243| 				Pointer:      p,
 244| 				RepositoryID: repoID,
 245| 			}); err != nil {
 246| 				log.Warn("failed to insert LFS meta object %-v for repo_id: %d into database, err=%v", p, repoID, err)
 247| 			}
 248| 		}
 249| 		return nil
 250| 	})
 251| }

 280| // IterateRepositoryIDsWithLFSMetaObjects iterates across the repositories that have LFSMetaObjects
 281| func IterateRepositoryIDsWithLFSMetaObjects(ctx context.Context, f func(ctx context.Context, repoID, count int64) error) error {
 282| 	batchSize := setting.Database.IterateBufferSize
 283| 	sess := db.GetEngine(ctx)
 284| 	id := int64(0)
 285| 	type RepositoryCount struct {
 286| 		RepositoryID int64
 287| 		Count        int64
 288| 	}
 289| 	for {
 290| 		counts := make([]*RepositoryCount, 0, batchSize)
 291| 		sess.Select("repository_id, COUNT(id) AS count").
 292| 			Table("lfs_meta_object").
 293| 			Where("repository_id > ?", id).
 294| 			GroupBy("repository_id").
 295| 			OrderBy("repository_id ASC")
 296| 
 297| 		if err := sess.Limit(batchSize, 0).Find(&counts); err != nil {
 298| 			return err
 299| 		}
 300| 		if len(counts) == 0 {
 301| 			return nil
 302| 		}
 303| 
 304| 		for _, count := range counts {

 308| 		}

 319| // IterateLFSMetaObjectsForRepo provides a iterator for LFSMetaObjects per Repo
 320| func IterateLFSMetaObjectsForRepo(ctx context.Context, repoID int64, f func(context.Context, *LFSMetaObject, int64) error, opts *IterateLFSMetaObjectsForRepoOptions) error {
 321| 	batchSize := setting.Database.IterateBufferSize
 322| 	engine := db.GetEngine(ctx)
 323| 	type CountLFSMetaObject struct {
 324| 		Count         int64
 325| 		LFSMetaObject `xorm:"extends"`
 326| 	}
 327| 
 328| 	lastID := int64(0)
 329| 
 330| 	for {
 331| 		beans := make([]*CountLFSMetaObject, 0, batchSize)
 332| 		sess := engine.Table("lfs_meta_object").Select("`lfs_meta_object`.*, COUNT(`l1`.oid) AS `count`").
 333| 			Join("INNER", "`lfs_meta_object` AS l1", "`lfs_meta_object`.oid = `l1`.oid").
 334| 			Where("`lfs_meta_object`.repository_id = ?", repoID)
 335| 		if !opts.OlderThan.IsZero() {
 336| 			sess.And("`lfs_meta_object`.created_unix < ?", opts.OlderThan)
 337| 		}
 338| 		if !opts.UpdatedLessRecentlyThan.IsZero() {
 339| 			sess.And("`lfs_meta_object`.updated_unix < ?", opts.UpdatedLessRecentlyThan)
 340| 		}
 341| 		sess.GroupBy("`lfs_meta_object`.id").
 342| 			And("`lfs_meta_object`.id > ?", lastID).
 343| 			OrderBy("`lfs_meta_object`.id ASC")
 344| 
 345| 		if err := sess.Limit(batchSize).Find(&beans); err != nil {
 346| 			return err
 347| 		}
 348| 		if len(beans) == 0 {
 349| 			return nil
 350| 		}
 351| 
 352| 		for _, bean := range beans {
 353| 			if err := f(ctx, &bean.LFSMetaObject, bean.Count); err != nil {
 354| 				return err
 355| 			}
 356| 		}
 357| 		lastID = beans[len(beans)-1].ID
 358| 	}
 359| }
