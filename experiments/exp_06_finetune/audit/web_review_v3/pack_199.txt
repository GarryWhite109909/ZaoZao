 116| // Clone clones original repository to target path.

 129| 	}
 130| 	if opts.Mirror {
 131| 		cmd.AddArguments("--mirror")
 132| 	}
 133| 	if opts.Bare {
 134| 		cmd.AddArguments("--bare")
 135| 	}
 136| 	if opts.Quiet {
 137| 		cmd.AddArguments("--quiet")
 138| 	}
 139| 	if opts.Shared {
 140| 		cmd.AddArguments("-s")
 141| 	}
 142| 	if opts.NoCheckout {
 143| 		cmd.AddArguments("--no-checkout")
 144| 	}
 145| 	if opts.Depth > 0 {
 146| 		cmd.AddArguments("--depth").AddDynamicArguments(strconv.Itoa(opts.Depth))
 147| 	}
 148| 	if opts.Filter != "" {
 149| 		cmd.AddArguments("--filter").AddDynamicArguments(opts.Filter)
 150| 	}
 151| 	if opts.SingleBranch {

 182| 	LocalRefName   string
 183| 	Branch         string

 191| // Push pushs local commits to given remote branch.
 192| func Push(ctx context.Context, repoPath string, opts PushOptions) error {
 193| 	cmd := gitcmd.NewCommand("push")
 194| 	if opts.ForceWithLease != "" {
 195| 		cmd.AddOptionFormat("--force-with-lease=%s", opts.ForceWithLease)
 196| 	} else if opts.Force {
 197| 		cmd.AddArguments("-f")
 198| 	}
 199| 	if opts.Mirror {
 200| 		cmd.AddArguments("--mirror")
 201| 	}
 202| 	remoteBranchArgs := []string{opts.Remote}
 203| 	if len(opts.Branch) > 0 {
 204| 		var refspec string
 205| 		if opts.LocalRefName != "" {
 206| 			refspec = fmt.Sprintf("%s:%s", opts.LocalRefName, opts.Branch)
 207| 		} else {
 208| 			refspec = opts.Branch
 209| 		}
 210| 		remoteBranchArgs = append(remoteBranchArgs, refspec)
 211| 	}
 212| 	cmd.AddDashesAndList(remoteBranchArgs...)
 213| 
 214| 	stdout, stderr, err := cmd.WithEnv(opts.Env).WithTimeout(opts.Timeout).WithDir(repoPath).RunStdString(ctx)
 215| 	if err != nil {
 216| 		if strings.Contains(stderr, "non-fast-forward") {
 217| 			return &ErrPushOutOfDate{StdOut: stdout, StdErr: stderr, Err: err}
 218| 		} else if strings.Contains(stderr, "! [remote rejected]") || strings.Contains(stderr, "! [rejected]") {
 219| 			err := &ErrPushRejected{StdOut: stdout, StdErr: stderr, Err: err}
 220| 			err.GenerateMessage()
 221| 			return err
 222| 		} else if strings.Contains(stderr, "matches more than one") {
 223| 			return &ErrMoreThanOne{StdOut: stdout, StdErr: stderr, Err: err}
 224| 		}
 225| 		return fmt.Errorf("push failed: %w - %s\n%s", err, stderr, stdout)
 226| 	}
 227| 
 228| 	return nil
 229| }
