  43| /**
  44|  * {@link List}-like implementation that has copy-on-write semantics.
  45|  *
  46|  * <p>
  47|  * This class is suitable where highly concurrent access is needed, yet
  48|  * the write operation is relatively uncommon.
  49|  *
  50|  * @author Kohsuke Kawaguchi
  51|  */
  52| public class CopyOnWriteList<E> implements Iterable<E> {
  53|     private volatile List<? extends E> core;
  54| 
  55|     public CopyOnWriteList(List<E> core) {
  56|         this(core, false);
  57|     }
  58| 
  59|     private CopyOnWriteList(List<E> core, boolean noCopy) {
  60|         this.core = noCopy ? core : new ArrayList<>(core);
  61|     }

  63|     public CopyOnWriteList() {
  64|         this.core = Collections.emptyList();
  65|     }
  66|

  68|         List<E> n = new ArrayList<>(core);
  69|         n.add(e);
  70|         core = n;
  71|     }
  72|

  74|         List<E> n = new ArrayList<>(core);
  75|         n.addAll(items);
  76|         core = n;
  77|     }

  80|      * Removes an item from the list.

  84|      *      in which case there's no change.
  85|      */
  86|     public synchronized boolean remove(E e) {
  87|         List<E> n = new ArrayList<>(core);
  88|         boolean r = n.remove(e);
  89|         core = n;

  91|     }

 158| func (t *ExecTool) guardCommand(command, cwd string) string {
 159| 	cmd := strings.TrimSpace(command)
 160| 	lower := strings.ToLower(cmd)
 161| 
 162| 	for _, pattern := range t.denyPatterns {
 163| 		if pattern.MatchString(lower) {
 164| 			return "Command blocked by safety guard (dangerous pattern detected)"
 165| 		}
 166| 	}
 167| 
 168| 	if len(t.allowPatterns) > 0 {
 169| 		allowed := false
 170| 		for _, pattern := range t.allowPatterns {
 171| 			if pattern.MatchString(lower) {
 172| 				allowed = true
 173| 				break
 174| 			}
 175| 		}
 176| 		if !allowed {
 177| 			return "Command blocked by safety guard (not in allowlist)"
 178| 		}
 179| 	}
 180| 
 181| 	if t.restrictToWorkspace {
 182| 		if strings.Contains(cmd, "..\\") || strings.Contains(cmd, "../") {
 183| 			return "Command blocked by safety guard (path traversal detected)"
 184| 		}
 185| 
 186| 		cwdPath, err := filepath.Abs(cwd)
 187| 		if err != nil {
 188| 			return ""
 189| 		}
 190| 
 191| 		pathPattern := regexp.MustCompile(`[A-Za-z]:\\[^\\\"']+|/[^\s\"']+`)
 192| 		matches := pathPattern.FindAllString(cmd, -1)
 193| 
 194| 		for _, raw := range matches {
 195| 			p, err := filepath.Abs(raw)
 196| 			if err != nil {
 197| 				continue
 198| 			}
 199| 
 200| 			rel, err := filepath.Rel(cwdPath, p)
 201| 			if err != nil {
 202| 				continue
 203| 			}
 204| 
 205| 			if strings.HasPrefix(rel, "..") {
 206| 				return "Command blocked by safety guard (path outside working dir)"
 207| 			}
 208| 		}
 209| 	}
 210| 
 211| 	return ""
 212| }
