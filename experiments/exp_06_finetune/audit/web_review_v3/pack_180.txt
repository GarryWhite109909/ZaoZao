 171| func QuoteBytes(dst, buf []byte) []byte {
 172| 	if len(buf) == 0 {
 173| 		return append(dst, `'\x'`...)
 174| 	}
 175| 
 176| 	// Calculate required length
 177| 	requiredLen := 3 + hex.EncodedLen(len(buf)) + 1
 178| 
 179| 	// Ensure dst has enough capacity
 180| 	if cap(dst)-len(dst) < requiredLen {
 181| 		newDst := make([]byte, len(dst), len(dst)+requiredLen)
 182| 		copy(newDst, dst)
 183| 		dst = newDst
 184| 	}
 185| 
 186| 	// Record original length and extend slice
 187| 	origLen := len(dst)
 188| 	dst = dst[:origLen+requiredLen]
 189| 
 190| 	// Add prefix
 191| 	dst[origLen] = '\''
 192| 	dst[origLen+1] = '\\'
 193| 	dst[origLen+2] = 'x'
 194| 
 195| 	// Encode bytes directly into dst
 196| 	hex.Encode(dst[origLen+3:len(dst)-1], buf)
 197| 
 198| 	// Add suffix
 199| 	dst[len(dst)-1] = '\''
 200| 
 201| 	return dst
 202| }
