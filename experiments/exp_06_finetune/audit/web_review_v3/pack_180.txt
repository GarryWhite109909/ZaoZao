 371| // generateOSCaseClauses reads complexity.OSDifficultyScores and generates SQL WHEN clauses.
 372| func generateOSCaseClauses() string {
 373| 	scoreToLevel := map[int]string{
 374| 		1: "easy",
 375| 		2: "medium",
 376| 		3: "hard",
 377| 		4: "database",
 378| 	}
 379| 	var clauses []string
 380| 	for keyword, score := range complexity.OSDifficultyScores {
 381| 		level := scoreToLevel[score]
 382| 		clauses = append(clauses, fmt.Sprintf(
 383| 			"            WHEN LOWER(effective_os) LIKE '%%%s%%' THEN '%s'", strings.ToLower(keyword), level))
 384| 	}
 385| 	sort.Strings(clauses) // deterministic output
 386| 	return strings.Join(clauses, "\n")
 387| }

 389| // generateComplexityMatrixCaseClauses reads complexity.ComplexityMatrix and generates
 390| // SQL WHEN clauses for the combined OS+disk complexity CASE expression.
 391| func generateComplexityMatrixCaseClauses() string {
 392| 	osScoreToLevel := map[complexity.Score]string{
 393| 		0: "unsupported",
 394| 		1: "easy",
 395| 		2: "medium",
 396| 		3: "hard",
 397| 		4: "database",
 398| 	}
 399| 	diskScoreToLevel := map[complexity.Score]string{
 400| 		1: "easy",
 401| 		2: "medium",
 402| 		3: "hard",
 403| 		4: "whiteglove",
 404| 	}
 405| 
 406| 	// Collect and sort outer (OS) keys for deterministic output.
 407| 	osKeys := make([]complexity.Score, 0, len(complexity.ComplexityMatrix))
 408| 	for osScore := range complexity.ComplexityMatrix {
 409| 		osKeys = append(osKeys, osScore)
 410| 	}
 411| 	sort.Ints(osKeys)
 412| 
 413| 	var clauses []string
 414| 	for _, osScore := range osKeys {
 415| 		inner := complexity.ComplexityMatrix[osScore]
 416| 		osLevel := osScoreToLevel[osScore]
 417| 
 418| 		// Collect and sort inner (disk) keys for deterministic output.
 419| 		diskKeys := make([]complexity.Score, 0, len(inner))
 420| 		for diskScore := range inner {
 421| 			diskKeys = append(diskKeys, diskScore)
 422| 		}
 423| 		sort.Ints(diskKeys)
 424| 
 425| 		for _, diskScore := range diskKeys {
 426| 			combined := inner[diskScore]
 427| 			diskLevel := diskScoreToLevel[diskScore]
 428| 			clauses = append(clauses, fmt.Sprintf(
 429| 				"            WHEN o.os_level = '%s' AND dl.disk_level = '%s' THEN %d",
 430| 				osLevel, diskLevel, combined))
 431| 		}
 432| 	}
 433| 	return strings.Join(clauses, "\n")
 434| }

 445| // buildVMListFilter creates a SQL IN clause for filtering by VM IDs.
 446| // Returns an empty string if the VMList is empty.
 447| // tableAlias is the optional table alias (e.g., "i") - use empty string for no alias.
 448| func buildVMListFilter(vmList []string, tableAlias string) string {
 449| 	if len(vmList) == 0 {
 450| 		return ""
 451| 	}
 452| 
 453| 	// Escape and quote each VM ID
 454| 	quoted := make([]string, len(vmList))
 455| 	for i, vmID := range vmList {
 456| 		// Escape single quotes by doubling them
 457| 		escaped := strings.ReplaceAll(vmID, "'", "''")
 458| 		quoted[i] = fmt.Sprintf("'%s'", escaped)
 459| 	}
 460| 
 461| 	// Build column reference with or without alias
 462| 	columnRef := "\"VM ID\""
 463| 	if tableAlias != "" {
 464| 		columnRef = fmt.Sprintf("%s.\"VM ID\"", tableAlias)
 465| 	}
 466| 
 467| 	return fmt.Sprintf(" AND %s IN (%s)", columnRef, strings.Join(quoted, ", "))
 468| }

 470| func (b *QueryBuilder) buildQuery(name, tmplContent string, params any) (string, error) {
 471| 	tmpl, err := template.New(name).Parse(tmplContent)
 472| 	if err != nil {
 473| 		return "", err
 474| 	}
 475| 	var buf bytes.Buffer
 476| 	if err := tmpl.Execute(&buf, params); err != nil {
 477| 		return "", err
 478| 	}
 479| 	return strings.TrimSpace(buf.String()), nil
 480| }
