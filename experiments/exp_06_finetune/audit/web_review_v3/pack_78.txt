  88| class RankFilter(Filter):
  89|     """
  90|     Create a rank filter.  The rank filter sorts all pixels in
  91|     a window of the given size, and returns the ``rank``'th value.
  92| 
  93|     :param size: The kernel size, in pixels.
  94|     :param rank: What pixel value to pick.  Use 0 for a min filter,
  95|                  ``size * size / 2`` for a median filter, ``size * size - 1``
  96|                  for a max filter, etc.
  97|     """
  98| 
  99|     name = "Rank"
 100| 
 101|     def __init__(self, size: int, rank: int) -> None:
 102|         if size % 2 == 0:
 103|             msg = "bad filter size"
 104|             raise ValueError(msg)
 105|         if size * size * 4 > (2**31 - 1):
 106|             msg = "filter size too large"
 107|             raise ValueError(msg)
 108|         if rank < 0 or rank >= size * size:
 109|             msg = "bad rank value"
 110|             raise ValueError(msg)
 111|         self.size = size
 112|         self.rank = rank

 382| class Color3DLUT(MultibandFilter):
 383|     """Three-dimensional color lookup table.
 384| 
 385|     Transforms 3-channel pixels using the values of the channels as coordinates
 386|     in the 3D lookup table and interpolating the nearest elements.
 387| 
 388|     This method allows you to apply almost any color transformation
 389|     in constant time by using pre-calculated decimated tables.
 390| 
 391|     .. versionadded:: 5.2.0
 392| 
 393|     :param size: Size of the table. One int or tuple of (int, int, int).
 394|                  Minimal size in any dimension is 2, maximum is 65.
 395|     :param table: Flat lookup table. A list of ``channels * size**3``
 396|                   float elements or a list of ``size**3`` channels-sized
 397|                   tuples with floats. Channels are changed first,
 398|                   then first dimension, then second, then third.
 399|                   Value 0.0 corresponds lowest value of output, 1.0 highest.
 400|     :param channels: Number of channels in the table. Could be 3 or 4.
 401|                      Default is 3.
 402|     :param target_mode: A mode for the result image. Should have not less
 403|                         than ``channels`` channels. Default is ``None``,
 404|                         which means that mode wouldn't be changed.
 405|     """
 406| 
 407|     name = "Color 3D LUT"
 408| 
 409|     def __init__(
 410|         self,
 411|         size: int | tuple[int, int, int],
 412|         table: Sequence[float] | Sequence[Sequence[int]] | NumpyArray,
 413|         channels: int = 3,
 414|         target_mode: str | None = None,
 415|         **kwargs: bool,
 416|     ) -> None:
 417|         if channels not in (3, 4):
 418|             msg = "Only 3 or 4 output channels are supported"
 419|             raise ValueError(msg)
 420|         self.size = size = self._check_size(size)
 421|         self.channels = channels
 422|         self.mode = target_mode
 423| 
 424|         # Hidden flag `_copy_table=False` could be used to avoid extra copying
 425|         # of the table if the table is specially made for the constructor.
 426|         copy_table = kwargs.get("_copy_table", True)
 427|         items = size[0] * size[1] * size[2]
 428|         wrong_size = False
 429| 
 430|         numpy: ModuleType | None = None
 431|         if hasattr(table, "shape"):
 432|             try:
 433|                 import numpy
 434|             except ImportError:
 435|                 pass
 436| 
 437|         if numpy and isinstance(table, numpy.ndarray):
 438|             numpy_table: NumpyArray = table
 439|             if copy_table:
 440|                 numpy_table = numpy_table.copy()
 441| 
 442|             if numpy_table.shape in [
 443|                 (items * channels,),
 444|                 (items, channels),
 445|                 (size[2], size[1], size[0], channels),
 446|             ]:
 447|                 table = numpy_table.reshape(items * channels)
 448|             else:
 449|                 wrong_size = True
 450| 
 451|         else:
 452|             if copy_table:
 453|                 table = list(table)
 454| 
 455|             # Convert to a flat list
 456|             if table and isinstance(table[0], (list, tuple)):
 457|                 raw_table = cast(Sequence[Sequence[int]], table)
 458|                 flat_table: list[int] = []
 459|                 for pixel in raw_table:
 460|                     if len(pixel) != channels:
 461|                         msg = (
 462|                             "The elements of the table should "
 463|                             f"have a length of {channels}."
 464|                         )
 465|                         raise ValueError(msg)
 466|                     flat_table.extend(pixel)
 467|                 table = flat_table
 468| 
 469|         if wrong_size or len(table) != items * channels:
 470|             msg = (
 471|                 "The table should have either channels * size**3 float items "
 472|                 "or size**3 items of channels-sized tuples with floats. "
 473|                 f"Table should be: {channels}x{size[0]}x{size[1]}x{size[2]}. "
 474|                 f"Actual length: {len(table)}"
 475|             )
 476|             raise ValueError(msg)
 477|         self.table = table
 478| 
 479|     @staticmethod
 480|     def _check_size(size: Any) -> tuple[int, int, int]:
 481|         try:
 482|             _, _, _ = size
 483|         except ValueError as e:
 484|             msg = "Size should be either an integer or a tuple of three integers."
 485|             raise ValueError(msg) from e
 486|         except TypeError:
 487|             size = (size, size, size)
 488|         size = tuple(int(x) for x in size)
 489|         for size_1d in size:
 490|             if not 2 <= size_1d <= 65:
 491|                 msg = "Size should be in [2, 65] range."
 492|                 raise ValueError(msg)
 493|         return size
 494| 
 495|     @classmethod
 496|     def generate(
 497|         cls,
 498|         size: int | tuple[int, int, int],
 499|         callback: Callable[[float, float, float], tuple[float, ...]],
 500|         channels: int = 3,
 501|         target_mode: str | None = None,
 502|     ) -> Color3DLUT:
 503|         """Generates new LUT using provided callback.
 504| 
 505|         :param size: Size of the table. Passed to the constructor.
 506|         :param callback: Function with three parameters which correspond
 507|                          three color channels. Will be called ``size**3``
 508|                          times with values from 0.0 to 1.0 and should return
 509|                          a tuple with ``channels`` elements.
 510|         :param channels: The number of channels which should return callback.
 511|         :param target_mode: Passed to the constructor of the resulting
 512|                             lookup table.
 513|         """
 514|         size_1d, size_2d, size_3d = cls._check_size(size)
 515|         if channels not in (3, 4):
 516|             msg = "Only 3 or 4 output channels are supported"
 517|             raise ValueError(msg)
 518| 
 519|         table: list[float] = [0] * (size_1d * size_2d * size_3d * channels)
 520|         idx_out = 0
 521|         for b in range(size_3d):
 522|             for g in range(size_2d):
 523|                 for r in range(size_1d):
 524|                     table[idx_out : idx_out + channels] = callback(
 525|                         r / (size_1d - 1), g / (size_2d - 1), b / (size_3d - 1)
 526|                     )
 527|                     idx_out += channels
 528| 
 529|         return cls(
 530|             (size_1d, size_2d, size_3d),
 531|             table,
 532|             channels=channels,
 533|             target_mode=target_mode,
 534|             _copy_table=False,
 535|         )

 537|     def transform(
 538|         self,
 539|         callback: Callable[..., tuple[float, ...]],
 540|         with_normals: bool = False,
 541|         channels: int | None = None,
 542|         target_mode: str | None = None,
 543|     ) -> Color3DLUT:
 544|         """Transforms the table values using provided callback and returns
 545|         a new LUT with altered values.
 546| 
 547|         :param callback: A function which takes old lookup table values
 548|                          and returns a new set of values. The number
 549|                          of arguments which function should take is
 550|                          ``self.channels`` or ``3 + self.channels``
 551|                          if ``with_normals`` flag is set.
 552|                          Should return a tuple of ``self.channels`` or
 553|                          ``channels`` elements if it is set.
 554|         :param with_normals: If true, ``callback`` will be called with
 555|                              coordinates in the color cube as the first
 556|                              three arguments. Otherwise, ``callback``
 557|                              will be called only with actual color values.
 558|         :param channels: The number of channels in the resulting lookup table.
 559|         :param target_mode: Passed to the constructor of the resulting
 560|                             lookup table.
 561|         """
 562|         if channels not in (None, 3, 4):
 563|             msg = "Only 3 or 4 output channels are supported"
 564|             raise ValueError(msg)
 565|         ch_in = self.channels
 566|         ch_out = channels or ch_in
 567|         size_1d, size_2d, size_3d = self.size
 568| 
 569|         table: list[float] = [0] * (size_1d * size_2d * size_3d * ch_out)
 570|         idx_in = 0
 571|         idx_out = 0
 572|         for b in range(size_3d):
 573|             for g in range(size_2d):
 574|                 for r in range(size_1d):
 575|                     values = self.table[idx_in : idx_in + ch_in]
 576|                     if with_normals:
 577|                         values = callback(
 578|                             r / (size_1d - 1),
 579|                             g / (size_2d - 1),
 580|                             b / (size_3d - 1),
 581|                             *values,
 582|                         )
 583|                     else:
 584|                         values = callback(*values)
 585|                     table[idx_out : idx_out + ch_out] = values
 586|                     idx_in += ch_in
 587|                     idx_out += ch_out
 588| 
 589|         return type(self)(
 590|             self.size,
 591|             table,
 592|             channels=ch_out,
 593|             target_mode=target_mode or self.mode,
 594|             _copy_table=False,
 595|         )
