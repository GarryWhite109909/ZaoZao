 404| func (p *PostgreSQL) defaultDeleteUser(ctx context.Context, username string) error {
 405| 	db, err := p.getConnection(ctx)
 406| 	if err != nil {
 407| 		return err
 408| 	}
 409| 
 410| 	// Check if the role exists
 411| 	var exists bool
 412| 	err = db.QueryRowContext(ctx, "SELECT exists (SELECT rolname FROM pg_roles WHERE rolname=$1);", username).Scan(&exists)
 413| 	if err != nil && err != sql.ErrNoRows {
 414| 		return err
 415| 	}
 416| 
 417| 	if !exists {
 418| 		return nil
 419| 	}
 420| 
 421| 	// Query for permissions; we need to revoke permissions before we can drop
 422| 	// the role
 423| 	// This isn't done in a transaction because even if we fail along the way,
 424| 	// we want to remove as much access as possible
 425| 	stmt, err := db.PrepareContext(ctx, "SELECT DISTINCT table_schema FROM information_schema.role_column_grants WHERE grantee=$1;")
 426| 	if err != nil {
 427| 		return err
 428| 	}
 429| 	defer stmt.Close()
 430| 
 431| 	rows, err := stmt.QueryContext(ctx, username)
 432| 	if err != nil {
 433| 		return err
 434| 	}
 435| 	defer rows.Close()
 436| 
 437| 	const initialNumRevocations = 16
 438| 	revocationStmts := make([]string, 0, initialNumRevocations)
 439| 	for rows.Next() {
 440| 		var schema string
 441| 		err = rows.Scan(&schema)
 442| 		if err != nil {
 443| 			// keep going; remove as many permissions as possible right now
 444| 			continue
 445| 		}
 446| 		revocationStmts = append(revocationStmts, fmt.Sprintf(
 447| 			`REVOKE ALL PRIVILEGES ON ALL TABLES IN SCHEMA %s FROM %s;`,
 448| 			(schema),
 449| 			dbutil.QuoteIdentifier(username)))
 450| 
 451| 		revocationStmts = append(revocationStmts, fmt.Sprintf(
 452| 			`REVOKE USAGE ON SCHEMA %s FROM %s;`,
 453| 			dbutil.QuoteIdentifier(schema),
 454| 			dbutil.QuoteIdentifier(username)))
 455| 	}
 456| 
 457| 	// for good measure, revoke all privileges and usage on schema public
 458| 	revocationStmts = append(revocationStmts, fmt.Sprintf(
 459| 		`REVOKE ALL PRIVILEGES ON ALL TABLES IN SCHEMA public FROM %s;`,
 460| 		dbutil.QuoteIdentifier(username)))
 461| 
 462| 	revocationStmts = append(revocationStmts, fmt.Sprintf(
 463| 		"REVOKE ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public FROM %s;",
 464| 		dbutil.QuoteIdentifier(username)))
 465| 
 466| 	revocationStmts = append(revocationStmts, fmt.Sprintf(
 467| 		"REVOKE USAGE ON SCHEMA public FROM %s;",
 468| 		dbutil.QuoteIdentifier(username)))
 469| 
 470| 	// get the current database name so we can issue a REVOKE CONNECT for
 471| 	// this username
 472| 	var dbname sql.NullString
 473| 	if err := db.QueryRowContext(ctx, "SELECT current_database();").Scan(&dbname); err != nil {
 474| 		return err
 475| 	}
 476| 
 477| 	if dbname.Valid {
 478| 		revocationStmts = append(revocationStmts, fmt.Sprintf(
 479| 			`REVOKE CONNECT ON DATABASE %s FROM %s;`,
 480| 			dbutil.QuoteIdentifier(dbname.String),
 481| 			dbutil.QuoteIdentifier(username)))
 482| 	}
 483| 
 484| 	// again, here, we do not stop on error, as we want to remove as
 485| 	// many permissions as possible right now
 486| 	var lastStmtError error
 487| 	for _, query := range revocationStmts {
 488| 		if err := dbtxn.ExecuteDBQueryDirect(ctx, db, nil, query); err != nil {
 489| 			lastStmtError = err
 490| 		}
 491| 	}
 492| 
 493| 	// can't drop if not all privileges are revoked
 494| 	if rows.Err() != nil {
 495| 		return fmt.Errorf("could not generate revocation statements for all rows: %w", rows.Err())
 496| 	}
 497| 	if lastStmtError != nil {
 498| 		return fmt.Errorf("could not perform all revocation statements: %w", lastStmtError)
 499| 	}
 500| 
 501| 	// Drop this user
 502| 	stmt, err = db.PrepareContext(ctx, fmt.Sprintf(
 503| 		`DROP ROLE IF EXISTS %s;`, dbutil.QuoteIdentifier(username)))
 504| 	if err != nil {
 505| 		return err
 506| 	}
 507| 	defer stmt.Close()
 508| 	if _, err := stmt.ExecContext(ctx); err != nil {
 509| 		return err
 510| 	}
 511| 
 512| 	return nil
 513| }

 525| // containsMultilineStatement is a best effort to determine whether
 526| // a particular statement is multiline, and therefore should not be
 527| // split upon semicolons. If it's unsure, it defaults to false.
 528| func containsMultilineStatement(stmt string) bool {
 529| 	// We're going to look for the word "END", but first let's ignore
 530| 	// anything the user provided within single or double quotes since
 531| 	// we're looking for an "END" within the Postgres syntax.
 532| 	literals, err := extractQuotedStrings(stmt)
 533| 	if err != nil {
 534| 		return false
 535| 	}
 536| 	stmtWithoutLiterals := stmt
 537| 	for _, literal := range literals {
 538| 		stmtWithoutLiterals = strings.ReplaceAll(stmt, literal, "")
 539| 	}
 540| 	// Now look for the word "END" specifically. This will miss any
 541| 	// representations of END that aren't surrounded by spaces, but
 542| 	// it should be easy to change on the user's side.
 543| 	return postgresEndStatement.MatchString(stmtWithoutLiterals)
 544| }
 545| 
 546| // extractQuotedStrings extracts 0 or many substrings
 547| // that have been single- or double-quoted. Ex:
 548| // `"Hello", silly 'elephant' from the "zoo".`
 549| // returns [ `Hello`, `'elephant'`, `"zoo"` ]
 550| func extractQuotedStrings(s string) ([]string, error) {
 551| 	var found []string
 552| 	toFind := []*regexp.Regexp{
 553| 		doubleQuotedPhrases,
 554| 		singleQuotedPhrases,
 555| 	}
 556| 	for _, typeOfPhrase := range toFind {
 557| 		found = append(found, typeOfPhrase.FindAllString(s, -1)...)
 558| 	}
 559| 	return found, nil
 560| }

 525| // containsMultilineStatement is a best effort to determine whether
 526| // a particular statement is multiline, and therefore should not be
 527| // split upon semicolons. If it's unsure, it defaults to false.
 528| func containsMultilineStatement(stmt string) bool {
 529| 	// We're going to look for the word "END", but first let's ignore
 530| 	// anything the user provided within single or double quotes since
 531| 	// we're looking for an "END" within the Postgres syntax.
 532| 	literals, err := extractQuotedStrings(stmt)
 533| 	if err != nil {
 534| 		return false
 535| 	}
 536| 	stmtWithoutLiterals := stmt
 537| 	for _, literal := range literals {
 538| 		stmtWithoutLiterals = strings.ReplaceAll(stmt, literal, "")
 539| 	}
 540| 	// Now look for the word "END" specifically. This will miss any
 541| 	// representations of END that aren't surrounded by spaces, but
 542| 	// it should be easy to change on the user's side.
 543| 	return postgresEndStatement.MatchString(stmtWithoutLiterals)
 544| }

 546| // extractQuotedStrings extracts 0 or many substrings
 547| // that have been single- or double-quoted. Ex:
 548| // `"Hello", silly 'elephant' from the "zoo".`
 549| // returns [ `Hello`, `'elephant'`, `"zoo"` ]
 550| func extractQuotedStrings(s string) ([]string, error) {
 551| 	var found []string
 552| 	toFind := []*regexp.Regexp{
 553| 		doubleQuotedPhrases,
 554| 		singleQuotedPhrases,
 555| 	}
 556| 	for _, typeOfPhrase := range toFind {
 557| 		found = append(found, typeOfPhrase.FindAllString(s, -1)...)
 558| 	}
 559| 	return found, nil
 560| }
