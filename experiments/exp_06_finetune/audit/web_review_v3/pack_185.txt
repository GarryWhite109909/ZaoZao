  97|     def validate_url(self, url: str) -> None:
  98|         """Validate a URL for security issues.
  99| 
 100|         This method resolves the hostname and validates all resolved IPs to prevent SSRF attacks.

 102|         To mitigate DNS rebinding attacks, this validation is called both at initialization and
 103|         immediately before each fetch operation, minimizing the TOCTOU window.
 104| 
 105|         Args:
 106|             url: The URL to validate
 107| 
 108|         Raises:
 109|             TrestleError: If the URL is deemed unsafe
 110|         """

 112|         # hostname is guaranteed to be non-None by _parse_and_validate_url
 113|         hostname = parsed.hostname.lower()  # type: ignore
 114|

 116|         self._check_domain_allowlist(hostname)
 117| 
 118|         ip_addresses = self._resolve_hostname(hostname)

 121|             ip_addr = self._parse_ip_address(ip_str, hostname)

 127|     def _parse_and_validate_url(self, url: str) -> parse.ParseResult:
 128|         """Parse and validate basic URL structure."""
 129|         try:
 130|             parsed = parse.urlparse(url)
 131|         except Exception as e:
 132|             raise TrestleError(f'Invalid URL format: {url}') from e
 133| 
 134|         if not parsed.scheme or not parsed.hostname:
 135|             raise TrestleError(f'URL must include scheme and hostname: {url}')
 136| 
 137|         if parsed.scheme not in ['https', 'sftp']:
 138|             raise TrestleError(f'Only HTTPS or SFTP schemes are allowed for remote URLs, got: {parsed.scheme}')
 139| 
 140|         return parsed

 142|     def _check_metadata_endpoints(self, hostname: str) -> None:

 150|     def _check_domain_allowlist(self, hostname: str) -> None:
 151|         """Check if hostname is in the allowed domains list."""

 153|             if hostname not in self.allowed_domains:
 154|                 raise TrestleError(

 156|                     f'Allowed domains: {", ".join(sorted(self.allowed_domains))}'

 159|     def _resolve_hostname(self, hostname: str) -> list:
 160|         """Resolve hostname to IP addresses."""
 161|         try:
 162|             addr_info = socket.getaddrinfo(hostname, None)
 163|             ip_addresses = [str(info[4][0]) for info in addr_info]
 164|         except socket.gaierror as e:
 165|             raise TrestleError(f'Unable to resolve hostname {hostname}: {e}') from e
 166| 
 167|         if not ip_addresses:
 168|             raise TrestleError(f'No IP addresses resolved for hostname {hostname}')
 169| 
 170|         return ip_addresses

 236|     def validate_url_path_for_cache(url_path: str) -> None:

 243|         Args:

 251|             >>> PathSecurityValidator.validate_url_path_for_cache('/../../../etc/passwd')  # Raises TrestleError

 255|             raise TrestleError(
 256|                 f'Security violation: Path traversal blocked. '
 257|                 f'URL path "{url_path}" contains ".." sequences which could '

 261|     @staticmethod
 262|     def validate_cache_path(cache_path: pathlib.Path, cache_root: pathlib.Path) -> None:
 263|         """
 264|         Validate that a cache file path stays within the cache directory.
 265| 
 266|         Uses path resolution and relative_to() to ensure the resolved cache path
 267|         is actually within the cache root directory, preventing path traversal attacks.
 268| 
 269|         Args:
 270|             cache_path: The proposed cache file path to validate
 271|             cache_root: The root cache directory that must contain the cache_path
 272| 
 273|         Raises:
 274|             TrestleError: If cache_path resolves outside cache_root
 275| 
 276|         Example:
 277|             >>> cache_root = pathlib.Path('/home/user/.trestle/cache')
 278|             >>> cache_path = cache_root / 'evil.com' / '..' / '..' / 'etc' / 'passwd'
 279|             >>> validate_cache_path(cache_path, cache_root)  # Raises TrestleError
 280|         """
 281|         # Resolve both paths to absolute, normalized paths
 282|         resolved_cache = cache_path.resolve()
 283|         resolved_root = cache_root.resolve()
 284| 
 285|         try:
 286|             # Check if cache path is relative to (within) cache root
 287|             resolved_cache.relative_to(resolved_root)
 288| 
 289|         except ValueError as e:
 290|             # relative_to() raises ValueError if path is not relative to root
 291|             raise TrestleError(
 292|                 f'Security violation: Cache path traversal blocked. '
 293|                 f'Attempted to write to "{resolved_cache}" which is outside '
 294|                 f'the cache directory "{resolved_root}"'
 295|             ) from e
 296|         except Exception as e:
 297|             raise TrestleError(f'Error validating cache path "{cache_path}": {e}') from e

 363|     @staticmethod
 364|     def validate_local_file_path(

 366|     ) -> None:
 367|         """Validate that a local file path is safe to access.
 368| 
 369|         This method provides defense-in-depth protection against arbitrary file access
 370|         by validating both workspace boundaries and blocking known sensitive system files.
 371| 
 372|         Args:
 373|             workspace_root: The trestle workspace root directory
 374|             file_path: The file path to validate
 375|             allow_outside_workspace: If True, allow access to files outside workspace (default: False)
 376| 
 377|         Raises:
 378|             TrestleError: If the path is deemed unsafe
 379| 
 380|         Example:
 381|             >>> workspace = pathlib.Path('/home/user/trestle-workspace')
 382|             >>> file_path = pathlib.Path('/etc/passwd')
 383|             >>> validate_local_file_path(workspace, file_path, allow_outside_workspace=False)
 384|             # Raises TrestleError: Access to files outside workspace not allowed
 385|             >>> validate_local_file_path(workspace, file_path, allow_outside_workspace=True)
 386|             # Raises TrestleError: Attempt to access sensitive system file
 387|         """
 388|         resolved_workspace = workspace_root.resolve()
 389|         resolved_file = file_path.resolve()
 390| 
 391|         try:
 392|             if not allow_outside_workspace:
 393|                 # Ensure file is within workspace
 394|                 resolved_file.relative_to(resolved_workspace)
 395|         except ValueError as e:
 396|             if not allow_outside_workspace:
 397|                 raise TrestleError(

 403|         # This provides defense-in-depth even when allow_outside_workspace=True
