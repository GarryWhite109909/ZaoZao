 328|     /**
 329|      * Returns the target path as relative reference from the base path.
 330|      *
 331|      * Only the URIs path component (no schema, host etc.) is relevant and must be given, starting with a slash.
 332|      * Both paths must be absolute and not contain relative parts.
 333|      * Relative URLs from one resource to another are useful when generating self-contained downloadable document archives.
 334|      * Furthermore, they can be used to reduce the link size in documents.
 335|      *
 336|      * Example target paths, given a base path of "/a/b/c/d":
 337|      * - "/a/b/c/d"     -> ""
 338|      * - "/a/b/c/"      -> "./"
 339|      * - "/a/b/"        -> "../"
 340|      * - "/a/b/c/other" -> "other"
 341|      * - "/a/x/y"       -> "../../x/y"
 342|      *
 343|      * @param string $basePath   The base path
 344|      * @param string $targetPath The target path
 345|      *
 346|      * @return string
 347|      */
 348|     public static function getRelativePath(string $basePath, string $targetPath)
 349|     {
 350|         if ($basePath === $targetPath) {
 351|             return '';
 352|         }
 353| 
 354|         $sourceDirs = explode('/', isset($basePath[0]) && '/' === $basePath[0] ? substr($basePath, 1) : $basePath);
 355|         $targetDirs = explode('/', isset($targetPath[0]) && '/' === $targetPath[0] ? substr($targetPath, 1) : $targetPath);
 356|         array_pop($sourceDirs);
 357|         $targetFile = array_pop($targetDirs);
 358| 
 359|         foreach ($sourceDirs as $i => $dir) {
 360|             if (isset($targetDirs[$i]) && $dir === $targetDirs[$i]) {
 361|                 unset($sourceDirs[$i], $targetDirs[$i]);
 362|             } else {
 363|                 break;
 364|             }
 365|         }
 366| 
 367|         $targetDirs[] = $targetFile;
 368|         $path = str_repeat('../', \count($sourceDirs)).implode('/', $targetDirs);
 369| 
 370|         // A reference to the same base directory or an empty subdirectory must be prefixed with "./".
 371|         // This also applies to a segment with a colon character (e.g., "file:colon") that cannot be used
 372|         // as the first segment of a relative-path reference, as it would be mistaken for a scheme name
 373|         // (see http://tools.ietf.org/html/rfc3986#section-4.2).
 374|         return '' === $path || '/' === $path[0]
 375|             || false !== ($colonPos = strpos($path, ':')) && ($colonPos < ($slashPos = strpos($path, '/')) || false === $slashPos)
 376|             ? "./$path" : $path;
 377|     }
