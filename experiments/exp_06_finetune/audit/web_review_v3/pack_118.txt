  74|         KeycloakUriInfo uriInfo = session.getContext().getUri();

  76| 
  77|         if (redirectUri == null) {
  78|             if (!requireRedirectUri) {
  79|                 redirectUri = getSingleValidRedirectUri(validRedirects);
  80|             }
  81| 
  82|             if (redirectUri == null) {
  83|                 logger.debug("No Redirect URI parameter specified");
  84|                 return null;
  85|             }
  86|         } else if (validRedirects.isEmpty()) {
  87|             logger.debug("No Redirect URIs supplied");
  88|             redirectUri = null;
  89|         } else {
  90|             URI originalRedirect = toUri(redirectUri);
  91|             if (originalRedirect == null) {
  92|                 // invalid URI passed as redirectUri
  93|                 return null;
  94|             }

  96|             // check if the passed URI allows wildcards
  97|             boolean allowWildcards = areWildcardsAllowed(originalRedirect);

  99|             String r = redirectUri;

 102|             String valid = matchesRedirects(resolveValidRedirects, r, allowWildcards);

 104|             if (valid == null && "http".equals(originalRedirect.getScheme()) && LOOPBACK_INTERFACES.contains(originalRedirect.getHost())) {
 105|                 String redirectWithDefaultPort = KeycloakUriBuilder.fromUri(originalRedirect).port(80).buildAsString();

 107|             }

 110|                 // return absolute if the original URI is relative
 111|                 if (!redirectUri.startsWith("/")) {

 174|     // return the String that matched the redirect or null if not matched
 175|     private static String matchesRedirects(Set<String> validRedirects, String redirect, boolean allowWildcards) {
 176|         logger.tracef("matchesRedirects: redirect URL to check: %s, allow wildcards: %b, Configured valid redirect URLs: %s", redirect, allowWildcards, validRedirects);
 177|         for (String validRedirect : validRedirects) {
 178|             if ("*".equals(validRedirect)) {
 179|                 // the valid redirect * is a full wildcard for http(s) even if the redirect URI does not allow wildcards
 180|                 return validRedirect;
 181|             } else if (validRedirect.endsWith("*") && !validRedirect.contains("?") && allowWildcards) {
 182|                 // strip off the query or fragment components - we don't check them when wildcards are effective
 183|                 int idx = redirect.indexOf('?');
 184|                 if (idx == -1) {
 185|                     idx = redirect.indexOf('#');
 186|                 }
 187|                 String r = idx == -1 ? redirect : redirect.substring(0, idx);
 188|                 // strip off *
 189|                 int length = validRedirect.length() - 1;
 190|                 validRedirect = validRedirect.substring(0, length);
 191|                 if (r.startsWith(validRedirect)) return validRedirect;
 192|                 // strip off trailing '/'
 193|                 if (length - 1 > 0 && validRedirect.charAt(length - 1) == '/') length--;
 194|                 validRedirect = validRedirect.substring(0, length);
 195|                 if (validRedirect.equals(r)) return validRedirect;
 196|             } else if (validRedirect.equals(redirect)) return validRedirect;
 197|         }
 198|         return null;
 199|     }
