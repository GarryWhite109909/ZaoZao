 552| function setListBounds(list, begin, end) {
 553|   // Sanitize begin & end using this shorthand for ToInt32(argument)
 554|   // http://www.ecma-international.org/ecma-262/6.0/#sec-toint32
 555|   if (begin !== undefined) {
 556|     begin |= 0;
 557|   }
 558|   if (end !== undefined) {
 559|     end |= 0;
 560|   }
 561|   const owner = list.__ownerID || new OwnerID();
 562|   let oldOrigin = list._origin;
 563|   let oldCapacity = list._capacity;
 564|   let newOrigin = oldOrigin + begin;
 565|   let newCapacity =
 566|     end === undefined
 567|       ? oldCapacity
 568|       : end < 0
 569|         ? oldCapacity + end
 570|         : oldOrigin + end;
 571|   if (newOrigin === oldOrigin && newCapacity === oldCapacity) {
 572|     return list;
 573|   }
 574| 
 575|   // If it's going to end after it starts, it's empty.
 576|   if (newOrigin >= newCapacity) {
 577|     return list.clear();
 578|   }
 579| 
 580|   let newLevel = list._level;
 581|   let newRoot = list._root;
 582| 
 583|   // New origin might need creating a higher root.
 584|   let offsetShift = 0;
 585|   while (newOrigin + offsetShift < 0) {
 586|     newRoot = new VNode(
 587|       newRoot && newRoot.array.length ? [undefined, newRoot] : [],
 588|       owner
 589|     );
 590|     newLevel += SHIFT;
 591|     offsetShift += 1 << newLevel;
 592|   }
 593|   if (offsetShift) {
 594|     newOrigin += offsetShift;
 595|     oldOrigin += offsetShift;
 596|     newCapacity += offsetShift;
 597|     oldCapacity += offsetShift;
 598|   }
 599| 
 600|   const oldTailOffset = getTailOffset(oldCapacity);
 601|   const newTailOffset = getTailOffset(newCapacity);
 602| 
 603|   // New size might need creating a higher root.
 604|   while (newTailOffset >= 1 << (newLevel + SHIFT)) {
 605|     newRoot = new VNode(
 606|       newRoot && newRoot.array.length ? [newRoot] : [],
 607|       owner
 608|     );
 609|     newLevel += SHIFT;
 610|   }
 611| 
 612|   // Locate or create the new tail.
 613|   const oldTail = list._tail;
 614|   let newTail =
 615|     newTailOffset < oldTailOffset
 616|       ? listNodeFor(list, newCapacity - 1)
 617|       : newTailOffset > oldTailOffset
 618|         ? new VNode([], owner)
 619|         : oldTail;
 620| 
 621|   // Merge Tail into tree.
 622|   if (
 623|     oldTail &&
 624|     newTailOffset > oldTailOffset &&
 625|     newOrigin < oldCapacity &&
 626|     oldTail.array.length
 627|   ) {
 628|     newRoot = editableVNode(newRoot, owner);
 629|     let node = newRoot;
 630|     for (let level = newLevel; level > SHIFT; level -= SHIFT) {
 631|       const idx = (oldTailOffset >>> level) & MASK;
 632|       node = node.array[idx] = editableVNode(node.array[idx], owner);
 633|     }
 634|     node.array[(oldTailOffset >>> SHIFT) & MASK] = oldTail;
 635|   }
 636| 
 637|   // If the size has been reduced, there's a chance the tail needs to be trimmed.
 638|   if (newCapacity < oldCapacity) {
 639|     newTail = newTail && newTail.removeAfter(owner, 0, newCapacity);
 640|   }
 641| 
 642|   // If the new origin is within the tail, then we do not need a root.
 643|   if (newOrigin >= newTailOffset) {
 644|     newOrigin -= newTailOffset;
 645|     newCapacity -= newTailOffset;
 646|     newLevel = SHIFT;
 647|     newRoot = null;
 648|     newTail = newTail && newTail.removeBefore(owner, 0, newOrigin);
 649| 
 650|     // Otherwise, if the root has been trimmed, garbage collect.
 651|   } else if (newOrigin > oldOrigin || newTailOffset < oldTailOffset) {
 652|     offsetShift = 0;
 653| 
 654|     // Identify the new top root node of the subtree of the old root.
 655|     while (newRoot) {
 656|       const beginIndex = (newOrigin >>> newLevel) & MASK;
 657|       if ((beginIndex !== newTailOffset >>> newLevel) & MASK) {
 658|         break;
 659|       }

 671|     if (newRoot && newTailOffset < oldTailOffset) {

 673|         owner,
 674|         newLevel,
