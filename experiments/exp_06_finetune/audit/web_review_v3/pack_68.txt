 284| def render_object_endpoint(request, content_type_id, object_id, require_editable):
 285|     try:
 286|         content_type = ContentType.objects.get_for_id(content_type_id)
 287|     except ContentType.DoesNotExist as err:
 288|         raise Http404 from err

 291| 
 292|     if require_editable and not is_editable_model(model):
 293|         return HttpResponseBadRequest('Requested object does not support frontend rendering')
 294| 
 295|     try:
 296|         if issubclass(model, PageContent):
 297|             # An apphook might be attached to a PageContent object
 298|             content_type_obj = model.admin_manager.select_related("page").get(pk=object_id)
 299|             request.current_page = content_type_obj.page
 300|             if (
 301|                 content_type_obj.page.application_urls and
 302|                 content_type_obj.page.application_urls in dict(apphook_pool.get_apphooks())
 303|             ):
 304|                 try:
 305|                     # If so, try get the absolute URL and pass it to the toolbar as request_path
 306|                     # The apphook's view function will be called.
 307|                     absolute_url = content_type_obj.get_absolute_url()
 308|                     from cms.toolbar.toolbar import CMSToolbar
 309|                     request.toolbar = CMSToolbar(request, request_path=absolute_url)
 310|                     # Resolve the apphook's url to get its view function
 311|                     view_func, args, kwargs = resolve(absolute_url)
 312|                     if view_func is not details:
 313|                         return view_func(request, *args, **kwargs)

 320|         raise Http404 from err

 324|         try:

 326|         except (AttributeError, NoReverseMatch):
 327|             object_url = None

 330|             original_path_info = request.path_info

 332|                 # Temporarily patch the request object and use the appresolver
 333|                 request.path_info = object_url
