 220|         if (null != audience) {
 221|             jwtBuilder.setAudience(audience);

 223|         if (null != payload) {
 224|             jwtBuilder.setPayload(payload);

 227|             jwtBuilder.setNotBefore(new Date(notBefore * 1000));

 229|         if(null != headerMap) {
 230|             jwtBuilder.setHeader(headerMap);
