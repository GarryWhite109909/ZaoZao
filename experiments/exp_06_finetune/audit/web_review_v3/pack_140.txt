 379| func isImageLocallyAvailable(ctx context.Context, cli *client.Client, imageReference string) (bool, error) {
 380| 	f := filters.NewArgs()
 381| 	f.Add("reference", imageReference)
 382| 
 383| 	images, err := cli.ImageList(ctx, image.ListOptions{Filters: f})
 384| 	if err != nil {
 385| 		return false, fmt.Errorf("listing images: %w", err)
 386| 	}
 387| 
 388| 	for _, img := range images {
 389| 		for _, tag := range img.RepoTags {
 390| 			if tag == imageReference {
 391| 				return true, nil
 392| 			}
 393| 		}
 394| 		for _, digest := range img.RepoDigests {
 395| 			if digest == imageReference {
 396| 				return true, nil
 397| 			}
 398| 		}
 399| 	}
 400| 	return false, nil

 403| func ensureBuilderImage(ctx context.Context, cli *client.Client, builderImage string, builderImagePull string) error {
 404| 	switch builderImagePull {
 405| 	case "always":
 406| 		return pullImage(ctx, cli, builderImage)
 407| 	case "missing":
 408| 		localAvailable, err := isImageLocallyAvailable(ctx, cli, builderImage)
 409| 		if err != nil {
 410| 			return err
 411| 		}
 412| 		if !localAvailable {
 413| 			return pullImage(ctx, cli, builderImage)
 414| 		}
 415| 	case "never":
 416| 		localAvailable, err := isImageLocallyAvailable(ctx, cli, builderImage)
 417| 		if err != nil {
 418| 			return err
 419| 		}
 420| 		if !localAvailable {
 421| 			return fmt.Errorf("image %s is not available locally and pull is disabled", builderImage)
 422| 		}
 423| 	default:
 424| 		return fmt.Errorf("invalid --builder-image-pull value: %s", builderImagePull)
 425| 	}
 426| 	return nil
 427| }

 429| func buildInContainer(opts *cmdOpts, conf *buildFile) error {
 430| 	cwd, err := os.Getwd()
 431| 	if err != nil {
 432| 		return fmt.Errorf("getting current directory: %w", err)
 433| 	}
 434| 
 435| 	ctx := context.TODO()
 436| 	cli, err := client.NewClientWithOpts(client.FromEnv, client.WithAPIVersionNegotiation())
 437| 	if err != nil {
 438| 		return fmt.Errorf("creating docker client: %w", err)
 439| 	}
 440| 	defer cli.Close()
 441| 
 442| 	if err := ensureBuilderImage(ctx, cli, opts.builderImage, opts.builderImagePull); err != nil {
 443| 		return err
 444| 	}
 445| 
 446| 	// where the gadget source code is mounted in the container
 447| 	gadgetSourcePath := "/work"
 448| 	pathHost := cwd
 449| 
 450| 	inspektorGadetSrcPath := os.Getenv("IG_SOURCE_PATH")
 451| 	useInTreeHeaders := "false"
 452| 	if inspektorGadetSrcPath != "" {
 453| 		pathHost = inspektorGadetSrcPath
 454| 		// find the gadget relative path to the inspektor-gadget source
 455| 		if !strings.HasPrefix(cwd, inspektorGadetSrcPath) {
 456| 			return fmt.Errorf("the current directory %q is not under the inspektor-gadget source path %q", cwd, inspektorGadetSrcPath)
 457| 		}
 458| 		gadgetRelativePath := strings.TrimPrefix(cwd, inspektorGadetSrcPath)
 459| 		gadgetSourcePath = filepath.Join("/work", gadgetRelativePath)
 460| 
 461| 		// use in-tree headers too
 462| 		useInTreeHeaders = "true"
 463| 	}
 464| 
 465| 	buildOpts := buildOptions{
 466| 		outputDir:         "/out",
 467| 		btfgen:            opts.btfgen,
 468| 		btfHubArchivePath: "/btfhub-archive",
 469| 		useInTreeHeaders:  useInTreeHeaders,
 470| 	}
 471| 
 472| 	if conf.Wasm != "" {
 473| 		buildOpts.wasmSourcePath = filepath.Join(gadgetSourcePath, conf.Wasm)
 474| 	}
 475| 
 476| 	if conf.EBPFSource != "" {
 477| 		buildOpts.ebpfSourcePath = filepath.Join(gadgetSourcePath, conf.EBPFSource)
 478| 	}
 479| 
 480| 	if term.IsTerminal(int(os.Stdout.Fd())) && term.IsTerminal(int(os.Stderr.Fd())) {
 481| 		buildOpts.forceColorsFlag = "true"
 482| 	}
 483| 
 484| 	cmd := buildCmd(buildOpts)
 485| 
 486| 	// The work mount ReadOnly field is updated as false, to allow Cargo.lock to compiled in /work folder for rust source code.
 487| 	mounts := []mount.Mount{
 488| 		{
 489| 			Type:     mount.TypeBind,
 490| 			Target:   "/work",
 491| 			Source:   pathHost,
 492| 			ReadOnly: false,
 493| 		},
 494| 		{
 495| 			Type:   mount.TypeBind,
 496| 			Target: "/out",
 497| 			Source: opts.outputDir,
 498| 		},
 499| 	}
 500| 
 501| 	if opts.btfgen {
 502| 		mounts = append(mounts, mount.Mount{
 503| 			Type:   mount.TypeBind,
 504| 			Target: "/btfhub-archive",
 505| 			Source: opts.btfhubarchive,
 506| 		})
 507| 	}
 508| 
 509| 	resp, err := cli.ContainerCreate(
 510| 		ctx,
 511| 		&container.Config{
 512| 			Image:      opts.builderImage,
 513| 			Cmd:        cmd,
 514| 			WorkingDir: gadgetSourcePath,
 515| 			User:       fmt.Sprintf("%d:%d", os.Getuid(), os.Getgid()),
 516| 		},
 517| 		&container.HostConfig{
 518| 			Mounts: mounts,
 519| 		},
 520| 		nil, nil, "",
 521| 	)
 522| 	if err != nil {
 523| 		return fmt.Errorf("creating builder container: %w", err)
 524| 	}
 525| 	defer func() {
 526| 		if err := cli.ContainerRemove(ctx, resp.ID, container.RemoveOptions{}); err != nil {
 527| 			fmt.Printf("Failed to remove builder container: %s\n", err)
 528| 		}
 529| 	}()
 530| 
 531| 	if err := cli.ContainerStart(ctx, resp.ID, container.StartOptions{}); err != nil {
 532| 		return fmt.Errorf("starting builder container: %w", err)
 533| 	}
 534| 
 535| 	var status container.WaitResponse
 536| 
 537| 	statusCh, errCh := cli.ContainerWait(ctx, resp.ID, container.WaitConditionNotRunning)
 538| 	select {
 539| 	case err := <-errCh:
 540| 		if err != nil {
 541| 			return fmt.Errorf("waiting for builder container: %w", err)
 542| 		}
 543| 	case status = <-statusCh:
 544| 	}
 545| 
 546| 	outputOpts := container.LogsOptions{ShowStderr: true}
 547| 
 548| 	if status.StatusCode != 0 || common.Verbose {
 549| 		outputOpts.ShowStdout = true
 550| 	}
 551| 
 552| 	out, err := cli.ContainerLogs(ctx, resp.ID, outputOpts)
 553| 	if err != nil {
 554| 		return fmt.Errorf("getting builder container logs: %w", err)
 555| 	}
 556| 
 557| 	fmt.Println("Build logs start:")
 558| 	stdcopy.StdCopy(os.Stdout, os.Stderr, out)
 559| 	fmt.Println("Build logs end")
 560| 
 561| 	if status.StatusCode != 0 {
 562| 		return fmt.Errorf("builder container exited with status %d", status.StatusCode)
 563| 	}
 564| 
 565| 	return nil
 566| }
