  53| 	} else { // 0x02

  62| func uncompressKey(compressedBytes []byte, priv []byte) (*big.Int, *big.Int) {
  63| 	// Split the sign byte from the rest
  64| 	signByte := uint(compressedBytes[0])
  65| 	xBytes := compressedBytes[1:]
  66| 
  67| 	x := new(big.Int).SetBytes(xBytes)
  68| 	three := big.NewInt(3)
  69| 
  70| 	// The params for P256
  71| 	c := elliptic.P256().Params()
  72| 
  73| 	// The equation is y^2 = x^3 - 3x + b
  74| 	// x^3, mod P
  75| 	xCubed := new(big.Int).Exp(x, three, c.P)
  76| 
  77| 	// 3x, mod P
  78| 	threeX := new(big.Int).Mul(x, three)
  79| 	threeX.Mod(threeX, c.P)
  80| 
  81| 	// x^3 - 3x + b mod P
  82| 	ySquared := new(big.Int).Sub(xCubed, threeX)
  83| 	ySquared.Add(ySquared, c.B)
  84| 	ySquared.Mod(ySquared, c.P)
  85| 
  86| 	// find the square root mod P
  87| 	y := new(big.Int).ModSqrt(ySquared, c.P)
  88| 	if y == nil {
  89| 		// If this happens then you're dealing with an invalid point.
  90| 		logger.SuciLog.Errorln("Uncompressed key with invalid point")
  91| 		return nil, nil
  92| 	}
  93| 
  94| 	// Finally, check if you have the correct root. If not you want -y mod P
  95| 	if y.Bit(0) != signByte&1 {
  96| 		y.Neg(y)
  97| 		y.Mod(y, c.P)
  98| 	}
