func exportHandler(w http.ResponseWriter, r *http.Request) {

func fetchRemoteData(endpoint string) string {
	parsed, err := url.Parse(endpoint)
	if err != nil {
		return fmt.Sprintf("error: invalid URL")
	}

	// 仅允许 HTTPS
	if parsed.Scheme != "https" {
		return fmt.Sprintf("error: only https allowed")
	}

	// 白名单校验主机名
	if !allowedHosts[parsed.Hostname()] {
		return fmt.Sprintf("error: host not allowed")
	}

	// 解析 IP 并阻止内网地址
	ips, err := net.LookupIP(parsed.Hostname())
	if err != nil {
		return fmt.Sprintf("error: cannot resolve host")
	}
	for _, ip := range ips {
		if ip.IsPrivate() || ip.IsLoopback() || ip.IsLinkLocalUnicast() {
			return fmt.Sprintf("error: internal IP blocked")
		}
	}

	// 设置超时，防止 SSRF 用于端口扫描
	client := &http.Client{Timeout: 5 * time.Second}
	resp, err := client.Get(endpoint)
	if err != nil {
		return fmt.Sprintf("error: %v", err)
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return fmt.Sprintf("error: %v", err)
	}
	return string(body)
}
