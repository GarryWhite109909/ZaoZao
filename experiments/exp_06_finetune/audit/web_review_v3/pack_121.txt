 129|     public void parse(InputStream stream, ContentHandler baseHandler, Metadata metadata,
 130|                       ParseContext context) throws IOException, SAXException, TikaException {
 131| 
 132|         EmbeddedDocumentUtil embeddedDocumentUtil = new EmbeddedDocumentUtil(context);
 133| 
 134|         // Open the Zip stream
 135|         // Use a File if we can, and an already open zip is even better
 136|         ZipFile zipFile = null;
 137|         ZipInputStream zipStream = null;
 138|         if (stream instanceof TikaInputStream) {
 139|             TikaInputStream tis = (TikaInputStream) stream;
 140|             Object container = ((TikaInputStream) stream).getOpenContainer();
 141|             if (container instanceof ZipFile) {
 142|                 zipFile = (ZipFile) container;
 143|             } else if (tis.hasFile()) {
 144|                 zipFile = new ZipFile(tis.getFile());
 145|             } else {
 146|                 zipStream = new ZipInputStream(stream);
 147|             }
 148|         } else {
 149|             zipStream = new ZipInputStream(stream);
 150|         }
 151| 
 152|         // Prepare to handle the content
 153|         XHTMLContentHandler xhtml = new XHTMLContentHandler(baseHandler, metadata);
 154|         xhtml.startDocument();
 155|         // As we don't know which of the metadata or the content
 156|         //  we'll hit first, catch the endDocument call initially
 157|         EndDocumentShieldingContentHandler handler = new EndDocumentShieldingContentHandler(xhtml);
 158| 
 159|         try {
 160|             if (zipFile != null) {
 161|                 try {
 162|                     handleZipFile(zipFile, metadata, context, handler, embeddedDocumentUtil);
 163|                 } finally {
 164|                     //Do we want to close silently == catch an exception here?
 165|                     zipFile.close();
 166|                 }
 167|             } else {
 168|                 try {
 169|                     handleZipStream(zipStream, metadata, context, handler, embeddedDocumentUtil);
 170|                 } finally {
 171|                     //Do we want to close silently == catch an exception here?
 172|                     zipStream.close();
 173|                 }
 174|             }
 175|         } catch (SAXException e) {
 176|             if (e.getCause() instanceof EncryptedDocumentException) {
 177|                 throw (EncryptedDocumentException)e.getCause();
 178|             }
 179|             throw e;
 180|         }
 181| 
 182|         // Only now call the end document
 183|         if (handler.isEndDocumentWasCalled()) {
 184|             handler.reallyEndDocument();
 185|         }

 197|     private void handleZipStream(ZipInputStream zipStream, Metadata metadata, ParseContext context,
 198|                                  EndDocumentShieldingContentHandler handler,
 199|                                  EmbeddedDocumentUtil embeddedDocumentUtil)
 200|             throws IOException, TikaException, SAXException {
 201|         ZipEntry entry = zipStream.getNextEntry();
 202|         if (entry == null) {
 203|             throw new IOException("No entries found in ZipInputStream");
 204|         }
 205|         List<SAXException> exceptions = new ArrayList<>();
 206|         do {
 207|             try {
 208|                 handleZipEntry(entry, zipStream, metadata, context, handler,
 209|                         embeddedDocumentUtil);
 210|             } catch (SAXException e) {
 211|                 WriteLimitReachedException.throwIfWriteLimitReached(e);
 212|                 if (e.getCause() instanceof EncryptedDocumentException) {
 213|                     throw (EncryptedDocumentException)e.getCause();
 214|                 } else {
 215|                     exceptions.add(e);
 216|                 }
 217|             }
 218|             entry = zipStream.getNextEntry();
 219|         } while (entry != null);
 220| 
 221|         if (exceptions.size() > 0) {
 222|             throw exceptions.get(0);
 223|         }
 224|     }

 226|     private void handleZipFile(ZipFile zipFile, Metadata metadata, ParseContext context,
 227|                                EndDocumentShieldingContentHandler handler,
 228|                                EmbeddedDocumentUtil embeddedDocumentUtil)
 229|             throws IOException, TikaException, SAXException {
 230|         // If we can, process the metadata first, then the
 231|         //  rest of the file afterwards (TIKA-1353)
 232|         // Only possible to guarantee that when opened from a file not a stream
 233| 
 234|         ZipEntry entry = zipFile.getEntry(MANIFEST_NAME);
 235|         if (entry != null) {
 236|             handleZipEntry(entry, zipFile.getInputStream(entry), metadata, context,
 237|                     handler, embeddedDocumentUtil);
 238|         }
 239| 
 240|         entry = zipFile.getEntry(META_NAME);
 241|         if (entry != null) {
 242|             handleZipEntry(entry, zipFile.getInputStream(entry), metadata, context,
 243|                     handler, embeddedDocumentUtil);
 244|         }
 245| 
 246|         Enumeration<? extends ZipEntry> entries = zipFile.entries();
 247|         while (entries.hasMoreElements()) {
 248|             entry = entries.nextElement();
 249|             if (!META_NAME.equals(entry.getName())) {
 250|                 handleZipEntry(entry, zipFile.getInputStream(entry), metadata,
 251|                         context, handler, embeddedDocumentUtil);
 252|             }
 253|         }

 256|     private void handleZipEntry(ZipEntry entry, InputStream zip, Metadata metadata,
 257|                                 ParseContext context, ContentHandler handler,
 258|                                 EmbeddedDocumentUtil embeddedDocumentUtil)
 259|             throws IOException, SAXException, TikaException {
 260| 
 261|         if (entry.isDirectory()) {
 262|             return;
 263|         }
 264|         if (entry.getName().contains("manifest.xml")) {
 265|             checkForEncryption(zip, context);
 266|         } else if (entry.getName().equals("mimetype")) {
 267|             String type = IOUtils.toString(zip, UTF_8);
 268|             metadata.set(Metadata.CONTENT_TYPE, type);
 269|         } else if (entry.getName().equals(META_NAME)) {
 270|             meta.parse(zip, new DefaultHandler(), metadata, context);
 271|         } else if (entry.getName().endsWith("content.xml")) {
 272|             if (content instanceof OpenDocumentContentParser) {
 273|                 ((OpenDocumentContentParser) content)
 274|                         .parseInternal(zip, handler, metadata, context);
 275|             } else {
 276|                 // Foreign content parser was set:
 277|                 content.parse(zip, handler, metadata, context);
 278|             }
 279|         } else if (entry.getName().endsWith("styles.xml")) {
 280|             if (content instanceof OpenDocumentContentParser) {
 281|                 ((OpenDocumentContentParser) content)
 282|                         .parseInternal(zip, handler, metadata, context);
 283|             } else {
 284|                 // Foreign content parser was set:
 285|                 content.parse(zip, handler, metadata, context);
 286|             }
 287|         } else {
 288|             String embeddedName = entry.getName();
 289|             //scrape everything under Thumbnails/ and Pictures/
 290|             if (embeddedName.contains("Thumbnails/") || embeddedName.contains("Pictures/")) {
 291| 
 292|                 Metadata embeddedMetadata = new Metadata();
 293|                 TikaInputStream stream = TikaInputStream.get(zip);
 294| 
 295|                 embeddedMetadata.set(TikaCoreProperties.RESOURCE_NAME_KEY, entry.getName());
 296|                 if (embeddedName.startsWith("Thumbnails/")) {
 297|                     embeddedMetadata.set(TikaCoreProperties.EMBEDDED_RESOURCE_TYPE,
 298|                             TikaCoreProperties.EmbeddedResourceType.THUMBNAIL.toString());
 299|                 }
 300| 
 301|                 if (embeddedName.contains("Pictures/")) {
 302|                     embeddedMetadata.set(TikaCoreProperties.EMBEDDED_RESOURCE_TYPE,
 303|                             TikaCoreProperties.EmbeddedResourceType.INLINE.toString());
 304| 
 305|                     MediaType embeddedMimeType =
 306|                             embeddedDocumentUtil.getDetector().detect(stream, embeddedMetadata);
 307|                     if (embeddedMimeType != null) {
 308|                         embeddedMetadata.set(Metadata.CONTENT_TYPE, embeddedMimeType.toString());
 309|                     }
 310|                     stream.reset();
 311|                 }
 312| 
 313|                 if (embeddedDocumentUtil.shouldParseEmbedded(embeddedMetadata)) {
 314|                     embeddedDocumentUtil.parseEmbedded(stream, new EmbeddedContentHandler(handler),
 315|                             embeddedMetadata, false);
 316|                 }
 317|             } else if (extractMacros && embeddedName.contains("Basic/")) {
 318|                 //process all files under Basic/; let maybeHandleMacro figure
 319|                 //out if it is a macro or not
 320|                 maybeHandleMacro(zip, embeddedName, handler, context);
 321|             }
 322| 
 323|         }
 324|     }
