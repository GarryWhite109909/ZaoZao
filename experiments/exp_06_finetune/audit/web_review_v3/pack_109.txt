 163|     def _do_expiration(self) -> None:
 164|         """Remove expired cookies."""
 165|         if not (expire_heap_len := len(self._expire_heap)):
 166|             return
 167| 
 168|         # If the expiration heap grows larger than the number expirations
 169|         # times two, we clean it up to avoid keeping expired entries in
 170|         # the heap and consuming memory. We guard this with a minimum
 171|         # threshold to avoid cleaning up the heap too often when there are
 172|         # only a few scheduled expirations.
 173|         if (
 174|             expire_heap_len > _MIN_SCHEDULED_COOKIE_EXPIRATION
 175|             and expire_heap_len > len(self._expirations) * 2
 176|         ):
 177|             # Remove any expired entries from the expiration heap
 178|             # that do not match the expiration time in the expirations
 179|             # as it means the cookie has been re-added to the heap
 180|             # with a different expiration time.
 181|             self._expire_heap = [
 182|                 entry
 183|                 for entry in self._expire_heap
 184|                 if self._expirations.get(entry[1]) == entry[0]
 185|             ]
 186|             heapq.heapify(self._expire_heap)

 190|         # Find any expired cookies and add them to the to-delete list
 191|         while self._expire_heap:
 192|             when, cookie_key = self._expire_heap[0]
 193|             if when > now:

 196|             # Check if the cookie hasn't been re-added to the heap
 197|             # with a different expiration time as it will be removed
 198|             # later when it reaches the top of the heap and its
 199|             # expiration time is met.
 200|             if self._expirations.get(cookie_key) == when:
