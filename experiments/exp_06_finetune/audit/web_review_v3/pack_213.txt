  90| // IsPathWithinDir reports whether targetPath resolves to a location strictly inside dir.
  91| // Both arguments are resolved to absolute paths first, so the check is robust against
  92| // "../" traversal sequences in targetPath. A path equal to dir is not considered within it.
  93| // Use this to prevent path traversal (CWE-22) before writing to a user-influenced path.
  94| func IsPathWithinDir(targetPath, dir string) (bool, error) {
  95| 	absTargetPath, err := filepath.Abs(targetPath)
  96| 	if err != nil {
  97| 		return false, errors.Wrap(err, "Failed to resolve target path")
  98| 	}

 100| 	if err != nil {
 101| 		return false, errors.Wrap(err, "Failed to resolve dir")
 102| 	}

 105| 	if err != nil {
 106| 		// the paths share no common base (e.g. different Windows volumes) - not within

 111| 		relPath != ".." &&
 112| 		!strings.HasPrefix(relPath, ".."+string(os.PathSeparator)), nil

 260| // retryUntilSuccessful calls callback every interval until duration as long as it should retry
 261| func retryUntilSuccessful(duration time.Duration,
 262| 	interval time.Duration,
 263| 	callback func(int) (bool, error)) error {
 264| 	var lastErr error
 265| 	timedOutErrorMessage := "Timed out waiting until successful"
 266| 	deadline := time.Now().Add(duration)
 267| 
 268| 	// while we haven't passed the deadline
 269| 	var retryCounter int
 270| 	for !time.Now().After(deadline) {
 271| 		shouldRetry, err := callback(retryCounter)
 272| 		lastErr = err
 273| 		if !shouldRetry {
 274| 			return err
 275| 		}
 276| 		retryCounter++
 277| 		time.Sleep(interval)
 278| 		continue
 279| 
 280| 	}
 281| 	if lastErr != nil {
 282| 
 283| 		// wrap last error
 284| 		return errors.Wrapf(lastErr, "%s", timedOutErrorMessage)
 285| 	}
 286| 
 287| 	// duration expired without any last error
 288| 	return errors.Errorf("%s", timedOutErrorMessage)
 289| }
