 491|                             $msg .= 'File already exists.';
 492|                             break;
 493| 
 494|                         case ZipArchive::ER_INCONS:
 495|                             $msg .= 'Zip archive inconsistent.';
 496|                             break;
 497| 
 498|                         case ZipArchive::ER_MEMORY:
 499|                             $msg .= 'Memory allocation failure.';
 500|                             break;
 501| 
 502|                         case ZipArchive::ER_NOENT:
 503|                             $msg .= 'No such file.';
 504|                             break;
 505| 
 506|                         case ZipArchive::ER_NOZIP:
 507|                             $msg .= 'Not a zip archive.';
 508|                             break;
 509| 
 510|                         case ZipArchive::ER_OPEN:
 511|                             $msg .= "Can't open file.";
 512|                             break;
 513| 
 514|                         case ZipArchive::ER_READ:
 515|                             $msg .= 'Read error.';
 516|                             break;
 517| 
 518|                         case ZipArchive::ER_SEEK:
 519|                             $msg .= 'Seek error.';
 520|                             break;
 521|                     }
 522|                 }
 523|                 break;
 524| 
 525|             case self::INVALID_SOURCE:
 526|                 $msg = 'Invalid source file';
 527|                 break;
 528| 
 529|             default:
