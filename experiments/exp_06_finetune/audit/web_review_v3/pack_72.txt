  88| class RankFilter(Filter):
  89|     """
  90|     Create a rank filter.  The rank filter sorts all pixels in
  91|     a window of the given size, and returns the ``rank``'th value.
  92| 
  93|     :param size: The kernel size, in pixels.

 379| class Color3DLUT(MultibandFilter):
 380|     """Three-dimensional color lookup table.
 381| 
 382|     Transforms 3-channel pixels using the values of the channels as coordinates
 383|     in the 3D lookup table and interpolating the nearest elements.
 384| 
 385|     This method allows you to apply almost any color transformation
 386|     in constant time by using pre-calculated decimated tables.
 387| 
 388|     .. versionadded:: 5.2.0
 389| 
 390|     :param size: Size of the table. One int or tuple of (int, int, int).
 391|                  Minimal size in any dimension is 2, maximum is 65.
 392|     :param table: Flat lookup table. A list of ``channels * size**3``
 393|                   float elements or a list of ``size**3`` channels-sized
 394|                   tuples with floats. Channels are changed first,
 395|                   then first dimension, then second, then third.
 396|                   Value 0.0 corresponds lowest value of output, 1.0 highest.
 397|     :param channels: Number of channels in the table. Could be 3 or 4.
 398|                      Default is 3.
 399|     :param target_mode: A mode for the result image. Should have not less
 400|                         than ``channels`` channels. Default is ``None``,
 401|                         which means that mode wouldn't be changed.
 402|     """
 403| 
 404|     name = "Color 3D LUT"
 405| 
 406|     def __init__(
 407|         self,
 408|         size: int | tuple[int, int, int],
 409|         table: Sequence[float] | Sequence[Sequence[int]] | NumpyArray,
 410|         channels: int = 3,
 411|         target_mode: str | None = None,
 412|         **kwargs: bool,
 413|     ) -> None:
 414|         if channels not in (3, 4):
 415|             msg = "Only 3 or 4 output channels are supported"
 416|             raise ValueError(msg)
 417|         self.size = size = self._check_size(size)
 418|         self.channels = channels
 419|         self.mode = target_mode
 420| 
 421|         # Hidden flag `_copy_table=False` could be used to avoid extra copying
 422|         # of the table if the table is specially made for the constructor.
 423|         copy_table = kwargs.get("_copy_table", True)
 424|         items = size[0] * size[1] * size[2]
 425|         wrong_size = False
 426| 
 427|         numpy: ModuleType | None = None
 428|         if hasattr(table, "shape"):
 429|             try:
 430|                 import numpy
 431|             except ImportError:
 432|                 pass
 433| 
 434|         if numpy and isinstance(table, numpy.ndarray):
 435|             numpy_table: NumpyArray = table
 436|             if copy_table:
 437|                 numpy_table = numpy_table.copy()
 438| 
 439|             if numpy_table.shape in [
 440|                 (items * channels,),
 441|                 (items, channels),
 442|                 (size[2], size[1], size[0], channels),
 443|             ]:
 444|                 table = numpy_table.reshape(items * channels)
 445|             else:
 446|                 wrong_size = True
 447| 
 448|         else:
 449|             if copy_table:
 450|                 table = list(table)
 451| 
 452|             # Convert to a flat list
 453|             if table and isinstance(table[0], (list, tuple)):
 454|                 raw_table = cast(Sequence[Sequence[int]], table)
 455|                 flat_table: list[int] = []
 456|                 for pixel in raw_table:
 457|                     if len(pixel) != channels:
 458|                         msg = (
 459|                             "The elements of the table should "
 460|                             f"have a length of {channels}."
 461|                         )
 462|                         raise ValueError(msg)
 463|                     flat_table.extend(pixel)
 464|                 table = flat_table
 465| 
 466|         if wrong_size or len(table) != items * channels:
 467|             msg = (
 468|                 "The table should have either channels * size**3 float items "
 469|                 "or size**3 items of channels-sized tuples with floats. "
 470|                 f"Table should be: {channels}x{size[0]}x{size[1]}x{size[2]}. "
 471|                 f"Actual length: {len(table)}"
 472|             )
 473|             raise ValueError(msg)
 474|         self.table = table
 475| 
 476|     @staticmethod
 477|     def _check_size(size: Any) -> tuple[int, int, int]:
 478|         try:
 479|             _, _, _ = size
 480|         except ValueError as e:
 481|             msg = "Size should be either an integer or a tuple of three integers."
 482|             raise ValueError(msg) from e
 483|         except TypeError:
 484|             size = (size, size, size)
 485|         size = tuple(int(x) for x in size)
 486|         for size_1d in size:
 487|             if not 2 <= size_1d <= 65:
 488|                 msg = "Size should be in [2, 65] range."
 489|                 raise ValueError(msg)
 490|         return size
 491| 
 492|     @classmethod
 493|     def generate(
 494|         cls,
 495|         size: int | tuple[int, int, int],
 496|         callback: Callable[[float, float, float], tuple[float, ...]],
 497|         channels: int = 3,
 498|         target_mode: str | None = None,
 499|     ) -> Color3DLUT:
 500|         """Generates new LUT using provided callback.
 501| 
 502|         :param size: Size of the table. Passed to the constructor.
 503|         :param callback: Function with three parameters which correspond
 504|                          three color channels. Will be called ``size**3``
 505|                          times with values from 0.0 to 1.0 and should return
 506|                          a tuple with ``channels`` elements.
 507|         :param channels: The number of channels which should return callback.
 508|         :param target_mode: Passed to the constructor of the resulting
 509|                             lookup table.
 510|         """
 511|         size_1d, size_2d, size_3d = cls._check_size(size)
 512|         if channels not in (3, 4):
 513|             msg = "Only 3 or 4 output channels are supported"
 514|             raise ValueError(msg)
 515| 
 516|         table: list[float] = [0] * (size_1d * size_2d * size_3d * channels)
 517|         idx_out = 0
 518|         for b in range(size_3d):
 519|             for g in range(size_2d):
 520|                 for r in range(size_1d):
 521|                     table[idx_out : idx_out + channels] = callback(
 522|                         r / (size_1d - 1), g / (size_2d - 1), b / (size_3d - 1)
 523|                     )
 524|                     idx_out += channels
 525| 
 526|         return cls(
 527|             (size_1d, size_2d, size_3d),
 528|             table,
 529|             channels=channels,
 530|             target_mode=target_mode,
 531|             _copy_table=False,
 532|         )

 534|     def transform(
 535|         self,
 536|         callback: Callable[..., tuple[float, ...]],
 537|         with_normals: bool = False,
 538|         channels: int | None = None,
 539|         target_mode: str | None = None,
 540|     ) -> Color3DLUT:
 541|         """Transforms the table values using provided callback and returns
 542|         a new LUT with altered values.
 543| 
 544|         :param callback: A function which takes old lookup table values
 545|                          and returns a new set of values. The number
 546|                          of arguments which function should take is
 547|                          ``self.channels`` or ``3 + self.channels``
 548|                          if ``with_normals`` flag is set.
 549|                          Should return a tuple of ``self.channels`` or
 550|                          ``channels`` elements if it is set.
 551|         :param with_normals: If true, ``callback`` will be called with
 552|                              coordinates in the color cube as the first
 553|                              three arguments. Otherwise, ``callback``
 554|                              will be called only with actual color values.
 555|         :param channels: The number of channels in the resulting lookup table.
 556|         :param target_mode: Passed to the constructor of the resulting
 557|                             lookup table.
 558|         """
 559|         if channels not in (None, 3, 4):
 560|             msg = "Only 3 or 4 output channels are supported"
 561|             raise ValueError(msg)
 562|         ch_in = self.channels
 563|         ch_out = channels or ch_in
 564|         size_1d, size_2d, size_3d = self.size
 565| 
 566|         table: list[float] = [0] * (size_1d * size_2d * size_3d * ch_out)
 567|         idx_in = 0
 568|         idx_out = 0
 569|         for b in range(size_3d):
 570|             for g in range(size_2d):
 571|                 for r in range(size_1d):
 572|                     values = self.table[idx_in : idx_in + ch_in]
 573|                     if with_normals:
 574|                         values = callback(
 575|                             r / (size_1d - 1),
 576|                             g / (size_2d - 1),
 577|                             b / (size_3d - 1),
 578|                             *values,
 579|                         )
 580|                     else:
 581|                         values = callback(*values)
 582|                     table[idx_out : idx_out + ch_out] = values
 583|                     idx_in += ch_in
 584|                     idx_out += ch_out
 585| 
 586|         return type(self)(
 587|             self.size,
 588|             table,
 589|             channels=ch_out,
 590|             target_mode=target_mode or self.mode,
 591|             _copy_table=False,
 592|         )
