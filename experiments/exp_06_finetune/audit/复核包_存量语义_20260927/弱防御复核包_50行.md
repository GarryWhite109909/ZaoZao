func checkRelativePath(p string) error {

// checkName is a whitelist for per-file names coming from BatchMove: a name
// must be one plain path element, with no separators and no dot-dot token.
func checkName(name string) error {
	if name == "" || name == "." || name == ".." {
		return ErrInvalidPath
	}
	if strings.ContainsRune(name, '/') || strings.ContainsRune(name, '\\') {
		return ErrInvalidPath
	}
	return nil

func resolveWithin(base, rel string) (string, error) {
	if err := checkRelativePath(rel); err != nil {
		return "", err
	}
	joined := path.Join(base, rel)
	if joined != base && !strings.HasPrefix(joined, base+"/") {
		return "", ErrInvalidPath
	}
	return joined, nil
}
