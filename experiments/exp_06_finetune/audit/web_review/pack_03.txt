 235| // retryUntilSuccessful calls callback every interval until duration as long as it should retry
 236| func retryUntilSuccessful(duration time.Duration,
 237| 	interval time.Duration,
 238| 	callback func(int) (bool, error)) error {
 239| 	var lastErr error
 240| 	timedOutErrorMessage := "Timed out waiting until successful"
 241| 	deadline := time.Now().Add(duration)
 242| 
 243| 	// while we haven't passed the deadline
 244| 	var retryCounter int
 245| 	for !time.Now().After(deadline) {
 246| 		shouldRetry, err := callback(retryCounter)
 247| 		lastErr = err
 248| 		if !shouldRetry {
 249| 			return err
 250| 		}
 251| 		retryCounter++
 252| 		time.Sleep(interval)
 253| 		continue
 254| 
 255| 	}
 256| 	if lastErr != nil {
 257| 
 258| 		// wrap last error
 259| 		return errors.Wrapf(lastErr, "%s", timedOutErrorMessage)
 260| 	}
 261| 
 262| 	// duration expired without any last error
 263| 	return errors.Errorf("%s", timedOutErrorMessage)
 264| }
