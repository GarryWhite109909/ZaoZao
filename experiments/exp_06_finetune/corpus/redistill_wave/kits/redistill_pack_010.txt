	"status": true,
	"version": true,
	"health": true,

// verifyToken 使用 HMAC-SHA256 验证请求来源
func verifyToken(payload RequestPayload, secret string) bool {
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(payload.Command))
	expectedMAC := mac.Sum(nil)
	receivedMAC, err := base64.StdEncoding.DecodeString(payload.Token)
	if err != nil {
		return false
	}
	return hmac.Equal(expectedMAC, receivedMAC)
}

func handler(w http.ResponseWriter, r *http.Request) {
	var payload RequestPayload
	if err := json.NewDecoder(r.Body).Decode(&payload); err != nil {
		http.Error(w, "invalid request", http.StatusBadRequest)
		return
	}

	// 第 1 道防线：HMAC 认证，拒绝未授权请求

	}

    void produce() {
        std::unique_lock<std::mutex> lock(mtx_);
        data_ = new int(42);  // line 11: 分配资源

        cv_.notify_one();
        // 模拟一些额外工作

        std::unique_lock<std::mutex> lock(mtx_);
        cv_.wait(lock, [this] { return ready_; });
        
        // 消费者使用 data_ 后释放
        if (data_) {
            std::cout << "Consumed: " << *data_ << std::endl;
            delete data_;      // line 23: 消费者释放
            data_ = nullptr;   // line 24: 置空防御
        }
    }
    
    void reset() {
        std::unique_lock<std::mutex> lock(mtx_);
        // 重置状态，但未检查 data_ 是否已被释放
        ready_ = false;
        // 这里存在潜在问题：如果 consume 还没执行，data_ 仍指向有效内存

    }

    std::condition_variable cv_;
    int* data_;
    bool ready_;
};

    buf.produce();
}

void worker_consumer(SharedBuffer& buf) {
    buf.consume();

import (

	"io"
	"net/http"

// ProxyRequest 定义API网关代理请求体
type ProxyRequest struct {
	Target string `json:"target"` // 目标URL，如 http://internal.service/api
	Method string `json:"method"` // HTTP方法
}

// InternalHosts 内部服务白名单前缀
var InternalHosts = []string{
	"http://api.internal.svc",
	"https://auth.internal.svc",
}

// isAllowedHost 校验目标URL是否在白名单内
func isAllowedHost(target string) bool {
	for _, prefix := range InternalHosts {
		if strings.HasPrefix(target, prefix) {
			return true
		}
	}
	return false
}

// fetchInternalResource 代理请求到内部服务
func fetchInternalResource(target string) (string, error) {
	// 解析URL并校验主机
	parsed, err := url.Parse(target)
	if err != nil {
		return "", err
	}

	// 仅允许HTTP/HTTPS协议
	if parsed.Scheme != "http" && parsed.Scheme != "https" {
		return "", fmt.Errorf("unsupported scheme: %s", parsed.Scheme)
	}

	// 白名单校验
	if !isAllowedHost(target) {
		return "", fmt.Errorf("host not allowed: %s", parsed.Host)
	}

	// 发起请求
	resp, err := http.Get(target)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return "", err
	}
	return string(body), nil
}

// proxyHandler 处理API网关转发请求
func proxyHandler(w http.ResponseWriter, r *http.Request) {
	var req ProxyRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "bad request", http.StatusBadRequest)
		return
	}

	// 调用内部资源获取
	result, err := fetchInternalResource(req.Target)
	if err != nil {
		http.Error(w, err.Error(), http.StatusBadGateway)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.Write([]byte(result))
}

func main() {
	http.HandleFunc("/proxy", proxyHandler)
	http.ListenAndServe(":8080", nil)
}
