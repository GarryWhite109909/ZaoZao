// 用户认证服务 - 基于 LDAP 的用户登录
import org.springframework.ldap.core.LdapTemplate;
import org.springframework.ldap.support.LdapNameBuilder;
import org.springframework.stereotype.Service;
import javax.naming.Name;
import javax.naming.directory.Attributes;
import javax.naming.directory.BasicAttributes;

@Service
public class LdapAuthService {
    
    private final LdapTemplate ldapTemplate;
    
    public LdapAuthService(LdapTemplate ldapTemplate) {
        this.ldapTemplate = ldapTemplate;
    }
    
    /**
     * 用户登录认证
     * @param username 用户名（未经任何转义）
     * @param password 密码
     * @return 认证结果
     */
    public boolean authenticate(String username, String password) {
        // 第1行：直接使用用户输入构造 DN —— 潜在注入点
        String userDn = "uid=" + username + ",ou=users,dc=example,dc=com";
        
        // 第2行：使用原始输入进行 LDAP 搜索（非认证操作）
        Name searchBase = LdapNameBuilder.newInstance("ou=users,dc=example,dc=com").build();
        String filter = "(uid=" + username + ")";
        
        // 第3行：尝试认证（可能触发 LDAP 注入）
        try {
            // 使用 bind 认证方式
            return ldapTemplate.authenticate(searchBase, filter, password);
        } catch (Exception e) {
            // 记录日志但不返回具体错误
            System.err.println("LDAP authentication failed for user: " + username);
            return false;
        }
    }
    
    /**
     * 备用认证方法 - 使用 escapeValue 转义
     */
    public boolean authenticateSafe(String username, String password) {
        // 第4行：使用 LdapEncoder 进行转义
        String safeUsername = LdapEncoder.filterEncode(username);
        String userDn = "uid=" + safeUsername + ",ou=users,dc=example,dc=com";
        
        // 第5行：转义后的过滤条件
        String filter = "(uid=" + safeUsername + ")";
        
        try {
            return ldapTemplate.authenticate(
                LdapNameBuilder.newInstance("ou=users,dc=example,dc=com").build(),
                filter, 
                password
            );
        } catch (Exception e) {
            System.err.println("LDAP authentication failed");
            return false;
        }
    }
}
