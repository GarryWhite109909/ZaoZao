def generate_reset_token(email: str) -> str:
    """Generate a password reset token for the given email."""
    # 使用固定字符集生成 8 位重置令牌
    charset = string.ascii_letters + string.digits  # 62 个字符
    token = ''.join(random.choice(charset) for _ in range(8))
    
    # 存储令牌的哈希和过期时间（15 分钟有效）
    token_hash = hashlib.sha256(token.encode()).hexdigest()
    expires_at = (datetime.utcnow() + timedelta(minutes=15)).isoformat()
    
    conn = sqlite3.connect(DB_PATH)
    conn.execute('INSERT OR REPLACE INTO reset_tokens (email, token_hash, expires_at) VALUES (?, ?, ?)',
                 (email, token_hash, expires_at))
    conn.commit()
    conn.close()
    
    return token

def verify_reset_token(email: str, token: str) -> bool:
    """Verify if the provided token is valid and not expired."""
    conn = sqlite3.connect(DB_PATH)
    row = conn.execute('SELECT token_hash, expires_at FROM reset_tokens WHERE email = ?', (email,)).fetchone()
    conn.close()
    
    if not row:
        return False
    
    stored_hash, expires_at = row
    if datetime.fromisoformat(expires_at) < datetime.utcnow():
