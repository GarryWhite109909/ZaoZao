package main

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/base64"
	"encoding/json"
	"fmt"
	"net/http"
	"os/exec"
	"strings"
)

// RequestPayload 定义 API 网关接收的请求体
type RequestPayload struct {
	Command string `json:"command"`
	Token   string `json:"token"`
}

// allowedCommands 白名单：仅允许执行预定义的只读命令
var allowedCommands = map[string]bool{
	"status": true,
	"version": true,
	"health": true,
}

// verifyToken 使用 HMAC-SHA256 验证请求来源
func verifyToken(payload RequestPayload, secret string) bool {
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(payload.Command))
	expectedMAC := mac.Sum(nil)
	receivedMAC, err := base64.StdEncoding.DecodeString(payload.Token)
	if err != nil {
		return false
	}
	return hmac.Equal(expectedMAC, receivedMAC)
}

func handler(w http.ResponseWriter, r *http.Request) {
	var payload RequestPayload
	if err := json.NewDecoder(r.Body).Decode(&payload); err != nil {
		http.Error(w, "invalid request", http.StatusBadRequest)
		return
	}

	// 第 1 道防线：HMAC 认证，拒绝未授权请求
	if !verifyToken(payload, "gateway-secret-key") {
		http.Error(w, "unauthorized", http.StatusUnauthorized)
		return
	}

	// 第 2 道防线：命令白名单校验，完全阻断任意命令执行
	if !allowedCommands[payload.Command] {
		http.Error(w, "command not allowed", http.StatusForbidden)
		return
	}

	// 第 3 道防线：使用参数列表形式（无 shell），杜绝拼接注入
	// 白名单中的命令均为固定字符串，不包含用户可控参数
	cmd := exec.Command("gateway-cli", payload.Command)
	output, err := cmd.CombinedOutput()
	if err != nil {
		http.Error(w, fmt.Sprintf("exec failed: %v", err), http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.Write(output)
}

func main() {
	http.HandleFunc("/api/v1/operation", handler)
	http.ListenAndServe(":8080", nil)
}

        app.ClientConnection.broadcast(resource="events", cmd="reset")

        options_dict = optmanager.dump_dicts(self.options, updated)
        app.ClientConnection.broadcast(
            resource="options", cmd="update", data=options_dict

        app.ClientConnection.broadcast(

            cmd="update",

                    s.mode.full_spec: s.to_json() for s in self.proxyserver.servers

        tornado.ioloop.IOLoop.current()

Try specifying a different port by using `--set web_port={self.options.web_port + 2}`."

        logger.info(
            f"Web server listening at http://{self.options.web_host}:{self.options.web_port}/",
        )

const (
	hashInputDelimiter = byte('$')

// SHA-512/256 is protected against length extension attacks and is more performant than SHA-256 on 64-bit architectures.
// https://en.wikipedia.org/wiki/Template:Comparison_of_SHA_functions
func SHA512_256(in ...[]byte) []byte {
	var data []byte
	state := crypto.SHA512_256.New()
	inLen := len(in)
	if inLen == 0 {
		return nil
	}
	bzSize := 0
	// prevent hash collisions with this prefix containing the block count
	inLenBz := make([]byte, 64/8)
	// converting between int and uint64 doesn't change the sign bit, but it may be interpreted as a larger value.
	// this prefix is never read/interpreted, so that doesn't matter.
	binary.LittleEndian.PutUint64(inLenBz, uint64(inLen))
	for _, bz := range in {
		bzSize += len(bz)
	}
	data = make([]byte, 0, len(inLenBz)+bzSize+inLen)
	data = append(data, inLenBz...)
	for _, bz := range in {
		data = append(data, bz...)
		data = append(data, hashInputDelimiter) // safety delimiter
		l := []byte(strconv.Itoa(len(bz)))
		data = append(data, l...) // Security audit: length of each byte buffer should be added after
		// each security delimiters in order to enforce proper domain separation
	}
	// n < len(data) or an error will never happen.
	// see: https://golang.org/pkg/hash/#Hash and https://github.com/golang/go/wiki/Hashing#the-hashhash-interface
	if _, err := state.Write(data); err != nil {
		Logger.Errorf("SHA512_256 Write() failed: %v", err)
		return nil
	}
	return state.Sum(nil)
}

func SHA512_256i(in ...*big.Int) *big.Int {
	var data []byte
	state := crypto.SHA512_256.New()
	inLen := len(in)
	if inLen == 0 {
		return nil
	}
	bzSize := 0
	// prevent hash collisions with this prefix containing the block count
	inLenBz := make([]byte, 64/8)
	// converting between int and uint64 doesn't change the sign bit, but it may be interpreted as a larger value.
	// this prefix is never read/interpreted, so that doesn't matter.
	binary.LittleEndian.PutUint64(inLenBz, uint64(inLen))
	ptrs := make([][]byte, inLen)
	for i, n := range in {
		ptrs[i] = append(n.Bytes(), byte(n.Sign()))
		bzSize += len(ptrs[i])
	}
	data = make([]byte, 0, len(inLenBz)+bzSize+inLen)
	data = append(data, inLenBz...)
	for i := range in {
		data = append(data, ptrs[i]...)
		data = append(data, hashInputDelimiter) // safety delimiter
		l := []byte(strconv.Itoa(len(ptrs[i])))
		data = append(data, l...) // Security audit: length of each byte buffer should be added after
		// each security delimiters in order to enforce proper domain separation
	}
	// n < len(data) or an error will never happen.
	// see: https://golang.org/pkg/hash/#Hash and https://github.com/golang/go/wiki/Hashing#the-hashhash-interface
	if _, err := state.Write(data); err != nil {
		Logger.Errorf("SHA512_256i Write() failed: %v", err)
		return nil
	}
	return new(big.Int).SetBytes(state.Sum(nil))
}

/**
 * Smarty Internal Plugin Resource Stream
 * Implements the streams as resource for Smarty template
 *
 * @link       https://php.net/streams

class StreamPlugin extends RecompiledPlugin {

	/**
	 * populate Source Object with meta data from Resource
	 *
	 * @param Source $source source object
	 * @param Template $_template template object
	 *
	 * @return void
	 */
	public function populate(Source $source, ?Template $_template = null) {
		$source->uid = false;
		$source->content = $this->getContent($source);
		$source->timestamp = $source->exists = !!$source->content;
	}

	/**
	 * Load template's source from stream into current template object
	 *
	 * @param Source $source source object
	 *
	 * @return string template source
	 */
	public function getContent(Source $source) {

		if (strpos($source->getResourceName(), '://') !== false) {
			$filepath = $source->getResourceName();
		} else {
			$filepath = str_replace(':', '://', $source->getFullResourceName());
		}

		$t = '';
		// the availability of the stream has already been checked in Smarty\Resource\Base::fetch()
		$fp = fopen($filepath, 'r+');
		if ($fp) {
			while (!feof($fp) && ($current_line = fgets($fp)) !== false) {
				$t .= $current_line;
			}
			fclose($fp);
			return $t;
		} else {
			return false;
		}
	}
