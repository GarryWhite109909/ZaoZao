#define MAX_DATA_SIZE 1024
#define SAFE_FREE(ptr) do { if (ptr) { free(ptr); (ptr) = NULL; } } while (0)

typedef struct {
    char *buffer;
    size_t length;
} DataBuffer;

static int init_buffer(DataBuffer *db, size_t size) {
    if (!db || size == 0 || size > MAX_DATA_SIZE) {
        return -1;
    }
    db->buffer = (char *)malloc(size);
    if (!db->buffer) {
        return -1;
    }
    db->length = size;
    memset(db->buffer, 0, size);
    return 0;
}

static void destroy_buffer(DataBuffer *db) {
    if (db) {
        SAFE_FREE(db->buffer);
        db->length = 0;
    }
}

static int copy_to_buffer(DataBuffer *db, const char *src, size_t src_len) {
    if (!db || !src || !db->buffer) {
        return -1;
    }
    if (src_len >= db->length) {
        return -1;
    }
    memcpy(db->buffer, src, src_len);
    db->buffer[src_len] = '\0';
    return 0;
}

int main(int argc, char *argv[]) {
    DataBuffer db;
    memset(&db, 0, sizeof(db));

    if (init_buffer(&db, 256) != 0) {
        fprintf(stderr, "Buffer init failed\n");
        return 1;
    }

    const char *user_input = "Hello, secure world!";
    size_t input_len = strlen(user_input);

    if (copy_to_buffer(&db, user_input, input_len) != 0) {
        fprintf(stderr, "Copy failed\n");
        destroy_buffer(&db);
        return 1;
    }

    printf("Buffer content: %s\n", db.buffer);
    destroy_buffer(&db);

type LogEntry struct {

	Level     string
	Message   string
	Metadata  map[string]interface{}
}

// LogProcessor 从网络连接读取日志并处理
type LogProcessor struct {
	conn net.Conn
	out  *os.File
}

// NewLogProcessor 创建日志处理器
func NewLogProcessor(conn net.Conn) *LogProcessor {
	return &LogProcessor{
		conn: conn,
		out:  os.Stdout,
	}
}

// ProcessLogs 持续接收并处理日志流
func (lp *LogProcessor) ProcessLogs() error {
	dec := gob.NewDecoder(lp.conn)
	for {
		var entry LogEntry
		// line 31: 反序列化来自网络的数据
		if err := dec.Decode(&entry); err != nil {
			if err == io.EOF {
				return nil
			}
			return fmt.Errorf("decode error: %w", err)
		}
		lp.writeLog(entry)
	}
}

// writeLog 将日志写入输出流
func (lp *LogProcessor) writeLog(entry LogEntry) {
	msg := fmt.Sprintf("[%s] %s", entry.Level, entry.Message)
	if len(entry.Metadata) > 0 {
		msg += fmt.Sprintf(" %v", entry.Metadata)
	}
	fmt.Fprintln(lp.out, msg)
	log.Printf("processed log: %s", entry.Message)
}

func main() {
	if len(os.Args) < 2 {
		log.Fatal("usage: logserver <port>")
	}
	ln, err := net.Listen("tcp", ":"+os.Args[1])
	if err != nil {
		log.Fatal(err)
	}
	for {
		conn, err := ln.Accept()
		if err != nil {
			log.Println(err)
			continue
		}
		go func(c net.Conn) {
			defer c.Close()
			lp := NewLogProcessor(c)
			if err := lp.ProcessLogs(); err != nil {
				log.Printf("processor error: %v", err)

package main

import (

	"strings"
)

type fetchRequest struct {
	TargetURL string `json:"target_url"`
}

// fetchInternalResource forwards a request to an internal service
// after validating the URL against a whitelist of internal hosts.
func fetchInternalResource(w http.ResponseWriter, r *http.Request) {
	var req fetchRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "invalid request", http.StatusBadRequest)
		return
	}

	parsedURL, err := url.Parse(req.TargetURL)
	if err != nil {
		http.Error(w, "invalid url", http.StatusBadRequest)
		return
	}

	// Whitelist check: only allow requests to internal.example.com
	allowedHost := "internal.example.com"
	if parsedURL.Host != allowedHost {
		http.Error(w, "host not allowed", http.StatusForbidden)
		return
	}

	// Perform the request to the internal service
	client := &http.Client{}
	resp, err := client.Get(parsedURL.String())
	if err != nil {
		http.Error(w, "upstream error", http.StatusBadGateway)
		return
	}
	defer resp.Body.Close()

	// Copy response back to the client
	body, _ := io.ReadAll(resp.Body)
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(resp.StatusCode)
	w.Write(body)
}

func main() {
	http.HandleFunc("/api/internal/fetch", fetchInternalResource)
	port := os.Getenv("PORT")
	if port == "" {
		port = "8080"
	}
	fmt.Printf("API gateway listening on :%s\n", port)
	http.ListenAndServe(":"+port, nil)
}

    if (g_shared) {
