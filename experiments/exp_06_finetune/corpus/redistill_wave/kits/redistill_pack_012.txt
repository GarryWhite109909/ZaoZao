class LDAPAuth {
    private $conn;
    private $base_dn = 'ou=users,dc=example,dc=com';
    
    public function __construct($host = 'ldap://localhost') {
        $this->conn = ldap_connect($host);
        ldap_set_option($this->conn, LDAP_OPT_PROTOCOL_VERSION, 3);
        ldap_set_option($this->conn, LDAP_OPT_REFERRALS, 0);
    }
    
    public function authenticate($username, $password) {
        // 先绑定管理员账号验证用户凭据
        $bind = @ldap_bind($this->conn, "cn=admin,dc=example,dc=com", "admin_pass_123");
        if (!$bind) {
            return ['error' => 'LDAP bind failed'];
        }
        
        // 使用用户提供的过滤条件搜索
        $filter = "(&(objectClass=person)(uid=" . $username . "))";
        $search = ldap_search($this->conn, $this->base_dn, $filter);
        
        if (!$search) {
            return ['error' => 'Search failed'];
        }
        
        $entries = ldap_get_entries($this->conn, $search);
        if ($entries['count'] == 0) {
            return ['error' => 'User not found'];
        }
        
        // 验证密码
        $user_dn = $entries[0]['dn'];
        $pass_check = @ldap_bind($this->conn, $user_dn, $password);
        if ($pass_check) {
            $_SESSION['user'] = $entries[0]['uid'][0];
            return ['success' => true];
        }
        
        return ['error' => 'Invalid credentials'];
    }
}

// 处理请求
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = $_POST['username'] ?? '';
    $password = $_POST['password'] ?? '';
    
    // 简单过滤，看似安全
    $username = str_replace(['(', ')', '*', '\\'], '', $username);
    
    $ldap = new LDAPAuth();
    $result = $ldap->authenticate($username, $password);

import csv
import io
from django.http import HttpResponse
from django.contrib.auth.decorators import login_required
from django.views.decorators.http import require_GET
from .models import ActivityLog

@login_required
@require_GET
def export_activity_logs(request):
    """
    导出当前用户的活动日志，格式为 CSV。
    支持按操作类型筛选，操作类型通过 query 参数传递。
    """
    user = request.user
    # 从 query string 获取操作类型，默认 'all' 表示全部
    action_filter = request.GET.get('action', 'all')

    # 防御迷惑点：看似用 replace 清理，实际未阻断注入
    # 实际操作类型是预定义的，但此处未做严格校验
    action_filter = action_filter.replace(';', '').replace('--', '')
    # 注意：这里没有调用数据库，而是用 Python 过滤，所以 SQL 注入不适用
    # 但 CWE-943 关注的是数据导出逻辑的注入，比如 CSV 公式注入

    # 查询当前用户的活动日志
    logs = ActivityLog.objects.filter(user=user)

    # 如果指定了操作类型，则按操作类型过滤
    if action_filter != 'all':
        # 使用 startswith 做前缀匹配，看似安全，但未规范化路径
        # 实际业务中操作类型是 'login', 'logout', 'download' 等
        # 此处直接使用用户输入做前缀匹配，可能导致逻辑绕过
        logs = [log for log in logs if log.action.startswith(action_filter)]

    # 生成 CSV
    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow(['时间', '操作', '详情'])

    for log in logs:
        # 防御：对 CSV 内容进行转义，防止公式注入
        def safe_csv_field(value):
            # 将值转为字符串
            value = str(value)
            # 防御：如果以 =, +, -, @ 开头，则加单引号前缀
            if value.startswith(('=', '+', '-', '@')):
                return "'" + value
            return value

        # 注意：详情字段可能包含用户输入，但这里做了转义
        writer.writerow([
            safe_csv_field(log.timestamp),
            safe_csv_field(log.action),
            safe_csv_field(log.details)
        ])

    # 返回 CSV 响应
    response = HttpResponse(output.getvalue(), content_type='text/csv')
    response['Content-Disposition'] = 'attachment; filename="activity_logs.csv"'
    return response

import io

def parse_xml(xml_data):
    """解析用户上传的 XML 并提取导出字段"""
    # 防御：移除 DOCTYPE 声明（迷惑点 1）
    xml_data = xml_data.replace("<!DOCTYPE", "<!--")
    xml_data = xml_data.replace("<!ENTITY", "<!--")
    
    root = ET.fromstring(xml_data)
    records = []
    for record in root.findall(".//record"):
        fields = {}
        for field in record:
            fields[field.tag] = field.text or ""
        records.append(fields)
    return records

@app.route("/api/export", methods=["POST"])
def export_data():
    """接收 XML 数据，解析后导出为 CSV"""
    try:
        xml_content = request.data.decode("utf-8")
        if not xml_content.strip():
            return jsonify({"error": "Empty XML"}), 400
        
        # 防御：检查是否包含外部实体关键词（迷惑点 2）
        if "<!DOCTYPE" in xml_content or "<!ENTITY" in xml_content:
            return jsonify({"error": "Invalid XML: DOCTYPE/ENTITY not allowed"}), 400
        
        # 解析 XML
        records = parse_xml(xml_content)
        
        # 生成 CSV 导出
        csv_buffer = io.StringIO()
        if records:
            headers = list(records[0].keys())
            csv_buffer.write(",".join(headers) + "\n")
            for rec in records:
                csv_buffer.write(",".join(rec.get(h, "") for h in headers) + "\n")
        
        csv_buffer.seek(0)
        return send_file(
            io.BytesIO(csv_buffer.getvalue().encode("utf-8")),
            mimetype="text/csv",
            as_attachment=True,
            download_name="export.csv"
        )

@app.route("/plugins/install", methods=["POST"])
def install_plugin():

    plugin_name = request.form.get("plugin_name")
    archive = request.files.get("archive")

    if not plugin_name or not archive:
        return jsonify({"error": "missing fields"}), 400
    if not NAME_RE.match(plugin_name):
        return jsonify({"error": "invalid plugin name"}), 400


    target_dir = os.path.join(PLUGIN_ROOT, plugin_name)
    zip_path = target_dir + ".zip"

    os.makedirs(target_dir, exist_ok=True)
    archive.save(zip_path)

    with zipfile.ZipFile(zip_path) as zf:
        for m in zf.namelist():
            if ".." in m or m.startswith(("/", "\")):
                raise zipfile.BadZipFile("zip slip")
            zf.extract(m, target_dir)

    manifest_path = os.path.join(target_dir, "plugin.yaml")
    if os.path.exists(manifest_path):
        with open(manifest_path) as f:
            pass  # 读取 manifest 但不校验其中的路径字段
