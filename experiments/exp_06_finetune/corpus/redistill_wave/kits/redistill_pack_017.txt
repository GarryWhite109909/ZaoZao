bool decode_scanline(FrameBuffer& fb, const uint8_t* compressed, size_t comp_len, size_t offset) {
    if (offset + comp_len > fb.size) {
        return false; // bounds check before write
    }
    for (size_t i = 0; i < comp_len; ++i) {
        uint8_t tag = compressed[i];
        if (tag & 0x80) {
            // RLE run: next byte is the value, repeat count is tag & 0x7F
            if (i + 1 >= comp_len) return false;
            uint8_t value = compressed[++i];

            if (offset + run_len > fb.size) return false;

// ReplacePathRegex is a middleware used to replace the path of a URL request with a regular expression.
type replacePathRegex struct {
	next        http.Handler
	regexp      *regexp.Regexp
	replacement string
	name        string
}

// New creates a new replace path regex middleware.
func New(ctx context.Context, next http.Handler, config dynamic.ReplacePathRegex, name string) (http.Handler, error) {
	log.FromContext(middlewares.GetLoggerCtx(ctx, name, typeName)).Debug("Creating middleware")

	exp, err := regexp.Compile(strings.TrimSpace(config.Regex))
	if err != nil {
		return nil, fmt.Errorf("error compiling regular expression %s: %w", config.Regex, err)
	}

	return &replacePathRegex{
		regexp:      exp,
		replacement: strings.TrimSpace(config.Replacement),
		next:        next,
		name:        name,
	}, nil
}

func (rp *replacePathRegex) GetTracingInformation() (string, ext.SpanKindEnum) {
	return rp.name, tracing.SpanKindNoneEnum
}

func (rp *replacePathRegex) ServeHTTP(rw http.ResponseWriter, req *http.Request) {
	currentPath := req.URL.RawPath
	if currentPath == "" {
		currentPath = req.URL.EscapedPath()
	}

	if rp.regexp != nil && rp.regexp.MatchString(currentPath) {
		req.Header.Add(replacepath.ReplacedPathHeader, currentPath)
		req.URL.RawPath = rp.regexp.ReplaceAllString(currentPath, rp.replacement)

		// as replacement can introduce escaped characters
		// Path must remain an unescaped version of RawPath
		// Doesn't handle multiple times encoded replacement (`/` => `%2F` => `%252F` => ...)
		var err error
		req.URL.Path, err = url.PathUnescape(req.URL.RawPath)
		if err != nil {
			log.FromContext(middlewares.GetLoggerCtx(context.Background(), rp.name, typeName)).Error(err)
			http.Error(rw, err.Error(), http.StatusInternalServerError)
			return
		}

		req.RequestURI = req.URL.RequestURI()
	}

func NewAPIGateway() *APIGateway {
	return &APIGateway{
		allowedImages: map[string]bool{
			"nginx:1.25":   true,
			"redis:7.2":    true,
			"postgres:16":  true,
		},
		timeout: 5 * time.Second,
	}

// ContainerRequest is the API payload for starting a container
type ContainerRequest struct {
	Image string `json:"image"`
	Name  string `json:"name"`
}

// startContainer is the core handler that executes the docker command
func (g *APIGateway) startContainer(w http.ResponseWriter, r *http.Request) {
	var req ContainerRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "invalid request body", http.StatusBadRequest)
		return
	}

	// Defense 1: Whitelist validation - only allow pre-approved images
	if !g.allowedImages[req.Image] {
		http.Error(w, "image not allowed", http.StatusForbidden)
		return
	}

	// Defense 2: Reject any name with shell metacharacters before it reaches exec
	// We use a character whitelist approach for the container name
	for _, ch := range req.Name {
		if !(ch >= 'a' && ch <= 'z') && !(ch >= '0' && ch <= '9') && ch != '-' && ch != '_' {
			http.Error(w, "invalid container name", http.StatusBadRequest)
			return
		}
	}
	if len(req.Name) == 0 {
		http.Error(w, "container name required", http.StatusBadRequest)
		return
	}

	// Defense 3: Use exec.CommandContext with explicit argument list (no shell)
	// The image is validated by whitelist, name is validated by character whitelist
	ctx, cancel := context.WithTimeout(r.Context(), g.timeout)
	defer cancel()

	cmd := exec.CommandContext(ctx, "docker", "run", "-d", "--name", req.Name, req.Image)
	output, err := cmd.CombinedOutput()
	if err != nil {
		log.Printf("docker run failed: %v, output: %s", err, string(output))
		http.Error(w, "container start failed", http.StatusInternalServerError)
		return
	}

	fmt.Fprintf(w, "container started: %s", req.Name)
}

#include <cstdint>
#include <cstring>
#include <vector>
#include <cstdio>

#define MAX_CHANNELS 4
#define CHUNK_HEADER_SIZE 8

struct AudioChunk {
    uint32_t sample_rate;
    uint16_t channels;
    uint16_t bits_per_sample;
    uint32_t data_size;
    const uint8_t* data;
};

// Decodes a single interleaved frame into per-channel samples.
// frame_index: 0-based frame number; channel: 0..channels-1
int decode_frame_sample(const AudioChunk* chunk, uint32_t frame_index, uint16_t channel) {
    if (chunk->channels == 0 || chunk->channels > MAX_CHANNELS) return -1;
    if (channel >= chunk->channels) return -1;

    uint32_t bytes_per_sample = chunk->bits_per_sample / 8;
    if (bytes_per_sample == 0 || bytes_per_sample > 4) return -1;

    // frame_size = channels * bytes_per_sample
    uint64_t frame_size = static_cast<uint64_t>(chunk->channels) * bytes_per_sample;
    if (frame_size == 0) return -1;

    uint64_t offset = static_cast<uint64_t>(frame_index) * frame_size + channel * bytes_per_sample;
    if (offset + bytes_per_sample > chunk->data_size) return -1;

    int32_t sample = 0;
    memcpy(&sample, chunk->data + offset, bytes_per_sample);
    return static_cast<int>(sample);
}

// Parses a raw audio container and extracts chunk references.
// Returns 0 on success, -1 on malformed header.
int parse_audio_container(const uint8_t* buffer, size_t buffer_len, AudioChunk* out) {
    if (buffer_len < CHUNK_HEADER_SIZE) return -1;

    uint32_t sample_rate;
    uint16_t channels, bits;
    uint32_t data_size;
    memcpy(&sample_rate, buffer, 4);
    memcpy(&channels, buffer + 4, 2);
    memcpy(&bits, buffer + 6, 2);

    if (channels == 0 || channels > MAX_CHANNELS) return -1;
    if (bits % 8 != 0 || bits > 32) return -1;

    out->sample_rate = sample_rate;
    out->channels = channels;
    out->bits_per_sample = bits;

    // Data section follows the 8-byte header; length is inferred from buffer_len.
    out->data = buffer + CHUNK_HEADER_SIZE;
    out->data_size = static_cast<uint32_t>(buffer_len - CHUNK_HEADER_SIZE);

    // Note: data_size is truncated to 32-bit; buffer_len may exceed 4GB on 64-bit.
    return 0;
}

// Public entry point: decodes a specific frame from a raw buffer.
int get_frame_sample(const uint8_t* container, size_t len, uint32_t frame_idx, uint16_t ch) {
    AudioChunk chunk;
    if (parse_audio_container(container, len, &chunk) != 0) return -1;
    return decode_frame_sample(&chunk, frame_idx, ch);
}
