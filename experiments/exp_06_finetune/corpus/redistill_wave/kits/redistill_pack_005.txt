		http.Error(w, "missing order_id", http.StatusBadRequest)
		return
	}

		http.Error(w, "failed to fetch order", http.StatusInternalServerError)
		return
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		http.Error(w, "failed to read response", http.StatusInternalServerError)
		return
	}

	// 解析并返回订单信息
	var order OrderInfo
	if err := json.Unmarshal(body, &order); err != nil {
		http.Error(w, "invalid order data", http.StatusInternalServerError)
		return
	}

	// 检查响应状态
	if resp.StatusCode != http.StatusOK {
		http.Error(w, "order service error", http.StatusBadGateway)
		return
	}

	// 返回订单信息
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(order)

    char full_path[MAX_PATH_LEN];
    FILE* fp;
    char* content;
    long file_size;

    /* 拼接完整路径 */
    snprintf(full_path, sizeof(full_path), "%s%s", CONFIG_DIR, filename);
    
    /* 打开配置文件 */
    fp = fopen(full_path, "r");
    if (fp == NULL) {
        return NULL;
    }

    /* 获取文件大小 */
    fseek(fp, 0, SEEK_END);
    file_size = ftell(fp);
    fseek(fp, 0, SEEK_SET);

    /* 分配内存并读取内容 */
    content = (char*)malloc(file_size + 1);
    if (content == NULL) {
        fclose(fp);
        return NULL;
    }

    size_t read_size = fread(content, 1, file_size, fp);
    content[read_size] = '\0';
    fclose(fp);

    return content;
}

    if (argc < 2) {
        printf("Usage: %s <config_name>\n", argv[0]);

    if (config_content == NULL) {
        printf("Failed to load config: %s\n", argv[1]);
        return 1;

package main

import (
	"encoding/gob"
	"fmt"
	"io"
	"net/http"
	"os"
)

type UserProfile struct {
	Name    string
	Age     int
	IsAdmin bool
}

func uploadHandler(w http.ResponseWriter, r *http.Request) {
	file, _, err := r.FormFile("profile")
	if err != nil {
		http.Error(w, "Failed to read file", http.StatusBadRequest)
		return
	}
	defer file.Close()

	// 创建临时文件存储上传内容
	tmpFile, err := os.CreateTemp("", "upload-*.dat")
	if err != nil {
		http.Error(w, "Failed to create temp file", http.StatusInternalServerError)
		return
	}
	defer os.Remove(tmpFile.Name())

	_, err = io.Copy(tmpFile, file)
	if err != nil {
		http.Error(w, "Failed to save file", http.StatusInternalServerError)
		return
	}
	tmpFile.Close()

	// 反序列化用户上传的gob数据
	var profile UserProfile
	dataFile, err := os.Open(tmpFile.Name())
	if err != nil {
		http.Error(w, "Failed to open file", http.StatusInternalServerError)
		return
	}
	defer dataFile.Close()

	decoder := gob.NewDecoder(dataFile)
	if err := decoder.Decode(&profile); err != nil {
		http.Error(w, "Failed to decode profile", http.StatusBadRequest)
		return
	}

	fmt.Fprintf(w, "User: %s, Age: %d, Admin: %v\n", profile.Name, profile.Age, profile.IsAdmin)
}

func main() {
	http.HandleFunc("/upload", uploadHandler)
	http.ListenAndServe(":8080", nil)
}
