public class PasswordResetService {
    private static final String PASSWORD_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    private static final int PASSWORD_LENGTH = 8;
    private static final long RESET_TOKEN_VALIDITY_MS = 15 * 60 * 1000; // 15 minutes
    
    private Map<String, ResetRequest> resetRequests = new HashMap<>();
    private Random random = new Random(); // 使用 java.util.Random
    
    public String generateResetCode(String userId) {
        StringBuilder code = new StringBuilder(PASSWORD_LENGTH);
        for (int i = 0; i < PASSWORD_LENGTH; i++) {
            int index = random.nextInt(PASSWORD_CHARS.length());
            code.append(PASSWORD_CHARS.charAt(index));
        }
        String resetCode = code.toString();
        
        resetRequests.put(userId, new ResetRequest(resetCode, System.currentTimeMillis()));
        return resetCode;
    }
    
    public boolean verifyResetCode(String userId, String code) {
        ResetRequest request = resetRequests.get(userId);
        if (request == null) {
            return false;
        }
        
        if (System.currentTimeMillis() - request.timestamp > RESET_TOKEN_VALIDITY_MS) {
            resetRequests.remove(userId);
            return false;
        }
        
        return request.code.equals(code);
    }
    
    public void resetPassword(String userId, String newPassword) {
        if (verifyResetCode(userId, newPassword)) {
            // 实际项目中这里会更新用户密码
            System.out.println("Password reset for user: " + userId);
            resetRequests.remove(userId);
        }
    }
    
    private static class ResetRequest {
        String code;
        long timestamp;
        
        ResetRequest(String code, long timestamp) {
            this.code = code;
            this.timestamp = timestamp;
        }
    }
}

    local request_path="$1"
    local request_method="$2"

    # 仅支持GET请求
    if [[ "$request_method" != "GET" ]]; then
        log_request "$request_method" "$request_path" "405"
        echo "HTTP/1.1 405 Method Not Allowed"
        echo "Content-Type: text/plain"
        echo ""
        echo "Method not allowed"
        return

    if [[ "$request_path" =~ ^/container/status/([a-zA-Z0-9_-]+)$ ]]; then
        local container_name="${BASH_REMATCH[1]}"

        local is_allowed=0
        for allowed in "${ALLOWED_CONTAINERS[@]}"; do
            if [[ "$container_name" == "$allowed" ]]; then
                is_allowed=1
                break
            fi
        done

        if [[ $is_allowed -eq 0 ]]; then
            log_request "$request_method" "$request_path" "403"
            echo "HTTP/1.1 403 Forbidden"
            echo "Content-Type: text/plain"
            echo ""
            echo "Container not allowed"

static int __init drv_init(void)
{
    k_buf = kzalloc(BUF_SIZE, GFP_KERNEL);
    if (!k_buf)
        return -ENOMEM;

    drv = kzalloc(sizeof(*drv), GFP_KERNEL);
    if (!drv) {
        kfree(k_buf);
        return -ENOMEM;
    }

    cdev_init(&drv->cdev, &fops);
    // 实际注册省略（alloc_chrdev_region 等）
    return 0;
}

static void __exit drv_exit(void)
{
    kfree(k_buf);
    kfree(drv);
}

module_init(drv_init);
module_exit(drv_exit);
MODULE_LICENSE("GPL");
