const crypto = require('crypto');
const { createClient } = require('redis');

const redis = createClient({ url: 'redis://localhost:6379' });
const TOKEN_TTL = 3600; // 1 hour

async function requestPasswordReset(req, res) {
  const { email } = req.body;

  // 1. 校验邮箱格式
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailRegex.test(email)) {
    return res.status(400).json({ error: 'Invalid email format' });
  }

  // 2. 生成重置令牌
  const token = crypto.randomBytes(32).toString('hex');
  const tokenHash = crypto.createHash('sha256').update(token).digest('hex');

  // 3. 存储令牌（带过期时间）
  await redis.set(`pwd_reset:${tokenHash}`, email, 'EX', TOKEN_TTL);

  // 4. 发送重置邮件（模拟）

  console.log(`[EMAIL] To: ${email}, Link: ${resetLink}`);

  res.json({ message: 'Reset link sent if email exists' });
}

async function resetPassword(req, res) {
  const { token, newPassword } = req.body;

  if (!token || !newPassword) {
    return res.status(400).json({ error: 'Missing token or new password' });
  }

  // 5. 验证密码强度
  if (newPassword.length < 8 || !/[A-Z]/.test(newPassword) || !/[0-9]/.test(newPassword)) {
    return res.status(400).json({ error: 'Password must be at least 8 chars, contain uppercase and digit' });
  }

  // 6. 哈希令牌并查询
  const tokenHash = crypto.createHash('sha256').update(token).digest('hex');
  const email = await redis.get(`pwd_reset:${tokenHash}`);

  if (!email) {
    return res.status(400).json({ error: 'Invalid or expired token' });
  }

  // 7. 更新密码（模拟）
  const passwordHash = crypto.createHash('sha256').update(newPassword).digest('hex');
  console.log(`[DB] UPDATE users SET password_hash='${passwordHash}' WHERE email='${email}'`);

  // 8. 删除令牌（一次性使用）
  await redis.del(`pwd_reset:${tokenHash}`);

  res.json({ message: 'Password updated successfully' });
}

#define MAX_CHANNELS 4
#define CHUNK_HEADER_SIZE 8

struct AudioChunk {
    uint32_t sample_rate;
    uint16_t channels;
    uint16_t bits_per_sample;
    uint32_t data_size;
    const uint8_t* data;
};

// Decodes a single interleaved frame into per-channel samples.
// frame_index: 0-based frame number; channel: 0..channels-1
int decode_frame_sample(const AudioChunk* chunk, uint32_t frame_index, uint16_t channel) {
    if (chunk->channels == 0 || chunk->channels > MAX_CHANNELS) return -1;
    if (channel >= chunk->channels) return -1;

    uint32_t bytes_per_sample = chunk->bits_per_sample / 8;
    if (bytes_per_sample == 0 || bytes_per_sample > 4) return -1;

    // frame_size = channels * bytes_per_sample
    uint64_t frame_size = static_cast<uint64_t>(chunk->channels) * bytes_per_sample;
    if (frame_size == 0) return -1;

    uint64_t offset = static_cast<uint64_t>(frame_index) * frame_size + channel * bytes_per_sample;
    if (offset + bytes_per_sample > chunk->data_size) return -1;

    int32_t sample = 0;
    memcpy(&sample, chunk->data + offset, bytes_per_sample);
    return static_cast<int>(sample);
}

// Parses a raw audio container and extracts chunk references.
// Returns 0 on success, -1 on malformed header.
int parse_audio_container(const uint8_t* buffer, size_t buffer_len, AudioChunk* out) {
    if (buffer_len < CHUNK_HEADER_SIZE) return -1;

    uint32_t sample_rate;
    uint16_t channels, bits;
    uint32_t data_size;
    memcpy(&sample_rate, buffer, 4);
    memcpy(&channels, buffer + 4, 2);
    memcpy(&bits, buffer + 6, 2);

    if (channels == 0 || channels > MAX_CHANNELS) return -1;
    if (bits % 8 != 0 || bits > 32) return -1;

    out->sample_rate = sample_rate;
    out->channels = channels;
    out->bits_per_sample = bits;

    // Data section follows the 8-byte header; length is inferred from buffer_len.
    out->data = buffer + CHUNK_HEADER_SIZE;
    out->data_size = static_cast<uint32_t>(buffer_len - CHUNK_HEADER_SIZE);

    // Note: data_size is truncated to 32-bit; buffer_len may exceed 4GB on 64-bit.
    return 0;
}

// Public entry point: decodes a specific frame from a raw buffer.
int get_frame_sample(const uint8_t* container, size_t len, uint32_t frame_idx, uint16_t ch) {
    AudioChunk chunk;
    if (parse_audio_container(container, len, &chunk) != 0) return -1;
    return decode_frame_sample(&chunk, frame_idx, ch);
}

static int copy_at_offset(Buffer *dst, const char *src, int offset, int src_len) {
    if (offset < 0 || offset > dst->len) {
        return -1;
    }
    /* 缺少剩余空间检查：offset+src_len可能超过dst->len */
    memcpy(dst->data + offset, src, src_len);
    return 0;
}

/* 外部接口：合并两个缓冲区 */
int merge_buffers(Buffer *dst, const Buffer *src1, const Buffer *src2) {
    if (!dst || !src1 || !src2) {
        return -1;
    }
    int total = src1->len + src2->len;
    if (total > MAX_BUF) {
        return -1;
    }

    char *tmp = (char *)malloc(total);
    if (!tmp) {
        return -1;
    }
    memcpy(tmp, src1->data, src1->len);
    memcpy(tmp + src1->len, src2->data, src2->len);

    /* 先拷贝到临时缓冲区，安全 */
    copy_at_offset(dst, tmp, 0, total);

    free(tmp);
    return 0;
}

int main() {
    Buffer a = {NULL, 0}, b = {NULL, 0}, out = {NULL, 0};
    char buf1[64] = "hello";
    char buf2[64] = " world!";

    a.data = buf1; a.len = strlen(buf1);
    b.data = buf2; b.len = strlen(buf2);

    }

        if (m_data) {
            free(m_data);
            m_data = nullptr;
        }
    }

    // 重建备份：重新读取文件内容
    bool reload() {
        if (m_data) {
            free(m_data);  // line 21: 释放旧数据
            m_data = nullptr;
        }
        load_file();
        return m_data != nullptr;
    }

    size_t size() const { return m_size; }

private:
    void load_file() {
        int fd = open(m_path.c_str(), O_RDONLY);
        if (fd < 0) return;

        struct stat st;
        if (fstat(fd, &st) != 0 || st.st_size <= 0) {
            close(fd);
            return;
        }

        m_size = static_cast<size_t>(st.st_size);
        m_data = static_cast<char*>(malloc(m_size + 1));
        if (!m_data) {
            close(fd);
            return;
        }

        ssize_t bytes = read(fd, m_data, m_size);
        close(fd);
        if (bytes != static_cast<ssize_t>(m_size)) {
            free(m_data);  // line 45: 读取失败，释放已分配内存
            m_data = nullptr;
            m_size = 0;
            return;
        }
        m_data[m_size] = '\0';
    }

    std::string m_path;
    char* m_data;
    size_t m_size;
};
