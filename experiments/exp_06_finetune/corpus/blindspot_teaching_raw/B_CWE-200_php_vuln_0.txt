<?php

namespace App\Http\Controllers\Export;

use App\Models\Product;
use Illuminate\Http\Request;
use Illuminate\Routing\Controller;
use Illuminate\Support\Facades\Response;

class ProductExportController extends Controller
{
    // GET /api/export/products?format=csv&debug=1
    public function export(Request $request)
    {
        $format = $request->query('format', 'json');
        $debug  = $request->query('debug', '0');

        $rows = Product::all()->map(function (Product $p) {
            return [
                'sku'              => $p->sku,
                'name'             => $p->name,
                'retail_price'     => $p->retail_price,
                'supplier_cost'    => $p->supplier_cost,
                'internal_margin'  => $p->internal_margin,
                'supplier_contact' => $p->supplier_contact,
                'internal_note'    => $p->internal_note,
            ];
        });

        if ($debug === '1') {
            return Response::json([
                'rows'        => $rows,
                'db_host'     => config('database.connections.mysql.host'),
                'db_database' => config('database.connections.mysql.database'),
                'app_env'     => config('app.env'),
                'app_version' => app()->version(),
            ]);
        }

        if ($format === 'csv') {
            $csv = implode(',', array_keys($rows->first()->toArray())) . "\n";
            foreach ($rows as $r) {
                $csv .= implode(',', $r->toArray()) . "\n";
            }
            return Response::make($csv, 200, ['Content-Type' => 'text/csv']);
        }

        return Response::json(['rows' => $rows]);
    }
}
