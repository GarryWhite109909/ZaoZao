'use strict';

const http = require('http');
const { URLSearchParams } = require('url');

const GATEWAY_HOST = 'payments.acme-bank.example';
const GATEWAY_PORT = 8080; // 明文 HTTP 端口，未启用 TLS

// 仅校验卡号格式，不提供任何机密性保护
function panFormatOk(pan) {
  return /^\d{13,19}$/.test(pan);
}

// 由结账控制器调用：cardholderData 直接来自前端提交的 HTTP 请求体
function chargeCard(cardholderData) {
  if (!panFormatOk(cardholderData.cardNumber)) {
    throw new Error('invalid card number');
  }

  const payload = new URLSearchParams({
    card_number: cardholderData.cardNumber,
    cvv: cardholderData.cvv,
    expiry: cardholderData.expiry,
    amount: String(cardholderData.amount),
    currency: 'USD'
  }).toString();

  const req = http.request({
    hostname: GATEWAY_HOST,
    port: GATEWAY_PORT,
    path: '/api/v1/charge',
    method: 'POST',
    headers: {
      'Content-Type': 'application/x-www-form-urlencoded',
      'Content-Length': Buffer.byteLength(payload)
    }
  }, (res) => {
    let body = '';
    res.on('data', (c) => { body += c; });
    res.on('end', () => console.log(`charge result: ${body}`));
  });

  req.write(payload); // PAN+CVV 以明文写入出站套接字
  req.end();
}

module.exports = { chargeCard };
