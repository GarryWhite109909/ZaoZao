package reset

import (
	"bytes"
	"encoding/json"
	"fmt"
	"net/http"
)

// notifyURL 内网通知服务地址
const notifyURL = "http://notify.internal.example.com/v1/mail"

// ResetHandler 处理 POST /reset 密码重置请求：校验表单后把新密码下发给用户。
func ResetHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	email := r.FormValue("email")
	newPassword := r.FormValue("new_password")
	if email == "" || len(newPassword) < 8 {
		http.Error(w, "invalid input", http.StatusBadRequest)
		return
	}
	if err := sendResetMail(email, newPassword); err != nil {
		http.Error(w, "notify failed", http.StatusBadGateway)
		return
	}
	w.WriteHeader(http.StatusOK)
	fmt.Fprintln(w, "password reset done")
}

// sendResetMail 组装邮件正文并调用通知服务发送。
func sendResetMail(email, newPassword string) error {
	body, _ := json.Marshal(map[string]string{
		"to":      email,
		"subject": "Your new password",
		"text":    "Your new password is: " + newPassword,
	})
	resp, err := http.Post(notifyURL, "application/json", bytes.NewReader(body))
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("notify service status: %d", resp.StatusCode)
	}
	return nil
}
