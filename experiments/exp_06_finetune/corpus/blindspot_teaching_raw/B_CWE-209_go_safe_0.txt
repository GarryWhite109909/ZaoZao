```code
package userapi

import (
	"database/sql"
	"encoding/json"
	"errors"
	"log"
	"net/http"

	_ "github.com/go-sql-driver/mysql"
)

// UserService 提供用户资料查询能力。
type UserService struct {
	db     *sql.DB
	logger *log.Logger
}

// NewUserService 注入数据库句柄与服务端日志器。
func NewUserService(db *sql.DB, logger *log.Logger) *UserService {
	return &UserService{db: db, logger: logger}
}

// HandleGetUser 处理 GET /api/user?user_id=xxx。
func (s *UserService) HandleGetUser(w http.ResponseWriter, r *http.Request) {
	userID := r.URL.Query().Get("user_id")

	var name, email string
	err := s.db.QueryRow("SELECT name, email FROM users WHERE id = ?", userID).Scan(&name, &email)
	if errors.Is(err, sql.ErrNoRows) {
		w.WriteHeader(http.StatusNotFound)
		_ = json.NewEncoder(w).Encode(map[string]string{"error": "user not found"})
		return
	}
	if err != nil {
		// 详细错误（含 SQL 片段、表名、驱动与连接细节）只进服务端日志
		s.logger.Printf("query user failed: user_id=%q err=%v", userID, err)
		// 响应体只由常量构成，err 不进入任何写出调用
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusInternalServerError)
		_ = json.NewEncoder(w).Encode(map[string]string{"error": "internal server error"})
		return
	}

	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(map[string]string{"name": name, "email": email})
}
```
