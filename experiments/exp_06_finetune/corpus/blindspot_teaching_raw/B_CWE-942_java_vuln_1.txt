package com.example.spabackend.api;

import java.util.LinkedHashMap;
import java.util.Map;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

/**
 * SPA 后端账户接口。前端站点与 API 不同源，开发图省事直接回显请求 Origin 放行跨域。
 */
@RestController
@RequestMapping("/api/account")
public class AccountController {

    @GetMapping("/profile")
    public Map<String, Object> profile(HttpServletRequest request, HttpServletResponse response) {
        echoOrigin(request, response);
        HttpSession session = request.getSession(false);
        Map<String, Object> data = new LinkedHashMap<>();
        if (session != null && session.getAttribute("loginUser") != null) {
            data.put("username", session.getAttribute("loginUser"));
            data.put("email", session.getAttribute("email"));
            data.put("balance", session.getAttribute("balance"));
        }
        return data;
    }

    @GetMapping("/orders")
    public Map<String, Object> orders(HttpServletRequest request, HttpServletResponse response) {
        echoOrigin(request, response);
        Map<String, Object> data = new LinkedHashMap<>();
        data.put("latestOrder", "ORD-20240517-0093");
        data.put("amount", 1299.00);
        return data;
    }

    private void echoOrigin(HttpServletRequest request, HttpServletResponse response) {
        String origin = request.getHeader("Origin");
        if (origin != null && !origin.isEmpty()) {
            response.setHeader("Access-Control-Allow-Origin", origin);
            response.setHeader("Access-Control-Allow-Credentials", "true");
        }
    }
}

1. 第 1 步（本类核心特征）：第 44 行把第 42 行读取的请求头 Origin 原样写入 Access-Control-Allow-Origin，第 45 行再叠加 Access-Control-Allow-Credentials: true——这正是 CWE-942 Permissive Cross-domain Policy 的典型形态“反射 Origin + 凭据放行”：任意源都能携带 Cookie 跨域读取受保护响应。必须明确：该缺陷属于跨域策略（CORS）配置过宽，不是命令注入、SQL 注入、XSS 或越权逻辑问题，问题出在响应头的跨域授权语义上。

2. 第 2 步（输入点枚举与可达性）：本文件外部输入点共两类。①第 42 行 request.getHeader('Origin')：由请求方浏览器/攻击页面完全控制，经第 21 行（/profile，第 19 行映射）与第 34 行（/orders，第 32 行映射）两处无条件调用 echoOrigin 直达第 44 行 sink，中间无任何转换；②隐式输入为浏览器自动附带的会话 Cookie（第 22 行 getSession 读取），由于第 44-45 行声明允许任意源携带凭据，跨域请求同样会带上它，成为敏感数据的取数凭据。文件内无其他请求参数参与 CORS 头生成，两条路由到 sink 的可达性均为 100%。

3. 第 3 步（防御类型/位置/覆盖面核查）：对两条 source→sink 流逐一核查——唯一条件判断在第 43 行 origin != null && !origin.isEmpty()，位置虽在 sink 之前，但只是判空守卫，不构成可信域白名单、正则匹配、后缀/协议校验或任何规范化；全文件不存在域名白名单或过滤逻辑（即便存在黑名单、简单字符串替换类过滤也属可绕过、不算有效防御，而此处连这类过滤都没有）。因此第 21 行与第 34 行两条流的防御覆盖面均为零，任意 Origin（如 https://evil.example）被逐字回显。

4. 第 4 步（暴露链）：攻击者在任意域托管页面执行 fetch('https://api.example.com/api/account/profile', {credentials: 'include'})；GET 属简单请求无需预检，浏览器自动携带受害者会话 Cookie；服务端第 24-27 行把登录用户的用户名/邮箱/余额写入 JSON 响应，第 44 行使 ACAO 与请求 Origin 完全一致、第 45 行允许凭据，浏览器据此放行攻击脚本读取完整响应体——已认证敏感数据被任意源窃取。/orders 同理经第 34 行到达同一 sink，泄露第 36-37 行的订单号与金额。

5. 第 5 步（第二入口/替代通道结论）：存在第二入口——第 32-34 行 /api/account/orders 与 /profile 共享第 41 行的 echoOrigin 到达同一 sink，说明漏洞并非孤立于单一路由；同时核查确认本文件没有 Servlet Filter、HandlerInterceptor 或其他间接设置 CORS 头的替代通道，全部过宽策略均出自第 44-45 行两处 setHeader。据此结论：CWE-942 成立，修复必须同时覆盖第 21、34 两个调用点（或直接收敛 echoOrigin 本身）。
