"""Tag search endpoint for the catalog service.

The legacy implementation validated queries with the stdlib pattern
r'^(\s*\w+\s*)*$' (nested quantifiers): a query such as 'a a a ... a!'
made the matcher backtrack exponentially (ReDoS, CWE-400). This
module keeps the feature but bounds matcher work three ways.
"""
import regex  # third-party 'regex' package: atomic groups + timeouts
from flask import Flask, request, jsonify

app = Flask(__name__)

MAX_QUERY_LEN = 128      # bound 1: cap the subject string length
MATCH_TIMEOUT_S = 0.2    # bound 3: wall-clock budget per match

# bound 2: atomic groups (?>...) never give characters back, so the
# comma-separated token grammar below matches in linear time.
QUERY_RE = regex.compile(r'\s*(?>[\w.-]+)(?>\s*,\s*[\w.-]+)*\s*')


def validate_query(query):
    """Return the parsed tag list, or None if the query is rejected."""
    if len(query) > MAX_QUERY_LEN:
        return None
    try:
        if QUERY_RE.fullmatch(query, timeout=MATCH_TIMEOUT_S) is None:
            return None
    except regex.TimeoutError:
        return None
    return [t for t in (s.strip() for s in query.split(',')) if t]


@app.route('/search')
def search():
    query = request.args.get('q', '')
    tags = validate_query(query)
    if tags is None:
        return jsonify(error='invalid query'), 400
    hits = []
    with open('catalog.txt', encoding='utf-8') as fh:
        for line in fh:
            low = line.lower()
            if all(t.lower() in low for t in tags):
                hits.append(line.rstrip('\n'))
    return jsonify(results=hits[:50])


if __name__ == '__main__':
    app.run()
