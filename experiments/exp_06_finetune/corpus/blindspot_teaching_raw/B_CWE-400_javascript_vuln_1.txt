```code
'use strict';

const express = require('express');
const router = express.Router();
const catalog = require('../services/catalog');

// 搜索词格式白名单：仅允许字母数字词，词间用单个空格或连字符分隔
const TERM_PATTERN = /^([a-zA-Z0-9]+[- ]?)+$/;

// 将查询词编译为高亮正则：多个词合并为交替分支后整体叠加 + 量词
function buildHighlightRegex(query) {
  const words = query.trim().split(/\s+/).filter(Boolean);
  const escaped = words.map(w => w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'));
  return new RegExp('(' + escaped.join('|') + ')+', 'i');
}

router.get('/search', (req, res) => {
  const q = String(req.query.q || '');
  const limit = Math.min(parseInt(req.query.limit, 10) || 20, 100);

  // 输入长度上限与格式校验
  if (q.length > 256) {
    return res.status(400).json({ error: 'query too long' });
  }
  if (!TERM_PATTERN.test(q)) {
    return res.status(400).json({ error: 'invalid query' });
  }

  const highlight = buildHighlightRegex(q);
  const pool = catalog.listAll(); // 全量商品，含卖家自定义名称与简介（用户生成内容）
  const hits = pool.filter(p => highlight.test(p.name) || highlight.test(p.summary));

  res.json({ total: hits.length, items: hits.slice(0, limit) });
});

module.exports = router;
```
