<?php
// app/Http/Resources/UserProfileResource.php

declare(strict_types=1);

namespace App\Http\Resources;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;
use Illuminate\Support\Arr;

/**
 * 用户资料接口序列化层：以字段白名单 + 脱敏掩码结构性阻断 CWE-200 敏感信息泄露。
 */
final class UserProfileResource extends JsonResource
{
    /** 服务端唯一允许离开响应边界的字段集合（allowlist，而非黑名单） */
    private const PUBLIC_FIELDS = ['id', 'username', 'avatar_url', 'email', 'created_at'];

    public function toArray(Request $request): array
    {
        // 外部输入点：GET /api/users/{id}/profile?fields=... 客户端可选字段列表
        $requested = array_map('trim', explode(',', (string) $request->query('fields', '')));

        // 防御核心：客户端所选字段必须先与服务端白名单求交集，未列入白名单的键一律丢弃
        $fields = array_values(array_intersect($requested, self::PUBLIC_FIELDS));
        if ($fields === []) {
            $fields = self::PUBLIC_FIELDS;
        }

        // 模型原始属性含 password_hash/remember_token/api_key，但只按白名单取值
        $data = Arr::only($this->resource->getAttributes(), $fields);

        // 半敏感字段 email 二次脱敏：掩码后才允许进入响应
        if (isset($data['email']) && is_string($data['email'])) {
            $data['email'] = $this->maskEmail($data['email']);
        }

        return $data;
    }

    private function maskEmail(string $email): string
    {
        [$local, $domain] = explode('@', $email, 2);
        $keep = min(2, strlen($local));

        return substr($local, 0, $keep) . str_repeat('*', max(strlen($local) - $keep, 1)) . '@' . $domain;
    }
}
