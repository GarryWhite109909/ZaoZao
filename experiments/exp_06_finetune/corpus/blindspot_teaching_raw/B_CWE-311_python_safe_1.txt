"""PCI-DSS aligned client for the external payment gateway.

Cardholder data (PAN) must never cross a trust boundary in plaintext
(CWE-311). This module hard-fails any non-TLS endpoint and enforces
TLS 1.2+ with certificate verification before any card byte moves.
"""

import ssl
from urllib.parse import urlparse

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.ssl_ import create_urllib3_context


class _TLS12Adapter(HTTPAdapter):
    """Force TLS 1.2+ with full certificate-chain verification."""

    def init_poolmanager(self, *args, **kwargs):
        ctx = create_urllib3_context()
        ctx.minimum_version = ssl.TLSVersion.TLSv1_2
        kwargs["ssl_context"] = ctx
        return super().init_poolmanager(*args, **kwargs)


class PaymentGatewayClient:
    def __init__(self, base_url: str, api_key: str) -> None:
        scheme = urlparse(base_url).scheme.lower()
        if scheme != "https":
            raise ValueError("cardholder data requires TLS: http:// endpoint rejected")
        self._endpoint = base_url.rstrip("/") + "/v1/charges"
        self._session = requests.Session()
        self._session.mount("https://", _TLS12Adapter())
        self._session.headers["Authorization"] = "Bearer " + api_key

    def charge(self, card_number: str, expiry: str, amount_cents: int) -> dict:
        payload = {"pan": card_number, "exp": expiry, "amount": amount_cents}
        response = self._session.post(
            self._endpoint, json=payload, timeout=10, verify=True
        )
        response.raise_for_status()
        return {"status": response.status_code, "auth_id": response.json().get("id")}
