"""Password reset service: reset tokens travel over enforced-TLS channels only."""
import secrets
import smtplib
import ssl
from email.message import EmailMessage

from argon2 import PasswordHasher

RESET_BASE_URL = "https://accounts.example.com/reset"   # https only, never http
SMTP_HOST, SMTP_PORT = "smtp.internal.example.com", 465

_hasher = PasswordHasher()            # Argon2id, per-hash random salt
_tls = ssl.create_default_context()   # CERT_REQUIRED + hostname verification
_digests: dict[str, str] = {}         # email -> argon2 digest (DB stand-in)


def save_new_password(user_email: str, new_password: str) -> None:
    """Persist only the Argon2id digest; cleartext never reaches storage."""
    _digests[user_email] = _hasher.hash(new_password)


def deliver_reset_email(user_email: str, token: str) -> None:
    msg = EmailMessage()
    msg["From"] = "noreply@example.com"
    msg["To"] = user_email
    msg["Subject"] = "Password reset"
    msg.set_content(f"Reset link: {RESET_BASE_URL}?token={token}")
    # Sink: the token crosses the trust boundary here. SMTP_SSL pins TLS from
    # the first byte; _tls rejects plaintext fallback and untrusted certificates.
    with smtplib.SMTP_SSL(SMTP_HOST, SMTP_PORT, context=_tls) as smtp:
        smtp.send_message(msg)


def handle_reset_request(user_email: str, new_password: str) -> None:
    """Entry point for POST /account/reset; OTP verification happens upstream."""
    token = secrets.token_urlsafe(32)  # 256-bit CSPRNG token
    save_new_password(user_email, new_password)
    deliver_reset_email(user_email, token)
