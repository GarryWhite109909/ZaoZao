```code
'use strict';
const express = require('express');
const multer = require('multer');
const fs = require('fs');

const app = express();
const upload = multer({ dest: '/tmp/uploads', limits: { fileSize: 10 * 1024 * 1024 } });

// 服务端日志通道:仅写 stderr,与 HTTP 响应物理隔离
function logError(scope, meta) {
  process.stderr.write(`[error] ${scope} ${JSON.stringify(meta)}\n`);
}

// POST /api/files/parse —— 接收上传文件并统计非空行数
app.post('/api/files/parse', upload.single('file'), (req, res) => {
  if (!req.file) {
    return res.status(400).json({ error: 'no file provided' });
  }
  try {
    const content = fs.readFileSync(req.file.path, 'utf8');
    const rows = content.split(/\r?\n/).filter(Boolean).length;
    fs.unlinkSync(req.file.path);
    return res.json({ rows });
  } catch (err) {
    // 完整异常细节(消息/堆栈/临时路径)只进服务端日志
    logError('file-parse', { message: err.message, stack: err.stack, tmp: req.file.path });
    // 客户端仅收到固定通用文案,不回显任何异常内容
    return res.status(500).json({ error: 'internal server error' });
  }
});

// 兜底错误中间件:接住 multer 等中间件抛出的异常,同样只回通用文案
app.use((err, req, res, next) => {
  logError('middleware', { message: err.message, stack: err.stack });
  return res.status(500).json({ error: 'internal server error' });
});

module.exports = app;
```
