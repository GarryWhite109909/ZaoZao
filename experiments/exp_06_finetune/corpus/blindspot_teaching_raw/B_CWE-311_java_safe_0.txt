package com.example.account.reset;

import java.net.URI;
import java.security.SecureRandom;
import java.time.Instant;
import java.util.Base64;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

@Service
public class PasswordResetService {
    private final UserRepository userRepository;
    private final BCryptPasswordEncoder hasher = new BCryptPasswordEncoder(12);
    private final SecureRandom rng = new SecureRandom();
    private final Map<String, String> pendingTokenHashes = new ConcurrentHashMap<>();
    private final Map<String, Instant> pendingExpiry = new ConcurrentHashMap<>();

    public PasswordResetService(UserRepository userRepository) {
        this.userRepository = userRepository;
    }

    public String createResetLink(String email, String baseUrl) {
        URI base = URI.create(baseUrl);
        if (!"https".equalsIgnoreCase(base.getScheme())) {
            throw new IllegalArgumentException("reset link must be served over TLS");
        }
        byte[] raw = new byte[32];
        rng.nextBytes(raw);
        String token = Base64.getUrlEncoder().withoutPadding().encodeToString(raw);
        pendingTokenHashes.put(email, hasher.encode(token));
        pendingExpiry.put(email, Instant.now().plusSeconds(900));
        return base + "/reset/confirm?token=" + token;
    }

    public boolean applyReset(String email, String token, String newPassword) {
        Instant expiry = pendingExpiry.get(email);
        String storedHash = pendingTokenHashes.get(email);
        if (expiry == null || Instant.now().isAfter(expiry) || !hasher.matches(token, storedHash)) {
            return false;
        }
        pendingTokenHashes.remove(email);
        pendingExpiry.remove(email);
        String passwordHash = hasher.encode(newPassword);
        return userRepository.updatePasswordHash(email, passwordHash);
    }
}
