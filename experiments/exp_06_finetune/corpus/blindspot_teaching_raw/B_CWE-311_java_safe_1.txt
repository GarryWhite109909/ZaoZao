package com.acme.pay.gateway;

import javax.net.ssl.HttpsURLConnection;
import javax.net.ssl.SSLContext;
import javax.net.ssl.SSLSocketFactory;
import java.io.IOException;
import java.io.OutputStream;
import java.net.URL;
import java.nio.charset.StandardCharsets;
import java.security.SecureRandom;

/** Submits cardholder data to the external payment processor; PAN must never transit in plaintext (CWE-311). */
public final class PaymentGatewayClient {

    private static final String ENDPOINT = "https://gateway.acmepay.example/v1/charges";

    private final SSLSocketFactory tlsFactory;

    public PaymentGatewayClient() throws Exception {
        SSLContext ctx = SSLContext.getInstance("TLSv1.3");
        ctx.init(null, null, new SecureRandom());
        this.tlsFactory = ctx.getSocketFactory();
    }

    public String charge(String pan, String expiry, int amountCents) throws IOException {
        URL url = new URL(ENDPOINT);
        if (!"https".equalsIgnoreCase(url.getProtocol())) {
            throw new SecurityException("cardholder data requires TLS transport");
        }
        HttpsURLConnection conn = (HttpsURLConnection) url.openConnection();
        conn.setSSLSocketFactory(tlsFactory);
        conn.setHostnameVerifier(HttpsURLConnection.getDefaultHostnameVerifier());
        conn.setInstanceFollowRedirects(false);
        conn.setRequestMethod("POST");
        conn.setDoOutput(true);
        conn.setRequestProperty("Content-Type", "application/json");
        String body = "{\"pan\":\"" + esc(pan) + "\",\"expiry\":\"" + esc(expiry)
                + "\",\"amount\":" + amountCents + "}";
        try (OutputStream out = conn.getOutputStream()) {
            out.write(body.getBytes(StandardCharsets.UTF_8));
        }
        int status = conn.getResponseCode();
        return (status >= 200 && status < 300) ? "approved" : "declined";
    }

    private static String esc(String v) {
        return v.replace("\\", "\\\\").replace("\"", "\\\"");
    }
}

2. **输入点枚举与可达性**：外部输入点仅第 25 行 `charge(String pan, String expiry, int amountCents)`，其中 pan/expiry 为持卡人敏感数据，amountCents 为非敏感整型。可达性确认：pan/expiry 在第 37-38 行拼入请求体 body，第 39-40 行经 `conn.getOutputStream()` 写出，source 到 sink（网络套接字）链路完整可达，因此信道加密必须逐一验证。

3. **主数据流防御验证（类型/位置/覆盖面）**：唯一信道由第 30 行 `url.openConnection()` 创建，URL 来自第 15 行私有常量 `ENDPOINT`，硬编码为 `https://`。第 27-29 行在打开连接前做协议结构校验：`getProtocol()` 非 https 即抛 SecurityException，fail-closed——这是对 URL scheme 的结构性白名单，不是可绕过的字符串替换/正则过滤。第 31 行将连接绑定到第 20 行创建的仅 TLSv1.3 上下文：加密类型正确（传输层强加密）、位置正确（施加于连接对象、输出流打开之前）、覆盖完整（第 40 行写出的全部字节含 pan/expiry 均走该加密流，无明文出网路径）。

4. **覆盖面补强**：第 32 行显式保留默认 HostnameVerifier（证书身份校验，未被宽松的 `return true` 实现覆盖），TLS 同时提供机密性与网关身份认证，防中间人；第 33 行 `setInstanceFollowRedirects(false)` 关闭自动重定向，封死 https→http 降级重定向这一 CWE-311 常见旁路。此外 pan 仅存在于内存与加密信道，无日志输出、无明文持久化，存储侧无缺口。

5. **第二入口/替代通道检查**：类仅暴露一个公开方法 charge；`ENDPOINT` 为 private static final，无 setter、无配置注入、无反射调用点可将 URL 换成 http；类内无第二个连接构造点、无失败重试降级逻辑。替代通道检查为阴性，支撑最终结论：无 CWE-311 漏洞。
