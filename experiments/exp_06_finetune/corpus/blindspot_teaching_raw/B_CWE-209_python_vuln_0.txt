"""Product search API for the storefront service.

Queries the catalog database and returns matching products as JSON.
"""
import os

import psycopg2
from flask import Flask, jsonify, request

app = Flask(__name__)

DB_CONFIG = {
    "host": os.environ.get("DB_HOST", "10.0.2.15"),
    "port": 5432,
    "dbname": "shop",
    "user": "app_rw",
    "password": os.environ.get("DB_PASSWORD", ""),
}


@app.route("/api/products/search", methods=["GET"])
def search_products():
    keyword = request.args.get("q", "").strip()
    category = request.args.get("category", "all")
    conn = None
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        cur = conn.cursor()
        sql = "SELECT id, name, price FROM products WHERE name ILIKE %s AND category = %s"
        cur.execute(sql, ("%" + keyword + "%", category))
        rows = cur.fetchall()
        cur.close()
        return jsonify({"items": [dict(zip(("id", "name", "price"), r)) for r in rows]})
    except psycopg2.Error as e:
        return jsonify({"status": "error", "message": str(e)}), 500
    finally:
        if conn is not None:
            conn.close()


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)

4. **逐流防御核查**：流 1——第 23 行 `.strip()` 仅去除空白，属输入规范化而非泄露防御；第 30 行参数绑定是针对 SQL 注入的有效防御，但防御类别与 CWE-209 不重叠，对错误消息内容零覆盖。流 2——连接配置来自第 12-18 行环境变量，无任何脱敏。两条流共用的第 34-35 行 except 块没有消息白名单、错误码映射或细节剥离，也没有把 `str(e)` 改为仅写服务端日志——即对错误信息泄露全程不存在有效防御。
