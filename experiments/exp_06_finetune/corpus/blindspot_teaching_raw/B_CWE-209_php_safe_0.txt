<?php

declare(strict_types=1);

namespace App\Api;

use PDO;
use Psr\Http\Message\ResponseInterface as Response;
use Psr\Http\Message\ServerRequestInterface as Request;
use Psr\Log\LoggerInterface;
use Throwable;

/** GET /api/user?email=... ；数据库故障只回通用文案，细节仅写服务端日志 */
final class UserLookupAction
{
    public function __construct(
        private readonly PDO $db, // 容器装配时已设 ERRMODE_EXCEPTION
        private readonly LoggerInterface $logger
    ) {
    }

    public function __invoke(Request $request, Response $response): Response
    {
        $email = (string)($request->getQueryParams()['email'] ?? '');
        try {
            $stmt = $this->db->prepare(
                'SELECT id, email, created_at FROM users WHERE email = :email'
            );
            $stmt->execute(['email' => $email]);
            $row = $stmt->fetch(PDO::FETCH_ASSOC);
            $response->getBody()->write(
                (string)json_encode(['code' => 0, 'data' => $row === false ? null : $row])
            );
            return $response->withStatus(200);
        } catch (Throwable $e) {
            // 驱动错误（含 SQL、连接细节、堆栈）只落服务端日志，不进响应体
            $this->logger->error('user lookup failed', [
                'exception' => get_class($e) . ': ' . $e->getMessage(),
                'trace' => $e->getTraceAsString(),
            ]);
            $response->getBody()->write(
                (string)json_encode(['code' => 500, 'msg' => '系统繁忙，请稍后重试'])
            );
            return $response->withStatus(500);
        }
    }
}
