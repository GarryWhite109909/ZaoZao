'use strict';
// services/exportService.js — 订单导出接口：仅导出公开字段，内部成本/毛利不外泄
const express = require('express');
const router = express.Router();

// 模拟数据层：内部记录含成本价、内部毛利率、供应商联系人等敏感列
const db = {
  findOrdersByTenant: (tenantId) => [
    {
      id: 'ORD-1001',
      product: 'Industrial Widget',
      listPrice: 199.0,
      costPrice: 62.5,
      internalMargin: 0.686,
      supplierContact: 'chen@partner.example',
    },
  ],
};

// 导出字段白名单：仅公开目录价，成本/毛利/供应商信息被显式排除
const EXPORTABLE_FIELDS = ['id', 'product', 'listPrice'];

function toPublicRow(row) {
  const out = {};
  for (const key of EXPORTABLE_FIELDS) {
    if (Object.prototype.hasOwnProperty.call(row, key)) {
      out[key] = row[key];
    }
  }
  return out;
}

router.get('/api/export/orders', (req, res) => {
  const tenantId = String(req.query.tenantId || '');
  if (!tenantId) {
    return res.status(400).json({ error: 'tenantId is required' });
  }
  const rows = db.findOrdersByTenant(tenantId);
  const payload = rows.map(toPublicRow);
  res.setHeader('Content-Disposition', 'attachment; filename="orders.json"');
  return res.status(200).json({ exportedAt: new Date().toISOString(), rows: payload });
});

module.exports = router;
