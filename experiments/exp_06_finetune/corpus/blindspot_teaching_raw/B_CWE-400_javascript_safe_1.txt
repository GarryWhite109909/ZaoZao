```code
// modules/catalog/searchFilter.js
// 商品目录搜索过滤: 校验用户查询词后检索内存目录
'use strict';

const express = require('express');
const router = express.Router();

const MAX_QUERY_LENGTH = 128; // 防御层1: 输入长度上限, 约束进入正则引擎的规模
// 防御层2: 线性时间正则。反例 /^(\w+[ ]*)+$/ 的外层与内层量词可消费相同字符,
// 尾部失配时按 2^n 指数回溯; 本模式强制每次外层迭代以空格开头, 字符类互斥, 无二义分解
const SAFE_TERM_RE = /^[a-zA-Z0-9_]+(?:[ ]+[a-zA-Z0-9_]+)*$/;

const CATALOG = [
  'steel water bottle',
  'bamboo toothbrush',
  'solar power bank',
  'ceramic coffee mug',
];

router.get('/search', function (req, res) {
  const raw = req.query.q; // 唯一外部输入点(source)

  if (typeof raw !== 'string' || raw.length === 0) {
    return res.status(400).json({ error: 'query parameter q is required' });
  }
  if (raw.length > MAX_QUERY_LENGTH) { // 在 sink 之前拒绝超长输入
    return res.status(413).json({ error: 'query too long' });
  }

  if (!SAFE_TERM_RE.test(raw)) { // sink: 正则引擎回溯消耗 CPU
    return res.status(400).json({ error: 'invalid search syntax' });
  }

  const terms = raw.toLowerCase().split(/ +/);
  const hits = CATALOG.filter((name) =>
    terms.some((term) => name.includes(term))
  );
  return res.json({ count: hits.length, hits });
});

module.exports = router;
```
