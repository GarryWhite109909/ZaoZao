<?php

namespace App\Modules\FileProcessing;

use App\Core\Database;
use App\Core\Response;
use App\Exception\StorageException;

class DocumentImportHandler
{
    private Database $db;

    public function __construct(Database $db)
    {
        $this->db = $db;
    }

    public function handle(): void
    {
        try {
            $file = $_FILES['attachment'] ?? null;
            if ($file === null || $file['error'] !== UPLOAD_ERR_OK) {
                Response::json(['status' => 'error', 'message' => 'no file uploaded'], 400);
                return;
            }

            $targetPath = __DIR__ . '/../../storage/uploads/' . basename($file['name']);
            if (!move_uploaded_file($file['tmp_name'], $targetPath)) {
                throw new StorageException('failed to persist upload to ' . $targetPath);
            }

            $stmt = $this->db->pdo->prepare('INSERT INTO documents (name, path, size) VALUES (:n, :p, :s)');
            $stmt->execute(['n' => $file['name'], 'p' => $targetPath, 's' => $file['size']]);
            Response::json(['status' => 'ok', 'id' => $this->db->pdo->lastInsertId()], 200);
        } catch (\PDOException $e) {
            Response::json([
                'status' => 'error',
                'message' => $e->getMessage(),
                'sql' => $e->errorInfo[2] ?? '',
                'trace' => $e->getTraceAsString(),
            ], 500);
        } catch (\Exception $e) {
            Response::json([
                'status' => 'error',
                'message' => $e->getMessage(),
                'trace' => $e->getTraceAsString(),
            ], 500);
        }
    }
}
