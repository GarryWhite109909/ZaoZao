package main

import (
	"database/sql"
	"encoding/json"
	"fmt"
	"log"
	"net/http"

	_ "github.com/go-sql-driver/mysql"
)

var db *sql.DB

func main() {
	var err error
	db, err = sql.Open("mysql", "billing_rw:S3cret!@tcp(10.0.8.12:3306)/billing")
	if err != nil {
		log.Fatal(err)
	}
	http.HandleFunc("/api/invoices", invoiceHandler)
	log.Fatal(http.ListenAndServe(":8080", nil))
}

// GET /api/invoices?customer=acme
func invoiceHandler(w http.ResponseWriter, r *http.Request) {
	customer := r.URL.Query().Get("customer")
	query := "SELECT invoice_id, amount, bank_account FROM invoices WHERE customer_name = ?"
	rows, err := db.Query(query, customer)
	if err != nil {
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusInternalServerError)
		detail := fmt.Sprintf("db error: %v; sql: %s", err, query)
		json.NewEncoder(w).Encode(map[string]string{"error": detail})
		return
	}
	defer rows.Close()
	invoices := make([]map[string]interface{}, 0)
	for rows.Next() {
		var id string
		var amount float64
		if err := rows.Scan(&id, &amount); err != nil {
			json.NewEncoder(w).Encode(map[string]string{"error": err.Error()})
			return
		}
		invoices = append(invoices, map[string]interface{}{"invoice_id": id, "amount": amount})
	}
	json.NewEncoder(w).Encode(invoices)
}
