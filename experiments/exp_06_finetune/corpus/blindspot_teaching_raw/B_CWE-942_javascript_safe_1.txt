```code
'use strict';

const express = require('express');
const app = express();

// SPA 前端部署的精确 Origin 白名单：协议 + 域名 + 端口整体匹配
const ALLOWED_ORIGINS = new Set([
  'https://app.example.com',
  'https://staging.example.com',
]);

// 统一 CORS 中间件：替代「把请求 Origin 原样回显」的图省事写法
app.use('/api', (req, res, next) => {
  const origin = req.headers.origin;

  // 无论命中与否都声明 Vary，让共享缓存以 Origin 作为缓存键
  res.setHeader('Vary', 'Origin');

  if (origin && ALLOWED_ORIGINS.has(origin)) {
    // 仅对白名单命中的 Origin 精确回显，绝不使用通配符 '*'
    res.setHeader('Access-Control-Allow-Origin', origin);
    // 凭据模式：允许携带 Cookie，但仅在白名单分支内设置，严禁与 '*' 组合
    res.setHeader('Access-Control-Allow-Credentials', 'true');
  }

  if (req.method === 'OPTIONS') {
    res.setHeader('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization');
    res.setHeader('Access-Control-Max-Age', '600');
    return res.sendStatus(204);
  }

  next();
});

// 敏感数据端点（上游另有鉴权中间件挂载 req.user）：
// 响应可携带 Cookie 凭据，因此 CORS 策略必须严格
app.get('/api/user/profile', (req, res) => {
  res.json({ id: req.user.id, email: req.user.email });
});

module.exports = app;
```
