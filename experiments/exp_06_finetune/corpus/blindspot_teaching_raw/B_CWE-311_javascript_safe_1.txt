// src/payment/GatewayClient.js - PCI cardholder-data client (TLS enforced)
import axios from 'axios';
import https from 'https';

const PLAINTEXT_PORTS = new Set(['80', '8080', '8000', '3000']);

function assertTlsEndpoint(rawUrl) {
  const url = new URL(rawUrl);
  if (url.protocol !== 'https:') {
    throw new Error(`refusing cardholder data over ${url.protocol}: TLS required`);
  }
  if (PLAINTEXT_PORTS.has(url.port)) {
    throw new Error(`plaintext port ${url.port} forbidden for PAN/CVV traffic`);
  }
  return url;
}

export class GatewayClient {
  constructor(endpointUrl, apiKey) {
    this.endpoint = assertTlsEndpoint(endpointUrl);
    this.apiKey = apiKey;
    this.http = axios.create({
      baseURL: this.endpoint.origin,
      httpsAgent: new https.Agent({ rejectUnauthorized: true, minVersion: 'TLSv1.2' }),
      timeout: 10000,
      maxRedirects: 0,
    });
  }

  async charge(card) {
    const payload = {
      pan: String(card.number).replace(/\D/g, ''),
      expiry: `${card.expMonth}/${card.expYear}`,
      cvv: String(card.cvv),
      amountMinor: card.amountMinor,
      currency: card.currency,
    };
    const res = await this.http.post('/v1/charges', payload, {
      headers: { Authorization: `Bearer ${this.apiKey}` },
    });
    return { transactionId: res.data.id, status: res.data.status };
  }
}

4. 第 4 步（防御位置与覆盖面）：校验位于第 20 行构造函数内，即任何请求发出之前，不存在“先配置后绕过”的窗口；第 22-27 行创建的 axios 实例是该类唯一网络出口对象。第 26 行 `maxRedirects: 0` 封堵了重定向降级到 http 的经典绕过路径；第 38 行请求路径为硬编码相对路径 '/v1/charges'，调用方无法按次覆盖主机。因此 PAN、CVV、有效期（第 38 行请求体）与 apiKey（第 39 行请求头）全部敏感字段均被同一 TLS 信道覆盖，无遗漏字段。
