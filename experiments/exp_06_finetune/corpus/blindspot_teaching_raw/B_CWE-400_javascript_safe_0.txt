```javascript
'use strict';

const express = require('express');
const RE2 = require('re2');

const router = express.Router();

const MAX_PATTERN_LENGTH = 128;
const MAX_SUBJECT_LENGTH = 4096;

router.post('/regex/validate', (req, res) => {
  const { pattern, subject } = req.body || {};

  if (typeof pattern !== 'string' || typeof subject !== 'string') {
    return res.status(400).json({ error: 'pattern and subject must be strings' });
  }
  if (pattern.length > MAX_PATTERN_LENGTH) {
    return res.status(400).json({ error: 'pattern exceeds 128 chars' });
  }
  if (subject.length > MAX_SUBJECT_LENGTH) {
    return res.status(400).json({ error: 'subject exceeds 4096 chars' });
  }

  let re;
  try {
    re = new RE2(pattern);
  } catch (err) {
    return res.status(400).json({ error: 'invalid pattern' });
  }

  return res.json({ matched: re.test(subject) });
});

module.exports = router;
```
