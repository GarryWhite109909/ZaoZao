'use strict';

const express = require('express');
const router = express.Router();

// 表单设计器自定义校验规则模块：
// 用户为字段提交正则规则，服务端保存前同步试运行一次以确认规则可用。
const savedRules = new Map();

router.use(express.json());

// 保存并试运行校验规则
router.post('/api/form/validation-rule', function (req, res) {
  const field = req.body.field;
  const pattern = req.body.pattern;   // 用户提交的正则模式
  const sample = req.body.sample;     // 试运行样本

  if (typeof field !== 'string' || !field) {
    return res.status(400).json({ error: 'field required' });
  }
  // 拦截危险关键字，限制长度
  if (typeof pattern !== 'string' || pattern.length > 256 ||
      /eval|Function|require|child_process/.test(pattern)) {
    return res.status(400).json({ error: 'bad pattern' });
  }

  const re = new RegExp(pattern);
  const matched = re.test(sample);    // 保存前试运行
  savedRules.set(field, { pattern, matched });
  return res.json({ field, matched });
});

// 前端「测试规则」按钮的独立试运行接口
router.post('/api/form/validation-rule/test', function (req, res) {
  const re = new RegExp(req.body.pattern);
  return res.json({ matched: re.test(req.body.sample) });
});

module.exports = router;

第 3 步（防御有效性验证，入口 1）：唯一过滤位于第 22-25 行，类型为长度上限（≤256）加关键字黑名单（eval/Function/require/child_process），位置在 RegExp 构造之前。但覆盖面完全错位：黑名单针对的是代码执行关键字，而灾难性回溯只需 `(a+)+$` 这类 7 字符模式——不含任何黑名单词、远小于长度上限，可 100% 绕过；长度上限本身说明开发者只考虑了存储维度而未考虑回溯复杂度维度。第 16 行的 sample 则完全未经过滤直达第 28 行。故入口 1 全程无有效防御。

第 4 步（消耗点与攻击路径）：攻击者向入口 1 提交 pattern=`(a+)+$`、sample=`'a'.repeat(30)+'!'`：第 28 行 re.test 必须枚举 30 个 a 的全部分组方式（约 2^30 次回溯）才能判定不匹配，V8 主线程同步阻塞数十秒以上，样本每增加 10 个 a 耗时约放大 1000 倍。第 27-28 行无超时机制、无 worker 线程隔离、未使用 re2 等线性时间引擎，单个廉价 HTTP 请求即可让整个进程停止响应所有并发请求，构成无界资源消耗。

第 5 步（第二入口与替代通道结论）：第 34-36 行是独立注册的第二条路由，pattern/sample 从 req.body 直达第 35 行 new RegExp 与第 36 行 test，连第 22-25 行的形式过滤都不存在，确认同一 sink 存在零防御的第二入口；此外第 29 行将用户模式持久化进 savedRules（Map），若其他模块后续重新执行已保存规则则形成持久化攻击通道。两处入口加持久化存储共同支撑最终结论：本模块存在系统性 CWE-400 漏洞，风险等级 High。
