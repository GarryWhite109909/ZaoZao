<?php

declare(strict_types=1);

namespace App\Http\Controllers\Search;

use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

/**
 * Catalog search: extracts field:value filter tokens from a free-form query.
 */
final class SearchFilterController extends Controller
{
    private const MAX_QUERY_LENGTH = 256;
    private const MAX_TOKENS = 16;

    /**
     * Possessive quantifiers (++) plus the atomic group (?>...) remove
     * backtracking entirely: each character is consumed at most once, so a
     * match attempt is O(n). The naive form with nested quantifiers such as
     * ((\w+\s*)+ would backtrack exponentially on crafted input.
     */
    private const TOKEN_RE = '/\b(?>(?P<field>[A-Za-z]++):(?P<value>[A-Za-z0-9_.-]++))/';

    public function search(Request $request): JsonResponse
    {
        $query = (string) $request->input('q', '');

        // Defense 1: hard length cap bounds every preg_match subject to 256 chars.
        if (mb_strlen($query) > self::MAX_QUERY_LENGTH) {
            return response()->json(['error' => 'query_too_long'], 422);
        }

        $filters = [];
        if (preg_match_all(self::TOKEN_RE, $query, $m, PREG_SET_ORDER) > 0) {
            // Defense 3: cap extracted tokens to bound result memory and downstream cost.
            foreach (array_slice($m, 0, self::MAX_TOKENS) as $token) {
                $filters[$token['field']] = $token['value'];
            }
        }

        return response()->json(['filters' => $filters]);
    }
}

第4步（防御二：模式级根治，第 25 行）：模式使用占有量词 [A-Za-z]++、[A-Za-z0-9_.-]++ 并整体包在原子组 (?>...) 中，PCRE 引擎对每个字符至多消费一次、子匹配失败即整体放弃，不存在可回溯状态，单次尝试严格 O(n)；即使移除第 32 行的长度校验也不会指数爆炸。这是消除回溯机制本身的根治型防御，区别于黑名单/字符串替换等可绕过过滤。第 20–23 行注释同时点明朴素写法 ((\w+\s*)+ 的嵌套量词风险，形成教学对照。

第6步（第二入口/替代通道检查）：唯一公开入口是第 27 行的 search()（挂载单一路由）；TOKEN_RE 全文件仅在第 37 行使用一次；不存在 CLI 命令、队列 Job、其他控制器方法或间接调用以未校验输入触发该正则。结论：无第二入口、无替代通道，两道防御在唯一 source→sink 链上类型正确、位置正确、覆盖完整，判定安全。
