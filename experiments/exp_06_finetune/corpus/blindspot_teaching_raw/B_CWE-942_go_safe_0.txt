package gateway

import (
	"net/http"
	"strings"
)

// corsPolicy enforces a strict cross-origin policy for the REST API gateway.
// Only exact, pre-registered origins are eligible for credentialed CORS.
type corsPolicy struct {
	allowed map[string]bool
}

// NewCORSPolicy builds the whitelist at startup. Any entry containing a
// wildcard is dropped so a permissive value can never reach the runtime path.
func NewCORSPolicy(origins []string) *corsPolicy {
	m := make(map[string]bool, len(origins))
	for _, o := range origins {
		if o == "" || strings.Contains(o, "*") {
			continue
		}
		m[o] = true
	}
	return &corsPolicy{allowed: m}
}

// Middleware wraps every gateway route with the CORS decision.
func (p *corsPolicy) Middleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := r.Header.Get("Origin")
		if origin != "" {
			w.Header().Add("Vary", "Origin")
			if p.allowed[origin] {
				h := w.Header()
				h.Set("Access-Control-Allow-Origin", origin)
				h.Set("Access-Control-Allow-Credentials", "true")
				h.Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE")
				h.Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
			}
		}
		if r.Method == http.MethodOptions && origin != "" {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

4. 第 4 步——流二（配置→白名单）与凭据组合验证：第 19 行在启动期丢弃任何含 `*` 的条目及空串，即使运维误配置，通配值也无法进入运行时 map；第 36 行 Allow-Credentials 仅在第 33 行白名单命中分支内输出，因此 ACAO 通配与凭据并存的危险组合在本文件中结构性不可能——凭据头从不在未匹配来源上发出。

6. 第 6 步——第二入口/替代通道检查：预检 OPTIONS（第 41-43 行）由同一策略处理并返回 204，响应头同样受第 33 行门控；文件内不存在其他写 ACAO/ACAC 的代码路径，第 45 行是唯一向业务路由放行的通道，无其他路由、参数或间接调用可绕过该包裹。结论：防御类型、位置、覆盖面均正确，不存在 CWE-942 漏洞。
