package search

import (
	"net/http"
	"time"

	"github.com/dlclark/regexp2"
)

// maxQueryLen bounds the worst-case size of any string handed to the engine.
const maxQueryLen = 256

// querySyntax keeps the classic nested-quantifier shape, but every group is
// atomic (?>...): the backtracking engine can never re-enter a matched group,
// so matching stays linear even on adversarial input like aaaa...a!
var querySyntax = regexp2.MustCompile(`^(?>[A-Za-z0-9_]+)(?>[ \-](?>[A-Za-z0-9_]+))*$`, regexp2.None)

func init() {
	// Belt-and-braces: abort any match still running after 500ms.
	querySyntax.MatchTimeout = 500 * time.Millisecond
}

// SearchHandler serves GET /search?q=<term> and validates the term before use.
func SearchHandler(w http.ResponseWriter, r *http.Request) {
	query := r.URL.Query().Get("q")
	if len(query) > maxQueryLen {
		http.Error(w, "query too long", http.StatusBadRequest)
		return
	}
	ok, err := querySyntax.MatchString(query)
	if err != nil || !ok {
		http.Error(w, "malformed query", http.StatusBadRequest)
		return
	}
	// Query is a well-formed token list; hand off to the search core here.
	w.Write([]byte("search accepted"))
}

// Register wires the only route exposed by this package.
func Register(mux *http.ServeMux) {
	mux.HandleFunc("/search", SearchHandler)
}
