// routes/userProfile.js — 公开用户资料接口（安全版：响应白名单投影 + 邮箱脱敏）
const express = require('express');
const router = express.Router();

// 模拟存储层：实体含密码哈希、TOTP 密钥等内部敏感字段
const users = new Map([
  [42, {
    id: 42,
    username: 'alice',
    email: 'alice@example.com',
    passwordHash: '$2b$12$KIXQe9m1oGvOx',
    totpSecret: 'JBSWY3DPEHPK3PXP',
    internalScore: 97,
    createdAt: '2024-01-01T00:00:00Z',
  }],
]);

// 白名单：唯一允许离开服务边界的字段集合（默认拒绝，未列出即不输出）
const PUBLIC_PROFILE_FIELDS = ['id', 'username', 'email', 'createdAt'];

function maskEmail(email) {
  const at = email.indexOf('@');
  if (at <= 0) return '***';
  return email[0] + '***' + email.slice(at);
}

function toPublicProfile(user) {
  const dto = {};
  for (const field of PUBLIC_PROFILE_FIELDS) {
    if (user[field] === undefined) continue;
    dto[field] = field === 'email' ? maskEmail(user[field]) : user[field];
  }
  return dto;
}

router.get('/api/users/:id/profile', (req, res) => {
  const user = users.get(Number(req.params.id));
  if (!user) return res.status(404).json({ error: 'user not found' });
  return res.json(toPublicProfile(user)); // 唯一序列化出口：只输出白名单投影
});

module.exports = router;
