// Package main implements the document ingestion upload endpoint.
package main

import (
	"encoding/json"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"runtime/debug"
)

func reply(w http.ResponseWriter, code int, detail string) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(code)
	_ = json.NewEncoder(w).Encode(map[string]string{"status": "error", "detail": detail})
}

func uploadHandler(w http.ResponseWriter, r *http.Request) {
	defer func() {
		if rec := recover(); rec != nil {
			reply(w, http.StatusInternalServerError, string(debug.Stack()))
		}
	}()
	file, header, err := r.FormFile("doc")
	if err != nil {
		reply(w, http.StatusBadRequest, err.Error())
		return
	}
	defer file.Close()
	savePath := filepath.Join("/var/lib/docparser/inbox", filepath.Base(header.Filename))
	dst, err := os.Create(savePath)
	if err != nil {
		reply(w, http.StatusInternalServerError, err.Error())
		return
	}
	defer dst.Close()
	if _, err := io.Copy(dst, file); err != nil {
		reply(w, http.StatusInternalServerError, err.Error())
		return
	}
	_ = json.NewEncoder(w).Encode(map[string]string{"status": "ok", "detail": "stored"})
}

func main() {
	http.HandleFunc("/api/v1/upload", uploadHandler)
	_ = http.ListenAndServe("0.0.0.0:8080", nil)
}
