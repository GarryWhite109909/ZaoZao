```javascript
const express = require('express');
const mysql = require('mysql2/promise');

const app = express();
app.use(express.json());

const pool = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  connectionLimit: 10,
});

// GET /api/orders?userId=123 —— 查询某用户的订单列表
app.get('/api/orders', async (req, res) => {
  const userId = req.query.userId;
  if (!userId || !/^\d+$/.test(userId)) {
    return res.status(400).json({ error: 'invalid userId' });
  }
  try {
    const sql = 'SELECT id, amount, status FROM orders WHERE user_id = ? ORDER BY created_at DESC';
    const [rows] = await pool.query(sql, [userId]);
    return res.json({ orders: rows });
  } catch (err) {
    // 详细错误（驱动消息、原始 SQL、错误码、堆栈）只写服务端日志
    console.error('[orders] query failed for userId=%s', userId, {
      message: err.message,
      code: err.code,
      sql: err.sql,
      stack: err.stack,
    });
    // 客户端只收到通用文案，绝不回显 err.message / err.sql / err.stack
    return res.status(500).json({ error: 'internal server error' });
  }
});

app.listen(3000, () => console.log('listening on 3000'));
```

1. **第 1 步——本类核心特征**：CWE-209 的判定对象是“错误信息暴露通道”：数据库驱动失败时，异常对象通常携带 `err.sql`（mysql2 中含拼接后的完整 SQL 文本）、`err.sqlMessage`、`errno`、堆栈，甚至主机/端口/文件路径等内部细节；漏洞形态是把 `err.message`/`err.sql`/`err.stack` 或整个 `err` 对象写进对客户端的响应，或未捕获异常导致框架默认错误页（Express 非 production 模式回显堆栈）直接渲染。这与 SQL 注入（恶意输入改变查询语义）或越权（未授权访问数据）是不同类别——即使查询已参数化、无注入，错误回显仍构成 CWE-209。本文件为安全版：第 25 行 catch 后，敏感字段只进第 27 行服务端日志，客户端只拿第 34 行通用文案。

2. **第 2 步——输入点枚举与可达性**：全文件外部输入仅一处：第 16 行注册的 GET /api/orders 路由中第 17 行的 `req.query.userId`；不存在对 `req.body`/`req.params`/`req.headers` 的任何读取。可达性验证：userId 在第 23 行作为绑定参数进入 `pool.query`，当查询因驱动层错误失败（连接断开、死锁、表结构变更等，与输入是否恶意无关）时 Promise 拒绝被第 25 行 catch 捕获，即该输入点可达错误处理 sink（响应边界），必须逐一核查每条响应路径。

3. **第 3 步——敏感数据流与防御类型正确性**：失败路径上第 25 行的 `err` 携带 `err.message`/`err.code`/`err.sql`/`err.stack`（第 28-31 行显式取出）。逐一核对全部 res 输出点：第 19 行 400 响应为静态字符串 'invalid userId'；第 24 行成功响应只含业务查询结果 rows；第 34 行 500 响应为静态字符串 'internal server error'——没有任何一条响应包含 `err` 或其字段，敏感错误内容的唯一去向是第 27 行 `console.error`（stderr，服务端日志）。防御类型正确：这是结构性收敛（catch 后用固定通用文案整体替换响应），而非对 `err.message` 做黑名单/正则/字符串替换清洗——后者可被驱动错误文案的任意措辞变化绕过，前者对任何错误内容都成立。

4. **第 4 步——防御位置与覆盖面**：位置正确：try/catch 位于 handler 最外层，包裹第 23 行这个唯一可失败的异步操作，任何错误路径必须经过第 25 行才能到达响应；try 之外（第 17-20 行）为纯同步逻辑、无外部调用、不可能抛出需响应的异常。覆盖完整：文件内仅此一条 DB 调用链，第 34 行的通用 500 是唯一错误响应通道。需澄清：第 18 行 `/^\d+$/` 只是参数格式校验（防注入的纵深手段），并非 CWE-209 的防御——即便输入完全合法，连接层故障仍会触发 catch，说明本防御不依赖输入过滤，这正是其不可绕过的原因。

5. **第 5 步——第二入口/替代通道检查**：全文件仅第 16 行注册一个路由，无其他 app.get/post/use 请求处理器，无错误转发中间件；代码中没有任何 `next(err)` 调用，因此错误不会落入 Express 默认错误处理器——该默认处理器在非 production 环境会把 `err.stack` 渲染给客户端，是 CWE-209 最典型的替代泄露通道，本文件已从结构上封死。也不存在全局 unhandledRejection/uncaughtException 处理器向客户端回显错误，无第二个查询参数或间接调用路径。结论：不存在第二入口或替代泄露通道，防御类型、位置、覆盖面均完整，判定为安全版本。
