package com.acme.catalog.search;

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

/**
 * Storefront search: validates the free-text query, then filters the catalog.
 * ReDoS hardening (CWE-400): the legacy pattern ^(\w+[ .,-]*)*$ nested
 * quantifiers and backtracked exponentially on crafted input; this version
 * caps query length and matches an atomic-group pattern in linear time.
 */
public final class CatalogSearchService {
    public static final int MAX_QUERY_LENGTH = 64;
    // (?>...) freezes each token: the engine can drop whole iterations but
    // never re-split a word run, so the exponential path is removed.
    private static final Pattern QUERY_PATTERN = Pattern.compile("^(?>\\w+[ .,-]*)+$");

    private final List<String> catalog;

    public CatalogSearchService(List<String> catalog) {
        this.catalog = List.copyOf(catalog);
    }

    public List<String> search(String rawQuery) {
        if (rawQuery == null || rawQuery.length() > MAX_QUERY_LENGTH) {
            return List.of();                  // defense 1: hard length cap
        }
        if (!QUERY_PATTERN.matcher(rawQuery).matches()) {
            return List.of();                  // defense 2: atomic-group regex
        }
        String needle = rawQuery.trim().toLowerCase();
        List<String> hits = new ArrayList<>();
        for (String item : catalog) {
            if (item.toLowerCase().contains(needle)) {
                hits.add(item);
            }
        }
        return hits;
    }

    /** Bulk entry for the admin import job; funnels into the same guard. */
    public List<String> searchAll(List<String> rawQueries) {
        List<String> merged = new ArrayList<>();
        for (String q : rawQueries) {
            merged.addAll(search(q));
        }
        return merged;
    }
}
