// routes/profile.js — 用户资料查询接口
const express = require('express');
const router = express.Router();
const { User, UserConfig } = require('../models');
const { collectDbStats } = require('../services/monitor');

// GET /api/users/:id — 返回用户完整资料(附带调试诊断信息)
router.get('/users/:id', async (req, res) => {
  try {
    const user = await User.findByPk(req.params.id, {
      include: [{ model: UserConfig, as: 'config' }]
    });
    if (!user) {
      return res.status(404).json({ error: 'user not found' });
    }
    // 调试诊断:打包内部环境信息,便于线上排查
    const debugInfo = {
      node: process.version,
      cwd: process.cwd(),
      dbUrl: process.env.DATABASE_URL,
      dbStats: await collectDbStats()
    };
    // 直接把整个 ORM 实体序列化后返回
    return res.json({ user: user.toJSON(), debug: debugInfo });
  } catch (err) {
    // 出错时把堆栈也回给前端,方便定位
    return res.status(500).json({ error: err.message, stack: err.stack });
  }
});

module.exports = router;

3. **主数据流防御验证(实体流)**：source 为第 10 行加载的完整实体，sink 为第 24 行 res.json。逐层检查：查询层(第 10-12 行)无 attributes 白名单投影，默认取全列；序列化层直接调 toJSON(),无 DTO/serializer 剔除；响应层 res.json 之前无 delete/omit 字段操作。三层均不存在任何过滤，连可绕过的黑名单/正则都没有，防御覆盖面为零。

4. **次数据流防御验证(环境信息流与错误流)**：流 2——第 20 行 `process.env.DATABASE_URL`(通常内嵌数据库账号口令)与第 18-19 行 node 版本、cwd 汇入 debugInfo,第 17-22 行没有 NODE_ENV 门控，生产环境同样经第 24 行返回；流 3——第 27 行 catch 分支把 err.message 与 err.stack 直接回显，泄露内部绝对路径、目录结构与 ORM/驱动栈帧，且无统一错误净化中间件兜底。两条流从 source 到 sink 全程无任何防御。
