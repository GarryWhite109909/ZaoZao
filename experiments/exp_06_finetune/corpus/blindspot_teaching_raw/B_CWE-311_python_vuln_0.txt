# app/auth/password_reset.py
"""Password reset flow: token issuance and confirmation."""
import logging
import secrets
import smtplib
from email.message import EmailMessage

import requests
from flask import Flask, jsonify, request

app = Flask(__name__)
log = logging.getLogger(__name__)

SMTP_HOST, SMTP_PORT = "mail.corp.lan", 25
TOKEN_LOOKUP_URL = "http://token-service.internal/lookup"
SET_PASSWORD_URL = "http://user-profile.internal/api/set-password"


@app.route("/password/reset", methods=["POST"])
def issue_reset_token():
    email = request.form["email"]
    token = secrets.token_urlsafe(32)
    msg = EmailMessage()
    msg["To"] = email
    msg["Subject"] = "Your password reset token"
    msg.set_content(f"Use this token to reset your password: {token}")
    with smtplib.SMTP(SMTP_HOST, SMTP_PORT) as smtp:
        smtp.send_message(msg)
    log.info("issued reset token %s for %s", token, email)
    return jsonify({"status": "sent"})


@app.route("/password/confirm", methods=["POST"])
def confirm_reset():
    token = request.form["token"]
    new_password = request.form["password"]
    owner = requests.get(
        TOKEN_LOOKUP_URL, params={"token": token}, timeout=5
    ).json()
    requests.post(
        SET_PASSWORD_URL,
        json={"user_id": owner["user_id"], "password": new_password},
        timeout=5,
    )
    return jsonify({"status": "ok"})

3. 流 2（令牌与新密码 → 明文 HTTP）：入口为第 33 行注册的 POST /password/confirm，第 35 行 request.form['token'] 与第 36 行 request.form['password'] 为外部输入。第 37-39 行以查询参数把令牌发往 http://token-service.internal——http:// 即无 TLS，且令牌进入 URL 更易被代理与访问日志留存；第 40-44 行把新密码以 JSON 体 POST 到 http://user-profile.internal。'内网域名'不提供任何加密，网络路径上的任意观察点都能读到明文密码。

5. 第二入口/替代通道检查：本模块仅注册第 19 行 /password/reset 与第 33 行 /password/confirm 两个路由，无其他路由、无 GET 参数变体、无间接调用包装；入口从 request.form 换成 request.args 也不影响结论，因为漏洞在出站 sink 而非入口解析。两条路由是通往同一信任边界（邮件中继/内部服务）的替代通道，且每条通道都是明文——替代通道不构成缓解，反而扩大暴露面，支撑最终结论：存在 CWE-311，风险 High（重置令牌与新密码泄露可直接导致账户接管）。
