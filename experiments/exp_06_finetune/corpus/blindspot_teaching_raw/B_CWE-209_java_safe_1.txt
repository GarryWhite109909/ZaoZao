package com.example.filesvc.web;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.multipart.MultipartFile;

@RestController
public class FileProcessController {

    private static final Logger log = LoggerFactory.getLogger(FileProcessController.class);
    private static final Path BASE_DIR = Paths.get("/var/appdata/uploads");

    @PostMapping("/files/process")
    public ResponseEntity<String> process(@RequestParam("file") MultipartFile file) {
        try {
            Path target = BASE_DIR.resolve(file.getOriginalFilename()).normalize();
            if (!target.startsWith(BASE_DIR)) {
                return ResponseEntity.badRequest().body("Invalid file request.");
            }
            String content = Files.readString(target);
            return ResponseEntity.ok("Processed " + content.lines().count() + " lines.");
        } catch (Exception e) {
            // Full details (stack trace, internal paths) stay in server-side logs only;
            // the client receives a fixed generic message with no exception data.
            log.error("File processing failed for upload: {}", file.getOriginalFilename(), e);
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                    .body("Unable to process the file. Please try again later.");
        }
    }
}
