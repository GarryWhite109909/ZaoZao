```code
'use strict';

const express = require('express');
const crypto = require('crypto');
const axios = require('axios');
const nodemailer = require('nodemailer');
const User = require('../models/user');

const router = express.Router();

// 明文 SMTP 传输器：25 端口、secure:false、不校验证书
const mailer = nodemailer.createTransport({
  host: 'smtp.internal.example.com',
  port: 25,
  secure: false,
  tls: { rejectUnauthorized: false }
});

// 密码重置：生成令牌并经明文信道分发
router.post('/forgot-password', async (req, res) => {
  const email = req.body.email;
  const user = await User.findOne({ where: { email } });
  if (!user) {
    return res.status(404).json({ error: 'user not found' });
  }

  const resetToken = crypto.randomBytes(32).toString('hex');
  await user.update({ resetToken, resetTokenExpiry: Date.now() + 3600000 });

  // 通道一：令牌明文写入 SMTP 邮件
  await mailer.sendMail({
    from: 'noreply@example.com',
    to: email,
    subject: 'Reset your password',
    text: 'Your reset token: ' + resetToken
  });

  // 通道二：令牌经 http:// 明文转发给内部通知服务
  await axios.post('http://notify.internal.example.com/reset', {
    userId: user.id,
    token: resetToken
  });

  return res.json({ message: 'reset instructions sent' });
});

module.exports = router;
```

2. **第 2 步（输入点枚举与可达性）**：外部输入点仅一处——第 21 行 req.body.email，是唯一攻击者可控输入，但它只作为查询键和收件人，本身不是秘密；真正的敏感数据是第 27 行服务端生成的 resetToken，它不依赖外部输入即属敏感。从第 20 行路由入口起，每个 POST /forgot-password 请求都无条件把令牌送达三个 sink：第 28 行（落库）、第 35 行（邮件正文）、第 41 行（http 请求体），可达性 100%。

3. **第 3 步（逐流防御核验）**：流A（第 27→28 行）：ORM update 直接写原始令牌，无任何哈希/加密调用；流B（第 27→35 行，经第 31-36 行 sendMail）：传输器第 14 行用 25 端口、第 15 行 secure:false 表示无隐式 TLS，第 16 行 rejectUnauthorized:false 即使发生 opportunistic STARTTLS 也接受任意证书，链路等同明文；流C（第 27→41 行，经第 39 行 axios.post）：目标硬编码为 http://，全程无 TLS。全文件不存在任何白名单/正则/过滤逻辑（也就不存在可绕过的伪防御），第 23-25 行的用户存在性检查只是业务前置条件，与机密性无关——三条流均无有效防御。

4. **第 4 步（危害链）**：能嗅探应用与 smtp.internal.example.com:25 或 notify.internal.example.com:80 之间流量的攻击者，可直接恢复出有效的 32 字节重置令牌，随即调用重置接口接管任意用户账号；数据库被拖走时同样泄露全部在途令牌。第 27 行 crypto.randomBytes 的高熵无法弥补明文传输——熵只防猜测，不防窃听。

5. **第 5 步（第二入口/替代通道检查）**：本模块仅导出一个路由（第 20 行 POST /forgot-password），无其他路由、无其他 req 读取点、无 EventEmitter/消息队列等间接分发令牌的调用；两个 sink 均内联在处理器内，三条明文流在本文件内闭环，无需依赖未证实的替代路径即可确认 CWE-311 成立，风险等级 High（明文信道泄露等同凭据的令牌导致账号接管）。
