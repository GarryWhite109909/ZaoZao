"""REST API gateway with a strict CORS policy (CWE-942 safe reference).

Only exact-allowlisted origins receive CORS grants; credentials are never
combined with a wildcard, and every response varies on Origin.
"""
from flask import Flask, jsonify, request, Response

ALLOWED_ORIGINS = frozenset({
    "https://app.example.com",
    "https://dashboard.example.com",
})


def apply_cors(resp: Response) -> Response:
    origin = request.headers.get("Origin", "")
    resp.vary.add("Origin")  # shared caches must not reuse one origin's response
    if origin in ALLOWED_ORIGINS:  # exact scheme+host match, no wildcard/regex/suffix
        resp.headers["Access-Control-Allow-Origin"] = origin
        resp.headers["Access-Control-Allow-Credentials"] = "true"
    return resp


app = Flask(__name__)


@app.after_request
def cors_hook(resp: Response) -> Response:
    return apply_cors(resp)


@app.route("/api/gateway/data", methods=["GET", "POST", "OPTIONS"])
def gateway_data():
    if request.method == "OPTIONS":  # preflight; grant still added by cors_hook
        resp = Response(status=204)
        resp.headers["Access-Control-Allow-Methods"] = "GET, POST"
        resp.headers["Access-Control-Allow-Headers"] = "Content-Type, Authorization"
        resp.headers["Access-Control-Max-Age"] = "600"
        return resp
    payload = {"service": "gateway", "user": request.headers.get("X-User", "anon")}
    return jsonify(payload)


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8443)
