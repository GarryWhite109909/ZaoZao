package com.example.spabackend.web;

import jakarta.servlet.Filter;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.ServletRequest;
import jakarta.servlet.ServletResponse;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.util.Set;

/**
 * CORS guard for the SPA backend: the client-controlled Origin header is
 * echoed only on an exact whitelist match, and Vary: Origin keeps shared
 * caches from replaying one origin's CORS decision to another origin.
 */
@Component
public class CorsGuardFilter implements Filter {

    private static final Set<String> TRUSTED_ORIGINS = Set.of(
            "https://app.example.com",
            "https://admin.example.com");

    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {
        HttpServletRequest req = (HttpServletRequest) request;
        HttpServletResponse res = (HttpServletResponse) response;

        String origin = req.getHeader("Origin");
        boolean trusted = origin != null && TRUSTED_ORIGINS.contains(origin);
        if (trusted) {
            res.setHeader("Access-Control-Allow-Origin", origin);
            res.setHeader("Access-Control-Allow-Credentials", "true");
        }
        res.addHeader("Vary", "Origin");
        if (trusted && "OPTIONS".equals(req.getMethod())) {
            res.setHeader("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE");
            res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");
            res.setStatus(HttpServletResponse.SC_NO_CONTENT);
            return;
        }
        chain.doFilter(request, response);
    }
}

2. **输入点枚举与可达性**：全文件唯一外部输入是第 33 行 `req.getHeader("Origin")`，完全攻击者可控；第 40 行 `req.getMethod()` 仅与常量 OPTIONS 比较，不进入任何输出。可达性成立：第 20 行 @Component 使 Spring Boot 将该 Filter 自动注册到 /*，第 28 行 doFilter 对每个请求执行，因此 Origin 在每个携带该头的请求上都会到达第 36-37 行的 setHeader sink，防御必须且只能落在这一条路径上。

3. **防御类型正确性**：第 23-25 行用 `Set.of` 建立常量精确白名单，第 34 行 `TRUSTED_ORIGINS.contains(origin)` 是整串全等比较（区分大小写、无子串/前缀/正则语义），不属于黑名单、正则、字符串替换这类可绕过过滤——`https://app.example.com.attacker.com`、`https://app-example.com`、大小写变体乃至字面量 null 均无法命中。未命中时第 35 行分支不执行，不输出任何 ACAO，浏览器按同源策略拦截跨源读取，行为 fail-closed；Origin 值除命中白名单时回显（即等于常量本身）外不进入任何业务逻辑或存储。

4. **防御位置与覆盖面 + 凭据组合**：位置上，第 34 行判定先于第 46 行 chain.doFilter，位于所有 Controller 之前的门卫位，且紧贴第 36-37 行唯一的 ACAO/凭据写出点；覆盖面上，Filter 注册路径为 /*，全部 API 路由无例外经过，本文件内不存在第二个 Access-Control-* 写出点。第 39 行对每个响应补 Vary: Origin，使 CDN/浏览器共享缓存以 Origin 为缓存键，防止『带 app.example.com 的 ACAO 的缓存响应』被回放给其他 Origin（无 Vary 时缓存投毒可绕过白名单）；第 37 行 Allow-Credentials: true 只出现在白名单命中分支内，全文件无 * 字面量，『通配符+凭据』这一 CWE-942 最危险组合在结构上不可能成立；预检分支第 40-45 行同样受 trusted 门控，非白名单 Origin 连预检批准也拿不到。

5. **第二入口/替代通道结论**：逐项排查——本文件是唯一 CORS 模块，无 @CrossOrigin 注解、无 WebMvcConfigurer.addCorsMappings、无其他 Filter/Interceptor 写跨域头；Origin 值无第二 sink（不参与日志拼接、不入库、不进下游调用）；无 Origin 头（同源/非 CORS 请求）与非白名单 Origin 行为一致，均不颁发任何跨域许可。结论：单一入口、单一 sink 组，防御类型/位置/覆盖面均正确，判定无 CWE-942 漏洞。
