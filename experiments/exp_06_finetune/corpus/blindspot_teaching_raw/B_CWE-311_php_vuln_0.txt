<?php
declare(strict_types=1);

namespace App\Auth;

use PDO;

final class PasswordResetService
{
    public function __construct(
        private readonly PDO $pdo,
        private readonly string $notifyApi = 'http://notify.internal.example.com/api/v1/mail'
    ) {}

    /** POST /auth/reset-request  form: email */
    public function requestReset(string $email): void
    {
        $token = bin2hex(random_bytes(32));
        $stmt = $this->pdo->prepare(
            'INSERT INTO password_resets (email, token, expires_at) VALUES (?, ?, NOW() + INTERVAL 1 HOUR)'
        );
        $stmt->execute([$email, $token]);
        $this->sendMail($email, 'Reset your password',
            "Open http://shop.example.com/auth/reset-confirm?token={$token} to set a new password.");
    }

    /** POST /auth/reset-confirm  form: token, password */
    public function confirmReset(string $token, string $newPassword): void
    {
        $sel = $this->pdo->prepare(
            'SELECT email FROM password_resets WHERE token = ? AND expires_at > NOW()'
        );
        $sel->execute([$token]);
        $row = $sel->fetch(PDO::FETCH_ASSOC);
        if ($row === false) {
            throw new \RuntimeException('invalid or expired reset token');
        }
        $upd = $this->pdo->prepare('UPDATE users SET password = ? WHERE email = ?');
        $upd->execute([$newPassword, $row['email']]);
    }

    private function sendMail(string $to, string $subject, string $body): void
    {
        $ctx = stream_context_create(['http' => [
            'method'  => 'POST',
            'content' => http_build_query(['to' => $to, 'subject' => $subject, 'body' => $body]),
        ]]);
        file_get_contents($this->notifyApi, false, $ctx);
    }
}
