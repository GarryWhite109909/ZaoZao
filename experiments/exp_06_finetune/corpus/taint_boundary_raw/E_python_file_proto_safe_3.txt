```python
"""静态资源定位工具：解析资源索引 XML，按名称定位服务器上的文件。

本模块由上层后台服务调用（传入上传/同步目录中的索引文件路径），
XML 内容来自外部提交，必须视为不可信输入。
"""
import os
import posixpath
from defusedxml.ElementTree import parse as safe_parse

ASSET_ROOT = "/srv/static/assets"
ALLOWED_EXT = {".png", ".jpg", ".css", ".js", ".woff2"}


def locate_asset(index_path: str) -> str:
    """解析索引 XML 并返回目标资源文件的绝对路径。"""
    # defusedxml 禁用外部实体与 DTD 展开，从根上阻断 XXE（CWE-611）
    tree = safe_parse(index_path)
    name = tree.getroot().get("name") or ""
    ext = posixpath.splitext(name)[1].lower()

    if ext not in ALLOWED_EXT:
        raise ValueError("unsupported asset type")

    resolved = os.path.realpath(os.path.join(ASSET_ROOT, name))
    if not resolved.startswith(os.path.realpath(ASSET_ROOT) + os.sep):
        raise ValueError("path traversal rejected")

    return resolved
```

3. 防御验证（类型正确）：第 13 行采用 defusedxml 的 `safe_parse`，它在底层禁用了 DTD 处理、外部实体解析与实体展开（forbid_dtd / forbid_entities 默认开启），属于"原生安全 API"类防御；XXE 攻击面完全位于 XML 解析器内部，防御位置在污点进入程序的第一道关口（第 14 行），先于任何属性读取，覆盖完整——不存在绕过路径，因为恶意实体根本不会被实例化。
4. 补充防御：即使解析结果被污染，第 20 行扩展名白名单与第 23-25 行 realpath+前缀校验也阻断了路径穿越型二次利用，形成纵深防御。
5. 第二入口检查：模块仅暴露 `locate_asset` 一个函数，唯一参数是文件路径，无其他路由、参数或间接调用通道；`ASSET_ROOT` 为编译期常量不可控。所有 source→sink 路径均被第 14 行的解析器级防御拦截，故判定安全。
