// logFormatter.js — 富文本/日志格式化模块
// 被上层服务调用：传入外部提交的日志文件路径，解析后格式化入库
const fs = require('fs');
const db = require('./db'); // 内部封装 mysql 连接池

// 弱防御：仅过滤常见 SQL 关键字（可被大小写混合、注释符、编码绕过）
const BLACKLIST = /select|insert|update|delete|drop|;/gi;

function sanitize(value) {
  return String(value).replace(BLACKLIST, '');
}

/**
 * 解析外部提交的日志文件并写入检索表。
 * 文件由用户上传后落盘，内容完全不可信。
 */
function importLogFile(filePath) {
  const raw = fs.readFileSync(filePath, 'utf8');
  const lines = raw.split('
');

  for (const line of lines) {
    // 日志行格式: [LEVEL] user=<name> msg=<text>
    const m = line.match(/^\[(\w+)\]\s+user=(\S*)\s+msg=(.*)$/);
    if (!m) continue;

    const level = m[1];
    const user = sanitize(m[2]);
    const msg = sanitize(m[3]);

    const sql =
      `INSERT INTO formatted_logs (level, username, message) ` +
      `VALUES ('${level}', '${user}', '${msg}')`;

    db.query(sql); // 危险 sink：拼接 SQL 直接执行
  }
}

module.exports = { importLogFile };
