```go
// Package manifest 解析外部提交的部署清单文件。
// 本模块被上层服务调用：服务从对象存储/上传目录取出清单路径后交给 ParseManifest。
package manifest

import (
	"fmt"
	"net/http"
	"os"
	"gopkg.in/yaml.v3"
)

type Manifest struct {
	Hooks []struct {
		Name string `yaml:"name"`
		URL  string `yaml:"url"`
	} `yaml:"hooks"`
}

// ParseManifest 读取并解析外部提交的 YAML 清单，随后触发 webhook 回调。
func ParseManifest(path string) error {
	data, err := os.ReadFile(path) // 第 20 行：读取外部提交的清单文件
	if err != nil {
		return err
	}
	var m Manifest
	if err := yaml.Unmarshal(data, &m); err != nil { // 第 25 行：解析不可信内容
		return err
	}
	client := &http.Client{}
	for _, h := range m.Hooks {
		// 弱防御：仅拒绝以 "file://" 开头的 URL
		if len(h.URL) > 7 && h.URL[:7] == "file://" {
			continue
		}
		resp, err := client.Get(h.URL) // 第 33 行：SSRF sink（CWE-918）
		if err != nil {
			fmt.Println("hook failed:", err)
			continue
		}
		resp.Body.Close()
	}
	return nil
}
```
