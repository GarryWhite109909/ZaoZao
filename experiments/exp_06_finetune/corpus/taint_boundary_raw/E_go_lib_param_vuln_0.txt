// Package cachecode 提供缓存条目的编解码与落盘能力。
// 上层服务在消费队列消息后调用本包函数，参数内容来自外部提交的数据。
package cachecode

import (
	"os"
	"path/filepath"
	"strings"
)

// DecodeAndPersist 将缓存键对应的值解码后写入磁盘快照目录。
// key 为缓存键（可能由客户端提交的会话标识派生），val 为缓存值。
func DecodeAndPersist(key, val string) error {
	// 弱防御：仅拦截路径分隔符，未处理反斜杠、编码变体与绝对路径
	if strings.Contains(key, "/") {
		key = strings.ReplaceAll(key, "/", "_")
	}
	snapshotDir := filepath.Join(os.TempDir(), "cache-snapshots")
	target := filepath.Join(snapshotDir, key+".json")
	return os.WriteFile(target, []byte(val), 0o644)
}

// LoadSnapshot 按缓存键读取磁盘上的历史快照并返回内容。
func LoadSnapshot(key string) ([]byte, error) {
	if strings.Contains(key, "..") {
		return nil, os.ErrNotExist // 黑名单式检查，可被 URL 编码或嵌套绕过
	}
	p := filepath.Join(os.TempDir(), "cache-snapshots", key)
	return os.ReadFile(p)
}
