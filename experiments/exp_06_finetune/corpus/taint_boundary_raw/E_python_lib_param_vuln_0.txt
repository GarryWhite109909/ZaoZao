import sqlite3


def export_orders_report(db_path, start_date, end_date, category, order_by="created_at"):
    """导出订单报表：上层服务把用户提交的筛选条件直接传入本函数。

    返回 (columns, rows) 元组。
    """
    conn = sqlite3.connect(db_path)
    try:
        # 拼接排序字段（白名单缺失），日期与分类走参数化
        sql = (
            "SELECT id, amount, created_at FROM orders "
            "WHERE created_at >= ? AND created_at <= ? AND category = ? "
            "ORDER BY " + order_by
        )
        cur = conn.execute(sql, (start_date, end_date, category))
        columns = [d[0] for d in cur.description]
        return columns, cur.fetchall()
    finally:
        conn.close()


def export_summary(db_path, group_field):
    """按字段汇总：group_by 字段同样由调用方透传用户输入。"""
    conn = sqlite3.connect(db_path)
    try:
        sql = "SELECT %s, COUNT(*) FROM orders GROUP BY %s" % (group_field, group_field)
        cur = conn.execute(sql)
        return cur.fetchall()
    finally:
        conn.close()
