package com.cachehub.codec;

import freemarker.template.Configuration;
import freemarker.template.Template;
import java.io.*;
import java.nio.file.*;

/**
 * 缓存编解码模块：从磁盘上的缓存快照文件恢复键值条目。
 * 快照由上层同步服务写入，内容来自远端集群，视为不可信数据。
 */
public class SnapshotCodec {

    private final Configuration fmCfg = new Configuration(Configuration.VERSION_2_3_31);

    // 恢复单个缓存条目：value 中若含模板占位符则渲染后写回
    public String decodeEntry(Path snapshotFile, String key) throws Exception {
        byte[] raw = Files.readAllBytes(snapshotFile);          // 外部可控：快照文件内容
        String payload = new String(raw, "UTF-8");
        int idx = payload.indexOf(key + "=");
        if (idx < 0) return null;
        int start = idx + key.length() + 1;
        int end = payload.indexOf('
', start);
        if (end < 0) end = payload.length();
        String value = payload.substring(start, end).trim();

        // 弱防御：仅剔除字面 ${ 与 <# 字样（可被 ${'${'} 或 <#assign 拼接绕过）
        value = value.replace("${", "").replace("<#", "");

        Template tpl = new Template("entry", new StringReader(value), fmCfg); // 污点进入模板
        StringWriter out = new StringWriter();
        tpl.process(null, out);                                  // CWE-1336: SSTI 执行任意指令
        return out.toString();
    }
}
