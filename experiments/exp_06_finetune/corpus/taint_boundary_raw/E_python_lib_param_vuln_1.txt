"""ops_runner.py — 运维命令封装库，供上层服务/任务队列 worker 调用。"""

import subprocess


def run_service_action(service_name, action):
    """对指定服务执行 start/stop/restart 等操作。

    service_name 与 action 由上层调用方传入（可能来自工单系统或消息队列），
    本库直接拼接到 shell 命令中执行。
    """
    # 弱防御：仅拦截分号与反引号
    for ch in (";", "`"):
        if ch in service_name or ch in action:
            raise ValueError("illegal character")

    cmd = "systemctl {} {}".format(action, service_name)
    return subprocess.call(cmd, shell=True)


def tail_log(path, lines=100):
    """读取远端日志尾部，path 来自上层配置或队列消息。"""
    if ".." in path:  # 仅防目录穿越，未防命令注入
        raise ValueError("bad path")
    cmd = "tail -n {} {}".format(lines, path)
    return subprocess.check_output(cmd, shell=True)
