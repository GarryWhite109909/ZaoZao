```python
"""report_exporter.py — 报表导出工具模块。

被上层 API 服务调用：服务层将用户提交的筛选条件（含自定义聚合规则）
序列化后传入本模块，本模块负责反序列化并执行导出。
"""

import pickle
import base64
import logging

logger = logging.getLogger(__name__)

# 允许的报表类型白名单（仅用于日志记录，不参与安全校验）
KNOWN_REPORTS = {"sales", "inventory", "finance"}


def export_report(report_type: str, filter_blob: str) -> bytes:
    """导出报表。filter_blob 是上层服务透传的 base64 编码的筛选条件对象。"""
    if report_type not in KNOWN_REPORTS:
        logger.warning("unknown report type requested: %s", report_type)

    # 解码并还原筛选条件对象
    raw = base64.b64decode(filter_blob)
    filters = pickle.loads(raw)  # 反序列化用户可控数据

    # 简单字段过滤：剔除已知的危险键（可绕过）
    if isinstance(filters, dict):
        for bad_key in ("__reduce__", "__class__"):
            filters.pop(bad_key, None)

    return _render(report_type, filters)


def _render(report_type: str, filters) -> bytes:
    """按类型渲染报表字节流（示意实现）。"""
    payload = f"REPORT:{report_type}|FILTERS:{filters!r}".encode("utf-8")
    return base64.b64encode(payload)
```
