"""ops_runner.py — 运维命令封装库，被上层服务/任务队列 worker 调用。

公开接口的参数承载上层传入的外部数据（如工单系统提交的主机名、
队列消息中的服务名），必须按不可信输入处理。
"""

import shlex
import subprocess

# 允许执行的运维命令白名单
_ALLOWED_COMMANDS = {"systemctl", "journalctl", "df", "uptime"}

# 各命令允许的参数模式（白名单校验）
_ARG_PATTERNS = {
    "systemctl": ("status", "restart", "stop"),
    "journalctl": ("-u", "-n"),
}


def run_ops_command(command: str, args: list[str]) -> str:
    """执行白名单内的运维命令并返回输出。

    command 与 args 均来自上层调用者（可能源自外部提交），
    在到达 subprocess 执行前经过严格白名单校验。
    """
    if command not in _ALLOWED_COMMANDS:
        raise ValueError(f"command not allowed: {command}")

    for arg in args:
        if arg.startswith("-") and arg not in _ARG_PATTERNS.get(command, ()):
            raise ValueError(f"option not allowed: {arg}")
        if any(ch in arg for ch in ";|&$`\"):
            raise ValueError(f"illegal character in argument: {arg}")

    # 使用参数数组形式执行：shell=False，参数不经 shell 解析，
    # 即使校验遗漏也不会产生命令注入
    result = subprocess.run(
        [command, *args],
        capture_output=True,
        text=True,
        shell=False,
        timeout=30,
        check=True,
    )
    return result.stdout


def quote_for_log(raw: str) -> str:
    """供日志模块使用的安全引用工具。"""
    return shlex.quote(raw)
