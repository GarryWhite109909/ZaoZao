package archiver

// ArchiveEvent 是队列消息体，由外部提交方生成，字段内容不可信。
type ArchiveEvent struct {
	ArchiveURL string `json:"archive_url"`
}

// HandleArchiveEvent 是消息队列消费者回调（被 worker 注册为 handler），
// 外部数据仅通过 event 参数进入本函数。
func HandleArchiveEvent(event ArchiveEvent) error {
	u, err := url.Parse(event.ArchiveURL)
	if err != nil {
		return fmt.Errorf("invalid url: %w", err)
	}
	if u.Scheme != "https" && u.Scheme != "http" {
		return fmt.Errorf("scheme not allowed")
	}
	host := u.Hostname()
	if !isAllowedHost(host) {
		return fmt.Errorf("host not allowed")
	}
	ip, err := net.LookupIP(host)
	if err != nil || len(ip) == 0 {
		return fmt.Errorf("dns resolve failed")
	}
	for _, a := range ip {
		if isPrivate(a) {
			return fmt.Errorf("private address blocked")
		}
	}
	resp, err := http.Get(u.String()) // sink：发起出站请求
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	return fetchArchive(resp)
}

// 白名单：仅允许合作方归档存储域名
func isAllowedHost(host string) bool {
	for _, h := range []string{"archives.partner-cdn.com", "files.trusted-mirror.org"} {
		if strings.EqualFold(host, h) {
			return true
		}
	}
	return false
}

func isPrivate(ip net.IP) bool {
	return ip.IsLoopback() || ip.IsPrivate() || ip.IsLinkLocalUnicast() || ip.IsUnspecified()
}
