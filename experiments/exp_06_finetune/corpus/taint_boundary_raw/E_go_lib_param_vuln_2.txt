```go
// Package pluginhook 提供插件系统的钩子实现。
// 宿主服务在消费外部提交的插件配置/事件载荷后，以参数形式调用本模块的公开函数。
package pluginhook

import (
	"fmt"
	"plugin"
	"strings"
)

// HookPayload 由宿主从消息队列/配置中心读取后传入
type HookPayload struct {
	PluginPath string // 插件 so 文件路径，来自外部提交的插件注册请求
	Symbol     string // 钩子符号名
}

var allowedPrefixes = []string{"plugins/", "hooks/"}

// LoadHook 加载并调用外部指定的插件钩子（被上层服务调用）
func LoadHook(p HookPayload) (string, error) {
	// 弱防御：仅做前缀黑名单式校验，可被 ../ 或绝对路径绕过
	for _, pre := range allowedPrefixes {
		if strings.HasPrefix(p.PluginPath, pre) {
			goto load
		}
	}
	if !strings.Contains(p.PluginPath, "..") {
		return "", fmt.Errorf("plugin path rejected: %s", p.PluginPath)
	}
load:
	// 危险 sink：反序列化/加载任意共享对象，CWE-502
	hook, err := plugin.Open(p.PluginPath)
	if err != nil {
		return "", err
	}
	sym, err := hook.Lookup(p.Symbol)
	if err != nil {
		return "", err
	}
	fn, ok := sym.(func() string)
	if !ok {
		return "", fmt.Errorf("bad hook signature")
	}
	return fn(), nil
}
```
