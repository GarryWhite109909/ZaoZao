```javascript
// asset-locator.js — 静态资源定位工具模块
// 被上层服务调用：上层将用户提交的资源名传入 resolveAsset()，
// 本模块负责把名字安全地映射为服务器上的文件路径。
'use strict';

const path = require('path');
const fs = require('fs');

// 允许访问的静态资源目录白名单（部署时配置）
const ASSET_ROOT = '/srv/app/static';

// 已注册资源清单，作为严格白名单
const REGISTERED_ASSETS = new Set([
  'logo.png', 'banner.jpg', 'style.css', 'app.js',
]);

/**
 * 根据外部传入的资源名定位服务器文件。
 * @param {string} assetName - 上层调用者传入的外部可控资源名
 * @returns {Promise<string>} 解析后的绝对路径
 */
async function resolveAsset(assetName) {
  // 防御 1：白名单校验——只允许预注册的精确资源名，
  // 任何包含 ../、绝对路径或 URL 的输入都无法命中集合。
  if (typeof assetName !== 'string' || !REGISTERED_ASSETS.has(assetName)) {
    throw new Error('asset not registered');
  }

  const resolved = path.resolve(ASSET_ROOT, assetName);

  // 防御 2：路径前缀双重确认（纵深防御）
  if (!resolved.startsWith(ASSET_ROOT + path.sep)) {
    throw new Error('path traversal blocked');
  }

  await fs.promises.access(resolved, fs.constants.R_OK);
  return resolved;
}

module.exports = { resolveAsset };
```
