<?php
declare(strict_types=1);

namespace Ops\Toolbox;

/**
 * 运维命令封装库：构建并渲染命令执行结果报告。
 * 本模块被上层服务调用（如 CLI 调度器、消息队列消费者），
 * 上层传入的参数可能来自外部提交内容，必须按不可信数据处理。
 */
final class CommandReport
{
    /** @var array<string,string> 允许执行的运维命令白名单 */
    private const ALLOWED = [
        'status'  => '/usr/bin/systemctl status',
        'disk'    => '/bin/df -h',
        'uptime'  => '/usr/bin/uptime',
    ];

    /**
     * 渲染单条命令执行结果为 HTML 片段。
     *
     * @param string $commandKey 命令键名，由上层调用者传入
     * @param string $output     命令输出内容，由上层调用者传入
     * @return string 安全的 HTML 片段
     */
    public function renderResult(string $commandKey, string $output): string
    {
        if (!isset(self::ALLOWED[$commandKey])) {
            return '<p class="error">unknown command</p>';
        }

        $safeCmd   = htmlspecialchars(self::ALLOWED[$commandKey], ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
        $safeOut   = htmlspecialchars($output, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');

        return '<div class="cmd-result"><code>' . $safeCmd . '</code><pre>' . $safeOut . '</pre></div>';
    }
}
