```javascript
// ops-runner.js —— 运维命令封装库，被上层服务调用：消费任务队列消息并执行白名单运维命令，
// 执行结果会渲染进内部状态页（HTML），因此输出侧必须防 XSS。
'use strict';

const { execFile } = require('child_process');
const EventEmitter = require('events');

const COMMAND_WHITELIST = {
  'disk-usage': ['df', ['-h']],
  'service-status': ['systemctl', ['status']],
  'log-tail': ['tail', ['-n', '100']],
};

function escapeHtml(str) {
  return String(str)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

class OpsRunner extends EventEmitter {}

const runner = new OpsRunner();

// 队列消费者回调：message.payload 为外部提交的任务内容，必须按不可信数据处理
runner.on('job', (message) => {
  const task = message && message.payload;
  if (!task || typeof task.name !== 'string') return;

  const entry = COMMAND_WHITELIST[task.name]; // 第 26 行：白名单精确匹配，任意命令名直接被丢弃
  if (!entry) {
    runner.emit('error', new Error(`unknown command: ${task.name}`));
    return;
  }

  // 参数数组形式执行，不经 shell 解析；额外参数仅允许字母数字，防止注入
  const extraArgs = Array.isArray(task.args)
    ? task.args.filter((a) => /^[A-Za-z0-9._/-]+$/.test(String(a)))
    : [];

  execFile(entry[0], entry[1].concat(extraArgs), { shell: false }, (err, stdout) => {
    const html = `<pre>${escapeHtml(stdout || '')}</pre>`; // 输出进 HTML 前做完整实体转义
    runner.emit('report', html);
  });
});

module.exports = runner;
```
