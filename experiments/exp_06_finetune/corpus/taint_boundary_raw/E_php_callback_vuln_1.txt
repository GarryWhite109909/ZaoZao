<?php
// plugins/audit/hooks/AuditLogHook.php
// 插件系统钩子：宿主在收到外部事件（如第三方系统推送的审计事件）时调用本钩子
declare(strict_types=1);

namespace Plugins\Audit\Hooks;

use PDO;

class AuditLogHook implements \PluginSystem\HookInterface
{
    private PDO $db;

    public function __construct(PDO $db)
    {
        $this->db = $db;
    }

    // 宿主回调：$payload 为外部提交的事件载荷（JSON 字符串）
    public function onEvent(string $eventName, string $payload): void
    {
        // 弱防御：仅做黑名单关键字替换，可被大小写/编码绕过
        $cleaned = str_ireplace(['drop table', 'union select', '--'], '', $payload);
        $data = json_decode($cleaned, true);
        if (!is_array($data) || !isset($data['actor'], $data['resource'])) {
            return;
        }

        $sql = "INSERT INTO audit_log (event, actor, resource, created_at) "
             . "VALUES ('" . $eventName . "', '" . $data['actor'] . "', '" . $data['resource'] . "', NOW())";
        $this->db->exec($sql); // 危险 sink：字符串拼接 SQL 直接执行
    }
}

3. 第 24-27 行将载荷 JSON 解码后取出 `actor`、`resource` 字段，未做任何类型校验或转义，直接进入第 28-29 行的字符串拼接 SQL。
4. 第 30 行 `$this->db->exec($sql)` 是 CWE-89 sink：拼接后的 SQL 被直接执行，攻击者通过构造 `resource` 值如 `x', NOW()); DROP TABLE audit_log; --` 即可实现堆叠注入或任意数据篡改，source→sink 全程无有效防御，可达性确认。
