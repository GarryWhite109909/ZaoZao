```go
// Package jobctl 提供运维命令封装：从任务描述文件加载命令并安全执行。
// 上层调度服务会调用 LoadAndRun，任务文件由外部提交，内容必须视为不可信。
package jobctl

import (
	"encoding/json"
	"fmt"
	"os/exec"
)

// allowedCommands 是命令白名单：仅允许执行这些固定运维工具。
var allowedCommands = map[string]bool{
	"systemctl": true,
	"df":        true,
	"uptime":    true,
}

type Task struct {
	Cmd  string   `json:"cmd"`
	Args []string `json:"args"`
}

// LoadAndRun 读取任务文件（外部提交），解析后执行。
func LoadAndRun(path string) (string, error) {
	data, err := osReadFile(path)
	if err != nil {
		return "", err
	}
	var t Task
	if err := json.Unmarshal(data, &t); err != nil {
		return "", fmt.Errorf("bad task: %w", err)
	}
	if !allowedCommands[t.Cmd] { // 白名单校验：命令名必须命中固定集合
		return "", fmt.Errorf("command not allowed: %s", t.Cmd)
	}
	out, err := exec.Command(t.Cmd, t.Args...).CombinedOutput() // 参数数组形式，不经 shell 解析
	return string(out), err
}
```
