def export_report(report_type: str, filters: Dict[str, Any]) -> List[Dict[str, Any]]:
    """导出指定类型的报表。

    report_type 与 filters 均来自上层服务透传的用户筛选条件，
    按不可信数据处理。
    """
    # 白名单校验：report_type 必须命中预定义集合，杜绝任意 SQL 片段拼接
    if report_type not in _ALLOWED_REPORT_TYPES:
        raise ValueError(f"unsupported report type: {report_type}")

    template = _QUERY_TEMPLATES[report_type]

    # 仅提取该报表模板声明的过滤键，未知键一律丢弃

    clean_filters = {k: v for k, v in filters.items() if k in allowed_keys}

def deserialize_payload(raw: bytes) -> Dict[str, Any]:
    """反序列化上层消息队列投递的导出任务载荷。"""
    # 使用 json 而非 pickle：json 解析器不会实例化任意对象，无代码执行面
    payload = json.loads(raw.decode("utf-8"))
    if not isinstance(payload, dict):
        raise TypeError("payload must be a JSON object")
    return payload
