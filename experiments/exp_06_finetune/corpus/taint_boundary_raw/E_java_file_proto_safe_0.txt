package com.acme.cache.codec;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import org.owasp.html.PolicyFactory;
import org.owasp.html.Sanitizers;

/**
 * 缓存编解码模块：被上层缓存服务调用，读取磁盘上的缓存快照文件
 * （由对端节点通过同步协议落盘，内容视为不可信外部数据），
 * 解析出 key/value 并写入管理后台的 HTML 渲染缓冲。
 */
public final class CacheSnapshotCodec {

    // OWASP HTML Sanitizer 白名单策略：仅允许纯文本与基础格式标签
    private static final PolicyFactory POLICY = Sanitizers.FORMATTING
            .and(Sanitizers.BLOCKS)
            .and(Sanitizers.LINKS);

    /**
     * 解析缓存快照文件并返回可直接安全渲染的 value。
     * 快照文件格式：每行 "key<TAB>value"，value 可能来自外部节点提交。
     */
    public String decodeValue(Path snapshotFile, String key) throws IOException {
        for (String line : Files.readAllLines(snapshotFile, StandardCharsets.UTF_8)) { // 第 24 行：污点源
            int tab = line.indexOf('\t');
            if (tab <= 0) continue;
            String k = line.substring(0, tab);
            if (!k.equals(key)) continue;                       // 第 27 行：键匹配
            String rawValue = line.substring(tab + 1);          // 第 28 行：外部可控 value
            return POLICY.sanitize(rawValue);                   // 第 29 行：sink 前白名单净化
        }
        throw new IOException("key not found: " + key);
    }
}
