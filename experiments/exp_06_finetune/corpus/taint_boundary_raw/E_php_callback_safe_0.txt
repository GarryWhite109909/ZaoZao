<?php
declare(strict_types=1);

namespace App\Templating;

use Twig\Environment;
use Twig\Loader\ArrayLoader;

/**
 * 模板渲染辅助类。
 * renderJob() 由队列消费者回调调用，$payload 为消息体（外部提交的模板片段与变量），
 * 本文件不含任何 HTTP 入口。
 */
final class TemplateRenderHelper
{
    private Environment $twig;

    public function __construct()
    {
        // ArrayLoader + autoescape=true：Twig 默认对输出做上下文相关自动转义
        $this->twig = new Environment(new ArrayLoader([]), [
            'autoescape' => 'html',
            'sandbox'    => true,
        ]);
    }

    /** 队列消费者回调：消息体 $payload 承载外部数据 */
    public function renderJob(array $payload): string
    {
        // 白名单校验：仅允许预注册模板名，拒绝任意模板内容注入
        $allowed = ['welcome', 'invoice', 'digest'];
        $name = (string)($payload['template'] ?? '');
        if (!in_array($name, $allowed, true)) {
            throw new \InvalidArgumentException('unknown template');
        }

        // 变量仅作为数据传入，绝不拼接进模板源码；Twig 自动转义覆盖 HTML 上下文
        return $this->twig->render($name . '.html.twig', [
            'user'  => htmlspecialchars((string)($payload['user'] ?? ''), ENT_QUOTES, 'UTF-8'),
            'body'  => (string)($payload['body'] ?? ''),
        ]);
    }
}
