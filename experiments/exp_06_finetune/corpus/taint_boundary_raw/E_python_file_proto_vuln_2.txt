```python
# 被上层服务调用：从队列拉取业务消息，消息体为序列化后的任务对象
import pickle
import subprocess


def handle_message(payload: bytes):
    """消费队列中的业务消息，payload 为消息体字节流（外部生产者提交的内容）"""
    # 弱防御：仅做黑名单关键字检查，拦截明显的危险命令词
    blacklist = [b"rm -rf", b"shutdown", b"reboot"]
    for kw in blacklist:
        if kw in payload:
            raise ValueError("blocked keyword in message")

    task = pickle.loads(payload)  # 反序列化外部可控数据

    if task.get("type") == "shell":
        cmd = task["command"]
        # 二次弱校验：字符串替换过滤分号与反引号
        cmd = cmd.replace(";", "").replace("`", "")
        subprocess.run(cmd, shell=True)
```
