// Package richtext 提供富文本/日志格式化能力，被上层服务调用，
// 上层将外部提交的文本内容作为参数传入本模块。
package richtext

import (
	"encoding/xml"
	"fmt"
	"strings"
)

// RenderXML 将用户提交的富文本片段渲染为 XML 日志条目。
// title 与 body 均为外部提交内容，必须按不可信数据处理。
func RenderXML(title, body string) (string, error) {
	// 白名单校验：仅允许有限长度的标题，拒绝包含控制字符的内容
	if len(title) > 128 || len(body) > 8192 {
		return "", fmt.Errorf("content too long")
	}
	for _, r := range title + body {
		if r < 0x20 && r != '
' && r != '\t' {
			return "", fmt.Errorf("illegal control character")
		}
	}

	var sb strings.Builder
	sb.WriteString("<entry>")
	// xml.EscapeText 对 & < > ' " 及非法字符做完整转义，
	// 阻断通过闭合标签注入任意 XML/XXE 实体的路径
	sb.WriteString("<title>" + xml.EscapeText([]byte(title)) + "</title>")
	sb.WriteString("<body>" + xml.EscapeText([]byte(body)) + "</body>")
	sb.WriteString("</entry>")

	// 使用 encoding/xml 解码器做最终结构校验（禁用外部实体解析）
	dec := xml.NewDecoder(strings.NewReader(sb.String()))
	dec.Strict = true
	var probe map[string]interface{}
	if err := dec.Decode(&probe); err != nil {
		return "", fmt.Errorf("rendered xml invalid: %w", err)
	}
	return sb.String(), nil
}
