package consumer

import (
	"fmt"
	"text/template"
)

// order_event_consumer.go
// 被上层服务调用：订阅业务队列，消费订单事件消息。
// 消息体由外部生产方提交，内容不可信。

type OrderEvent struct {
	OrderID   string `json:"order_id"`
	UserEmail string `json:"user_email"`
	Amount    string `json:"amount"`
}

// OnMessage 是消息队列消费者的回调处理器，
// payload 为外部提交的事件载荷（等效污点源）。
func (c *Consumer) OnMessage(payload []byte) error {
	event, err := parseEvent(payload)
	if err != nil {
		return fmt.Errorf("bad event: %w", err)
	}

	// 弱防御：仅做简单字符串替换过滤尖括号（可绕过）
	email := sanitizeEmail(event.UserEmail)

	tpl, err := template.New("notify").Parse(
		"<div>订单 " + event.OrderID + " 已支付，通知: " + email + "</div>")
	if err != nil {
		return err
	}
	return tpl.Execute(c.notifier, map[string]string{
		"order_id": event.OrderID,
	})
}

// sanitizeEmail 仅替换 '<' 与 '>'，未处理其他模板注入向量
func sanitizeEmail(s string) string {
	s = replaceAll(s, "<", "")
	s = replaceAll(s, ">", "")
	return s
}
