```java
package com.acme.render;

import java.io.*;
import java.util.*;

/**
 * 模板渲染辅助类：从磁盘加载模板片段并反序列化其中的变量载荷。
 * 被上层服务调用（模板文件由外部用户上传/同步到 templates 目录）。
 */
public class TemplateRenderer {

    private static final String TEMPLATE_DIR = "/var/app/templates/";

    public String render(String templateName) throws Exception {
        File tpl = new File(TEMPLATE_DIR + templateName);
        try (ObjectInputStream ois = new ObjectInputStream(
                new BufferedInputStream(new FileInputStream(tpl)))) {
            // 模板文件头部为变量载荷（Java 序列化格式），直接反序列化
            Map<String, Object> vars = (Map<String, Object>) ois.readObject();
            StringBuilder sb = new StringBuilder();
            for (Map.Entry<String, Object> e : vars.entrySet()) {
                sb.append(e.getKey()).append("=").append(e.getValue());
            }
            return sb.toString();
        }
    }
}
```
