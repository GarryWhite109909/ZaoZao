```python
"""cache_codec.py — 缓存编解码模块，被上层服务调用以消费缓存同步队列消息。"""
import json
import sqlite3
import logging

logger = logging.getLogger(__name__)

# 允许被解码的缓存命名空间白名单
_ALLOWED_NAMESPACES = {"session", "profile", "cart"}

_conn = sqlite3.connect("cache_meta.db", check_same_thread=False)


def on_cache_message(message: dict) -> None:
    """消息队列消费者回调：message 为外部提交的载荷，必须按不可信数据处理。"""
    namespace = message.get("namespace")
    key = message.get("key")

    # 防御 1：白名单校验命名空间（类型正确、位置在 sink 前、覆盖全部插值点）
    if not isinstance(namespace, str) or namespace not in _ALLOWED_NAMESPACES:
        logger.warning("rejected unknown cache namespace: %r", namespace)
        return

    # 防御 2：键名严格字符集校验，杜绝引号/分号等 SQL 元字符
    if not isinstance(key, str) or not key.isascii() or not all(
        c.isalnum() or c in "-_." for c in key
    ):
        logger.warning("rejected malformed cache key: %r", key)
        return

    # 防御 3：参数化查询，值不参与 SQL 文本拼接
    row = _conn.execute(
        "SELECT encoded_value FROM cache_entries WHERE ns = ? AND cache_key = ?",
        (namespace, key),
    ).fetchone()

    if row is None:
        logger.info("cache miss for %s/%s", namespace, key)
        return

    payload = json.loads(row[0])
    logger.debug("decoded cache entry for %s/%s", namespace, key)
```
