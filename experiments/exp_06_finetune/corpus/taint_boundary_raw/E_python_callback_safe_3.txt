```python
"""日志格式化插件：消费消息队列中的富文本格式化任务。

被上层服务调用：worker 从队列拉取消息后调用 on_message。
"""

import logging
from string import Template

logger = logging.getLogger("fmt_worker")

# 仅允许的模板占位符白名单
_ALLOWED_FIELDS = frozenset({"title", "body", "author"})


def _sanitize(value: str) -> str:
    """转义 $ 字符，阻断 Template 占位符注入（CWE-1336）。"""
    return value.replace("$", "$$")


def on_message(payload: dict) -> str:
    """队列消费者回调：payload 为外部提交的格式化任务消息体。"""
    template_str = payload.get("template", "")
    fields = payload.get("fields", {})

    # 白名单校验：丢弃未知占位符，防止攻击者引入额外 substitution 点
    clean_fields = {k: v for k, v in fields.items() if k in _ALLOWED_FIELDS}

    # 防御点：对模板字符串本身做 $ 转义，使外部文本无法形成 ${...} 占位符
    safe_template = Template(_sanitize(template_str))

    rendered = safe_template.safe_substitute(clean_fields)
    logger.info("rendered log entry: %s", rendered)
    return rendered
```
