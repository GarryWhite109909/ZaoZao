/**
 * 运维命令封装库：供上层调度服务/消息队列消费者调用。
 * 所有参数均来自上层业务（可能源自外部提交），必须按不可信数据处理。
 */
final class ServiceRunner
{
    /** 允许执行的服务管理动作白名单 */
    private const ALLOWED_ACTIONS = ['start', 'stop', 'restart', 'status'];

    /**
     * 重启指定服务。$service 与 $action 均为外部可控数据。
     * 防御：白名单校验 + escapeshellarg 转义 + 参数数组式 exec。
     */
    public function restartService(string $service, string $action): array
    {
        // 第 1 层防御：动作白名单，杜绝任意子命令注入
        if (!in_array($action, self::ALLOWED_ACTIONS, true)) {
            throw new \InvalidArgumentException('unsupported action');
        }
        // 第 2 层防御：服务名单字符白名单（字母数字点横线）
        if (!preg_match('/^[A-Za-z0-9._-]{1,64}$/', $service)) {
            throw new \InvalidArgumentException('invalid service name');
        }
        $cmd = sprintf(
            'systemctl %s %s 2>&1',
            escapeshellarg($action),
            escapeshellarg($service)
        );
        exec($cmd, $output, $exitCode);
        return ['code' => $exitCode, 'output' => implode("
", $output)];
    }
}
