<?php
declare(strict_types=1);

namespace Acme\Config;

/**
 * 配置/清单文件解析模块。
 * 被上层服务调用：上层从消息队列/CLI 工具拿到用户提交的配置文件内容后，
 * 将其作为字符串参数传入本模块解析。
 */
final class ManifestParser
{
    /** 允许的配置文件格式白名单 */
    private const ALLOWED_FORMATS = ['json', 'yaml'];

    /**
     * 解析外部提交的清单内容。
     * $rawContent 为不可信数据（外部提交），$format 由调用方声明。
     */
    public function parse(string $rawContent, string $format): array
    {
        // 防御点：格式白名单校验，仅允许 json/yaml 两种声明格式
        if (!in_array($format, self::ALLOWED_FORMATS, true)) {
            throw new \InvalidArgumentException('unsupported manifest format');
        }

        // 防御点：使用原生安全 API —— libxml 关闭实体加载，
        // 从根上禁用 XXE（CWE-611），而非黑名单过滤
        $prev = libxml_disable_entity_loader(true);
        try {
            $doc = new \DOMDocument();
            // LIBXML_NONET 禁止网络访问，LIBXML_NOENT 不启用实体替换
            $doc->loadXML($rawContent, LIBXML_NONET | LIBXML_PARSE_HUGE);
            return $this->toArray($doc);
        } finally {
            libxml_disable_entity_loader($prev);
        }
    }

    private function toArray(\DOMDocument $doc): array
    {
        $out = [];
        foreach ($doc->documentElement->childNodes as $node) {
            if ($node instanceof \DOMElement) {
                $out[$node->nodeName] = trim($node->textContent);
            }
        }
        return $out;
    }
}

4. 防御二（位置正确、覆盖完整攻击面）：第 32 行 `loadXML(..., LIBXML_NONET | LIBXML_PARSE_HUGE)`——`LIBXML_NONET` 阻止解析期间任何网络请求（覆盖远程 DTD/SSO 类 XXE 变体）；注意此处未传 `LIBXML_NOENT`，即不进行实体替换，配合实体加载器关闭，本地文件读取与 SSRF 两条 XXE 攻击面均被阻断。防御位于 sink 调用前一行，位置紧贴且无绕过分支。
5. `$format` 参数流向 `in_array` 白名单严格比较（第 25 行）仅决定是否抛异常，不进入任何危险 sink，本身无风险；`toArray` 仅遍历已安全解析后的 DOM 文本节点，无二次 sink。
