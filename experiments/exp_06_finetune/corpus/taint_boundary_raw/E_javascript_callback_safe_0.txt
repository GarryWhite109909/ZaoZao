// outbound-webhook-dispatcher.js
// 被上层服务调用：消费消息队列中的 webhook 投递任务
const { Pool } = require('pg');
const axios = require('axios');

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

// 队列消费者回调：message.body 为外部提交的 webhook 配置载荷（不可信）
async function handleDelivery(message) {
  const { tenantId, eventType, payload } = message.body;

  // 白名单校验：eventType 只允许预定义集合，杜绝注入面
  const ALLOWED_EVENTS = ['order.created', 'order.refunded', 'user.signup'];
  if (!ALLOWED_EVENTS.includes(eventType)) {
    throw new Error(`event type not permitted: ${eventType}`);
  }

  // 参数化查询：tenantId 与 eventType 作为占位符参数传入，永不拼接 SQL
  const { rows } = await pool.query(
    'SELECT endpoint_url, secret FROM webhook_endpoints WHERE tenant_id = $1 AND event_type = $2',
    [tenantId, eventType]
  );

  for (const row of rows) {
    await axios.post(row.endpoint_url, payload, {
      headers: { 'X-Webhook-Signature': signPayload(payload, row.secret) }
    });
  }
}

function signPayload(body, secret) {
  return require('crypto').createHmac('sha256', secret).update(JSON.stringify(body)).digest('hex');
}

module.exports = { handleDelivery };
