// 宿主服务在收到外部提交的插件配置后，通过本包的公开函数加载并执行钩子。
package hookmgr

import (
	"fmt"
	"os/exec"
)

// AllowedPlugins 是经过安全评审的插件白名单，仅允许执行这些已签名的钩子程序。
var AllowedPlugins = map[string]bool{
	"backup":    true,
	"notify":    true,
	"compress":  true,
	"reportgen": true,
}

// RunHook 由宿主上层调用：hookName 与 payload 来自外部提交的插件配置，
// 必须按不可信数据处理。payload 作为参数数组传递给钩子进程，不做 shell 拼接。
func RunHook(hookName string, payload []string) error {
	// 白名单校验：拒绝任何不在评审清单内的钩子名，阻断任意命令注入面。
	if !AllowedPlugins[hookName] {
		return fmt.Errorf("hook %q not allowed", hookName)
	}

	// exec.Command 使用原生 API，不经过 shell 解释；
	// payload 以独立参数元素传入，即使包含分号、反引号、$() 等元字符也只会被当作字面量。
	cmd := exec.Command(hookName, payload...)
	cmd.Dir = "/var/lib/hooks"
	if out, err := cmd.CombinedOutput(); err != nil {
		return fmt.Errorf("hook failed: %v: %s", err, out)
	}
	return nil
}
