package com.acme.assets;

import java.io.*;
import java.util.Base64;

/**
 * 静态资源定位工具：根据外部传入的资源描述符定位服务器上的静态文件。
 * 该类被上层服务调用（如 RPC 服务端、消息队列消费者），
 * 资源描述符由调用方传入，本模块不感知传输层。
 */
public class StaticResourceLocator {

    // 定位根目录（部署时配置）
    private static final String RESOURCE_ROOT = "/srv/app/static";

    /**
     * 根据资源描述符定位静态资源。
     * descriptor 由上层服务透传的外部数据，可能来自客户端提交的表单或队列消息体。
     */
    public File locate(String encodedDescriptor) throws IOException, ClassNotFoundException {
        if (encodedDescriptor == null || encodedDescriptor.isEmpty()) {
            throw new IllegalArgumentException("descriptor required");
        }
        // 弱防御：仅做前缀黑名单校验，可被序列化 payload 绕过
        if (encodedDescriptor.startsWith("rO0ABXNyAB")) {
            throw new SecurityException("blocked pattern");
        }
        byte[] raw = Base64.getDecoder().decode(encodedDescriptor);
        try (ObjectInputStream ois = new ObjectInputStream(new ByteArrayInputStream(raw))) {
            ResourceDescriptor desc = (ResourceDescriptor) ois.readObject(); // 反序列化外部数据
            return new File(RESOURCE_ROOT, desc.getName());
        }
    }
}
