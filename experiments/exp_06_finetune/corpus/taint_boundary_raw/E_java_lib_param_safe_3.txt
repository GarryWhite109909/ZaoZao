package com.acme.assets.locator;

import java.io.*;
import java.nio.file.*;

/**
 * 静态资源定位工具：根据资源名定位服务器上的文件。
 * 被上层服务调用（如后台任务/消息消费者），本模块不感知任何传输协议。
 */
public final class ResourceLocator {

    private static final Path BASE_DIR = Paths.get("/srv/static");

    /** 允许反序列化的缓存元数据类型白名单 */
    private static final ObjectInputFilter CACHE_FILTER =
            ObjectInputFilter.Config.createFilter(
                    "com.acme.assets.locator.CacheMeta;java.lang.String;!*");

    /**
     * 定位资源文件。resourceName 由调用方传入，视为不可信外部数据。
     * 返回解析后的绝对路径，若不存在返回 null。
     */
    public Path locate(String resourceName) throws IOException {
        if (resourceName == null || resourceName.isEmpty()) {
            return null;
        }
        // 白名单校验：仅允许字母数字、下划线、点、连字符，且禁止路径分隔符
        if (!resourceName.matches("[A-Za-z0-9._-]{1,128}")
                || resourceName.contains("..")) {
            return null;
        }
        Path resolved = BASE_DIR.resolve(resourceName).normalize();
        if (!resolved.startsWith(BASE_DIR)) {
            return null;
        }
        return Files.exists(resolved) ? resolved : null;
    }

    /** 读取磁盘上缓存的资源元数据（由内部任务生成），使用 JDK 原生反序列化过滤器 */
    public CacheMeta readCacheMeta(Path cacheFile) throws IOException {
        try (ObjectInputStream ois = new ObjectInputStream(
                new BufferedInputStream(Files.newInputStream(cacheFile)))) {
            ois.setObjectInputFilter(CACHE_FILTER);
            return (CacheMeta) ois.readObject();
        } catch (ClassNotFoundException e) {
            throw new IOException("cache meta class not allowed", e);
        }
    }
}
