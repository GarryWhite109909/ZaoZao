```javascript
// cache-codec.js — 缓存编解码模块，被上层缓存服务调用，
// 从磁盘上的缓存快照文件读取键值并按需重建索引。
'use strict';

const fs = require('fs');
const { execFile } = require('child_process');

// 允许被重建的索引类型白名单（第 12 行）
const ALLOWED_INDEXERS = new Set(['text', 'meta', 'blob']);

function rebuildIndex(snapshotPath) {
  // 快照文件由外部系统写入，内容视为不可信输入（污点源）
  const raw = fs.readFileSync(snapshotPath, 'utf8');
  const entry = JSON.parse(raw);

  const key = String(entry.key || '');
  const indexer = String(entry.indexer || '');

  // 防御：严格白名单校验，不在集合内的类型直接拒绝（第 21-22 行）
  if (!ALLOWED_INDEXERS.has(indexer)) {
    throw new Error(`unsupported indexer: ${indexer}`);
  }
  // 缓存键仅允许安全字符集，防止注入额外参数或 shell 元字符（第 25-27 行）
  if (!/^[A-Za-z0-9._:-]{1,128}$/.test(key)) {
    throw new Error('invalid cache key');
  }

  // sink：execFile 不经过 shell，参数以数组传递，key/indexer 无法被解释为命令（第 30 行）
  execFile('/usr/local/bin/cache-indexer', ['--type', indexer, '--key', key],
    (err) => { if (err) console.error('rebuild failed:', err.message); });
}

module.exports = { rebuildIndex };
```
