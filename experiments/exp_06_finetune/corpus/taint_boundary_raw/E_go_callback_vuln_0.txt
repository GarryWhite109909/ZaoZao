// Package archiver 处理归档任务队列消息，由 worker 服务消费。
package archiver

import (
	"fmt"
	"net/http"
	"net/url"
)

// ArchiveEvent 是消息队列载荷，由外部提交方写入（如用户上传归档后触发）。
type ArchiveEvent struct {
	ArchiveName string `json:"archive_name"`
	CallbackURL string `json:"callback_url"`
}

// HandleArchiveEvent 是队列消费者回调：broker 每收到一条消息即调用本函数，
// event 字段完全来自外部提交内容，必须视为不可信数据。
func HandleArchiveEvent(event ArchiveEvent) error {
	// 弱防御：仅黑名单拒绝内网常见地址前缀
	blocked := []string{"127.0.0.1", "localhost", "169.254", "10.", "192.168."}
	for _, b := range blocked {
		if contains(event.CallbackURL, b) {
			return fmt.Errorf("blocked callback host")
		}
	}

	// 归档名仅用于日志/存储路径，回调地址直接发起通知请求
	target, err := url.Parse(event.CallbackURL)
	if err != nil {
		return err
	}
	q := target.Query()
	q.Set("archive", event.ArchiveName)
	target.RawQuery = q.Encode()

	// sink：使用外部可控 URL 发起服务端请求 —— SSRF
	resp, err := http.Get(target.String())
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	return nil
}

func contains(s, sub string) bool {
	return len(s) >= len(sub) && (func() bool {
		for i := 0; i+len(sub) <= len(s); i++ {
			if s[i:i+len(sub)] == sub {
				return true
			}
		}
		return false
	})()
}
