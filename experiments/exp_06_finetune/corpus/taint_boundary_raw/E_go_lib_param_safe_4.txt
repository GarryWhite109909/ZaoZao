// Package cachecodec 提供缓存条目的编解码与回源校验能力。
// 本模块被上层服务调用（如缓存刷新 worker、消息队列消费者），
// 外部数据通过函数参数传入，本文件不注册任何路由或 HTTP 处理器。
package cachecodec

import (
	"fmt"
	"net"
	"net/url"
	"strings"
)

// CacheEntry 表示一条待回源校验的缓存记录。
type CacheEntry struct {
	Key   string
	Value []byte
}

// allowedSchemes 是回源地址的协议白名单。
var allowedSchemes = map[string]bool{"http": true, "https": true}

// ValidateOrigin 校验缓存条目中携带的回源 URL，并对其发起探测请求。
// entry.Key / entry.Value 由上层从外部数据反序列化而来，视为不可信输入。
func ValidateOrigin(entry CacheEntry) (string, error) {
	raw := strings.TrimSpace(string(entry.Value))
	u, err := url.Parse(raw)
	if err != nil {
		return "", fmt.Errorf("invalid origin url: %w", err)
	}
	// 防御 1：协议白名单，仅允许 http/https，阻断 file/gopher/dict 等 SSRF 常用协议
	if !allowedSchemes[u.Scheme] {
		return "", fmt.Errorf("scheme not allowed: %s", u.Scheme)
	}
	host := u.Hostname()
	// 防御 2：主机白名单，仅允许内部回源域名后缀，且拒绝直接使用 IP 字面量
	if host == "" || net.ParseIP(host) != nil || !strings.HasSuffix(host, ".internal.example.com") {
		return "", fmt.Errorf("host not allowed: %s", host)
	}
	// 防御 3：解析后的 IP 必须落在内网网段，防止 DNS rebinding 指向公网/元数据地址
	ips, err := net.LookupIP(host)
	if err != nil {
		return "", fmt.Errorf("resolve failed: %w", err)
	}
	for _, ip := range ips {
		if !ip.IsPrivate() {
			return "", fmt.Errorf("resolved ip not private: %s", ip.String())
		}
	}
	return probe(u.String())
}

// probe 使用标准库 http 客户端访问已通过三重校验的 URL。
func probe(target string) (string, error) {
	resp, err := httpGet(target)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	return "ok", nil
}
