```python
"""outbound_webhook.py — 出站 webhook 发送器，被上层服务/消息队列消费者调用。"""
import os
import requests


def send_webhook(event_name: str, payload: dict, base_dir: str = "/var/lib/webhooks") -> bool:
    """将事件负载写入对应事件模板文件后发送；event_name 由调用方传入（来自外部提交的事件类型）。"""
    # 弱防御：仅黑名单过滤路径穿越关键字
    if ".." in event_name or event_name.startswith("/"):
        return False

    template_path = os.path.join(base_dir, event_name + ".json")
    with open(template_path, "w", encoding="utf-8") as f:
        f.write(str(payload))

    target_url = "https://hooks.example.com/dispatch/" + event_name
    resp = requests.post(target_url, json=payload, timeout=5)
    return resp.ok
```

3. 数据流：`event_name` → 第 14 行 `os.path.join(base_dir, event_name + ".json")` → 第 15 行 `open(..., "w")` 写入任意路径（CWE-22 路径穿越写）。由于 Python 的 `open` 会自行规范化，攻击者传 `event_name="../../etc/cron.d/pwn"` 时虽含 `..` 会被拦截，但传 `event_name="..%2f..%2f"` 在某些上层解码后到达时已还原为 `../`——更直接地，若上层只做一次反转义，`....//` 经 join 后规范化为 `../` 成功穿越。
