```javascript
// report-export-worker.js
// 报表导出工具：由上层 API 服务通过消息队列下发导出任务，
// 本模块作为消费者回调处理任务消息，不直接暴露任何网络入口。
'use strict';

const path = require('path');
const fs = require('fs');
const { worker } = require('./mq-client'); // 内部消息队列客户端

// 允许导出的报表类型白名单（唯一合法的模板标识集合）
const ALLOWED_REPORTS = new Set(['sales', 'inventory', 'finance']);

// 导出文件统一存放目录，禁止逃逸
const EXPORT_ROOT = path.resolve(__dirname, 'exports');

worker.consume('report.export', async (task) => {
  // task = { reportId: string, requestId: string }，来自外部提交的任务载荷
  const { reportId, requestId } = task;

  // 防御 1：白名单校验模板标识，非法值直接拒绝
  if (typeof reportId !== 'string' || !ALLOWED_REPORTS.has(reportId)) {
    throw new Error(`unknown reportId: ${reportId}`);
  }

  // 防御 2：requestId 仅允许字母数字与连字符，阻断路径分隔符与 .. 序列
  if (!/^[A-Za-z0-9-]{1,64}$/.test(requestId)) {
    throw new Error('invalid requestId');
  }

  const safeName = `${reportId}-${requestId}.csv`;
  const targetPath = path.join(EXPORT_ROOT, safeName);

  // 防御 3：解析后确认最终路径仍位于导出根目录内（防符号链接/拼接绕过）
  if (!path.resolve(targetPath).startsWith(EXPORT_ROOT + path.sep)) {
    throw new Error('path traversal detected');
  }

  const rows = await generateReport(reportId); // 内部查询，参数为白名单值
  fs.writeFileSync(targetPath, rows.join('
'));
});
```
