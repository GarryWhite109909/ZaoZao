package com.ops.exec;

import java.util.List;
import java.util.Map;
import java.util.Set;

/**
 * 运维命令封装库：被消息队列消费者调用，处理部署/巡检指令消息。
 * 消息体为 JSON 载荷，由上层消费者反序列化后传入 handle()。
 */
public class OpsCommandRunner {

    // 仅允许的运维操作白名单，每个操作绑定固定的参数模板
    private static final Map<String, List<String>> ALLOWED = Map.of(
        "disk_usage",  List.of("df", "-h"),
        "service_status", List.of("systemctl", "status"),
        "log_tail",    List.of("tail", "-n", "100")
    );

    private static final Set<String> VALID_UNITS =
        Set.of("nginx", "redis", "postgres", "app-server");

    /**
     * 回调入口：message 为队列消息体（外部提交的载荷），不可信。
     * message 结构: {"action": "...", "target": "..."}
     */
    public String handle(Map<String, String> message) throws Exception {
        String action = message.get("action");
        String target = message.get("target");

        List<String> template = ALLOWED.get(action);          // 第 27 行：白名单查表
        if (template == null) {
            throw new IllegalArgumentException("unknown action");
        }
        if (!VALID_UNITS.contains(target)) {                  // 第 31 行：目标白名单
            throw new IllegalArgumentException("invalid target");
        }

        ProcessBuilder pb = new ProcessBuilder();             // 第 35 行
        pb.command(appendArg(template, target));              // 参数数组，无 shell 解析
        pb.redirectErrorStream(true);
        Process p = pb.start();
        return new String(p.getInputStream().readAllBytes());
    }

    private static List<String> appendArg(List<String> t, String arg) {
        return new java.util.ArrayList<>(t) {{ add(arg); }};
    }
}
