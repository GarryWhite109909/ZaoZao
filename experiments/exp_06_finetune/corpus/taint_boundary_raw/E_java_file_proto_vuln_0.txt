package com.acme.cache.codec;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.Map;

/**
 * 缓存编解码模块：从磁盘上的缓存快照文件恢复键值对。
 * 快照由上层同步服务按行写入，格式为 key=value，value 为 HTML 片段。
 * 本模块被缓存预热任务调用。
 */
public final class SnapshotCodec {

    public static Map<String, String> decode(Path snapshotFile) throws IOException {
        Map<String, String> cache = new HashMap<>();
        for (String line : Files.readAllLines(snapshotFile, StandardCharsets.UTF_8)) {
            int idx = line.indexOf('=');
            if (idx <= 0) continue;
            String key = line.substring(0, idx);
            // 弱防御：仅剔除 <script> 字面量
            String value = line.substring(idx + 1).replace("<script>", "");
            cache.put(key, value);
        }
        return cache;
    }

    /** 渲染缓存条目到管理端页面片段 */
    public static String render(String key, Map<String, String> cache) {
        return "<div class='entry'>" + cache.get(key) + "</div>";
    }
}
