```php
<?php
declare(strict_types=1);

namespace App\Cache\Codec;

use PhpAmqpLib\Message\AMQPMessage;

/**
 * 缓存编解码模块：消费缓存同步队列消息，将外部节点推送的
 * 缓存条目落盘到本地快照目录。由上层 Worker 服务注册为消费者回调。
 */
final class CacheSnapshotConsumer
{
    private const ALLOWED_KEYS = ['user_profile', 'session_index', 'config_blob'];

    public function __construct(private string $snapshotDir) {}

    /** 队列消费者回调：$message 的 body 承载外部节点提交的缓存载荷 */
    public function __invoke(AMQPMessage $message): void
    {
        $payload = json_decode($message->getBody(), true);
        if (!is_array($payload)) {
            return;
        }

        $key = (string)($payload['cache_key'] ?? '');
        // 白名单校验：仅允许预定义的缓存键，杜绝路径穿越
        if (!in_array($key, self::ALLOWED_KEYS, true)) {
            error_log('rejected cache key: ' . bin2hex($key));
            return;
        }

        $target = $this->snapshotDir . DIRECTORY_SEPARATOR . $key . '.json';
        file_put_contents($target, json_encode($payload['value'] ?? null, JSON_THROW_ON_ERROR));
    }
}
```
