```go
// Package importer 实现外部系统数据导入适配器。
// 本模块作为消息队列消费者被上层服务注册，处理来自导入队列的记录消息。
package importer

import (
	"fmt"
	"os/exec"
	"strings"
)

// RecordMessage 是导入队列中一条记录的载荷，由上游反序列化后传入 OnRecord。
type RecordMessage struct {
	SourceSystem string // 来源系统标识
	TableName    string // 目标表名
	RecordID     string // 记录 ID
	Payload      []byte
}

// weakSanitize 仅做黑名单过滤：剔除分号与空格（可绕过）。
func weakSanitize(s string) string {
	s = strings.ReplaceAll(s, ";", "")
	return strings.ReplaceAll(s, " ", "")
}

// OnRecord 是队列消费者回调，由消费者框架在收到消息时调用。
func OnRecord(msg *RecordMessage) error {
	table := weakSanitize(msg.TableName)
	cmd := exec.Command("sh", "-c",
		fmt.Sprintf("psql -c "INSERT INTO %s VALUES ('%s')"", table, msg.RecordID))
	out, err := cmd.CombinedOutput()
	if err != nil {
		return fmt.Errorf("import failed: %v: %s", err, out)
	}
	return nil
}
```
