```python
"""outbound_webhook.py — 出站 webhook 发送器，被上层服务/消息队列消费者调用。"""
import os
import urllib.parse

# 允许推送的固定资源集合（白名单）
_ALLOWED_EVENTS = {"build.finished", "deploy.done", "alert.fired"}
_BASE_DIR = "/var/lib/webhook/templates"


def send_event(event_name: str, payload: str) -> bool:
    """向预配置端点发送事件通知；event_name 来自外部提交内容，视为不可信。"""
    if event_name not in _ALLOWED_EVENTS:
        return False  # 白名单校验：拒绝一切未登记的事件名

    # 模板文件路径由白名单键映射生成，不拼接用户输入
    template_path = os.path.join(_BASE_DIR, event_name.replace(".", "_") + ".json")
    if not os.path.realpath(template_path).startswith(os.path.realpath(_BASE_DIR) + os.sep):
        return False  # 纵深防御：realpath 归一化后确认仍在基目录内

    endpoint = "https://hooks.internal.example.com/v1/" + urllib.parse.quote(event_name, safe="")
    # requests.post(endpoint, json=payload) —— 实际发送逻辑由上层注入
    return True
```
