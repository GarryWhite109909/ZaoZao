```python
# consumer/message_handler.py
# 消息消费者业务处理器：由上层服务从队列拉取业务消息后调用本模块处理。
import json
import logging

logger = logging.getLogger(__name__)

# 允许执行的业务动作白名单（第 10 行）
ALLOWED_ACTIONS = {"refresh_cache", "sync_inventory", "send_report"}

# 动作到内部处理函数的映射，全部为模块内静态函数（第 13 行）
ACTION_HANDLERS = {
    "refresh_cache": lambda payload: logger.info("cache refreshed: %s", payload.get("scope")),
    "sync_inventory": lambda payload: logger.info("inventory synced: %s", payload.get("warehouse")),
    "send_report": lambda payload: logger.info("report sent: %s", payload.get("report_id")),
}


def handle_message(raw_bytes: bytes) -> None:
    """处理队列消息体（字节流），消息内容视为不可信外部输入。"""
    try:
        # 第 21 行：反序列化外部消息体 —— 污点源
        message = json.loads(raw_bytes.decode("utf-8"))
    except (ValueError, UnicodeDecodeError):
        logger.warning("malformed message dropped")
        return

    action = message.get("action")
    # 第 27 行：白名单校验，未在 ALLOWED_ACTIONS 中的动作直接拒绝
    if not isinstance(action, str) or action not in ALLOWED_ACTIONS:
        logger.warning("action rejected: %r", action)
        return

    handler = ACTION_HANDLERS[action]
    # 第 33 行：仅调用白名单映射出的静态函数，payload 只进入日志/业务参数
    handler(message.get("payload") or {})
```
