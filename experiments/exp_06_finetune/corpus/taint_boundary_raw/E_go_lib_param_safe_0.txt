// 本模块被上层服务调用：上层将外部提交的缓存键/值传入后，
// 由本包负责校验、编码并写入磁盘缓存文件。
package cachecodez

import (
	"fmt"
	"path/filepath"
	"strings"
)

// allowedKeyChars 是缓存键的白名单字符集（字母、数字、下划线、点）。
const allowedKeyChars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_."

// SaveCacheEntry 将 key/value 编码后写入缓存目录下的对应文件。
// key 与 value 均来自上层调用者，视为不可信数据。
func SaveCacheEntry(baseDir, key, value string) error {
	if !validKey(key) {
		return fmt.Errorf("invalid cache key")
	}
	// filepath.Base 消除任何路径分隔符残留，双重防御
	safeName := filepath.Base(key)
	path := filepath.Join(baseDir, safeName)
	return writeFile(path, []byte(value))
}

func validKey(k string) bool {
	if k == "" || len(k) > 64 {
		return false
	}
	for _, c := range k {
		if !strings.ContainsRune(allowedKeyChars, c) {
			return false
		}
	}
	return true
}

func writeFile(path string, data []byte) error {
	f, err := osCreate(path)
	if err != nil {
		return err
	}
	defer f.Close()
	_, err = f.Write(data)
	return err
}

// osCreate 为可测试性抽象的 os.Create 包装。
var osCreate = func(p string) (*osFile, error) { return osOpenFile(p) }
