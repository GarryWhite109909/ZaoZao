```go
// Package reportexport 报表导出模块：被上层 API 服务以用户筛选条件调用，
// 从磁盘读取报表模板配置文件并渲染导出。
package reportexport

import (
	"bytes"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"text/template"
)

// allowedTemplates 是模板白名单：只允许预置的报表模板名，
// 模板文件本身由运维部署，不来自外部提交内容。
var allowedTemplates = map[string]bool{
	"sales_summary.tmpl":   true,
	"inventory_daily.tmpl": true,
}

// ExportReport 读取指定模板名的配置文件，用筛选数据渲染后返回导出字节流。
func ExportReport(templateName string, filters map[string]string) ([]byte, error) {
	// 防御点：白名单校验，拒绝任何不在白名单内的模板名
	if !allowedTemplates[templateName] {
		return nil, fmt.Errorf("template not allowed: %s", templateName)
	}
	// 二次防御：路径净化，阻断目录穿越拼接
	clean := filepath.Base(filepath.Clean("/etc/report/templates/" + templateName))
	if strings.Contains(clean, "..") || !strings.HasSuffix(clean, ".tmpl") {
		return nil, fmt.Errorf("invalid template path")
	}
	raw, err := os.ReadFile(clean)
	if err != nil {
		return nil, err
	}
	tpl, err := template.New("report").Parse(string(raw))
	if err != nil {
		return nil, err
	}
	var buf bytes.Buffer
	if err := tpl.Execute(&buf, filters); err != nil {
		return nil, err
	}
	return buf.Bytes(), nil
}
```
