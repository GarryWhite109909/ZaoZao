package com.acme.render;

import java.io.IOException;
import java.io.ObjectInputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Base64;
import java.util.Collections;
import java.util.HashSet;
import java.util.Set;

/**
 * 模板渲染辅助类：从磁盘上的模板描述文件加载变量映射，
 * 供上层渲染服务消费（文件由外部提交方上传，内容不可信）。
 */
public final class TemplateContextLoader {

    // 白名单：仅允许这些模板变量名进入渲染上下文
    private static final Set<String> ALLOWED_KEYS =
            new HashSet<>(java.util.Arrays.asList("title", "body", "footer", "user.name"));

    /**
     * 读取 base64 编码的序列化属性文件并构建只读上下文。
     */
    public static java.util.Map<String, String> load(Path file) throws IOException {
        byte[] raw = Base64.getMimeDecoder().decode(Files.readAllBytes(file)); // 第 24 行：污点源
        java.util.Properties props = new java.util.Properties();
        props.load(new java.io.ByteArrayInputStream(raw));                     // 第 26 行：解析不可信内容

        java.util.Map<String, String> ctx = new java.util.HashMap<>();
        for (String name : props.stringPropertyNames()) {                      // 第 28 行
            if (!ALLOWED_KEYS.contains(name)) {                                // 第 29 行：白名单校验
                continue;                                                      // 丢弃未授权键
            }
            String value = props.getProperty(name, "");
            if (value.length() > 4096) {                                       // 长度上限
                throw new IOException("template variable too large");
            }
            ctx.put(name, value);                                              // 仅白名单键入上下文
        }
        return Collections.unmodifiableMap(ctx);
    }
}

5. **第二入口检查**：类中唯一公开方法是 `load(Path)`，无其他重载、无静态初始化读取外部数据、无常驻监听；`ALLOWED_KEYS` 为编译期常量不可篡改。不存在替代通道使未校验数据到达 sink。
