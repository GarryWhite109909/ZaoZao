```java
package com.acme.confighub.parser;

import org.apache.velocity.app.VelocityEngine;
import org.apache.velocity.VelocityContext;
import org.apache.velocity.Template;
import java.io.StringWriter;
import java.util.Map;
import java.util.Set;

/**
 * 配置清单渲染模块：由消息队列消费者回调触发，
 * 消费 config.submitted 主题，载荷为外部提交的 YAML/JSON 清单内容。
 */
public class ManifestRenderService {

    // 仅允许渲染内置模板，禁止外部指定模板名或模板体
    private static final Set<String> ALLOWED_TEMPLATES = Set.of("service-default", "job-default", "cron-default");

    private final VelocityEngine engine;
    private final TemplateStore templateStore; // 内部只读模板仓库

    public ManifestRenderService(VelocityEngine engine, TemplateStore templateStore) {
        this.engine = engine;
        this.templateStore = templateStore;
    }

    // 消息队列消费者回调入口：payload 为外部提交的配置清单（不可信数据）
    public void onConfigSubmitted(String payload) {
        Map<String, Object> fields = ConfigYaml.safeParse(payload);   // 第 24 行：解析为纯键值对
        String tplName = String.valueOf(fields.getOrDefault("template", "service-default"));
        if (!ALLOWED_TEMPLATES.contains(tplName)) {                    // 第 26 行：白名单校验
            throw new IllegalArgumentException("template not allowed: " + tplName);
        }
        VelocityContext ctx = new VelocityContext(fields);             // 外部字段仅作为数据变量
        Template tpl = templateStore.loadSigned(tplName);              // 模板来自签名校验的内部仓库
        StringWriter out = new StringWriter();
        engine.evaluate(ctx, out, "manifest", tpl.getData());          // 第 32 行：sink
    }
}
```
