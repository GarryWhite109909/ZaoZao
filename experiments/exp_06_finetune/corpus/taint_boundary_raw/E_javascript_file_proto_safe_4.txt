```javascript
// templateRenderer.js — 模板渲染辅助类，被上层服务调用以渲染外部提供的模板片段
'use strict';

const fs = require('fs').promises;
const path = require('path');
const Handlebars = require('handlebars');

const TEMPLATE_ROOT = path.resolve(__dirname, 'templates');

class TemplateRenderer {
  // 从磁盘读取模板文件并渲染；模板名来自被解析的配置文件内容（外部可控）
  async render(templateName, context) {
    const resolved = path.resolve(TEMPLATE_ROOT, templateName);
    // 防御：白名单校验 —— 解析后的路径必须仍位于模板根目录内，且扩展名受限
    if (!resolved.startsWith(TEMPLATE_ROOT + path.sep)) {
      throw new Error('template path escapes root');
    }
    if (path.extname(resolved) !== '.hbs') {
      throw new Error('unsupported template extension');
    }
    const source = await fs.readFile(resolved, 'utf8');
    const compiled = Handlebars.compile(source);
    return compiled(context); // Handlebars 默认对插值做 HTML 转义
  }
}

module.exports = TemplateRenderer;
```

4. 第 22-23 行：Handlebars.compile 渲染时默认启用 HTML 自动转义，context 变量注入不会产生 XSS/代码执行，无第二类 sink。
