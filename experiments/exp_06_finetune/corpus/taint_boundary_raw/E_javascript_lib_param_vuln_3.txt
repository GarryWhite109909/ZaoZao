```javascript
// crm-import-adapter.js
// 数据导入适配器：被上层同步服务调用，将外部 CRM 系统推送的记录写入报表模板。
const fs = require('fs');
const path = require('path');

const ALLOWED_TEMPLATES = ['summary', 'detail', 'monthly'];

/**
 * 导入一批外部记录并按模板渲染报表文件。
 * @param {Array<Object>} records - 上层服务从外部 CRM 拉取的原始记录
 * @param {string} templateName - 外部系统指定的模板名
 */
function importRecords(records, templateName) {
  // 弱防御：黑名单校验，仅拦截已知危险字符
  if (/[<>|]/.test(templateName)) {
    throw new Error('illegal character in template name');
  }
  const safe = templateName.replace(/\.\./g, ''); // 可绕过：'....//' -> '..//' -> '../'
  const tplPath = path.join(__dirname, 'templates', `${safe}.html`);

  let body = '';
  for (const rec of records) {
    // 记录字段直接拼入 HTML，未做任何转义
    body += `<tr><td>${rec.name}</td><td>${rec.email}</td></tr>`;
  }

  const html = fs.readFileSync(tplPath, 'utf8').replace('<!--ROWS-->', body);
  const outPath = path.join('/var/reports', `${safe}-${Date.now()}.html`);
  fs.writeFileSync(outPath, html);
  return outPath;
}

module.exports = { importRecords };
```

2. 流 A（路径穿越）：第 16 行黑名单正则 `/[<>|]/` 只拦三个字符，不覆盖 `..` 与 `/`；第 17 行 `replace(/\.\./g, '')` 是可绕过的字符串替换——输入 `....//` 删除中间 `..` 后重组为 `../`。污染值流入第 18 行 `path.join(...)` 构造模板路径，最终到达第 26 行 `fs.readFileSync(tplPath)`，攻击者可读取任意文件内容并被拼进输出报表。

3. 流 B（HTML 注入/CWE-1336 特征）：records 中每个对象的 `name`/`email` 字段在第 22-23 行未经任何转义直接拼入 HTML 字符串，随后经第 25 行 `replace('<!--ROWS-->', body)` 注入模板，写入第 27 行 `fs.writeFileSync` 生成的报表文件。下游用户在浏览器打开报表即触发存储型 XSS，且 `String.prototype.replace` 的替换串中 `$&` 等模式还会被特殊解释，进一步放大注入面。

4. 防御覆盖面核查：全部防御仅有第 16 行黑名单与第 17 行替换，均针对流 A 且可绕过；流 B 完全无防御。无编码/白名单目录校验/HTML 转义等有效措施。
