package com.acme.mq.handler;

import org.owasp.encoder.Encode;
import java.util.Map;

/**
 * 消息消费者业务处理器：由上层消息服务从队列拉取业务消息后调用，
 * 本模块不注册任何路由，外部数据仅通过函数参数进入。
 */
public class NotificationMessageHandler {

    /** 允许的通知渠道白名单 */
    private static final Map<String, String> ALLOWED_CHANNELS = Map.of(
            "email", "email",
            "sms", "sms",
            "push", "push"
    );

    /**
     * 处理一条通知消息。payload 由队列消费方传入，视为不可信外部数据。
     * 返回渲染后的通知正文，供下游投递。
     */
    public String handle(String channel, String payload) {
        // 防御1：渠道参数走严格白名单校验，非白名单值直接拒绝
        if (channel == null || !ALLOWED_CHANNELS.containsKey(channel)) {
            throw new IllegalArgumentException("unsupported channel");
        }
        // 防御2：payload 在拼入 HTML 通知模板前经 OWASP Encoder 做 HTML 实体转义
        String safePayload = Encode.forHtml(payload == null ? "" : payload);
        return "<div class='notice' data-channel='" + Encode.forHtmlAttribute(channel)
                + "'>" + safePayload + "</div>";
    }
}
