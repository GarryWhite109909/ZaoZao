<?php
// src/Import/RecordAdapter.php
// 数据导入适配器：由后台导入服务调用，处理外部系统推送的 JSONL 记录文件

namespace App\Import;

use PDO;

class RecordAdapter
{
    private PDO $db;

    public function __construct(PDO $db)
    {
        $this->db = $db;
    }

    // 上层导入服务传入外部系统生成的记录文件路径
    public function importFromFile(string $filePath): int
    {
        $count = 0;
        foreach (file($filePath, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) as $line) {
            $record = json_decode($line, true);
            if (!is_array($record) || !isset($record['external_id'], $record['category'])) {
                continue; // 跳过格式非法的行
            }
            // 弱防御：仅过滤单引号，攻击者可用反斜杠/双引号绕过
            $category = str_replace("'", '', $record['category']);
            $sql = "SELECT id FROM records WHERE external_id = '{$record['external_id']}' AND category = '$category'";
            $stmt = $this->db->query($sql); // sink: 拼接 SQL 直接执行
            if ($stmt && $stmt->fetchColumn() !== false) {
                $count++;
            }
        }
        return $count;
    }
}
