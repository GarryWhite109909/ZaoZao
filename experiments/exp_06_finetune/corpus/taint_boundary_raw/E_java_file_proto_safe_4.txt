```java
package com.acme.cache.codec;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.regex.Pattern;

/**
 * 缓存编解码模块：被上层缓存服务调用，读取磁盘上的缓存快照文件
 * （由对端节点通过同步协议写入，内容视为不可信外部数据），
 * 解析出 key/value 后重建本地缓存条目。
 */
public final class SnapshotCodec {

    // 仅允许字母数字、点、下划线、连字符组成的缓存键
    private static final Pattern SAFE_KEY = Pattern.compile("^[A-Za-z0-9._-]{1,128}$");

    public void load(Path snapshotFile) throws IOException {
        String content = new String(Files.readAllBytes(snapshotFile), StandardCharsets.UTF_8);
        for (String line : content.split("
")) {
            int idx = line.indexOf('=');
            if (idx <= 0) continue;
            String key = line.substring(0, idx).trim();
            String value = line.substring(idx + 1);
            if (!SAFE_KEY.matcher(key).matches()) {
                continue; // 拒绝非法键，不进入后续操作
            }
            CacheRegistry.register(key, value);
        }
    }
}
```

4. value 未做校验，但 register 的 value 参数仅作为纯数据存储，不进入命令执行、SQL、模板渲染等危险 sink，故 value 流不存在可达的危险终点。
