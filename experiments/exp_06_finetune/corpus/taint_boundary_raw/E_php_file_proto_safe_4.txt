```php
<?php
// ArchiveExtractor.php — 归档解包模块，被上层后台服务调用（消费任务队列消息）
declare(strict_types=1);

namespace App\Archive;

class ArchiveExtractor
{
    // 允许解包的归档扩展名白名单
    private const ALLOWED_EXT = ['zip', 'tar', 'tgz'];

    /**
     * 处理一个归档文件：$archivePath 由队列消息携带的外部提交归档名解析而来，
     * 归档内部条目名同样视为不可信外部数据。
     */
    public function extract(string $archivePath, string $destDir): void
    {
        $ext = strtolower(pathinfo($archivePath, PATHINFO_EXTENSION));
        if (!in_array($ext, self::ALLOWED_EXT, true)) {
            throw new \InvalidArgumentException('unsupported archive type');
        }

        $realArchive = realpath($archivePath);
        if ($realArchive === false || !is_file($realArchive)) {
            throw new \RuntimeException('archive not found');
        }

        $za = new \ZipArchive();
        if ($za->open($realArchive) !== true) {
            throw new \RuntimeException('cannot open archive');
        }

        for ($i = 0; $i < $za->numFiles; $i++) {
            $entry = $za->getNameIndex($i);          // 外部可控：归档内条目名
            $safe  = basename(str_replace('\', '/', $entry)); // 去目录穿越
            if ($safe === '' || str_starts_with($safe, '.')) {
                continue;                              // 拒绝隐藏/相对路径条目
            }
            $target = rtrim($destDir, '/') . '/' . $safe;
            if (!str_starts_with(realpath(dirname($target)) ?: '', realpath($destDir))) {
                continue;                              // 二次确认不逃逸目标目录
            }
            copy('zip://' . $realArchive . '#' . rawurlencode($entry), $target);
        }
        $za->close();
    }
}
```
