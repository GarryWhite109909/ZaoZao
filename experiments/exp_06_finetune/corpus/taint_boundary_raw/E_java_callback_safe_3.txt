package com.acme.report.export;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.amqp.rabbit.annotation.RabbitListener;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;

/**
 * 报表导出消费者：上层 API 服务将用户筛选条件发布到 report.export 队列，
 * 本组件消费消息并执行导出查询。
 */
@Component
public class ReportExportConsumer {

    private final JdbcTemplate jdbcTemplate;
    private final ObjectMapper objectMapper = new ObjectMapper();

    public ReportExportConsumer(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    @RabbitListener(queues = "report.export")
    public void onExportMessage(String messageBody) {
        try {
            JsonNode payload = objectMapper.readTree(messageBody);
            String region = payload.path("region").asText();
            String sql = "SELECT order_id, amount FROM orders WHERE region = ?";
            jdbcTemplate.query(sql, rs -> rs.getLong("order_id"), region);
        } catch (Exception e) {
            // 记录并丢弃非法消息，避免毒丸消息阻塞队列
        }
    }
}
