<?php
declare(strict_types=1);

namespace App\Consumer;

use App\Service\HttpCallbackClient;

/**
 * 消息消费者业务处理器：消费队列中的业务消息。
 * 消息体为 JSON 协议报文，由上层服务投递，本模块仅负责解析与处理。
 */
final class WebhookNotifyConsumer
{
    private const ALLOWED_EVENTS = ['order.paid', 'order.refunded', 'user.registered'];

    public function __construct(private HttpCallbackClient $client) {}

    /** @param string $payload 队列消息原始字节（协议报文，外部可控） */
    public function handle(string $payload): void
    {
        $msg = json_decode($payload, true, 512, JSON_THROW_ON_ERROR);
        $event = (string)($msg['event'] ?? '');
        if (!in_array($event, self::ALLOWED_EVENTS, true)) {
            return; // 白名单校验事件类型
        }
        $url = filter_var((string)($msg['callback_url'] ?? ''), FILTER_VALIDATE_URL);
        if ($url === false) {
            return;
        }
        $host = parse_url($url, PHP_URL_HOST);
        $ip = gethostbyname((string)$host);
        // SSRF 防御：仅允许公网地址，拒绝内网/环回/链路本地等保留段
        if (!filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE)) {
            return;
        }
        $this->client->post($url, ['event' => $event, 'data' => $msg['data'] ?? []]);
    }
}
