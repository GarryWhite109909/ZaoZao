package consumer

import (
	"database/sql"
	"encoding/json"
	"log"

	_ "github.com/go-sql-driver/mysql"

// OrderEventConsumer 消费订单队列消息，由上层服务注册到 MQ broker。
// 回调参数 payload 为外部提交的消息体，必须按不可信数据处理。

type OrderEvent struct {
	OrderID string `json:"order_id"`
	Status  string `json:"status"`
}

type Consumer struct {
	db *sql.DB
}

func NewConsumer(db *sql.DB) *Consumer { return &Consumer{db: db} }

func (c *Consumer) HandleMessage(payload []byte) {
	var ev OrderEvent
	if err := json.Unmarshal(payload, &ev); err != nil {
		log.Printf("invalid message dropped: %v", err)
		return
	}
	allowed := map[string]bool{"created": true, "paid": true, "shipped": true, "cancelled": true}
	if !allowed[ev.Status] {
		log.Printf("illegal status rejected")
		return
	}
	if _, err := c.db.Exec(
		"UPDATE orders SET status = ? WHERE order_id = ?", ev.Status, ev.OrderID,
	); err != nil {
		log.Printf("update failed: %v", err)
	}
}
