```python
"""数据导入适配器：解析外部系统导出的记录文件并写入数据库。

本模块由上层导入服务调用，传入待处理的导出文件路径。
"""

import csv
import os
import sqlite3

# 允许导入的表名白名单（外部系统约定的目标表）
ALLOWED_TABLES = {"customers", "orders", "products"}

DB_PATH = "/var/lib/importer/app.db"


def import_records(file_path: str, target_table: str) -> int:
    """解析 CSV 导出文件，将记录批量插入目标表，返回导入条数。"""
    if target_table not in ALLOWED_TABLES:
        raise ValueError("unsupported table")

    conn = sqlite3.connect(DB_PATH)
    try:
        with open(file_path, newline="", encoding="utf-8") as fh:
            reader = csv.DictReader(fh)
            rows = []
            for record in reader:
                rows.append((record.get("id"), record.get("name")))
        # 使用参数化占位符执行插入，表名来自白名单
        sql = f"INSERT INTO {target_table} (ext_id, name) VALUES (?, ?)"
        conn.executemany(sql, rows)
        conn.commit()
        return len(rows)
    finally:
        conn.close()
```
