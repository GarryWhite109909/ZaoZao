package assetloc

import (
	"database/sql"
	"fmt"
	"log"
)

// AssetEvent 是队列消息载荷，由外部系统投递（如 CDN 预热/清理任务）。
type AssetEvent struct {
	AssetName string `json:"asset_name"`
	Env       string `json:"env"`
}

// Locator 负责根据资源名查询数据库定位服务器文件路径。
type Locator struct {
	db *sql.DB
}

// HandleAssetEvent 是消息队列消费者回调：broker 将消息体反序列化后调用本函数，
// e 的字段完全来自外部提交内容，必须按不可信数据处理。
func (l *Locator) HandleAssetEvent(e AssetEvent) error {
	// 白名单校验：资源名仅允许字母数字与点、下划线、连字符；环境限定枚举值。
	if !isSafeName(e.AssetName) {
		return fmt.Errorf("invalid asset name")
	}
	switch e.Env {
	case "prod", "staging", "dev":
	default:
		return fmt.Errorf("unknown env")
	}
	// 参数化查询：占位符由驱动预编译绑定，外部值永不拼接进 SQL 文本。
	const q = "SELECT server_path FROM assets WHERE name = ? AND env = ?"
	var path string
	if err := l.db.QueryRow(q, e.AssetName, e.Env).Scan(&path); err != nil {
		log.Printf("asset lookup failed: %v", err)
		return err
	}
	log.Printf("resolved asset %s to %s", e.AssetName, path)
	return nil
}

func isSafeName(s string) bool {
	for _, r := range s {
		ok := (r >= 'a' && r <= 'z') || (r >= 'A' && r <= 'Z') ||
			(r >= '0' && r <= '9') || r == '.' || r == '_' || r == '-'
		if !ok {
			return false
		}
	}
	return len(s) > 0 && len(s) < 128
}
