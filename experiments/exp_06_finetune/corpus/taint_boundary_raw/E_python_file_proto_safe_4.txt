```python
"""数据导入适配器：解析外部系统导出的 JSONL 记录文件。

本模块由上层导入服务调用，传入待处理的文件路径；
文件内容来自外部合作方系统，必须按不可信数据处理。
"""

import json
import ipaddress
import socket
from urllib.parse import urlparse

# 允许回调通知的目标主机白名单（运维配置，非用户输入）
ALLOWED_HOSTS = {"partner-a.internal.example.com", "partner-b.internal.example.com"}


def notify_partner(record: dict) -> None:
    """向记录中指定的回调地址发送确认（模拟）。"""
    raw_url = record.get("callback_url", "")
    parsed = urlparse(raw_url)
    host = parsed.hostname
    if not host or host not in ALLOWED_HOSTS:
        raise ValueError("callback host not allowed")
    # 防御 DNS 重绑定：解析后校验 IP 是否为内网段
    ip = socket.gethostbyname(host)
    if not ipaddress.ip_address(ip).is_private:
        raise ValueError("resolved address is not internal")
    # 仅使用白名单主机名重建 URL，端口/路径取自白名单常量
    safe_url = f"https://{host}/ack"
    _http_post(safe_url, payload=json.dumps({"id": record.get("id")}))


def _http_post(url: str, payload: str) -> None:
    sock = socket.create_connection((urlparse(url).hostname, 443), timeout=5)
    try:
        sock.sendall(f"POST / HTTP/1.1\r
Host: {urlparse(url).hostname}\r
\r
{payload}".encode())
    finally:
        sock.close()


def process_import_file(path: str) -> int:
    """逐行解析外部记录文件并触发回调。"""
    count = 0
    with open(path, "r", encoding="utf-8") as fh:
        for line in fh:
            record = json.loads(line)
            notify_partner(record)
            count += 1
    return count
```
