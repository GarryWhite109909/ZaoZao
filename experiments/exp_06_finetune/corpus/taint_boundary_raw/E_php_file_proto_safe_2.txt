<?php
declare(strict_types=1);

namespace App\Asset;

/**
 * 静态资源定位模块：读取资源清单文件（由运维平台导出、经文件通道投递），
 * 将清单中的模板行渲染为服务器上的实际路径。被上层服务调用。
 */
class AssetLocator
{
    private const ALLOWED_KEYS = ['asset_name', 'version', 'env'];
    private const NAME_RE = '/^[A-Za-z0-9_][A-Za-z0-9_.\/-]{0,127}$/';

    public function locate(string $manifestPath): array
    {
        $results = [];
        foreach (file($manifestPath, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) as $line) {
            // 清单行格式：key=value;key=value
            $pairs = [];
            foreach (explode(';', $line) as $kv) {
                [$k, $v] = array_pad(explode('=', $kv, 2), 2, '');
                if (!in_array($k, self::ALLOWED_KEYS, true)) {
                    continue; // 白名单键：清单中未声明的键一律丢弃，杜绝注入额外占位符
                }
                $pairs[$k] = $v;
            }
            if (!isset($pairs['asset_name']) || preg_match(self::NAME_RE, $pairs['asset_name']) !== 1) {
                continue;
            }
            // 不做任何字符串拼接 eval/模板求值；仅做固定格式插值 + 路径规范化校验
            $base = rtrim('/srv/static/' . $pairs['env'] ?? 'prod', '/');
            $candidate = $base . '/' . ltrim($pairs['asset_name'], '/');
            $real = realpath($candidate);
            if ($real === false || !str_starts_with($real, '/srv/static/')) {
                continue; // 目录穿越防御：解析后必须仍位于静态根目录内
            }
            $results[] = $real;
        }
        return $results;
    }
}
