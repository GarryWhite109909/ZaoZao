```javascript
// archive-extractor.js — 归档解包工具模块，被上层服务调用（如消费队列消息后传入归档名）
'use strict';

const fs = require('fs');
const path = require('path');
const { execFile } = require('child_process');

// 允许的归档扩展名白名单
const ALLOWED_EXTENSIONS = new Set(['.zip', '.tar', '.tar.gz', '.tgz']);

/**
 * 解包外部提交的归档文件。
 * @param {string} archiveName - 上层调用者传入的外部归档名（不可信）
 * @param {string} destDir     - 目标解压目录
 */
function extractArchive(archiveName, destDir) {
  // 防御 1：扩展名白名单校验，阻断非归档/伪装参数
  const ext = path.extname(archiveName).toLowerCase();
  const isTarGz = archiveName.toLowerCase().endsWith('.tar.gz') || archiveName.toLowerCase().endsWith('.tgz');
  if (!isTarGz && !ALLOWED_EXTENSIONS.has(ext)) {
    throw new Error(`unsupported archive type: ${ext}`);
  }

  // 防御 2：路径规范化 + 目录逃逸检查，阻断 ../ 注入
  const resolved = path.resolve(destDir, path.basename(archiveName));
  if (!resolved.startsWith(path.resolve(destDir) + path.sep)) {
    throw new Error('path traversal detected');
  }
  if (!fs.existsSync(resolved)) {
    throw new Error('archive not found');
  }

  // 防御 3：execFile 数组传参，不经 shell 解释，无命令注入面
  if (isTarGz) {
    execFile('tar', ['-xzf', resolved, '-C', destDir], (err) => {
      if (err) throw err;
    });
  } else {
    execFile('unzip', ['-o', resolved, '-d', destDir], (err) => {
      if (err) throw err;
    });
  }
}

module.exports = { extractArchive };
```
