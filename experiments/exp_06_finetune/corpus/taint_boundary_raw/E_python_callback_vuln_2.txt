```python
"""report_exporter 插件钩子：宿主插件系统在收到外部事件时调用 on_event。"""
import json
import urllib.request
from plugin_sdk import PluginHook


class ReportExporterHook(PluginHook):
    """导出报表插件：宿主以事件载荷调用本钩子，载荷来自外部提交的报表任务。"""

    def on_event(self, event_type, payload):
        # payload 形如 {"action": "export", "callback_url": "...", "report_id": "R123"}
        if event_type != "report.export":
            return {"status": "ignored"}

        data = json.loads(payload)
        callback_url = data.get("callback_url", "")
        report_id = data.get("report_id", "")

        # 弱防御：仅黑名单拦截内网关键字，可被十进制 IP / DNS rebinding 绕过
        blocked = ("127.0.0.1", "localhost", "169.254", "10.", "192.168.")
        if any(b in callback_url for b in blocked):
            return {"status": "blocked"}

        export_url = "{}/reports/{}".format(callback_url.rstrip("/"), report_id)
        with urllib.request.urlopen(export_url, timeout=5) as resp:  # SSRF sink
            body = resp.read()

        return {"status": "ok", "bytes": len(body)}
```

4. 第 23 行将可控 `callback_url` 拼入 URL，第 24 行 `urllib.request.urlopen(export_url)` 直接发起请求，即 CWE-918 的危险 sink；攻击者可借此访问内网服务、云元数据接口（169.254.169.254 的变体写法）或任意主机。
