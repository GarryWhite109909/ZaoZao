// config-ingest.js — 消费配置提交队列消息，解析并应用外部提交的 YAML 配置
const { execFileSync } = require('child_process');
const yaml = require('js-yaml');

// 允许通过配置触发的部署钩子命令白名单（仅限内部脚本名，不接受任意字符串）
const ALLOWED_HOOKS = new Set(['lint', 'validate', 'schema-check']);
// 钩子脚本所在目录固定，且白名单值不参与路径拼接
const HOOK_DIR = '/opt/deploy/hooks/';

function handleConfigMessage(message) {
  // message.body 为外部提交的配置文件内容（不可信）
  const body = message.body;
  if (typeof body !== 'string' || body.length > 65536) {
    throw new Error('invalid config payload');
  }

  const manifest = yaml.safeLoad(body); // safeLoad 不实例化任意对象

  const hook = manifest && manifest.post_deploy_hook;
  if (typeof hook !== 'string') {
    return { applied: false, reason: 'no hook' };
  }

  // 白名单校验：hook 必须完全匹配预定义集合成员
  if (!ALLOWED_HOOKS.has(hook)) {
    throw new Error('hook not allowed');
  }

  // execFileSync 使用数组参数形式，不经 shell 解析；
  // 且参数已被白名单约束为固定枚举值
  execFileSync(HOOK_DIR + 'run-hook.sh', [hook], {
    shell: false,
    timeout: 30000,
  });

  return { applied: true };
}

module.exports = { handleConfigMessage };
