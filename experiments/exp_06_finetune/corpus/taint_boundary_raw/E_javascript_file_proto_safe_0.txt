```javascript
// template-renderer.js
// 模板渲染辅助模块，被上层服务调用，用于渲染外部提供的模板片段与变量。
// 模板片段来自磁盘上的用户上传目录文件（不可信内容）。

const fs = require('fs');
const path = require('path');
const Handlebars = require('handlebars');

// 允许的模板变量白名单：仅这些键会被注入渲染上下文
const ALLOWED_CONTEXT_KEYS = new Set(['title', 'body', 'author', 'date']);

function renderTemplateFile(templatePath, rawContext) {
  // 防御1：路径白名单校验，防止任意文件读取/SSRF 式路径穿越
  const resolved = path.resolve(templatePath);
  const baseDir = path.resolve(__dirname, 'templates');
  if (!resolved.startsWith(baseDir + path.sep)) {
    throw new Error('template path outside allowed directory');
  }

  const source = fs.readFileSync(resolved, 'utf8');

  // 防御2：Handlebars 默认对插值 {{ }} 做 HTML 实体转义，
  // 编译时显式声明 noEscape: false，确保所有变量输出均被正确转义
  const compiled = Handlebars.compile(source, { noEscape: false });

  const safeContext = {};
  for (const key of Object.keys(rawContext)) {
    if (ALLOWED_CONTEXT_KEYS.has(key)) {
      safeContext[key] = String(rawContext[key]);
    }
  }
  return compiled(safeContext);
}

module.exports = { renderTemplateFile };
```
