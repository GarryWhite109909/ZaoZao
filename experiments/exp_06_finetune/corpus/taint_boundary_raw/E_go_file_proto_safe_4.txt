// 从任务清单文件读取待执行命令并登记到审计数据库。
package opscmd

import (
	"database/sql"
	"os/exec"
	"strings"
)

type Runner struct {
	DB *sql.DB // 审计库连接
}

// RunFromFile 解析任务清单文件（每行一条命令），内容视为不可信外部数据。
func (r *Runner) RunFromFile(path string) error {
	data, err := readFile(path)
	if err != nil {
		return err
	}
	for _, line := range strings.Split(string(data), "
") {
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}
		fields := strings.Fields(line) // 拆分为参数数组
		// sink1: 命令执行 —— 使用参数数组形式，不经 shell 解释
		cmd := exec.Command(fields[0], fields[1:]...)
		if err := cmd.Run(); err != nil {
			return err
		}
		// sink2: 审计入库 —— 预编译参数化查询，文件内容仅作绑定参数
		if _, err := r.DB.Exec(
			"INSERT INTO audit_log(cmdline, status) VALUES (?, ?)",
			line, "ok",
		); err != nil {
			return err
		}
	}
	return nil
}
