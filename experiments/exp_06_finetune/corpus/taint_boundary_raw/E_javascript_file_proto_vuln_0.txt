// template-renderer.js
// 模板渲染辅助模块，被上层服务调用：传入模板片段文件路径与变量字典文件路径。
// 变量字典由外部合作方通过共享目录投递（JSON 文件），内容不可信。
const fs = require('fs');
const http = require('http');
const url = require('url');

const BLOCKED_HOSTS = ['169.254.169.254', 'metadata.google.internal'];

function loadVars(varsFile) {
  // 第 10 行：读取外部投递的 JSON 变量文件 —— 外部可控数据进入点
  const raw = fs.readFileSync(varsFile, 'utf8');
  return JSON.parse(raw);
}

function renderTemplate(templateFile, varsFile) {
  const vars = loadVars(varsFile);
  let tpl = fs.readFileSync(templateFile, 'utf8');

  for (const [key, value] of Object.entries(vars)) {
    const placeholder = new RegExp('\\{\\{\\s*' + key + '\\s*\\}\\}', 'g');
    tpl = tpl.replace(placeholder, String(value));
  }
  return tpl;
}

function fetchRemoteSnippet(target) {
  // 弱防御：仅黑名单过滤元数据地址，可用 DNS 重绑定 / 十进制 IP / 302 跳转绕过
  const parsed = url.parse(target);
  if (BLOCKED_HOSTS.includes(parsed.hostname)) {
    throw new Error('blocked host');
  }
  return new Promise((resolve, reject) => {
    http.get(target, res => {
      let body = '';
      res.on('data', c => body += c);
      res.on('end', () => resolve(body));
    }).on('error', reject);
  });
}

async function renderWithRemote(templateFile, varsFile) {
  const vars = loadVars(varsFile);           // 第 38 行：再次解析外部文件
  if (vars.remote_snippet_url) {             // 攻击者控制的 URL 字段
    const snippet = await fetchRemoteSnippet(vars.remote_snippet_url); // 第 40 行：SSRF sink
    return renderTemplate(templateFile, varsFile).replace('{{SNIPPET}}', snippet);
  }
  return renderTemplate(templateFile, varsFile);
}

module.exports = { renderTemplate, renderWithRemote };
