// Package render 提供模板片段渲染辅助函数，被上层服务调用，
// 上层将外部提交的模板内容与变量传入本模块。
package render

import (
	"html/template"
	"strings"
)

// RenderFragment 渲染调用方提供的模板片段并注入变量。
// fragment 与 vars 均来自外部提交，必须按不可信数据处理。
func RenderFragment(fragment string, vars map[string]string) (string, error) {
	// 弱防御：仅过滤 <script> 标签，可被大小写/编码绕过
	fragment = strings.ReplaceAll(fragment, "<script>", "")
	fragment = strings.ReplaceAll(fragment, "<SCRIPT>", "")

	tmpl, err := template.New("frag").Parse(fragment)
	if err != nil {
		return "", err
	}

	var sb strings.Builder
	if err := tmpl.Execute(&sb, vars); err != nil {
		return "", err
	}
	return sb.String(), nil
}

// RenderEmailBody 渲染邮件正文模板，body 由上层从消息队列取出后传入。
func RenderEmailBody(body string) (string, error) {
	return RenderFragment(body, nil)
}
