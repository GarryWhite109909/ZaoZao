```javascript
// hooks/templateHook.js
// 插件系统钩子实现：宿主服务在插件触发时调用本模块，
// 模板文件内容（外部提交的插件包）被视为不可信输入。
'use strict';

const fs = require('fs').promises;
const path = require('path');
const ejs = require('ejs');

const PLUGIN_DIR = '/var/lib/app/plugins';
// 白名单：仅允许纯小写字母/数字/下划线/连字符组成的 .ejs 文件名
const SAFE_NAME = /^[a-z0-9_-]{1,64}\.ejs$/;

async function onRenderHook(pluginName, context) {
  // 防御点 1：白名单校验文件名，阻断路径穿越与任意文件读取
  if (typeof pluginName !== 'string' || !SAFE_NAME.test(pluginName)) {
    throw new Error('invalid plugin name');
  }
  const resolved = path.resolve(PLUGIN_DIR, pluginName);
  if (!resolved.startsWith(PLUGIN_DIR + path.sep)) {
    throw new Error('path traversal blocked');
  }

  const raw = await fs.readFile(resolved, 'utf8');

  // 防御点 2：外部内容只作为【数据】传入渲染引擎的 locals，
  // 绝不拼接进模板字符串本身，因此不存在模板注入面；
  // ejs 对输出默认做 HTML 转义（escape 参数显式开启）。
  const html = await ejs.render(raw, { ctx: context }, {
    async: true,
    escape: (s) => String(s).replace(/[&<>"']/g,
      c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])),
    root: PLUGIN_DIR,
    includePath: (p) => SAFE_NAME.test(p) ? p : null
  });
  return html;
}

module.exports = { onRenderHook };
```

3. 数据流二（CWE-1336 主链）：readFile 读出的模板字符串 `raw` → 第 30 行 `ejs.render(raw, ...)`。若外部数据被拼进模板字符串即构成服务端模板注入（RCE）。此处防御有效：外部可控的 `context` 仅作为第二个参数的数据对象传入，永不进入模板文本；且第 31-33 行显式提供 escape 函数对插值输出做 HTML 实体转义，覆盖 `<script>` 注入面。模板字符串本身来自经白名单校验路径的受控目录文件，非任意用户直传内容。

4. 数据流三：ejs 的 `<%- include(...) %>` 可加载任意模板形成二次注入通道。第 34-35 行通过 `root` 限定根目录、`includePath` 回调复用同一白名单 `SAFE_NAME` 校验，include 目标同样无法穿越或指向任意文件，替代通道被封死。
