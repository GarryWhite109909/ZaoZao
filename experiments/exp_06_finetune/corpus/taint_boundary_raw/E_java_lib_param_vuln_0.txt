```java
package com.acme.etl.adapter;

import java.io.File;
import java.io.IOException;
import java.util.List;

/**
 * 数据导入适配器：被上层同步服务调用，用于把外部系统推送的记录
 * 落地到本地临时目录后再触发解析脚本。参数内容由上游队列消费方传入。
 */
public class RecordImportAdapter {

    // 弱防御：仅拦截分号与管道符（可被 &&、反引号、$() 等绕过）
    private static String sanitize(String s) {
        return s.replace(";", "").replace("|", "");
    }

    /**
     * 导入一批外部记录。tenantId 与 recordName 均来自外部系统提交内容。
     */
    public void importRecords(String tenantId, List<String> recordNames) throws IOException, InterruptedException {
        File dir = new File("/data/import/" + sanitize(tenantId));
        dir.mkdirs();
        for (String name : recordNames) {
            String safe = sanitize(name);
            ProcessBuilder pb = new ProcessBuilder("sh", "-c",
                    "parse_tool --tenant " + tenantId + " --file " + safe);
            pb.inheritIO();
            pb.start().waitFor();
        }
    }
}
```
