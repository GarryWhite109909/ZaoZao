```python
"""cache_codec.py — 缓存编解码模块，被上层服务调用以消费缓存同步队列消息。"""
import json
import logging

import redis

logger = logging.getLogger(__name__)

# 允许反序列化的载荷类型白名单（仅纯数据结构）
ALLOWED_TYPES = {"snapshot", "counter", "manifest"}

_pool = redis.ConnectionPool(host="127.0.0.1", port=6379, db=0)
_client = redis.Redis(connection_pool=_pool)


def on_cache_message(payload: bytes) -> None:
    """消息队列消费者回调：payload 为外部提交的缓存同步消息体。"""
    try:
        message = json.loads(payload.decode("utf-8"))
    except (ValueError, UnicodeDecodeError):
        logger.warning("malformed cache message dropped")
        return

    # 白名单校验：只接受已知类型，且载荷必须是 dict
    if not isinstance(message, dict) or message.get("type") not in ALLOWED_TYPES:
        logger.warning("rejected cache message type")
        return

    key = "cache:%s:%s" % (message["type"], str(message.get("key", ""))[:128])
    value = json.dumps(message.get("value"), ensure_ascii=True)

    _client.set(key, value)
```
