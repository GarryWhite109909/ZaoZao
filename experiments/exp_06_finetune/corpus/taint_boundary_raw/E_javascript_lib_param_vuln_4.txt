// 弱防御：仅拦截常见 SQL 关键字，可被注释符/大小写/编码绕过
function sanitizeName(name) {
  const blacklist = ['select', 'insert', 'update', 'delete', 'drop', ';'];
  let cleaned = name;
  for (const kw of blacklist) {
    cleaned = cleaned.replace(new RegExp(kw, 'gi'), '');
  }
  return cleaned;
}
