```javascript
// outbound-webhook-dispatcher.js
// 被上层服务调用：消费消息队列中的 webhook 投递任务
const amqp = require('amqplib');
const crypto = require('crypto');
const { URL } = require('url');

// 允许投递的目标主机白名单（由运维配置，非外部可控）
const ALLOWED_HOSTS = new Set(['hooks.partner-a.com', 'hooks.partner-b.com']);
const ALLOWED_SCHEMES = new Set(['https']);

async function startConsumer() {
  const conn = await amqp.connect(process.env.AMQP_URL);
  const channel = await conn.createChannel();
  await channel.assertQueue('webhook.outbound', { durable: true });

  channel.consume('webhook.outbound', async (msg) => {
    // msg.content 是外部提交方写入队列的载荷，必须按不可信数据处理
    const payload = JSON.parse(msg.content.toString());
    // payload: { targetUrl, body, signature }

    // 防御 1：签名校验（HMAC），防止伪造任务
    const expected = crypto.createHmac('sha256', process.env.SIGNING_KEY)
      .update(payload.body).digest('hex');
    if (!crypto.timingSafeEqual(Buffer.from(expected), Buffer.from(payload.signature))) {
      channel.nack(msg, false, false);
      return;
    }

    // 防御 2：白名单校验目标地址（scheme + host 双重校验）
    let parsed;
    try {
      parsed = new URL(payload.targetUrl);
    } catch {
      channel.nack(msg, false, false);
      return;
    }
    if (!ALLOWED_SCHEMES.has(parsed.protocol.replace(':', '')) ||
        !ALLOWED_HOSTS.has(parsed.hostname)) {
      channel.nack(msg, false, false);
      return;
    }

    // sink：向校验后的地址发起出站请求；body 仅作为 JSON 字符串化数据传输，
    // 不进入任何命令执行/反序列化/模板引擎
    const res = await fetch(parsed.href, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ data: payload.body })
    });
    if (res.ok) channel.ack(msg);
    else channel.nack(msg, false, true);
  }, { noAck: false });
}

module.exports = { startConsumer };
```
