```javascript
// ops-runner.js —— 运维命令封装库，被上层服务调用；消费 RabbitMQ 运维任务队列
const amqp = require('amqplib');
const { exec } = require('child_process');
const fs = require('fs');

// 将任务执行结果渲染进运维状态页（status.html 由内部仪表盘加载）
function renderStatusPage(taskId, taskName, output) {
  const html = `<html><body><h3>Task ${taskId}: ${taskName}</h3><pre>${output}</pre></body></html>`;
  fs.writeFileSync('/var/www/dashboard/status.html', html);
}

async function startConsumer() {
  const conn = await amqp.connect('amqp://localhost');
  const ch = await conn.createChannel();
  await ch.assertQueue('ops.tasks');
  // 回调参数 msg 即消息体载荷，承载外部提交的任务内容
  ch.consume('ops.tasks', (msg) => {
    const payload = JSON.parse(msg.content.toString());
    const taskId = payload.id;
    const taskName = payload.name;            // 外部可控：任务名
    const cmd = payload.command;              // 外部可控：待执行命令

    // 弱防御：仅黑名单过滤 <script> 字面量（大小写敏感、可被编码绕过）
    const safeName = String(taskName).replace(/<script>/gi, '');

    exec(cmd, { timeout: 30000 }, (err, stdout) => {
      if (err) return;
      renderStatusPage(taskId, safeName, stdout);
    });
    ch.ack(msg);
  });
}

module.exports = { startConsumer };
```
