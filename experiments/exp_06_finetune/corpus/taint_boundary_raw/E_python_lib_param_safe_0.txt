```python
"""report_export.py — 报表导出工具模块，被上层 API 服务以用户筛选条件调用。"""

import sqlite3
from typing import Any, Iterable

# 允许排序/分组的列白名单（与 reports 表实际列一一对应）
_ALLOWED_COLUMNS = frozenset({
    "created_at", "region", "product_line", "amount", "status",
})

# 允许的聚合方式
_ALLOWED_AGGS = {"sum", "avg", "count", "min", "max"}


def export_report(conn: sqlite3.Connection,
                  columns: Iterable[str],
                  agg: str,
                  region_filter: str) -> list[dict[str, Any]]:
    """按调用方传入的筛选条件导出报表行。

    columns / agg / region_filter 均来自上层服务透传的用户输入，
    必须全部经过校验后才能进入 SQL。
    """
    cols = [c for c in columns if c in _ALLOWED_COLUMNS]
    if not cols:
        raise ValueError("no valid columns requested")
    if agg.lower() not in _ALLOWED_AGGS:
        raise ValueError("unsupported aggregation")

    placeholders = ", ".join("?" for _ in cols)
    sql = (
        f"SELECT region, {agg.upper()}({placeholders}) AS value "
        f"FROM reports WHERE region = ? GROUP BY region"
    )
    params: list[Any] = [*cols, region_filter]

    cur = conn.execute(sql, params)
    return [dict(zip(("region", "value"), row)) for row in cur.fetchall()]
```
