```go
// Package tmplutil 提供模板渲染辅助函数，被上层服务调用以渲染外部提交的模板片段。
package tmplutil

import (
	"html/template"
	"strings"
)

// RenderFragment 渲染上层调用者传入的模板片段与变量。
// fragment 与 vars 均来自外部提交内容，必须按不可信数据处理。
func RenderFragment(fragment string, vars map[string]string) (string, error) {
	// 白名单校验：仅允许纯文本片段，拒绝任何模板动作语法，
	// 防止攻击者通过 {{...}} 注入模板指令（SSTI）进而触发 XSS。
	if strings.ContainsAny(fragment, "{}") {
		return "", ErrUnsafeFragment
	}

	tmpl, err := template.New("fragment").Parse(fragment)
	if err != nil {
		return "", err
	}

	// html/template 在执行时对插入值做上下文感知自动转义，
	// vars 中的不可信数据经 {{.}} 输出时会被正确 HTML 转义。
	var sb strings.Builder
	if err := tmpl.Execute(&sb, vars); err != nil {
		return "", err
	}
	return sb.String(), nil
}

// ErrUnsafeFragment 表示片段包含被禁止的模板语法。
var ErrUnsafeFragment = errorString("fragment contains forbidden template syntax")

type errorString string

func (e errorString) Error() string { return string(e) }
```
