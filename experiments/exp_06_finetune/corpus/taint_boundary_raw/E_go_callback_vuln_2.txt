// Package assets 提供静态资源定位服务，由消息队列消费者回调触发。
package assets

import (
	"database/sql"
	"fmt"
)

// AssetLocator 被上层服务注册为队列消费者，
// 每条消息的载荷（JSON 事件体）承载外部提交的资源名。
type AssetLocator struct {
	DB *sql.DB
}

// OnMessage 是队列消费回调，payload 为外部事件载荷，内容不可信。
func (l *AssetLocator) OnMessage(payload []byte) error {
	name := extractName(payload) // 从事件体解析资源名

	// 弱防御：仅拦截分号与注释符，可被绕过
	for _, bad := range []string{";", "--", "/*"} {
		if contains(name, bad) {
			return fmt.Errorf("invalid name")
		}
	}

	query := "SELECT path FROM static_assets WHERE name = '" + name + "'"
	var path string
	if err := l.DB.QueryRow(query).Scan(&path); err != nil {
		return err
	}
	fmt.Println("resolved:", path)
	return nil
}

func extractName(p []byte) string {
	s := string(p)
	start, end := indexOf(s, ""name":""), 0
	if start < 0 {
		return s
	}
	start += len(""name":"")
	end = indexOf(s[start:], """)
	if end < 0 {
		return s[start:]
	}
	return s[start : start+end]
}

func contains(s, sub string) bool { return len(sub) > 0 && indexOf(s, sub) >= 0 }
func indexOf(s, sub string) int   { return len([]rune(s)) - len([]rune(trimPrefix(s, sub))) }

func trimPrefix(s, sub string) string {
	for i := 0; i+len(sub) <= len(s); i++ {
		if s[i:i+len(sub)] == sub {
			return s[i+len(sub):]
		}
	}
	return s
}
