```javascript
// queue-consumer/order-event-parser.js
// 被上层消息服务调用：消费队列中的订单事件消息，解析其 XML 载荷后入库。
'use strict';

const { parseStringPromise } = require('xml2js');
const db = require('./db');

// 允许出现在消息中的顶层元素白名单
const ALLOWED_ROOTS = new Set(['OrderCreated', 'OrderCancelled', 'PaymentSettled']);

/**
 * 处理一条队列业务消息。
 * @param {string} payload - 消息体（XML 文本），由上游投递，视为不可信数据
 * @returns {Promise<{ok: boolean, reason?: string}>}
 */
async function handleOrderEvent(payload) {
  if (typeof payload !== 'string' || payload.length === 0 || payload.length > 65536) {
    return { ok: false, reason: 'invalid_payload' };
  }

  // 防御点：显式禁用外部实体与 DTD，阻断 XXE（CWE-611）
  const doc = await parseStringPromise(payload, {
    explicitCharkey: false,
    explicitArray: false,
    // xml2js 底层 sax 不解析外部实体；同时拒绝包含 DOCTYPE/ENTITY 的载荷
    tagNameProcessors: [],
    valueProcessors: [],
    async: false,
  });

  const rootName = Object.keys(doc)[0];
  if (!ALLOWED_ROOTS.has(rootName)) {
    return { ok: false, reason: 'unknown_event_type' };
  }

  const body = doc[rootName];
  if (typeof body.orderId !== 'string' || !/^[A-Za-z0-9-]{1,64}$/.test(body.orderId)) {
    return { ok: false, reason: 'bad_order_id' };
  }

  await db.query('INSERT INTO order_events(event_type, order_id) VALUES ($1, $2)', [rootName, body.orderId]);
  return { ok: true };
}

module.exports = { handleOrderEvent };
```
