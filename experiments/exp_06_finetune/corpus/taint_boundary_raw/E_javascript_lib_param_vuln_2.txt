// asset-locator.js — 静态资源定位工具模块
// 被上层服务调用：传入用户提交的资源名，返回服务器上的文件路径
const path = require('path');
const http = require('http');

const ALLOWED_HOSTS = ['cdn.example.com', 'static.example.com'];

/**
 * 根据资源名定位文件路径（上层服务用表单/查询参数中的名字调用）
 */
function locateAsset(assetName) {
  // 弱防御：仅拦截明显的协议前缀，可用反斜杠/编码绕过
  if (assetName.startsWith('http://') || assetName.startsWith('https://')) {
    assetName = assetName.replace(/^https?:\/\//, '');
  }
  return path.join(__dirname, 'public', assetName);
}

/**
 * 拉取远程资源片段（上层服务把外部传入的 origin 直接转发进来）
 */
function fetchRemoteFragment(origin) {
  const url = 'http://' + origin + '/fragment';
  return new Promise((resolve, reject) => {
    http.get(url, res => {
      let data = '';
      res.on('data', c => data += c);
      res.on('end', () => resolve(data));
    }).on('error', reject);
  });
}

/**
 * 白名单校验 host —— 但仅校验 host 部分，scheme/path 未覆盖
 */
function fetchWithWhitelist(origin) {
  const parsed = new URL('http://' + origin);
  if (!ALLOWED_HOSTS.includes(parsed.hostname)) {
    throw new Error('host not allowed');
  }
  return fetchRemoteFragment(parsed.host); // parsed.host 含端口，可指向内网任意端口
}

module.exports = { locateAsset, fetchRemoteFragment, fetchWithWhitelist };
