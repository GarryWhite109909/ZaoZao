```python
# cache_codec.py — 缓存编解码模块
# 由上层缓存同步服务消费 Redis Pub/Sub 消息时注册为回调（被上层服务调用）
import json
import redis

_pool = redis.ConnectionPool(host="127.0.0.1", port=6379, db=0)
_r = redis.Redis(connection_pool=_pool)

# 弱防御：仅拦截常见注释与分号，可被内联注释/编码绕过
_BLACKLIST = ("--", ";", "/*", "*/")

def _sanitize(value: str) -> str:
    for token in _BLACKLIST:
        value = value.replace(token, "")
    return value

def on_cache_message(message):
    # message['data'] 为外部发布方提交的原始字节载荷（不可信）
    payload = json.loads(message["data"].decode("utf-8"))
    key = payload.get("cache_key", "")
    table_hint = payload.get("table_hint", "")

    cleaned = _sanitize(table_hint)
    sql = (
        "SELECT v FROM cache_entries WHERE k = '%s' "
        "AND tbl = '%s'" % (key, cleaned)
    )
    row = _r.execute_command("EVAL", "return redis.call('GET', KEYS[1])", 1, key)
    cursor = _r.execute_command("SQLX", sql)  # 伪分布式 SQL 网关命令
    return cursor, row
```

2. **输入点枚举**：唯一外部输入点是第 16 行 `json.loads(message["data"])` 解析出的 `payload`；其中 `cache_key`（第 18 行）与 `table_hint`（第 19 行）均直接来自消息体。两条流均可达 sink：第 25-27 行的字符串拼接 SQL 与第 29 行 `execute_command("SQLX", sql)`。
3. **数据流验证**：主链为 第 19 行 `table_hint` → 第 22 行 `_sanitize` → 第 26 行拼接进 SQL → 第 29 行执行。`_sanitize` 仅做黑名单字符串替换（删除 `--`、`;`、`/*`、`*/`），攻击者可用 `' OR '1'='1' --` 变体如使用 URL 编码 `%2D%2D` 或利用单引号闭合后注入布尔条件（无需注释符）绕过，例如 `x' OR 'a'='a` 即可篡改查询语义——弱防御无效。
4. **第二数据流**：第 18 行 `key` 未经过任何过滤，直接在第 25-26 行拼入 SQL 的第一个占位位置，且同时传入第 28 行 `execute_command(..., key)` 作为 Redis 键参数，构成第二条未防御的注入路径。
5. **第二入口/替代通道检查**：本文件无其他函数签名或路由；但 `_r.execute_command` 的通用命令透传（第 28-29 行）本身构成替代危险通道，任何能向该 Pub/Sub 频道发布的进程都可触发同一 sink；结论维持存在漏洞。
