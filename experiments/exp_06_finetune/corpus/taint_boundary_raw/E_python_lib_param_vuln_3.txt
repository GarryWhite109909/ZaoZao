```python
"""config_loader.py — 解析外部提交的 YAML 配置清单，渲染为 HTML 报告片段。
被上层服务调用：服务从消息队列/上传目录取出配置文件内容后传入本模块。"""

import html
from xml.sax.saxutils import escape


def parse_config(raw_text: str) -> dict:
    """解析原始配置文本，返回键值对。"""
    config = {}
    for line in raw_text.splitlines():
        if "=" not in line:
            continue
        key, _, value = line.partition("=")
        # 弱防御：仅过滤 <script> 字面量（大小写不敏感），可被 <img onerror>、<svg onload> 等绕过
        value = value.replace("<script", "").replace("</script", "")
        config[key.strip()] = value.strip()
    return config


def render_config_report(config: dict, title: str) -> str:
    """将配置渲染为 HTML 片段，供上层页面直接嵌入。"""
    parts = [f"<h2>{title}</h2><ul>"]
    for key, value in config.items():
        # 漏洞点：value 未做 HTML 转义，直接拼入 HTML
        parts.append(f"<li>{key} = {value}</li>")
    parts.append("</ul>")
    return "".join(parts)


def build_report(raw_text: str, title: str = "Config Report") -> str:
    """公开入口：上层把外部提交的配置文件内容原样传入。"""
    return render_config_report(parse_config(raw_text), title)
```
