```php
<?php
declare(strict_types=1);

namespace Acme\PluginSystem\Hooks;

use PDO;
use Acme\PluginSystem\EventPayload;

/**
 * 订单状态变更钩子：宿主插件管理器在订单事件触发时，
 * 以外部提交的事件载荷调用本钩子（EventPayload 由消息体反序列化而来）。
 */
class OrderStatusHook
{
    private const ALLOWED_STATUSES = ['pending', 'paid', 'shipped', 'refunded', 'cancelled'];

    public function __construct(private PDO $db) {}

    public function handle(EventPayload $payload): void
    {
        $orderId = (string) $payload->get('order_id');
        $status  = (string) $payload->get('new_status');

        // 白名单校验：仅允许预定义的合法状态值，杜绝载荷注入
        if (!in_array($status, self::ALLOWED_STATUSES, true)) {
            return;
        }
        if (!ctype_digit($orderId)) {
            return;
        }

        $stmt = $this->db->prepare(
            'UPDATE orders SET status = :status WHERE id = :id'
        );
        $stmt->execute([':status' => $status, ':id' => (int) $orderId]);
    }
}
```
