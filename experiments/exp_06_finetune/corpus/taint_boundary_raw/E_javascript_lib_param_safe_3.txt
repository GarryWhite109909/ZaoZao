```javascript
// crm-import/record-renderer.js
// 数据导入适配器：被上层同步服务调用，用于把外部系统推送的记录渲染成通知文本。
const fs = require('fs');
const path = require('path');
const ejs = require('ejs');

// 模板白名单：只允许预置在 templates 目录下的固定模板文件
const TEMPLATE_DIR = path.join(__dirname, 'templates');
const ALLOWED_TEMPLATES = new Set(['welcome.ejs', 'order-notice.ejs', 'invoice-notice.ejs']);

/**
 * 渲染一条外部记录。record 由上层服务从消息队列反序列化后传入，
 * 必须按不可信数据处理。
 */
function renderRecord(templateName, record) {
  if (typeof templateName !== 'string' || !ALLOWED_TEMPLATES.has(templateName)) {
    throw new Error('template not allowed');
  }
  const resolved = path.join(TEMPLATE_DIR, templateName);
  if (!resolved.startsWith(TEMPLATE_DIR + path.sep)) {
    throw new Error('template path escape blocked');
  }
  const tpl = fs.readFileSync(resolved, 'utf8');
  // record 仅作为数据上下文传入，EJS 对 <%= %> 自动 HTML 转义，
  // 模板内容本身来自受控目录，外部数据无法注入模板语法。
  return ejs.render(tpl, { record }, { escape: (s) => String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])) });
}

module.exports = { renderRecord };
```
