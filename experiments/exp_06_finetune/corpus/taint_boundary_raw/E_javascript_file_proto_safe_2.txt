// richTextFormatter.js — 富文本/日志格式化模块
// 被上层服务调用：传入外部提交的 .rtf/.log 文件路径，解析后写入审计数据库
const fs = require('fs');
const { Pool } = require('pg');

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

// 允许的日志级别白名单（唯一合法取值）
const ALLOWED_LEVELS = ['INFO', 'WARN', 'ERROR', 'DEBUG'];

async function formatAndStore(logFilePath) {
  // 第 12 行：读取外部提交的日志文件，内容视为不可信输入
  const raw = fs.readFileSync(logFilePath, 'utf8');

  for (const line of raw.split('
')) {
    if (!line.trim()) continue;
    const [ts, level, ...msgParts] = line.split('|');
    // 第 17 行：白名单校验——level 必须精确匹配枚举值，否则丢弃该行
    if (!ALLOWED_LEVELS.includes(level)) continue;
    const message = msgParts.join('|').slice(0, 2000);

    // 第 20 行：参数化查询（预编译占位符），消息内容仅作为绑定参数传递，
    // 不参与 SQL 文本拼接，注入载荷无法改变查询结构
    await pool.query(
      'INSERT INTO audit_logs (ts, level, message) VALUES ($1, $2, $3)',
      [ts, level, message]
    );
  }
}

module.exports = { formatAndStore };
