// archive-indexer.js — 归档索引工具模块，被上层服务调用（如消费队列消息后传入归档名）
'use strict';

const { Pool } = require('pg');

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

// 允许写入索引表的归档类型白名单
const ALLOWED_TYPES = new Set(['zip', 'tar', 'tar.gz', 'tgz']);

/**
 * 将外部提交的归档元数据登记到索引库。
 * archiveName / archiveType 均来自上层调用者（不可信数据）。
 */
async function registerArchive(archiveName, archiveType, sizeBytes) {
  if (typeof archiveName !== 'string' || archiveName.length === 0 || archiveName.length > 255) {
    throw new Error('invalid archive name');
  }
  // 白名单校验：archiveType 必须命中预定义集合，杜绝注入
  if (!ALLOWED_TYPES.has(archiveType)) {
    throw new Error('unsupported archive type');
  }
  const size = Number(sizeBytes);
  if (!Number.isFinite(size) || size < 0) {
    throw new Error('invalid size');
  }
  // 参数化查询：$1/$2/$3 占位符由 pg 驱动安全绑定，值不会拼入 SQL 文本
  await pool.query(
    'INSERT INTO archives (name, type, size_bytes) VALUES ($1, $2, $3)',
    [archiveName, archiveType, Math.floor(size)]
  );
}

module.exports = { registerArchive };
