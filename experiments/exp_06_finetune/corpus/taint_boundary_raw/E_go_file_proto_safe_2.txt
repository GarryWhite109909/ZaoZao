```go
// Package configparse 解析外部提交的部署清单文件（YAML）。
// 本模块被上层服务调用，用于消费上传的配置文件并预取其中的资源地址。
package configparse

import (
	"fmt"
	"io"
	"net/http"
	"net/url"
	"os"
	"strings"

	"gopkg.in/yaml.v3"
)

type Manifest struct {
	ResourceURL string `yaml:"resource_url"`
}

// 允许拉取资源的内部镜像域名白名单
var allowedHosts = map[string]bool{
	"artifacts.internal.example.com": true,
	"mirror.internal.example.com":    true,
}

func LoadAndFetch(path string) error {
	data, err := os.ReadFile(path)
	if err != nil {
		return fmt.Errorf("read manifest: %w", err)
	}
	var m Manifest
	if err := yaml.Unmarshal(data, &m); err != nil {
		return fmt.Errorf("parse manifest: %w", err)
	}
	u, err := url.Parse(m.ResourceURL) // 第 30 行：解析外部可控 URL
	if err != nil {
		return fmt.Errorf("invalid url: %w", err)
	}
	if u.Scheme != "https" || !allowedHosts[strings.ToLower(u.Hostname())] { // 第 34 行：白名单校验
		return fmt.Errorf("host not allowed")
	}
	resp, err := http.Get(u.String()) // 第 37 行：sink
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	_, _ = io.Copy(io.Discard, resp.Body)
	return nil
}
```
