// Package archiver 处理来自消息队列的归档解包任务。
// 上层服务消费队列消息后调用本模块的 HandleArchiveEvent。
package archiver

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// ArchiveEvent 是消息载荷，由外部提交方生成，内容不可信。
type ArchiveEvent struct {
	TaskID   string
	FileName string // 外部提交的归档文件名
}

var allowedExts = map[string]bool{
	".zip": true, ".tar": true, ".tar.gz": true, ".tgz": true,
}

// HandleArchiveEvent 是队列消费者回调入口（污点边界）。
func HandleArchiveEvent(ev ArchiveEvent) error {
	name := ev.FileName
	if name == "" || strings.ContainsAny(name, "\x00") {
		return fmt.Errorf("task %s: empty or invalid filename", ev.TaskID)
	}
	clean := filepath.Base(name) // 剥离任何路径分隔符与 .. 组件
	ext := strings.ToLower(filepath.Ext(clean))
	if ext == ".gz" && strings.HasSuffix(strings.ToLower(clean), ".tar.gz") {
		ext = ".tar.gz"
	}
	if !allowedExts[ext] { // 白名单校验：仅允许归档扩展名
		return fmt.Errorf("task %s: extension not allowed", ev.TaskID)
	}
	dest := filepath.Join("/srv/archive/incoming", clean)
	if !strings.HasPrefix(dest, "/srv/archive/incoming/") {
		return fmt.Errorf("task %s: path escape blocked", ev.TaskID)
	}
	f, err := os.Open(dest)
	if err != nil {
		return err
	}
	defer f.Close()
	return extract(f, dest)
}

func extract(f *os.File, dest string) error {
	fmt.Println("extracting", dest)
	return nil
}
