// archive_unpacker.php — 归档解包模块，被上层批处理服务调用，
// 处理外部提交的归档文件（tar 格式），解析归档内条目并执行清理钩子。

function unpackArchive(string $archivePath, string $destDir): array
{
    $phar = new PharData($archivePath);
    $extracted = [];

    foreach (new RecursiveIteratorIterator($phar) as $file) {
        $entryName = $file->getPathname();          // 第 12 行：条目名来自被解析的归档内容
        $target = rtrim($destDir, '/') . '/' . basename($entryName);

        // 弱防御：仅黑名单过滤分号与空格，认为可阻止命令注入
        $clean = str_replace([';', ' '], '', $entryName);   // 第 16 行：可绕过的字符串替换

        if (preg_match('/\.sh$/', $entryName)) {
            // 解包后对脚本条目执行权限修复钩子
            exec("chmod +x " . escapeshellarg($target));    // 第 20 行：此处已转义
            // 同时调用外部病毒扫描器扫描原始条目名
            system("clamscan --no-summary " . $clean);      // 第 22 行：SINK
        }

        $phar->extractTo($destDir, $entryName, true);
        $extracted[] = $target;
    }
    return $extracted;
}
