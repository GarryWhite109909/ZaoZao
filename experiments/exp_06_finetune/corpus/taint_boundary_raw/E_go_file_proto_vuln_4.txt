package opscmd

import (
	"database/sql"
	"encoding/json"
	"fmt"
	"os"
)

type TaskEntry struct {
	Name    string `json:"name"`
	Command string `json:"command"`
	Owner   string `json:"owner"`
}

// LoadTasks 解析任务清单 JSON 文件，并将每条任务写入审计表。
func LoadTasks(db *sql.DB, path string) error {
	data, err := os.ReadFile(path) // 第 20 行：污点源，文件内容外部可控
	if err != nil {
		return err
	}
	var tasks []TaskEntry
	if err := json.Unmarshal(data, &tasks); err != nil { // 第 24 行：解析不可信内容
		return err
	}
	for _, t := range tasks {
		owner := sanitize(t.Owner) // 第 27 行：弱防御，仅黑名单替换
		q := fmt.Sprintf("INSERT INTO audit_tasks(name, cmd, owner) VALUES('%s','%s','%s')",
			t.Name, t.Command, owner) // 第 29 行：字符串拼接构造 SQL
		if _, err := db.Exec(q); err != nil { // 第 31 行：sink，SQL 执行
			return err
		}
	}
	return nil
}

// sanitize 黑名单过滤单引号，可被双写绕过
func sanitize(s string) string {
	out := ""
	for _, c := range s {
		if c != ''' {
			out += string(c)
		}
	}
	return out
}
