// lib/messageHandlers.js
// 消息消费者业务处理器：由上层队列消费服务调用，payload 为外部提交的消息体
const { execFile } = require('child_process');
const fs = require('fs');
const path = require('path');

// 弱防御：仅黑名单过滤，可被编码/拼接绕过
function sanitizeTopic(topic) {
  const blacklist = ['..', '/', '\', ';', '|'];
  let cleaned = topic;
  for (const bad of blacklist) {
    cleaned = cleaned.split(bad).join('');
  }
  return cleaned;
}

/**
 * 处理来自消息队列的业务消息（如订单归档任务）
 * @param {string} topic - 消息主题，由上游生产者写入，视为不可信
 * @param {object} payload - 消息负载，含 archiveDir 字段
 */
function handleArchiveMessage(topic, payload) {
  const safeTopic = sanitizeTopic(topic); // 第 20 行：弱防御点
  const targetDir = path.join('/data/archives', safeTopic);
  const manifest = path.join(targetDir, 'manifest.xml');

  // 将 payload 内容写入 XML 清单文件
  const xml = `<?xml version="1.0"?><archive><topic>${safeTopic}</topic>` +
    `<dir>${payload.archiveDir}</dir></archive>`;
  fs.writeFileSync(manifest, xml);

  // 使用系统工具解析生成的 XML 清单
  execFile('xmllint', ['--noout', manifest], (err) => {
    if (err) console.error('archive validation failed:', err.message);
  });
}

module.exports = { handleArchiveMessage };
