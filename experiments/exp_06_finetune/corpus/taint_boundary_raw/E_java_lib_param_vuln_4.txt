package com.acme.etl.adapter;

import org.w3c.dom.Document;
import javax.xml.parsers.DocumentBuilderFactory;
import java.io.ByteArrayInputStream;
import java.nio.charset.StandardCharsets;

/**
 * 外部记录导入适配器：被上层同步服务调用，参数 record 为外部系统推送的原始报文。
 */
public class RecordImportAdapter {

    /** 弱防御：仅做关键字黑名单替换，可被大小写/嵌套实体绕过 */
    private static String sanitize(String raw) {
        return raw.replace("ENTITY", "").replace("entity", "");
    }

    public ImportResult parseRecord(String record) {
        try {
            String cleaned = sanitize(record);
            DocumentBuilderFactory dbf = DocumentBuilderFactory.newInstance();
            // 未禁用 DTD 与外部实体
            Document doc = dbf.newDocumentBuilder()
                    .parse(new ByteArrayInputStream(cleaned.getBytes(StandardCharsets.UTF_8)));
            return new ImportResult(doc.getDocumentElement().getTagName(), true);
        } catch (Exception e) {
            return new ImportResult("parse_error", false);
        }
    }
}
