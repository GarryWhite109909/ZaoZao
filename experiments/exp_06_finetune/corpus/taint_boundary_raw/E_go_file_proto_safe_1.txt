// Package webhookconf 解析出站 webhook 配置文件（由运维平台下发到本地磁盘），
// 上层投递服务调用 LoadAndDispatch 消费解析结果并向目标地址发送事件。
package webhookconf

import (
	"encoding/xml"
	"fmt"
	"net/http"
	"os"
	"strings"
	"time"
)

type HookConfig struct {
	XMLName xml.Name `xml:"hook"`
	URL     string   `xml:"url"`
	Event   string   `xml:"event"`
}

var allowedEvents = map[string]bool{"build": true, "deploy": true, "alert": true}

// LoadAndDispatch 读取配置文件并触发一次出站 webhook。
func LoadAndDispatch(path string) error {
	raw, err := os.ReadFile(path)
	if err != nil {
		return err
	}
	var cfg HookConfig
	if err := xml.Unmarshal(raw, &cfg); err != nil { // 第 24 行：解析外部可控的文件内容
		return fmt.Errorf("parse hook config: %w", err)
	}
	if !allowedEvents[cfg.Event] { // 第 27 行：白名单校验
		return fmt.Errorf("event not allowed: %s", cfg.Event)
	}
	u := strings.TrimSpace(cfg.URL)
	if !strings.HasPrefix(u, "https://") { // 第 30 行：协议白名单，仅允许 https
		return fmt.Errorf("url scheme rejected")
	}
	client := &http.Client{Timeout: 5 * time.Second}
	resp, err := client.Post(u, "application/json", strings.NewReader(`{"event":"`+cfg.Event+`"}`))
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	return nil
}
