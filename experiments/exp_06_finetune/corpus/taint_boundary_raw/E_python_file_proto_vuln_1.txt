```code
"""archive_indexer.py — 归档索引生成模块
被上层批处理服务调用，处理外部提交的归档文件。"""

import os
import tarfile
import html

INDEX_DIR = "/var/www/archive-index"

def _sanitize(name):
    # 弱防御：仅过滤 <script> 字面量
    return name.replace("<script>", "")

def build_index(archive_path):
    """解析归档，为每个成员生成 HTML 索引页"""
    with tarfile.open(archive_path, "r:*") as tf:
        for member in tf.getmembers():
            safe_name = _sanitize(member.name)
            page = "<html><body><h1>Archive: {}</h1></body></html>".format(safe_name)
            out = os.path.join(INDEX_DIR, os.path.basename(member.name) + ".html")
            with open(out, "w", encoding="utf-8") as f:
                f.write(page)
```

2. 第 19 行 `_sanitize` 仅做 `replace("<script>", "")` 字符串替换，可绕过：如 `<scr<script>ipt>alert(1)</script>` 去除后重组出 `<script>`，或直接用 `<img src=x onerror=alert(1)>`、大小写变体等不含该字面量的载荷。
3. 第 20 行将未净化的成员名经 `.format()` 直接拼入 HTML 正文，无任何转义（虽导入了 `html` 却未使用）。
4. 第 22-23 行将含污点的 HTML 写入 `/var/www/archive-index` 下的 .html 文件，该目录由 web 服务对外提供，形成存储型 XSS sink（CWE-79），source→sink 全程无有效防御。
