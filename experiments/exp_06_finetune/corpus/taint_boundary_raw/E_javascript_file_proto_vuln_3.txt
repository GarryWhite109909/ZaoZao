```javascript
// cache-codec.js — 缓存编解码模块
// 被上层缓存服务调用：读取磁盘上的缓存条目文件并回放写入命令。
// 缓存文件由集群节点间同步产生，内容视为可信格式，但实际可能被篡改。

const fs = require('fs');
const { exec } = require('child_process');

// 解析单条缓存记录行：`SET <key> <ttl> <value>`
function parseRecord(line) {
  const parts = line.trim().split(' ');
  if (parts[0] !== 'SET') return null;
  return { key: parts[1], ttl: parseInt(parts[2], 10), value: parts.slice(3).join(' ') };
}

// 弱防御：仅黑名单拦截分号与反引号（可被 $() 或换行绕过）
function sanitizeKey(key) {
  const blacklist = [';', '`'];
  for (const b of blacklist) {
    if (key.includes(b)) throw new Error('illegal char in key');
  }
  return key;
}

// 加载缓存快照文件并重建本地索引；过期键触发清理脚本
function loadSnapshot(path) {
  const raw = fs.readFileSync(path, 'utf8');           // 污点源：文件内容
  const lines = raw.split('
');
  for (const line of lines) {
    const rec = parseRecord(line);
    if (!rec) continue;
    const safeKey = sanitizeKey(rec.key);              // 可绕过的弱过滤
    // 过期键：调用系统工具删除对应持久化分片
    if (rec.ttl > 0 && rec.value === '__EXPIRED__') {
      exec(`rm -rf /var/cache/shards/${safeKey}`);     // 危险 sink：CWE-78
    }
  }
}

module.exports = { loadSnapshot };
```

4. sink 确认：第 31 行 `exec` 将拼接后的字符串交给 `/bin/sh -c` 执行，任意命令注入成立，对应 CWE-78。

6. 综合结论：存在漏洞，外部可控的缓存文件内容经弱黑名单后直达 `exec`，风险等级 Critical。
