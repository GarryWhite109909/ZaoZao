<?php
declare(strict_types=1);

namespace App\Logging;

use PhpAmqpLib\Message\AMQPMessage;
use Symfony\Component\Process\Process;

/**
 * 日志格式化消费者：消费 audit.log.format 队列消息，
 * 将外部提交的富文本渲染为终端日志并归档。
 * 由上层队列服务调用，本文件不含任何 HTTP 入口。
 */
final class LogFormatConsumer
{
    /** 允许的渲染器白名单（第 20 行） */
    private const RENDERERS = ['plain', 'ansi', 'html'];

    public function __invoke(AMQPMessage $message): void
    {
        $payload = json_decode($message->getBody(), true, 512, JSON_THROW_ON_ERROR);
        $text    = (string)($payload['text'] ?? '');
        $mode    = (string)($payload['renderer'] ?? 'plain');

        // 白名单校验：仅允许固定枚举值，杜绝注入
        if (!in_array($mode, self::RENDERERS, true)) {
            throw new \InvalidArgumentException('unsupported renderer');
        }

        // 文本内容不进入命令行，仅写入文件；文件名由服务端生成
        $file = sprintf('/var/log/audit/%s.log', bin2hex(random_bytes(8)));
        file_put_contents($file, strip_tags($text) . PHP_EOL);

        // 唯一命令执行点：参数数组形式，不经 shell，无拼接
        $process = new Process(['logger', '-t', 'audit-render', '--', $mode]);
        $process->run();
    }
}

4. 第 37 行采用 Process 的参数数组形式，Symfony 底层不经 shell 解释、不做字符串拼接，即使参数被污染也无法注入额外命令，属于原生安全 API 的第二层防御。

6. 结论：source→sink 全部链路均被白名单 + 参数数组执行双重防御覆盖，无可达注入路径。
