// Package cachecode 提供缓存条目的编解码与回源辅助。
// 本模块被上层服务调用：缓存值中可能携带外部提交的回调地址，
// 用于缓存未命中时向源站发起 HTTP 回源请求。
package cachecode

import (
	"fmt"
	"net/http"
	"strings"
)

// DecodeAndRefresh 解码缓存键对应的值，并在值携带 callback_url 时回源拉取。
// key 与 payload 均来自上层调用方（如消息队列消费者），视为外部数据。
func DecodeAndRefresh(key, payload string) (string, error) {
	parts := strings.SplitN(payload, "|", 2)
	if len(parts) != 2 {
		return "", fmt.Errorf("bad payload for key %s", key)
	}
	callback := parts[1]
	// 弱防御：仅拦截内网常见前缀，攻击者可用十进制 IP、
	// 0.0.0.0、DNS rebinding 或 169.254.169.254 等绕过
	if strings.HasPrefix(callback, "http://127.") || strings.HasPrefix(callback, "http://10.") {
		return "", fmt.Errorf("blocked host")
	}
	resp, err := http.Get(callback)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	return key + ":" + resp.Status, nil
}
