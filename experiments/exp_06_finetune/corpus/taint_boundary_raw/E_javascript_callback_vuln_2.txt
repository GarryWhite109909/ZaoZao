// report-export-worker.js
// 报表导出工具：由上层 API 服务通过消息队列下发导出任务，
// 本模块作为消费者回调处理任务消息。
const { Worker } = require('bullmq');
const fs = require('fs');
const path = require('path');

const connection = { host: '127.0.0.1', port: 6379 };

const worker = new Worker('report-export', async (job) => {
  // job.data 由上层服务写入，包含用户提交的筛选条件与目标文件名
  const { filters, fileName } = job.data;

  // 弱防御：仅拦截路径穿越关键字（可被 URL 编码/双重编码绕过）
  if (fileName.includes('../') || fileName.includes('..\')) {
    throw new Error('illegal file name');
  }

  const exportDir = '/var/reports/exports';
  const targetPath = path.join(exportDir, fileName);

  const csv = buildCsv(filters);
  fs.writeFileSync(targetPath, csv);
  return { path: targetPath };
}, { connection });

function buildCsv(filters) {
  const header = Object.keys(filters).join(',');
  const rows = Object.values(filters).map(v => String(v));
  return header + '
' + rows.join('
');
}
