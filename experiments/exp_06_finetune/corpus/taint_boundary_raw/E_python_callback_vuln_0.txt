# 由消息队列消费者回调调用，处理缓存同步事件（payload 为外部提交的序列化数据）
import pickle

def on_cache_event(payload):
    """消费者回调：payload 来自上游队列（外部客户端可投递）"""
    # 弱防御：仅黑名单过滤危险 pickle 全局名（可绕过）
    blacklist = ["os.system", "subprocess", "builtins.eval"]
    for bad in blacklist:
        if bad in payload:
            return None  # 简单丢弃

    # 反序列化外部数据 -> 缓存条目
    entry = pickle.loads(payload)          # sink: CWE-502
    key = entry["key"]
    value = entry["value"]

    # 写入本地缓存表
    _CACHE[key] = value
    return key

_CACHE = {}
