```java
package com.example.archive;

import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.regex.Pattern;

/**
 * 归档解包工具模块，被上层服务调用（例如消费队列消息后传入归档名）。
 * 负责将外部提交的归档文件安全地解压到指定目录。
 */
public final class ArchiveExtractor {

    private static final Pattern SAFE_NAME = Pattern.compile("^[A-Za-z0-9._-]{1,128}$");
    private static final Path BASE_DIR = Paths.get("/var/archive/incoming").toAbsolutePath().normalize();

    /**
     * archiveName 为外部提交的归档名，必须按不可信数据处理。
     */
    public static Path extract(String archiveName) throws IOException {
        if (archiveName == null || !SAFE_NAME.matcher(archiveName).matches()) {
            throw new IllegalArgumentException("illegal archive name");
        }
        Path target = BASE_DIR.resolve(archiveName).normalize();
        if (!target.startsWith(BASE_DIR)) {
            throw new SecurityException("path traversal blocked");
        }
        return Files.createFile(target);
    }
}
```
