package com.acme.etl.adapter;

import java.io.File;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Locale;
import java.util.Set;

/**
 * 数据导入适配器：被上层同步服务调用，将外部系统推送的记录
 * 转换为本地归档文件。本模块不感知任何传输协议。
 */
public final class RecordImportAdapter {

    /** 允许的目标数据集白名单（与后端注册表一致） */
    private static final Set<String> ALLOWED_DATASETS =
            Set.of("customers", "orders", "inventory");

    /**
     * 导入一条外部记录。dataset 与 payload 均来自外部系统，
     * 必须按不可信数据处理。
     *
     * @param dataset 目标数据集标识
     * @param payload 记录内容（已由上层反序列化为纯文本）
     * @return 归档文件路径
     */
    public File importRecord(String dataset, String payload) throws IOException {
        // 防御：严格白名单校验，仅允许预注册的数据集名
        if (dataset == null || !ALLOWED_DATASETS.contains(dataset.toLowerCase(Locale.ROOT))) {
            throw new IllegalArgumentException("unknown dataset: " + dataset);
        }
        String safeDataset = dataset.toLowerCase(Locale.ROOT);

        List<String> command = new ArrayList<>();
        command.add("/usr/local/bin/ingest");
        command.add("--dataset");
        command.add(safeDataset);
        command.add("--data");          // 参数数组形式，payload 不经过 shell 解析
        command.add(payload == null ? "" : payload);

        ProcessBuilder pb = new ProcessBuilder(command);
        pb.redirectErrorStream(true);
        Process proc = pb.start();
        try {
            int code = proc.waitFor();
            if (code != 0) {
                throw new IOException("ingest failed with exit code " + code);
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new IOException("interrupted during ingest", e);
        }
        return new File("/var/lib/etl/archive/" + safeDataset + ".dat");
    }
}
