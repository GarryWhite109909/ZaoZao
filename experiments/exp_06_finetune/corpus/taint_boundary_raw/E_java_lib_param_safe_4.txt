package com.acme.etl.adapter;

import org.w3c.dom.Document;
import javax.xml.XMLConstants;
import javax.xml.parsers.DocumentBuilderFactory;
import java.io.ByteArrayInputStream;
import java.nio.charset.StandardCharsets;

/**
 * 数据导入适配器：被上层同步服务调用，将外部系统推送的记录
 * （以 XML 字符串形式经参数传入）解析为内部 Document 对象。
 */
public final class RecordImportAdapter {

    /**
     * 解析一条外部记录。recordXml 由上游队列消费者/外部系统调用方传入，
     * 内容完全不可信。
     */
    public static Document parseRecord(String recordXml) throws Exception {
        if (recordXml == null || recordXml.isEmpty()) {
            throw new IllegalArgumentException("empty record");
        }
        if (recordXml.length() > 1 << 20) {
            throw new IllegalArgumentException("record too large");
        }

        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        // 禁用 DTD，彻底阻断 XXE（CWE-611）
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
        factory.setXIncludeAware(false);
        factory.setExpandEntityReferences(false);

        return factory.newDocumentBuilder()
                .parse(new ByteArrayInputStream(recordXml.getBytes(StandardCharsets.UTF_8)));
    }
}
