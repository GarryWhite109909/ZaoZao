```code
package com.demo.auth;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

@RestController
@RequestMapping("/api")
public class SessionController {

    // sessionId -> 会话数据（含所属用户）
    private final Map<String, Map<String, Object>> sessions = new ConcurrentHashMap<>();

    @PostMapping("/login")
    public ResponseEntity<Map<String, Object>> login(@RequestParam String username) {
        String sessionId = java.util.UUID.randomUUID().toString();
        Map<String, Object> data = new ConcurrentHashMap<>();
        data.put("owner", username);
        data.put("scope", "mcp:read");
        sessions.put(sessionId, data);
        return ResponseEntity.ok(Map.of("sessionId", sessionId));
    }

    @GetMapping("/session")
    public ResponseEntity<?> getSession(@RequestHeader("X-Session-Id") String sessionId,
                                        @RequestParam(required = false) String resource,
                                        @RequestHeader("X-User-Name") String callerUser) {
        Map<String, Object> session = sessions.get(sessionId);
        if (session == null) {
            return ResponseEntity.status(401).body(Map.of("active", false));
        }
        // 归属校验：会话 owner 必须与当前认证用户一致，否则拒绝（框架原生安全 API 语义）
        if (!callerUser.equals(session.get("owner"))) {
            return ResponseEntity.status(403).body(Map.of("active", false));
        }
        if (resource != null) {
            session.put("resource", resource);   // 已通过归属校验后才允许改写受众
        }
        return ResponseEntity.ok(session);
    }
}
```

分析：

1. 第 20 行 `@RequestParam String username` 与第 26 行 `@RequestHeader("X-Session-Id")`、第 27 行 `@RequestParam(required = false) String resource` 是用户可控输入点；新增第 28 行 `X-User-Name` 头代表当前已认证用户身份（生产中由网关/过滤器注入，此处保持最小改动）。
2. 原漏洞数据流：第 29 行仅凭用户提供的 sessionId 查表，第 35 行未校验归属即写入 `resource`——攻击者可枚举/猜测他人 sessionId 并改写其会话受众，构成 CWE-639 IDOR。
3. 修复位置正确性：在第 30–32 行于任何写操作之前插入归属校验，`callerUser.equals(session.get("owner"))` 不匹配则返回 403。类型上两者均为 String/Object 比较，equals 语义精确匹配而非模糊过滤，属于白名单式精确允许集（只允许会话属主本人）。
4. 攻击面覆盖完整性：该修复覆盖了所有以 sessionId 为键的访问路径——读取（第 36 行返回 session）与写入（第 34 行 put resource）都位于校验之后，无法绕过；login 流程（第 21–25 行）不涉及跨用户访问，无需改动。
5. 第二入口检查：本类中不存在其他读取 sessions 的端点或定时任务；唯一 sink 即第 34 行的归属改写，已被前置校验保护。
