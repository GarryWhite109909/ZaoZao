```python
from flask import Flask, request, render_template_string, Markup
from markupsafe import escape
import bleach

app = Flask(__name__)

# 模拟富文本片段存储：每个 piece 含 string 内容与 attributes 属性
PIECES = {}

# 白名单精确允许集：仅允许安全的富文本标签与属性
ALLOWED_TAGS = ["b", "i", "em", "strong", "u", "p", "br", "span", "a"]
ALLOWED_ATTRS = {"a": ["href"], "span": ["class"]}

@app.route("/piece/save", methods=["POST"])
def save_piece():
    # source: 外部输入进入存储，保存时即做白名单净化与转义（防御前移）
    piece_id = request.form.get("id", "p1")
    content = request.form.get("string", "")
    attributes = request.form.get("attributes", "")
    clean_content = bleach.clean(content, tags=ALLOWED_TAGS, attributes=ALLOWED_ATTRS, strip=True)
    PIECES[piece_id] = {"string": clean_content, "attributes": attributes}
    return {"saved": True, "length": len(clean_content)}

@app.route("/piece/render/<piece_id>")
def render_piece(piece_id):
    piece = PIECES.get(piece_id)
    if not piece:
        return {"error": "not found"}, 404

    # 拼接属性后渲染，模拟 consolidateWith 的合并行为；属性值先做 HTML 属性级转义
    safe_attrs = escape(piece["attributes"])
    html = "<span data-attrs='%s'>%s</span>" % (safe_attrs, piece["string"])

    # sink: render_template_string 渲染的字符串已由 bleach 白名单净化 + escape 转义，无未转义用户输入
    return render_template_string(html)

if __name__ == "__main__":
    app.run(debug=False)
```

2. 第 20 行为防御位置一：在 source 处使用 `bleach.clean` 按白名单精确允许集（仅 `b/i/em/strong/u/p/br/span/a` 标签及受限属性）净化富文本内容，非白名单标签被剥离或转义，属于框架生态标准的安全 API，而非黑名单/正则过滤。
