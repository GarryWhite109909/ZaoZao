```php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

class SourceMapController extends Controller
{
    // GET /sourcemap?cssFile=storage/css/app.css&mapUrl=app.css.map
    public function load(Request $request)
    {
        $cssFile = $request->query('cssFile');
        $mapUrl  = $request->query('mapUrl');

        if (!$cssFile || !$mapUrl) {
            return response()->json(['error' => 'missing params'], 422);
        }

        // 模拟 previous-map 的 loadAnnotation：从 CSS 注释提取 sourceMappingURL
        if (str_starts_with($mapUrl, 'data:application/json')) {
            $inline = substr($mapUrl, strpos($mapUrl, ',') + 1);
            return response()->json(['inline' => base64_decode($inline)]);
        }

        // 修复：白名单精确允许集 —— map 文件必须与 cssFile 同目录、同名、且以 .map 结尾，
        // 并通过 Storage 门面（框架原生安全 API）在受限磁盘根内解析，杜绝目录逃逸。
        $disk     = Storage::disk('local');
        $baseDir  = dirname($cssFile);                       // 例如 storage/css
        $expected = basename($cssFile, '.css') . '.map';     // app.css -> app.css.map

        if ($mapUrl !== $expected) {
            return response()->json(['error' => 'invalid map path'], 403);
        }

        $relativePath = ltrim($baseDir . '/' . $expected, '/');

        if (!$disk->exists($relativePath)) {
            return response()->json(['error' => 'map not found'], 404);
        }

        $content = $disk->get($relativePath);

        return response()->json([
            'mapFile' => basename($relativePath),
            'root'    => dirname($relativePath),
            'text'    => $content,
        ]);
    }
}
```

1. 第 14-15 行：用户可控输入点为 `$request->query('cssFile')` 与 `$request->query('mapUrl')`，二者均直接来自查询字符串，是本数据流唯一外部源；第 20 行 data URI 分支不触达文件系统，保持原样。
2. 原漏洞链：第 30-31 行将未校验的 `$mapUrl` 拼入路径，`../../.env.map` 可逃逸出 css 目录，第 34 行 `file_get_contents($real)` 即 sink，构成 CWE-22 路径穿越。
3. 修复位置正确性：在第 29-33 行于任何路径解析之前先做白名单精确允许集校验——`$mapUrl` 必须严格等于由 `cssFile` 推导出的同目录同名 `.map` 文件，类型为字符串全等比较，无绕过空间；`../`、绝对路径、URL 编码变体全部因不等于期望值而被拒绝。
4. 攻击面覆盖完整性：第 35 行改用框架原生安全 API `Storage::disk('local')->exists/get`，文件访问被限制在磁盘配置根目录之内，即使后续逻辑变化也无法越界读取；`realpath`/`file_get_contents` 直连文件系统的 sink 已被完全移除。
5. 数据流走向与业务场景不变：仍是 cssFile + mapUrl → 定位 sourcemap → 返回 JSON，仅替换了路径构造与读取方式，满足 minimal pair 局部对照。
6. 第二入口检查：data URI 内联分支（第 20-22 行）只做 base64 解码返回，不读文件系统，无路径穿越面；控制器无其他路由方法或额外请求字段，替代通道结论为不存在残余风险。
