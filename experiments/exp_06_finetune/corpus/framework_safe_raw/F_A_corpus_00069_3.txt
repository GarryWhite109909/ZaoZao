package com.example.auth;

import javax.servlet.*;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.crypto.Cipher;
import javax.crypto.spec.SecretKeySpec;
import javax.crypto.AEADBadTagException;
import java.io.IOException;
import java.util.Base64;
import java.util.Set;

public class TokenAuthFilter implements Filter {

    // 安全版本：仅允许 V2 配置，LEGACY 被白名单精确排除，
    // 不再存在任何回退到弱算法的代码路径
    private static final String SECURITY_PROFILE = System.getenv("SECURITY_PROFILE"); // 仅接受 V2
    private static final Set<String> ALLOWED_PROFILES = Set.of("V2");
    private static final byte[] KEY = "0123456789abcdef".getBytes();

    @Override
    public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain)
            throws IOException, ServletException {
        HttpServletRequest request = (HttpServletRequest) req;
        HttpServletResponse response = (HttpServletResponse) res;

        String token = request.getHeader("X-Auth-Token");
        if (token == null || token.isEmpty()) {
            response.sendError(401, "missing token");
            return;
        }

        try {
            String userId;
            if ("V2".equals(SECURITY_PROFILE) && ALLOWED_PROFILES.contains(SECURITY_PROFILE)) {
                userId = verifyV2(token);          // AES-GCM + 签名校验
            } else {
                response.sendError(401, "unsupported security profile");
                return;                            // 拒绝而非回退到弱算法
            }
            request.setAttribute("userId", userId);
            chain.doFilter(req, res);
        } catch (AEADBadTagException e) {
            response.sendError(401, "invalid token");
        } catch (Exception e) {
            response.sendError(401, "invalid token");
        }
    }

    private String verifyV2(String token) throws Exception {
        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
        cipher.init(Cipher.DECRYPT_MODE, new SecretKeySpec(KEY, "AES"));
        return new String(cipher.doFinal(Base64.getDecoder().decode(token)));
    }
}

2. 第 31-36 行：原漏洞的 sink 是第 44 行 `decryptLegacy` 内的 `Cipher.getInstance("AES/ECB/PKCS5Padding")`。安全版本在第 32 行将配置判断收紧为双重校验：环境变量必须等于 'V2' 且命中白名单集合 `ALLOWED_PROFILES`（第 15 行，`Set.of("V2")`），否则第 34-35 行直接返回 401，删除了整个 `decryptLegacy` 方法及其调用路径——这是框架原生安全 API（JCA 强算法）加白名单精确允许集的组合防御。
3. 类型与位置正确性：白名单校验发生在进入任何解密逻辑之前（第 32 行），位于控制流分叉点，保证不存在任何分支能到达弱算法；`AEADBadTagException` 单独捕获（第 39 行）确保 GCM 认证标签校验失败时拒绝令牌，防止篡改与重放绕过。
4. 攻击面覆盖完整性：ECB 的确定性加密、无认证、块重排三类攻击全部随 `decryptLegacy` 的移除而消除；GCM 模式自带完整性认证，密文被替换或重放会导致标签校验失败并落入 401 分支。
5. 第二入口检查：除 `X-Auth-Token` 头外，过滤器未读取 cookie、参数或 body；`SECURITY_PROFILE` 来自环境变量而非用户请求，不可被攻击者操控，且已被白名单约束，无替代通道可触达已删除的 ECB 路径。
