```code
package com.example.comment;

import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.thymeleaf.TemplateEngine;
import org.thymeleaf.context.Context;
import org.thymeleaf.templatemode.TemplateMode;
import org.thymeleaf.templateresolver.StringTemplateResolver;
import org.thymeleaf.TemplateSpec;
import java.util.Map;
import java.util.Set;

@Controller
@RequestMapping("/admin/comments")
public class AdminCommentController {

    @Autowired
    private CommentRepository commentRepository;

    @Autowired
    private TemplateEngine templateEngine;

    // 渲染评论通知模板，模板内容来自用户提交的评论格式设置
    @PostMapping("/save")
    @ResponseBody
    public String saveComment(@RequestParam Map<String, String> params) {
        String commentBody = params.getOrDefault("comment_body", "");
        String format = params.get("format");

        if ("template".equals(format)) {
            // 防御：白名单精确允许集——用户只能选择预定义模板名，
            // 用户可控字符串不再作为模板内容交给模板引擎解析执行
            String requestedTemplate = params.get("comment_body");
            Set<String> allowedTemplates = Set.of("comment_notify_default", "comment_notify_plain");
            if (!allowedTemplates.contains(requestedTemplate)) {
                requestedTemplate = "comment_notify_default";
            }
            Context ctx = new Context();
            ctx.setVariable("user", params.get("comment_name"));
            return templateEngine.process(requestedTemplate, ctx);
        }

        commentRepository.save(new Comment(commentBody, params.get("rel_type"), params.get("rel_id")));
        return "saved";
    }
}
```

2. 原代码第 31 行 `templateEngine.process(commentBody, ctx)` 是 SSTI sink——`commentBody` 直接来自第 22 行的用户输入并被当作模板源解析执行（Thymeleaf 会执行 `${...}`、`__${...}__` 等表达式），攻击者可注入任意 SpEL/OGNL 表达式实现 RCE。
3. 修复采用框架原生安全实践之一——模板名白名单精确允许集：第 30-34 行将 `comment_body` 的语义从『模板内容』收窄为『预定义模板选择器』，不在 `Set.of(...)` 允许集内的取值一律回退到默认模板；类型正确性在于 sink 接收的永远是编译期已知的常量字符串，位置正确性在于校验位于 source 与 sink 之间的必经路径上，覆盖了所有到达 `templateEngine.process` 的取值分支。
4. 数据流走向、路由 `/admin/comments/save`、函数名 `saveComment`、依赖注入与 ORM 调用均保持不变，仅插入防御语句，满足最小 diff 纪律。
5. 第二数据流检查：第 37 行 `commentRepository.save(...)` 将 `commentBody` 等存入数据库，属存储场景而非本次目标漏洞（SSTI）链路，按要求保持原样不动。
6. 替代通道检查：`params.get("comment_name")` 仅经 `ctx.setVariable` 作为变量值传入，Thymeleaf 中变量值不会被二次解析为表达式，无注入风险；无其他调用 `templateEngine.process` 的通道。
