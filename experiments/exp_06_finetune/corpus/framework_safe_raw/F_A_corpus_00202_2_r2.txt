```java
package com.example.ghost.controller;

import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.jdbc.core.JdbcTemplate;
import org.jsoup.Jsoup;
import org.jsoup.safety.Safelist;

@Controller
@RequestMapping("/admin/posts")
public class PostHistoryController {

    private final JdbcTemplate jdbc;

    public PostHistoryController(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    // 查看历史修订：将数据库中存储的 revision html 经白名单净化后注入模板
    @GetMapping("/{postId}/history")
    public String history(@PathVariable Long postId, Model model) {
        var revisions = jdbc.queryForList(
            "SELECT id, title, html FROM post_revisions WHERE post_id = ? ORDER BY created_at DESC", postId);
        for (var r : revisions) {
            r.put("html", Jsoup.clean((String) r.get("html"), Safelist.basicWithImages()));
        }
        model.addAttribute("revisions", revisions);
        return "post-history"; // Thymeleaf 模板中使用 th:utext="${r['html']}" 渲染
    }

    // 恢复修订：把 staff 用户提交的 revision html 经白名单净化后写回当前帖子的正文
    @PostMapping("/{postId}/restore")
    @ResponseBody
    public String restore(@PathVariable Long postId, @RequestParam("revisionId") long revisionId) {
        String html = jdbc.queryForObject(
            "SELECT html FROM post_revisions WHERE id = ? AND post_id = ?", String.class, revisionId, postId);
        // 白名单净化后再持久化为帖子正文，其他 staff 打开该帖不会触发脚本执行
        String safeHtml = Jsoup.clean(html, Safelist.basicWithImages());
        jdbc.update("UPDATE posts SET html = ? WHERE id = ?", safeHtml, postId);
        return "restored";
    }
}
```

2. 主漏洞链一（存储 XSS）：数据库中存储的 `html` 字段 → 第 25 行 `model.addAttribute` → Thymeleaf `th:utext` 原样输出。原代码未净化即渲染，攻击者可持久化 `<script>` 载荷。
3. 修复位置正确性：在第 26-28 行对每条修订的 `html` 用 `Jsoup.clean(..., Safelist.basicWithImages())` 白名单净化——只保留基本标签与图片，剥离 script/style/event handler 属性；这是框架无关的精确允许集方案，属于有效防御而非黑名单过滤。
4. 主漏洞链二（恢复写入）：第 34-35 行读出的 `html` 在第 38 行经同一白名单净化为 `safeHtml` 后才于第 39 行持久化到 posts 表，从源头阻断恶意载荷入库，覆盖了后续任何以 `th:utext` 渲染正文的读取通道。

6. 第二入口/替代通道检查：类内无其他写库或模板输出路径；SQL 均已参数化，无 SQL 注入；除上述两处外无新增用户可控输入点。
