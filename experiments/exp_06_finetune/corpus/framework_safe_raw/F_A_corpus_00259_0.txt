import re

from django.db import connection
from django.http import JsonResponse
from django.views.decorators.http import require_POST


def _validate_identifier(name):
    """Whitelist check: allow only safe SQL identifiers (table/column names)."""
    return bool(re.fullmatch(r"[A-Za-z_][A-Za-z0-9_.]*", name))


@require_POST
def column_config(request):
    """Builds a custom report query from user-supplied SQL fragments."""
    config = request.POST.get("sql", "")
    from_clause = request.POST.get("from", "")
    where_clause = request.POST.get("where", "")
    group_by = request.POST.get("groupby", "")

    sql = "SELECT "
    if not config.strip().upper().startswith("SELECT"):
        sql += "*"
    else:
        select_part = config.strip()[len("SELECT"):].strip()
        if not all(_validate_identifier(part.strip()) for part in select_part.split(",")):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += "SELECT " + select_part
    if from_clause:
        if not _validate_identifier(from_clause.strip()):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += " FROM " + from_clause.strip()
    if where_clause:
        cleaned = re.sub(r"^\s*WHERE\s*", "", where_clause, flags=re.I)
        # parameterized: replace user values with placeholders bound via params list
        params = []
        def _param_sub(match):
            params.append(match.group(1).strip("'""))
            return "%s"
        cleaned = re.sub(r"'([^']*)'|"([^"]*)"", _param_sub, cleaned)
        if not re.fullmatch(r"[\w\s=<>!(),.%s\[\]]+", cleaned):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += " WHERE (" + cleaned + ")"
    if group_by:
        parts = [p.strip() for p in group_by.split(",")]
        if not all(_validate_identifier(p) for p in parts):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += " GROUP BY " + ", ".join(parts)

    with connection.cursor() as cursor:
        cursor.execute(sql, params)
        row = cursor.fetchone()

    columns = list(row.keys()) if row else []
    return JsonResponse({"columns": columns})
