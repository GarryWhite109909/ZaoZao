package middleware

import (
	"net/http"
	"os"
	"path/filepath"
	"strings"

	"github.com/gin-gonic/gin"
)

// StaticFileAuth 鉴权中间件：校验 token 后按 X-Target-Path 头返回受保护目录下的文件
func StaticFileAuth(protectedRoot string) gin.HandlerFunc {
	return func(c *gin.Context) {
		token := c.GetHeader("X-Access-Token")
		if token != "expected-session-token" {
			c.AbortWithStatus(http.StatusUnauthorized)
			return
		}

		// 从请求头取目标文件路径（source）
		target := c.GetHeader("X-Target-Path")
		if target == "" {
			c.AbortWithStatus(http.StatusBadRequest)
			return
		}

		// 白名单精确允许集：仅放行受保护目录下已登记的合法相对路径
		allowed := map[string]bool{
			"reports/summary.pdf":  true,
			"docs/manual.pdf":      true,
			"data/export.csv":      true,
		}
		cleaned := strings.ReplaceAll(target, "<", "")
		cleaned = strings.ReplaceAll(cleaned, ">", "")
		if !allowed[cleaned] {
			c.AbortWithStatus(http.StatusForbidden)
			return
		}

		// 拼接后再次用 filepath.Rel 校验解析结果仍在根目录内（纵深防御）
		full := filepath.Join(protectedRoot, cleaned)
		rel, err := filepath.Rel(protectedRoot, full)
		if err != nil || rel == ".." || strings.HasPrefix(rel, ".."+string(filepath.Separator)) {
			c.AbortWithStatus(http.StatusForbidden)
			return
		}

		data, err := os.ReadFile(full)
		if err != nil {
			c.AbortWithStatus(http.StatusNotFound)
			return
		}
		c.Data(http.StatusOK, "application/octet-stream", data)
	}
}

func RegisterRoutes(r *gin.Engine) {
	r.GET("/protected/file", StaticFileAuth("/var/app/protected"))
}

2. 原代码第 29–30 行的 `strings.ReplaceAll` 仅删除 `<`/`>`，属于黑名单式表面清洗，无法阻止 `../../etc/passwd` 或绝对路径 `/etc/passwd`（`filepath.Join` 会丢弃根前缀），第 32 行 `os.ReadFile(full)` 即为任意文件读取 sink。
3. 修复在第 33–39 行插入白名单精确允许集：请求路径必须逐字命中预先登记的合法相对路径集合，未命中即返回 403。这是白名单精确允许集防御，类型正确（字符串精确匹配）、位置正确（在 Join 与 ReadFile 之前、清洗之后），完全覆盖 `..` 穿越、绝对路径、URL 编码变体等攻击面。
4. 第 42–46 行追加 `filepath.Rel` 纵深校验：确认最终解析路径相对 protectedRoot 不含 `..` 前缀，防止未来白名单条目被误配成穿越形式时仍能兜底；两道防御均位于 sink 之前，控制流其余部分不变。
5. 第二入口检查：路由注册第 58 行仅此一个 GET 入口挂载该中间件，无其他 handler 直接读取 protectedRoot；token 鉴权逻辑保持原样未改动，符合 minimal pair 局部对照要求。
