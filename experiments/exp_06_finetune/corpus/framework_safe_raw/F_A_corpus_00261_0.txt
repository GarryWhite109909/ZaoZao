from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.db import connection


@csrf_exempt
def rewrite_base(request):
    """Adjusts the RewriteBase and ErrorDocument 404 entries for the user's environment."""
    if request.method != 'POST':
        return JsonResponse({'error': 'method not allowed'}, status=405)

    server_path = request.POST.get('server_path', '/')
    error_code = request.POST.get('error_code', '404')

    with connection.cursor() as cursor:
        # Look up the current rewrite base record for this installation path
        cursor.execute(
            "SELECT id, rewrite_base FROM setup_htaccess WHERE server_path = %s",
            [server_path]
        )
        row = cursor.fetchone()
        if row is None:
            return JsonResponse({'error': 'no htaccess config found'}, status=404)

        # Update the rewrite base and the matching ErrorDocument directive
        new_404_path = server_path.rstrip('/') + '/index.php?action=404'
        cursor.execute(
            "UPDATE setup_htaccess SET rewrite_base = %s, "
            "error_document = %s WHERE server_path = %s",
            [server_path, error_code + ' ' + new_404_path, server_path]
        )

    return JsonResponse({'status': 'ok'})
