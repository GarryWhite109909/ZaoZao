```code
# cache_frontend/views.py
import hmac
import json
from django.http import JsonResponse, HttpResponseBadRequest
from django.views.decorators.csrf import csrf_exempt
from django.core.cache import cache
from django.conf import settings
from django.core import signing


@csrf_exempt
def cache_entry(request, entry_identifier):
    if request.method == 'POST':
        if not entry_identifier.isalnum():
            return HttpResponseBadRequest('invalid cache entry identifier')
        data = request.body
        tags = request.POST.getlist('tags')
        lifetime = request.POST.get('lifetime')

        # invoke registered signal hooks before storing (mirrors TYPO3 hook dispatch)
        for hook_path in settings.CACHE_SET_HOOKS:
            module_name, func_name = hook_path.rsplit('.', 1)
            __import__(module_name)
            getattr(__import__(module_name), func_name)(data)

        # persistent backend: serialize with framework-native signed serializer (no pickle)
        payload = signing.dumps({'data': data.decode('utf-8', 'replace'), 'tags': tags})
        cache.set(entry_identifier, payload, timeout=lifetime or None)
        return JsonResponse({'status': 'stored'})

    elif request.method == 'GET':
        raw = cache.get(entry_identifier)
        if raw is None:
            return JsonResponse({'found': False})
        # safe sink: verify HMAC signature via django.core.signing before trusting data
        try:
            obj = signing.loads(raw, serializer=signing.JSONSerializer)
        except signing.BadSignature:
            return HttpResponseBadRequest('tampered cache entry')
        return JsonResponse({'found': True, 'data': obj['data']})
```

1. 第 12 行 `entry_identifier`、第 15-17 行 `request.body`/`tags`/`lifetime` 为用户可控输入点；identifier 已有 isalnum 校验，lifetime 仅作超时参数不进入反序列化。
2. 原漏洞主链：第 19-22 行 hooks → 第 25 行 `pickle.dumps` 写入缓存 → GET 分支第 34 行 `pickle.loads(raw)` 为 sink；缓存后端可被攻击者污染（如 memcached 未鉴权），字节无完整性校验即反序列化，构成 CWE-502 RCE 链。
3. 修复位置正确性：写侧第 25 行改用 `django.core.signing.dumps`（框架原生安全 API），读侧第 33-37 行用 `signing.loads` 校验 HMAC 签名后才信任数据——类型为签名字符串而非任意 pickle 字节流，篡改或替换的载荷在 `BadSignature` 处被拒绝。
4. 攻击面覆盖：签名密钥来自 SECRET_KEY，攻击者无法伪造合法签名；JSONSerializer 替代 pickle 彻底消除对象实例化执行面，即使缓存后端完全被攻陷也无法触发代码执行。
5. 第二入口检查：POST 分支 hooks 调用（第 19-22 行）遍历的是 settings 白名单配置项而非用户输入，非本漏洞数据流，按 minimal diff 纪律保持原样；`__import__` 参数不可控，无新增风险。
