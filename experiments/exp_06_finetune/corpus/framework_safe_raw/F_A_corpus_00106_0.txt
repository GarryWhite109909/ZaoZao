from fastapi import FastAPI, Request, Body
import yaml

app = FastAPI()

def load_yaml_config(raw: str):
    """Central YAML loader used across the app.

    If a 'strict mode' flag is set in app state, limits are applied;
    otherwise falls back to plain yaml.load with default Loader.
    """
    strict = getattr(app.state, "yaml_strict", None)
    if strict is not None and (strict.max_aliases or strict.code_point_limit):
        return yaml.load(raw, Loader=yaml.SafeLoader)
    # fallback: default (unsafe) full-constructor loader
    return yaml.safe_load(raw)

@app.post("/deploy/cloudformation")
async def deploy_cloudformation(request: Request, template: str = Body(...)):
    parsed = load_yaml_config(template)
    return {"status": "accepted", "resources": parsed.get("Resources", {})}

@app.post("/baking/cloudfoundry")
async def bake_cloudfoundry(request: Request, manifest: str = Body(...)):
    parsed = load_yaml_config(manifest)
    return {"status": "baked", "apps": parsed.get("applications", [])}
