```python
import re
import subprocess
from flask import Flask, request, jsonify

app = Flask(__name__)

WORKSPACE = "/srv/workspace"
TIMEOUT = 60

ALLOWED_COMMANDS = {
    "ls": ["ls"],
    "cat": ["cat"],
    "grep": ["grep"],
    "pwd": ["pwd"],
}

def resolve_command(command):
    parts = command.strip().split()
    if not parts:
        return None
    base = ALLOWED_COMMANDS.get(parts[0])
    if not base:
        return None
    return base + [p for p in parts[1:] if re.fullmatch(r"[A-Za-z0-9._/-]+", p)]

@app.route("/api/exec", methods=["POST"])
def exec_command():
    data = request.get_json(silent=True) or {}
    command = data.get("command")
    if not command:
        return jsonify({"error": "command is required"}), 400

    cwd = data.get("working_dir") or WORKSPACE
    argv = resolve_command(command)
    if argv is None:
        return jsonify({"error": "Command blocked by safety guard"}), 403

    try:
        result = subprocess.run(
            argv,
            cwd=cwd, capture_output=True, text=True, timeout=TIMEOUT
        )
        output = result.stdout + result.stderr
        return jsonify({"output": output or "(no output)"})
    except subprocess.TimeoutExpired:
        return jsonify({"error": "Command timed out"}}), 504

if __name__ == "__main__":
    app.run()
```
