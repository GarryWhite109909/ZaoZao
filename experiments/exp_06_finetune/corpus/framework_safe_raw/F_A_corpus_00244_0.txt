from flask import Flask, request, jsonify, g
import sqlite3

app = Flask(__name__)

def get_db():
    conn = sqlite3.connect("gitea.db")
    conn.row_factory = sqlite3.Row
    return conn

@app.before_request
def load_current_user():
    uid = request.headers.get("X-User-Id")
    g.current_user_id = int(uid) if uid else None

def require_authenticated_user():
    if g.current_user_id is None:
        from flask import abort
        abort(401)

@app.route("/api/v1/user/starred")
def list_starred_repos():
    require_authenticated_user()
    include_private = request.args.get("private", "false") == "true"
    owner_id = request.args.get("owner_id", type=int)
    db = get_db()
    query = ("SELECT r.* FROM repository r INNER JOIN star s ON r.id = s.repo_id "
             "WHERE s.uid = ?")
    params = [g.current_user_id]
    if owner_id:
        query += " AND r.owner_id = ?"
        params.append(owner_id)
    # fixed: enforce authorization — non-owner may only see public repos;
    # private repos are only returned when the starred repo's owner is the
    # current user (server-side check, client param no longer trusted)
    if not include_private:
        query += " AND r.is_private = 0"
    else:
        query += " AND (r.is_private = 0 OR r.owner_id = ?)"
        params.append(g.current_user_id)
    rows = db.execute(query, params).fetchall()
    return jsonify([dict(r) for r in rows])

if __name__ == "__main__":
    app.run()

2. **追踪到 sink（第 33 行 `db.execute(query, params)`）**：数据流为 `request.args → include_private/owner_id → SQL 查询构造 → execute`。原代码在 `include_private=true` 时无任何服务端授权校验，任意已认证用户可枚举他人私有仓库。
3. **防御类型正确性**：修复采用框架原生 `abort(401)` 强制认证（新增辅助函数 `require_authenticated_user`），并在 SQL 层用参数化占位符 `?` 绑定 `g.current_user_id` 作为白名单式精确允许集——只有仓库属主本人才能通过私有过滤条件，属于服务端强制授权而非客户端参数信任。
4. **位置正确性**：授权检查插入在路由函数入口（认证）与查询构造处（授权谓词），位于 sink `execute` 之前，覆盖了所有到达私有数据的路径；未改动函数名、路由、变量名与整体控制流，满足最小 diff 纪律。
