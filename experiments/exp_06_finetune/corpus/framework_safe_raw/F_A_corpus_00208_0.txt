from flask import Flask, request, jsonify
import re
import bleach

app = Flask(__name__)

# 依赖此净化函数的下游会直接渲染返回的 SVG（信任其已无脚本）
# 白名单精确允许集：仅保留安全的 SVG 结构/绘图标签与属性，其余一律剥离或转义
SVG_ALLOWED_TAGS = [
    'svg', 'g', 'defs', 'title', 'desc', 'path', 'rect', 'circle', 'ellipse',
    'line', 'polyline', 'polygon', 'text', 'tspan', 'use', 'symbol'
]
SVG_ALLOWED_ATTRS = ['id', 'class', 'd', 'x', 'y', 'x1', 'y1', 'x2', 'y2',
                     'cx', 'cy', 'r', 'rx', 'ry', 'width', 'height',
                     'fill', 'stroke', 'stroke-width', 'transform',
                     'viewBox', 'points', 'text-anchor']
SVG_ALLOWED_PROTOCOLS = []  # 禁止所有 URL 协议，阻断 javascript:/data: 等

def sanitize_svg(svg_text):
    # 白名单式净化：非白名单标签（含 script、on* 事件属性、外链协议）全部被剥离/转义
    cleaned = bleach.clean(
        svg_text,
        tags=SVG_ALLOWED_TAGS,
        attributes=SVG_ALLOWED_ATTRS,
        protocols=SVG_ALLOWED_PROTOCOLS,
        strip=True,
    )
    return cleaned

@app.route('/api/svg/clean', methods=['POST'])
def clean_svg():
    svg_text = request.form.get('svg', '')
    if not svg_text:
        return jsonify({'error': 'missing svg field'}), 400
    result = sanitize_svg(svg_text)
    return result, 200, {'Content-Type': 'image/svg+xml'}

@app.route('/view/<doc_id>')
def view_doc(doc_id):
    # 第二入口：从存储读取并直接内联渲染用户上传的 SVG 片段
    snippet = load_user_snippet(doc_id)  # 假设来自数据库的用户内容
    return f'<div class="embed">{snippet}</div>'

def load_user_snippet(doc_id):
    import sqlite3
    conn = sqlite3.connect('docs.db')
    row = conn.execute('SELECT body FROM snippets WHERE id=?', (doc_id,)).fetchone()
    conn.close()
    return row[0] if row else ''
