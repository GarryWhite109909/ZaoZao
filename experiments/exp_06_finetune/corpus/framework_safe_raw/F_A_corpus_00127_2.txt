package main

import (
	"net/http"
	"net/url"
	"strings"

	"github.com/gin-gonic/gin"

// viewNameTranslator 把请求路径转换为模板视图名并重定向，
// 未显式指定视图名时使用（模拟 DefaultRequestToViewNameTranslator 语义）。
func viewNameTranslator(c *gin.Context) {
	prefix := ""
	suffix := ""
	stripLeadingSlash := true
	stripExtension := true

	// source: 攻击者完全可控的请求 URI 路径
	path := c.Request.URL.Path

	if stripLeadingSlash && strings.HasPrefix(path, "/") {
		path = path[1:]
	}
	if stripExtension {
		if idx := strings.LastIndex(path, "."); idx != -1 {
			path = path[:idx]
		}
	}

	target := prefix + path + suffix

	// 防御：白名单精确允许集——仅允许站内相对路径，拒绝任何含 scheme/主机成分的目标
	u, err := url.Parse(target)
	if err != nil || u.IsAbs() || u.Host != "" || strings.HasPrefix(target, "//") {
		c.String(http.StatusBadRequest, "invalid redirect target")
		return
	}

	// sink: 将经过白名单校验的相对路径作为 Location 目标发起 302 重定向
	c.Redirect(http.StatusFound, target)
}

func main() {
	r := gin.Default()
	r.GET("/*filepath", viewNameTranslator)
	r.Run(":8080")
}

2. 第 31 行拼接得到 target 后直接流入第 33 行 `c.Redirect`（sink），原代码无任何校验，构成 CWE-601 开放重定向。
3. 修复在第 34-38 行插入防御：先用标准库 `url.Parse` 做结构化解析（类型正确的框架原生 API），再以白名单式精确判定——绝对 URL、带 Host、协议相对路径 `//` 开头一律拒绝。
