```go
package main

import (
	"bytes"
	"crypto/hmac"
	"crypto/sha256"
	"encoding/gob"
	"net/http"

	"github.com/gin-gonic/gin"
)

// SessionState 是从缓存/客户端恢复的会话聚合状态，
// 与原 Java 侧从远端缓存读取后直接反序列化的结构等价。
type SessionState struct {
	UserID   string
	Roles    []string
	Payload  map[string]interface{}
}

var sessionMACKey = []byte("fixed-32-byte-session-mac-key-here!")

func verifyAndDecode(body []byte) (SessionState, error) {
	var state SessionState
	// 防御 1：分离 MAC（前 32 字节）与密文主体，先做 HMAC-SHA256 完整性/来源校验
	if len(body) < 33 {
		return state, errors.New("body too short")
	}
	mac, payload := body[:32], body[32:]
	expected := hmac.New(sha256.New, sessionMACKey)
	expected.Write(payload)
	if !hmac.Equal(mac, expected.Sum(nil)) {
		return state, errors.New("mac mismatch")
	}
	// 防御 2：仅对通过签名校验的字节执行 gob 反序列化
	dec := gob.NewDecoder(bytes.NewReader(payload))
	if err := dec.Decode(&state); err != nil {
		return state, err
	}
	// 防御 3：白名单精确允许集——Roles 只能取自固定角色集合
	allowed := map[string]bool{"admin": true, "user": true, "viewer": true}
	for _, r := range state.Roles {
		if !allowed[r] {
			return state, errors.New("role not allowed")
		}
	}
	return state, nil
}

func restoreSession(c *gin.Context) {
	// source: 客户端提交的原始序列化字节，未做任何校验
	body, err := c.GetRawData()
	if err != nil {
		c.String(http.StatusBadRequest, "bad body")
		return
	}

	state, derr := verifyAndDecode(body)
	if derr != nil {
		c.String(http.StatusInternalServerError, "decode failed")
		return
	}

	c.JSON(http.StatusOK, gin.H{"user": state.UserID, "roles": state.Roles})
}

func main() {
	r := gin.Default()
	r.POST("/session/restore", restoreSession)
	r.Run(":8080")
}
```

分析：

1. 第 30 行 `c.GetRawData()` 是唯一用户可控输入点：客户端 POST `/session/restore` 的原始请求体字节，完全攻击者可控，直接进入反序列化路径。
2. 原代码第 37 行 `dec.Decode(&state)` 是 sink：对不可信字节流直接 gob 反序列化，gob 可实例化任意注册类型并触发其 GobDecode 方法，等价于 Java 原生反序列化 RCE 面。修复版将解码逻辑移入 `verifyAndDecode` 辅助函数（防御必需的新增辅助函数，符合最小 diff 纪律），函数名、路由、变量名与控制流保持不变。
3. 类型正确性：HMAC 校验使用 `crypto/hmac` 的 `hmac.Equal` 常量时间比较，防止时序侧信道伪造；密钥为服务端持有，客户端无法为恶意 payload 计算合法 MAC。位置正确性：校验发生在 `gob.NewDecoder(...).Decode` 之前，未通过签名的字节流永远到达不了反序列化 sink，攻击面被完整覆盖。
4. 白名单精确允许集：第 47~52 行对恢复出的 `Roles` 逐项比对固定集合 `{admin,user,viewer}`，任何越权角色注入（即使带合法签名，如内部缓存被污染）也会被拒绝，属于精确允许集而非黑名单。
5. 第二入口检查：本文件仅有一个路由入口 `/session/restore`，无其他 handler、中间件或后台 goroutine 消费该字节流；`verifyAndDecode` 是唯一反序列化通道且已受保护，替代通道结论为不存在残余风险。
