import re

from django.db import connection
from django.http import JsonResponse
from django.views.decorators.http import require_POST


@require_POST
def column_config(request):
    """Builds a custom report query from user-supplied SQL fragments."""
    config = request.POST.get("sql", "")
    from_clause = request.POST.get("from", "")
    where_clause = request.POST.get("where", "")
    group_by = request.POST.get("groupby", "")

    # whitelist: only allow a fixed set of identifiers for each fragment
    def _is_safe_identifier(value):
        return bool(re.fullmatch(r"[A-Za-z_][A-Za-z0-9_.]*", value))

    sql = "SELECT "
    if not config.strip().upper().startswith("SELECT"):
        sql += "*"
    else:
        if not _is_safe_identifier(config.strip()[len("SELECT"):].strip()):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += config.strip()
    if from_clause:
        if not _is_safe_identifier(from_clause):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += " FROM " + from_clause
    if where_clause:
        cleaned = re.sub(r"^\s*WHERE\s*", "", where_clause, flags=re.I)
        if not re.fullmatch(r"[A-Za-z0-9_.' ]+", cleaned):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += " WHERE (" + cleaned + ")"
    if group_by:
        if not _is_safe_identifier(group_by):
            return JsonResponse({"error": "Only SELECT statements are allowed"}, status=400)
        sql += " GROUP BY " + group_by

    with connection.cursor() as cursor:
        cursor.execute(sql)
        row = cursor.fetchone()

    columns = list(row.keys()) if row else []
    return JsonResponse({"columns": columns})

4. 类型正确性：fullmatch 锚定整个字符串，无法通过前缀/后缀绕过；位置正确性：校验在每个片段拼入 `sql` 之前（第 22、27、32、37 行），任何非法片段直接返回 400，不进入 sink；覆盖完整性：四个用户可控数据流全部校验，其中 WHERE 片段额外允许数字、点号与单引号以支持字面量比较，但禁止双引号、分号、括号与注释符。
