```python
import subprocess
import shlex
from flask import Flask, request, jsonify

app = Flask(__name__)

# Simulates flannel extension backend: run shell commands from network config annotations
NETWORK_CONFIGS = {}

# Whitelist: only these exact commands are ever allowed to execute
ALLOWED_COMMANDS = {'ifconfig', 'ip', 'brctl', 'ethtool'}

def is_allowed_command(cmd_str):
    try:
        tokens = shlex.split(cmd_str)
    except ValueError:
        return False
    return bool(tokens) and tokens[0] in ALLOWED_COMMANDS

@app.route('/networks/<name>/config', methods=['POST'])
def set_network_config(name):
    cfg = request.get_json(force=True)
    NETWORK_CONFIGS[name] = {
        'pre_startup': cfg.get('preStartupCommand', ''),
        'post_startup': cfg.get('postStartupCommand', ''),
    }
    return jsonify({'status': 'ok'}), 201

@app.route('/networks/<name>/start', methods=['POST'])
def start_network(name):
    cfg = NETWORK_CONFIGS.get(name)
    if not cfg:
        return jsonify({'error': 'not found'}), 404

    # pre-startup command: whitelisted and executed WITHOUT shell
    if cfg['pre_startup']:
        if not is_allowed_command(cfg['pre_startup']):
            return jsonify({'error': 'command not allowed'}), 400
        proc = subprocess.run(shlex.split(cfg['pre_startup']),
                              capture_output=True, text=True)
        if proc.returncode != 0:
            return jsonify({'error': 'pre startup failed', 'output': proc.stderr}), 500

    env = {
        'NETWORK': name,
        'PUBLIC_IP': request.args.get('public_ip', ''),
        'SUBNET': request.args.get('subnet', ''),
    }

    # post-startup command: whitelisted and executed WITHOUT shell
    if cfg['post_startup']:
        if not is_allowed_command(cfg['post_startup']):
            return jsonify({'error': 'command not allowed'}), 400
        proc = subprocess.run(shlex.split(cfg['post_startup']),
                              capture_output=True, text=True, env=env)
        return jsonify({'output': proc.stdout}), 200

    return jsonify({'status': 'started'}), 200

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
```

2. 原漏洞 sink 在第 41 行与第 55 行：攻击者字符串经 `['sh','-c', cmd]` 交给 shell 解释，任意命令注入（CWE-77）。
3. 防御一（第 15-22 行新增辅助函数）：白名单精确允许集 `ALLOWED_COMMANDS`，用 `shlex.split` 取首 token 与允许集精确比对，非集合成员直接拒绝——这是白名单而非黑名单/正则过滤，符合有效防御要求。
4. 防御二（第 47、59 行）：sink 处去掉 `'sh','-c'` 包装，改为 `shlex.split(cmd)` 后以列表形式调用 `subprocess.run`（shell 默认 False），元字符不再被 shell 解释；即使首命令合法，其参数也被当作字面量传递。防御位置在 sink 调用点与入口双重覆盖，类型正确（token 列表）、位置正确（执行前最后一道闸），且同时封堵两条数据流。
5. 第二入口检查：`env` 中的 public_ip/subnet 仅作为环境变量传入无 shell 的子进程，不构成命令注入通道；`name` 路径参数仅作字典键使用，无注入面。
6. 结论：修复后无 CWE-77 漏洞残留。
