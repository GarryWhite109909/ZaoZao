package main

import (
	"bytes"
	"encoding/xml"
	"fmt"
	"io"
	"net/http"
)

type Doc struct {
	XMLName xml.Name `xml:"document"`
	Items   []Item   `xml:"item"`
}

type Item struct {
	Title string `xml:"title"`
	Body  string `xml:"body"`
}

// ensureNoEntityDecls 在正式解码前对同一份字节流做一次白名单式
// 令牌扫描：只允许普通元素、字符数据、注释以及目标为 xml 的处理指令，
// 任何 DTD 指令（DOCTYPE/ENTITY）一律拒绝，从根源上阻断 XXE。
func ensureNoEntityDecls(data []byte) error {
	scanner := xml.NewDecoder(bytes.NewReader(data))
	for {
		tok, err := scanner.Token()
		if err == io.EOF {
			return nil
		}
		if err != nil {
			return fmt.Errorf("xml scan error")
		}
		switch t := tok.(type) {
		case xml.Directive:
			return fmt.Errorf("dtd directives are not allowed")
		case xml.ProcInst:
			if t.Target != "xml" {
				return fmt.Errorf("processing instruction not allowed")
			}
		case xml.StartElement, xml.EndElement, xml.CharData, xml.Comment:
			// 白名单允许的令牌类型
		default:
			return fmt.Errorf("token type not allowed")
		}
	}
}

func uploadHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}

	file, _, err := r.FormFile("docfile")
	if err != nil {
		http.Error(w, "missing file", http.StatusBadRequest)
		return
	}
	defer file.Close()

	data, err := io.ReadAll(file)
	if err != nil {
		http.Error(w, "read error", http.StatusInternalServerError)
		return
	}

	if err := ensureNoEntityDecls(data); err != nil {
		http.Error(w, "invalid xml", http.StatusBadRequest)
		return
	}

	var doc Doc
	decoder := xml.NewDecoder(bytes.NewReader(data))
	if err := decoder.Decode(&doc); err != nil {
		http.Error(w, "invalid xml", http.StatusBadRequest)
		return
	}

	for _, item := range doc.Items {
		fmt.Fprintf(w, "%s: %s
", item.Title, item.Body)
	}
}

func main() {
	http.HandleFunc("/upload", uploadHandler)
	http.ListenAndServe(":8080", nil)
}
