package com.example.artifact.controller;

import org.springframework.web.bind.annotation.*;
import org.springframework.http.ResponseEntity;
import java.nio.file.*;
import java.util.stream.Collectors;

@RestController
@RequestMapping("/api/artifacts")
public class ArtifactViewController {

    private static final Path LOCK_DIR = Paths.get("/var/lib/app/lockfiles").toAbsolutePath().normalize();

    // Returns the content of a package lockfile by package name and version
    @GetMapping("/{packageName}/{version}")
    public ResponseEntity<String> getLockFile(@PathVariable String packageName,
                                              @PathVariable String version) throws Exception {
        // Defense: strict whitelist validation of path segments before any file resolution
        if (!isValidSegment(packageName) || !isValidSegment(version)) {
            return ResponseEntity.badRequest().build();
        }

        Path target = LOCK_DIR.resolve(packageName + "-" + version + ".lock").normalize();

        // Defense: canonical-path containment check against the base directory
        if (!target.startsWith(LOCK_DIR)) {
            return ResponseEntity.badRequest().build();
        }

        if (!Files.exists(target)) {
            return ResponseEntity.notFound().build();
        }

        String content = Files.readAllLines(target).stream()
                .collect(Collectors.joining("
"));

        return ResponseEntity.ok(content);
    }

    private static boolean isValidSegment(String segment) {
        if (segment == null || segment.isEmpty() || segment.length() > 128) {
            return false;
        }
        for (char c : segment.toCharArray()) {
            boolean ok = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
                    || (c >= '0' && c <= '9') || c == '.' || c == '_' || c == '-';
            if (!ok) {
                return false;
            }
        }
        // reject traversal-looking segments even though charset already excludes '/'
        if (segment.equals(".") || segment.equals("..")) {
            return false;
        }
        return true;
    }
}

2. 第 20-23 行：在 resolve 之前先调用 `isValidSegment` 做白名单校验——只允许字母、数字、`.`、`_`、`-`，并显式拒绝 `.`/`..`。这是精确允许集（白名单），不是黑名单或正则过滤，符合有效防御要求；位置正确（在任何文件系统操作之前），类型正确（作用于字符串段而非拼接后的路径）。
3. 第 25 行：`resolve(...).normalize()` 消除路径中的冗余段；第 28-30 行：用 `target.startsWith(LOCK_DIR)`（LOCK_DIR 已在第 12 行 `toAbsolutePath().normalize()`）做规范化后的包含性检查，确保即使未来校验被绕过，任何逃逸出基目录的路径都会被拒绝。双重防御覆盖了 `../`、绝对路径注入、URL 编码穿越等攻击面。
4. 第 32-34 行：仅当包含性检查通过后才执行 `Files.exists` 与第 37 行的 `Files.readAllLines(target)`，sink 的前置条件已完全受控，数据流走向与原代码一致（URL → resolve → read → 返回内容）。
5. 第二入口检查：该 Controller 仅此一个 handler，无其他映射；`LOCK_DIR` 为编译期常量不可控；无其他文件读取通道。替代通道（如直接访问文件系统）不属于 HTTP 入口范畴。结论：无残留漏洞。
