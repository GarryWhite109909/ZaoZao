package com.example.artifact.controller;

import org.springframework.web.bind.annotation.*;
import org.springframework.http.ResponseEntity;
import java.nio.file.*;
import java.util.stream.Collectors;

@RestController
@RequestMapping("/api/artifacts")
public class ArtifactViewController {

    private static final Path LOCK_DIR = Paths.get("/var/lib/app/lockfiles").toAbsolutePath().normalize();

    // Returns the content of a package lockfile by package name and version
    @GetMapping("/{packageName}/{version}")
    public ResponseEntity<String> getLockFile(@PathVariable String packageName,
                                              @PathVariable String version) throws Exception {
        // packageName/version come straight from the URL path
        Path target = LOCK_DIR.resolve(packageName + "-" + version + ".lock").normalize();

        if (!target.startsWith(LOCK_DIR) || !Files.exists(target)) {
            return ResponseEntity.notFound().build();
        }

        String content = Files.readAllLines(target).stream()
                .collect(Collectors.joining("
"));

        return ResponseEntity.ok(content);
    }
}
