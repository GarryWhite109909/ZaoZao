```php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;

class DevProxyController extends Controller
{
    // config/proxy.php: 'routes' => ['/' => 'http://backend.internal:3000'], 'ws' => true
    public function proxy(Request $request)
    {
        $context = $request->route('context', '/');
        $target = config("proxy.routes.{$context}", config('proxy.routes./'));

        if (!$target) {
            abort(404);
        }

        // Any request not matching a local static file is forwarded,
        // including the app's own /broadcasting/auth and websocket handshake.
        $localPath = public_path(ltrim($request->path(), '/'));
        if (file_exists($localPath)) {
            return response()->file($localPath);
        }

        // Defense 1: strict allowlist on the resolved target — only exact
        // values registered in config/proxy.php may be forwarded to.
        $allowedTargets = array_values((array) config('proxy.routes', []));
        if (!in_array($target, $allowedTargets, true)) {
            abort(404);
        }

        // Defense 2: never relay WebSocket Upgrade handshakes through the
        // HTTP proxy; the dev server's own HMR/broadcasting sockets must
        // not be hijacked by the upstream.
        if (strtolower($request->header('upgrade', '')) === 'websocket') {
            abort(400, 'WebSocket upgrade is not proxied.');
        }

        $forward = Http::withHeaders(
            collect($request->headers->all())
                ->except(['host'])
                ->map(fn ($v) => $v[0])
                ->toArray()
        );

        return $forward
            ->send($request->method(), rtrim($target, '/') . '/' . ltrim($request->path(), '/'),
                $request->getMethod() === 'GET' ? [] : $request->post());
    }
}
```

1. 第 22 行：用户可控输入点为 `$request->route('context')`（路由参数）、`$request->headers->all()`（全部请求头）、`$request->path()` 与 `$request->post()`；其中 `context` 经 `config("proxy.routes.{$context}")` 解析出 `$target`，是转发目标的主数据流起点。
2. 第 30–33 行：新增白名单精确允许集防御——将解析出的 `$target` 与 `config('proxy.routes')` 注册表做严格类型比较（`in_array(..., true)`），只有配置中显式登记的内部后端地址才被放行，杜绝通过 context 注入或配置回退把请求代理到任意主机。这是框架原生 config 白名单 + 精确匹配，非黑名单/正则。
3. 第 35–39 行：新增对 `Upgrade: websocket` 头的拦截，直接以 400 拒绝而非继续转发，修复 CWE-441 中 ws:true 语义导致的 HMR/广播 socket 被上游劫持的中间人通道；同时原第 41 行起保留的头部透传不再叠加 upgrade/connection 头，攻击面覆盖完整。
4. 类型正确性：`in_array` 第三参数 `true` 保证字符串严格比较，防止 `0`/数组等松散比较绕过；`array_values((array) ...)` 保证配置缺失时为空数组、默认拒绝。
5. 位置正确性：两处防御均位于 sink（第 46 行 `Http::send`）之前、且在本地静态文件分支之后，所有到达转发的路径都必须先通过校验，无旁路。
6. 第二入口检查：`$localPath` 分支（第 28 行）仅读取 public 目录内已存在文件并经 `response()->file` 返回，不构成对外代理；其余数据流（headers 透传、body 转发）按 minimal pair 要求保持原样未改动。
