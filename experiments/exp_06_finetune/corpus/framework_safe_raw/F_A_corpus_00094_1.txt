```php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use Illuminate\Validation\Rule;

class DevProxyController extends Controller
{
    // 允许代理的主机白名单（精确允许集，来自受信配置，而非用户输入）
    private function allowedTargets(): array
    {
        return array_values(array_filter((array) config('app.dev_proxy_allowlist')));
    }

    private function resolveTarget(Request $request): string
    {
        $target = (string) $request->input('proxy_target', config('app.dev_proxy'));

        $validated = validator(['proxy_target' => $target], [
            'proxy_target' => ['required', 'url', Rule::in($this->allowedTargets())],
        ])->validate();

        return $validated['proxy_target'];
    }

    public function proxy(Request $request)
    {
        $target = $this->resolveTarget($request);
        $path = $request->path();

        // 静态资源不代理，其余全部转发（含 WebSocket 升级请求）
        if ($this->isPublicAsset($path)) {
            return response()->file(public_path($path));
        }

        // 目标主机已通过白名单精确匹配校验，仅允许回环/内网开发服务
        $upstream = Http::withHeaders($request->headers->all())
            ->send($request->method(), rtrim($target, '/') . '/' . $path,
                ['body' => $request->getContent()]);

        return response($upstream->body(), $upstream->status())
            ->withHeaders($upstream->headers());
    }

    private function isPublicAsset(string $path): bool
    {
        return file_exists(public_path($path));
    }
}
```

1. 第 20 行：用户可控输入点为 `$request->input('proxy_target')`（框架参数绑定），这是唯一进入代理目标解析的数据流入口；`$request->path()`、请求头与 body 属于转发语义本身，非本漏洞数据流。
2. 第 22-27 行：原代码仅用 `preg_match('#^https?://#')` 做前缀黑名单式检查，攻击者可传 `http://169.254.169.254/` 等任意主机，构成 SSRF/未受限资源代理；修复改用 Laravel 原生 `validator` + `url` 规则 + `Rule::in()` 白名单精确允许集，类型上对字符串输入做整体精确匹配，位置上在目标被用于拼接 URL 之前完成。
3. 第 12-16 行：白名单来源为服务器端配置 `config('app.dev_proxy_allowlist')`（如 `['http://localhost:3000']`），用户无法注入或扩展该集合，覆盖了任意主机、云元数据地址、内网地址等全部攻击面；`url` 规则同时保证 scheme 合法性，替代了原正则。
4. 第 35 行：sink 为 `Http::...->send(...)` 发起上游请求处；到达 sink 的 `$target` 已在第 26 行经白名单验证通过，数据流上不存在绕过路径——`validate()` 失败会抛出 ValidationException 中止请求。
5. 其余数据流保持原样：第 31-33 行静态资源分支、第 36-39 行响应透传逻辑均未改动，保持 minimal pair 局部对照性。
6. 第二入口检查：无其他路由或方法调用 `resolveTarget` 或直接构造上游 URL；配置默认值 `config('app.dev_proxy')` 同样经过同一白名单校验，替代通道已覆盖。
