```python
import ipaddress
import socket
from urllib.parse import urlparse

import requests
from requests.adapters import HTTPAdapter
from flask import Flask, request, jsonify

app = Flask(__name__)

BLOCKED_NETS = [
    ipaddress.ip_network('10.0.0.0/8'),
    ipaddress.ip_network('172.16.0.0/12'),
    ipaddress.ip_network('192.168.0.0/16'),
    ipaddress.ip_network('127.0.0.0/8'),
    ipaddress.ip_network('169.254.0.0/16'),
    ipaddress.ip_network('100.64.0.0/10'),
    ipaddress.ip_network('0.0.0.0/8'),
    ipaddress.ip_network('240.0.0.0/4'),
]

ALLOWED_SCHEMES = {'http', 'https'}


def is_private_ip(addr):
    try:
        ip = ipaddress.ip_address(addr)
    except ValueError:
        return True
    return any(ip in net for net in BLOCKED_NETS)


class PinnedResolverAdapter(HTTPAdapter):
    """HTTPAdapter that pins DNS resolution to the pre-validated IP set,
    so the actual TCP connection cannot be re-bound to another address."""

    def __init__(self, hostname, pinned_ips, **kwargs):
        self.hostname = hostname
        self.pinned_ips = list(pinned_ips)
        super().__init__(**kwargs)

    def send(self, request, **kwargs):
        original_getaddrinfo = socket.getaddrinfo

        def pinned_getaddrinfo(host, port, *args, **kw):
            results = original_getaddrinfo(host, port, *args, **kw)
            if host == self.hostname:
                return [
                    (r[0], r[1], r[2], r[3], (ip, r[4][1] if len(r[4]) > 1 else port))
                    for ip in self.pinned_ips
                    for r in [results[0]]
                ]
            return results

        socket.getaddrinfo = pinned_getaddrinfo
        try:
            return super().send(request, **kwargs)
        finally:
            socket.getaddrinfo = original_getaddrinfo


@app.route('/fetch')
def fetch_url():
    target = request.args.get('url', '')
    parsed = urlparse(target)
    if not parsed.hostname:
        return jsonify(error='invalid url'), 400
    if parsed.scheme not in ALLOWED_SCHEMES:
        return jsonify(error='scheme not allowed'), 400

    # SSRF guard: resolve hostname and reject private addresses
    resolved = {ai[4][0] for ai in socket.getaddrinfo(parsed.hostname, None)}
    if any(is_private_ip(a) for a in resolved):
        return jsonify(error='blocked host'), 403

    # Anti-DNS-rebinding: pin the validated IPs onto the session used for the request,
    # and disable redirects so a 30x cannot bounce to an unvalidated host.
    session = requests.Session()
    adapter = PinnedResolverAdapter(parsed.hostname, resolved)
    session.mount('http://', adapter)
    session.mount('https://', adapter)
    resp = session.get(target, timeout=5, allow_redirects=False)
    return jsonify(status=resp.status_code, body=resp.text[:2000])


if __name__ == '__main__':
    app.run()
```

分析：

1. 第 47 行：用户可控输入点为 Flask 查询参数绑定 `request.args.get('url', '')`，攻击者完全控制目标 URL，这是唯一的数据流起点。
2. 第 51–54 行：对 scheme 施加白名单精确允许集（仅 http/https），阻断 file://、gopher:// 等替代协议通道；第 56 行解析 hostname 后在第 58–60 行做私网地址校验。
3. 原漏洞成因：第 62 行的校验与真正发起请求之间存在 TOCTOU 窗口——`requests.get(target)` 内部会再次独立解析 DNS，攻击者通过 DNS rebinding 让第一次解析返回公网 IP、第二次解析返回 127.0.0.1，即可绕过全部黑名单检查到达 sink。
4. 修复位置正确性：第 65–70 行在 sink 调用前创建 Session 并挂载 `PinnedResolverAdapter`，把已通过校验的 IP 集合固定下来；适配器内部（第 33–43 行）临时替换 `socket.getaddrinfo`，使该 hostname 的所有连接只能使用预验证过的地址，类型上保证连接层与校验层使用同一份解析结果，彻底消除 rebinding 窗口；finally 子句确保恢复全局解析器不污染其他请求。
5. 攻击面覆盖完整性：第 71 行同时设置 `allow_redirects=False`，防止服务端跟随 302 重定向跳向未验证主机形成第二跳 SSRF；timeout=5 保留原有限制。
6. 第二入口/替代通道检查：应用仅有 `/fetch` 一个路由，无其他读取 url 参数的端点；`is_private_ip` 与 BLOCKED_NETS 黑名单保留作为纵深防御第一层，但真正的有效性由 IP 固定机制（白名单式精确允许已验证地址集）保证。
