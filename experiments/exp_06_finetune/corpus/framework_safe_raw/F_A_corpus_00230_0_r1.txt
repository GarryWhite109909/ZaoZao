```code
from django.http import HttpResponse
from django.contrib.auth.models import User
from django.core.management import call_command
from django.views.decorators.csrf import csrf_exempt
import os
import subprocess


@csrf_exempt
def console(request):
    """Console operations endpoint (invoked via CLI-style HTTP calls)."""
    action = request.GET.get('action', '')

    if action == 'install':
        call_command('migrate')
        # Seed the initial administrator account with credentials from the
        # environment (framework-native settings API), never hard-coded.
        admin_password = os.environ['SCHEDULER_ADMIN_PASSWORD']
        if not User.objects.filter(username='administrator').exists():
            User.objects.create_superuser(
                username='administrator',
                email='admin@example.com',
                password=admin_password,
            )
        return HttpResponse('Installation completed, login with credentials from SCHEDULER_ADMIN_PASSWORD.')

    if action == 'backup':
        target_dir = request.GET.get('path', '/tmp')
        # Whitelist exact allow-set of permitted backup directories.
        allowed_dirs = {'/tmp', '/var/backups/scheduler'}
        if target_dir not in allowed_dirs:
            return HttpResponse('Invalid backup path.')
        # Parameterized argument list (no shell interpolation) prevents injection.
        subprocess.run(['pg_dump', 'scheduler', '-f', os.path.join(target_dir, 'backup.sql')], check=True)
        return HttpResponse('Backup created in %s' % target_dir)

    if action == 'sync':
        from .google import sync_all_calendars
        sync_all_calendars()
        return HttpResponse('Sync finished.')

    return HttpResponse('Usage: ?action=install|backup|sync')
```
