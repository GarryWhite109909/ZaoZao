```go
package main

import (
	"net/http"

	"github.com/gin-gonic/gin"
	libxml2 "github.com/lestrrat-go/libxml2"
	"github.com/lestrrat-go/libxml2/parser"
)

// POST /api/xml/format : pretty-print an uploaded XML document.
func formatXMLHandler(c *gin.Context) {
	raw, err := c.GetRawData()
	if err != nil {
		c.String(http.StatusBadRequest, "cannot read request body")
		return
	}

	// Parse the untrusted XML with hardened parser options:
	// disable DTD loading and external entity resolution entirely.
	p := parser.New(
		parser.XMLParseNoEnt(false),
		parser.XMLParseDTDLoad(false),
		parser.XMLParseDTDAttr(false),
		parser.XMLParseNoNet(false),
	)
	doc, err := p.Parse(raw)
	if err != nil {
		c.String(http.StatusBadRequest, "invalid XML")
		return
	}
	defer doc.Free()

	formatted, err := doc.ToString(2)
	if err != nil {
		c.String(http.StatusInternalServerError, "format failed")
		return
	}
	c.Data(http.StatusOK, "application/xml", []byte(formatted))
}

func main() {
	r := gin.Default()
	r.POST("/api/xml/format", formatXMLHandler)
	r.Run(":8080")
}
```

3. Sink 位于第 27 行 `p.Parse(raw)`：不可信字节流在此被完整解析，外部实体引用会被展开（可读本地文件或发起 SSRF），随后经第 33 行 `doc.ToString` 回显给客户端，构成完整的读取-回显链。
4. 修复在第 25-30 行以最小 diff 替换解析器构造语句：通过 libxml2 框架原生安全 API 显式禁用实体替换、DTD 加载、DTD 属性与网络访问。这是白名单式的精确禁用集（框架原生安全配置），而非黑名单/正则过滤；位置在 Parse 调用之前，类型正确且覆盖了 DTD 内部子集声明实体、外部参数实体与网络取回三类攻击面。
