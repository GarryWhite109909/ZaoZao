```go
package main

import (
	"bytes"
	"encoding/gob"
	"net/http"

	"github.com/gin-gonic/gin"
)

// TaskConfig 模拟 Jenkins 中可被反序列化的核心类型
type TaskConfig struct {
	Name    string
	Handler func() // 可执行载荷字段
}

type PluginEntry struct {
	Plugin string
	Config interface{}
}

// allowedPlugins 白名单精确允许集：仅允许注册过的插件名
var allowedPlugins = map[string]bool{
	"build-task": true,
	"deploy-task": true,
}

func loadConfig(c *gin.Context) {
	raw, err := c.GetRawData()
	if err != nil {
		c.String(http.StatusBadRequest, "read body failed")
		return
	}

	var entry PluginEntry
	dec := gob.NewDecoder(bytes.NewReader(raw))
	if err := dec.Decode(&entry); err != nil { // sink: 反序列化不可信数据到任意类型
		c.String(http.StatusBadRequest, "decode failed")
		return
	}

	// 防御 1：白名单校验插件名，拒绝未注册插件的任意 Config
	if !allowedPlugins[entry.Plugin] {
		c.String(http.StatusForbidden, "plugin not allowed")
		return
	}

	// 防御 2：Config 仅允许反序列化为安全的纯数据结构，禁止 func 字段
	var safeCfg struct {
		Name string
	}
	if tc, ok := entry.Config.(map[string]interface{}); ok {
		if name, ok := tc["Name"].(string); ok {
			safeCfg.Name = name
		}
	}
	_ = safeCfg // 后续仅使用纯数据字段，绝不调用任何来自反序列化的可执行值

	c.String(http.StatusOK, "loaded")
}

func main() {
	r := gin.Default()
	r.POST("/config", loadConfig)
	r.Run(":8080")
}
```

2. 第 33 行 `dec.Decode(&entry)` 为 sink：gob 将不可信字节流解码为含 `interface{}` 的 `PluginEntry`，攻击者可注入带 `func` 字段的 `TaskConfig`。
3. 原代码第 38-40 行将解码结果断言为 `TaskConfig` 并调用 `tc.Handler()`，构成完整的「请求体 → gob 解码 → 类型断言 → 函数调用」反序列化执行链（CWE-502）。
4. 修复位置正确性：防御插入在 sink 之后、危险消费点之前——第 37-40 行白名单精确允许集先拦截未注册插件名，属于框架无关的精确允许集（非黑名单/正则）；第 43-49 行将 `entry.Config` 的处理限定为仅提取纯字符串数据到匿名安全结构体，彻底移除对反序列化产物的可执行调用，类型上不可能再触发任意函数执行。
5. 攻击面覆盖完整性：即使白名单内插件被命中，`Config` 也只走 map→string 的纯数据提取路径，原 `tc.Handler()` 调用语句已被删除，无残留执行通道；数据流走向（GetRawData → Decode → 校验 → 响应）与原代码一致，函数名、路由 `/config`、变量名均未改动。
