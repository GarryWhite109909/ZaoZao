import json
import re
import shlex
import subprocess

from django.http import JsonResponse
from django.views.decorators.http import require_POST


FFMPEG_PREFIXES = ('ffmpeg', '/usr/bin/ffmpeg', '/bin/ffmpeg')


def sanitize_ffmpeg_command(command):
    # whitelist-style validation: split into argv, allow only ffmpeg binary
    try:
        argv = shlex.split(command)
    except ValueError:
        return []
    if not argv:
        return []
    if any(argv[0] == p for p in FFMPEG_PREFIXES):
        return argv
    return []


@require_POST
def ffmpeg_control(request):
    action = request.POST.get('action', '')

    if action == 'list':
        keyword = request.POST.get('keyword', '')
        proc = subprocess.run(
            ['ps', '-eo', 'pid,etime,%cpu,%mem,cmd'],
            capture_output=True, text=True)
        lines = proc.stdout.splitlines()
        if keyword:
            grep_proc = subprocess.run(
                ['grep', '-F', '-e', keyword],
                input=proc.stdout, capture_output=True, text=True)
            lines = grep_proc.stdout.splitlines()
        return JsonResponse({'output': lines})

    if action == 'kill':
        pid = request.POST.get('pid', '')
        if not pid.isdigit():
            return JsonResponse({'error': True, 'msg': 'Invalid PID'}, status=400)
        subprocess.run(['kill', '-9', pid])
        return JsonResponse({'error': False, 'msg': f'Process {pid} killed'})

    if action == 'start':
        raw_command = request.POST.get('command', '')
        safe_argv = sanitize_ffmpeg_command(raw_command)
        if not safe_argv:
            return JsonResponse({'error': True, 'msg': 'Sanitization failed'}, status=400)
        subprocess.Popen(safe_argv,
                         stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return JsonResponse({'error': False, 'msg': 'started'})

    return JsonResponse({'error': True, 'msg': 'Unknown action'}, status=400)
