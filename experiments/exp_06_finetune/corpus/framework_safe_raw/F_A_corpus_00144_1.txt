package com.example.notes;

import org.springframework.web.bind.annotation.*;
import org.springframework.http.ResponseEntity;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.DocumentBuilder;
import org.w3c.dom.*;
import java.io.File;
import java.util.*;

@RestController
@RequestMapping("/api/notes")
public class NoteImportController {

    // POST /api/notes/import?path=/data/uploads/notebook.enex
    @PostMapping("/import")
    public ResponseEntity<Map<String, Object>> importNotebook(@RequestParam String path) {
        Map<String, Object> result = new HashMap<>();
        try {
            DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
            // XXE 防御：禁用 DTD 与外部实体（框架原生安全 API）
            factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
            factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
            factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
            factory.setXIncludeAware(false);
            factory.setExpandEntityReferences(false);
            DocumentBuilder builder = factory.newDocumentBuilder();
            Document doc = builder.parse(new File(path)); // 已防御

            NodeList notes = doc.getElementsByTagName("note");
            List<Map<String, Object>> parsed = new ArrayList<>();
            for (int i = 0; i < notes.getLength(); i++) {
                Element note = (Element) notes.item(i);
                Map<String, Object> noteMap = new HashMap<>();
                noteMap.put("title", note.getElementsByTagName("title").item(0).getTextContent());
                noteMap.put("content", note.getElementsByTagName("content").item(0).getTextContent());
                parsed.add(noteMap);
            }
            result.put("notes", parsed);
            return ResponseEntity.ok(result);
        } catch (Exception e) {
            result.put("error", e.getMessage());
            return ResponseEntity.badRequest().body(result);
        }
    }
}
