<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;
use App\Models\Post;

class PostRenderController extends Controller
{
    /**
     * Force-rerender a post and backfill missing image dimensions
     * by fetching the image URL stored on each image card payload.
     */
    public function rerender(Request $request, int $postId)
    {
        $post = Post::findOrFail($postId);

        $mobiledoc = json_decode($post->mobiledoc_json, true);
        if (!is_array($mobiledoc) || !isset($mobiledoc['cards'])) {
            return response()->json(['error' => 'invalid mobiledoc'], 422);
        }

        foreach ($mobiledoc['cards'] as $card) {
            [$cardName, $payload] = $card;

            if ($cardName !== 'image' || empty($payload['src'])) {
                continue;
            }
            if (!empty($payload['width']) && !empty($payload['height'])) {
                continue;
            }

            try {
                // fetch the remote image to read its dimensions (SSRF-hardened)
                $url = $this->validateImageUrl((string) $payload['src']);
                if ($url === null) {
                    Log::warning('image url rejected by allowlist', ['url' => $payload['src']]);
                    continue;
                }
                // follow redirects but re-validate each hop against the allowlist
                $response = Http::timeout(5)
                    ->withOptions(['allow_redirects' => ['on_redirect' => function ($req, $res, $uri) {
                        if (!$this->isAllowedHost(parse_url((string) $uri, PHP_URL_HOST))) {
                            throw new \RuntimeException('redirect to disallowed host');
                        }
                    }]])
                    ->get($url);
                if ($response->successful()) {
                    $size = getimagesizefromstring($response->body());
                    if ($size !== false) {
                        $payload['width'] = $size[0];
                        $payload['height'] = $size[1];
                    }
                }
            } catch (\Throwable $e) {
                Log::error('image size fetch failed', ['url' => $payload['src']]);
            }
        }

        $post->update(['mobiledoc_json' => json_encode($mobiledoc)]);

        return response()->json(['status' => 'ok']);
    }

    /**
     * Defense: strict scheme+host allowlist before any outbound request.
     * Returns the validated URL string or null when rejected.
     */
    private function validateImageUrl(string $rawUrl): ?string
    {
        $parts = parse_url($rawUrl);
        if ($parts === false || !isset($parts['scheme'], $parts['host'])) {
            return null;
        }
        // only https is allowed; no http, no file/data/gopher/dict schemes
        if (strtolower($parts['scheme']) !== 'https') {
            return null;
        }
        // reject userinfo tricks like https://evil@internal-host/
        if (!empty($parts['user']) || !empty($parts['pass'])) {
            return null;
        }
        // reject port-based bypasses except default 443
        if (isset($parts['port']) && $parts['port'] !== 443) {
            return null;
        }
        if (!$this->isAllowedHost($parts['host'])) {
            return null;
        }
        return $rawUrl;
    }

    /**
     * Exact-match allowlist of permitted image CDN hosts.
     */
    private function isAllowedHost(?string $host): bool
    {
        if ($host === null || $host === '') {
            return false;
        }
        $host = strtolower(rtrim($host, '.'));
        static $allowed = [
            'cdn.example.com',
            'images.example.com',
            'media.example-cdn.net',
        ];
        return in_array($host, $allowed, true);
    }
}
