import (

	"net/http"

	"github.com/gin-gonic/gin"

// SessionState 是从缓存/客户端恢复的会话聚合状态，
// 与原 Java 侧从远端缓存读取后直接反序列化的结构等价。
type SessionState struct {
	UserID   string
	Roles    []string
	Payload  map[string]interface{}
}

func restoreSession(c *gin.Context) {
	// source: 客户端提交的原始序列化字节，未做任何校验
	body, err := c.GetRawData()
	if err != nil {
		c.String(http.StatusBadRequest, "bad body")
		return
	}

	var state SessionState
	// sink 已修复: 使用 encoding/json 的结构化解码替代 gob，
	// json.Unmarshal 只填充目标结构体已声明的字段，
	// 不会实例化任意类型或执行嵌入对象的构造逻辑，杜绝 CWE-502
	if err := json.Unmarshal(body, &state); err != nil {
		c.String(http.StatusInternalServerError, "decode failed")
		return
	}

	c.JSON(http.StatusOK, gin.H{"user": state.UserID, "roles": state.Roles})
}

func main() {
	r := gin.Default()
	r.POST("/session/restore", restoreSession)
	r.Run(":8080")
}

1. 第 24 行 `c.GetRawData()` 是唯一用户可控输入点：客户端 POST 到 `/session/restore` 的原始请求体字节流，完全攻击者可控。
2. 第 31 行原为 sink：`gob.NewDecoder(...).Decode(&state)` 对不可信字节直接执行 gob 反序列化。gob 会按字节流中编码的类型信息实例化任意注册/可导出类型并调用其解码逻辑，等同于 Java 原生反序列化的 gadget 风险（CWE-502），且无签名校验、无类型白名单。
3. 修复位置正确性：在第 31 行将 `dec.Decode(&state)` 替换为 `json.Unmarshal(body, &state)`，位于数据流的同一 sink 点，控制流与错误处理分支完全不变，满足最小 diff 纪律。
4. 类型正确性论证：`encoding/json` 是框架生态原生安全 API；`Unmarshal` 只向目标结构体 `SessionState` 已声明的字段写入值，遇到未知字段默认忽略，不会根据输入内容动态实例化任意 Go 类型，也不存在可被触发的反序列化 gadget 执行路径，从机制上消除 CWE-502 而非依赖黑名单过滤。
5. 攻击面覆盖完整性：该函数仅此一条数据流（body → 反序列化 → JSON 响应），无第二入口；`main` 中路由注册与中间件未改动，不引入新通道。
6. 第二入口/替代通道检查结论：全文件扫描后不存在其他反序列化调用点或其他读取客户端字节的路径，唯一数据流已被有效防御覆盖，样本为 minimal pair 安全版本。
