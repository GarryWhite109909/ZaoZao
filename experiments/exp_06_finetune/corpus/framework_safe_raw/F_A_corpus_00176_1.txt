<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\File;

class ComposerConfigController extends Controller
{
    // 防御辅助：composer 包名的白名单精确允许集（锚定全串匹配，仅允许官方命名规范字符）
    private function isValidPackageName(string $pkg): bool
    {
        return preg_match('/^[a-z0-9](?:[a-z0-9-]*[a-z0-9])?(?:\/[a-z0-9](?:[a-z0-9._-]*[a-z0-9])?)?$/', $pkg) === 1
            && strlen($pkg) <= 191;
    }

    public function update(Request $request)
    {
        $packages = $request->input('packages', []);
        $installScript = $request->input('post_install_cmd');

        $composerPath = base_path('composer.json');
        $backupPath = base_path('composer-bak.json');

        $composer = json_decode(File::get($composerPath), true);

        unset($composer['require-dev']);

        $deps = [];
        foreach ($packages as $pkg) {
            if (!is_string($pkg) || !$this->isValidPackageName($pkg)) {
                abort(422, 'Invalid package name');
            }
            $deps[$pkg] = '*';
        }
        $composer['require'] = $deps;

        // 防御：scripts 段不接受任何用户输入，post-install-cmd 固定为框架原生安全命令；
        // 用户提交的 post_install_cmd 仅作存在性确认，不进入 composer.json
        $composer['scripts'] = [
            'post-install-cmd' => 'php artisan optimize',
            'post-autoload-dump' => 'php artisan optimize',
        ];
        unset($installScript);

        File::copy($composerPath, $backupPath);
        File::delete($composerPath);
        File::put($composerPath, json_encode($composer, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES));

        return response()->json(['status' => 'updated']);
    }
}
