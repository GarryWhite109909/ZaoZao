<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Facades\Validator;

class ComposerConfigController extends Controller
{
    // 白名单：仅允许安装经过审核的内部包
    private const ALLOWED_PACKAGES = [
        'monolog/monolog',
        'guzzlehttp/guzzle',
        'laravel/horizon',
        'predis/predis',
    ];

    // 白名单：post-install-cmd 仅允许预定义的固定脚本集合
    private const ALLOWED_SCRIPTS = [
        'none' => '',
        'optimize' => '@php artisan optimize',
        'migrate' => '@php artisan migrate --force',
    ];

    public function update(Request $request)
    {
        $validated = Validator::make($request->all(), [
            'packages' => 'array',
            'packages.*' => 'string',
            'post_install_cmd' => 'required|string|in:' . implode(',', array_keys(self::ALLOWED_SCRIPTS)),
        ])->validate();

        $packages = $validated['packages'] ?? [];
        $installScriptKey = $validated['post_install_cmd'];

        $composerPath = base_path('composer.json');
        $backupPath = base_path('composer-bak.json');

        $composer = json_decode(File::get($composerPath), true);

        unset($composer['require-dev']);

        $deps = [];
        foreach ($packages as $pkg) {
            if (!in_array($pkg, self::ALLOWED_PACKAGES, true)) {
                return response()->json(['status' => 'rejected', 'package' => $pkg], 422);
            }
            $deps[$pkg] = '*';
        }
        $composer['require'] = $deps;

        // sink 处写入的是白名单映射出的固定字符串，而非用户原始输入
        $composer['scripts'] = [
            'post-install-cmd' => self::ALLOWED_SCRIPTS[$installScriptKey],
            'post-autoload-dump' => 'php artisan optimize',
        ];

        File::copy($composerPath, $backupPath);
        File::delete($composerPath);
        File::put($composerPath, json_encode($composer, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES));

        return response()->json(['status' => 'updated']);
    }
}

2. 第 26-40 行对 `post_install_cmd` 使用 Laravel 原生 Validator 的 `in:` 规则做精确允许集校验——只接受 `none/optimize/migrate` 三个键名而非脚本内容本身；第 12-18 行的 `ALLOWED_SCRIPTS` 映射表把键名翻译为服务端硬编码的固定命令字符串。类型正确性：校验对象是标量键名，映射值是常量，用户输入永远无法到达 sink 字符串内容；位置正确性：校验发生在数据进入业务逻辑之前（入口处）；攻击面覆盖：即使绕过 Validator 直接构造请求，第 60 行写入的仍是查表结果，不存在拼接路径。
3. 第 47-51 行对 packages 数组逐项执行 `in_array(..., ALLOWED_PACKAGES, true)` 严格比较白名单校验，未通过则返回 422 拒绝——防止攻击者注入恶意包名（如含 `/../` 或伪造 vendor 名导致依赖混淆/任意包安装）。严格模式 `true` 排除松散比较绕过。
4. 第 60 行 sink 处 `post-install-cmd` 的取值来源仅为常量数组查找，与原漏洞代码中直接赋值 `$installScript` 形成最小对照差异；其余逻辑（备份、删除、重写 composer.json、JSON 编码选项、响应结构）全部保持原样，满足 minimal pair 局部对照性。
5. 第二入口检查：`$deps[$pkg] = '*'` 中包名经白名单约束后作为 JSON 对象键写入，json_encode 会正确转义，不构成第二注入通道；`base_path()` 为框架函数不受输入影响；文件路径均为服务端常量，无路径遍历面。
