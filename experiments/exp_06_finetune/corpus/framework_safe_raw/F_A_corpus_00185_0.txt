```python
import os
import subprocess
import shlex
from flask import Flask, request, jsonify

app = Flask(__name__)

CONFIG_PATH = "/etc/sqls/config.yaml"

# 白名单精确允许集：仅允许已知的编辑器可执行名
ALLOWED_EDITORS = {"vim", "vi", "nano"}

def _resolve_editor(editor):
    # 防御：白名单精确匹配，拒绝任何含 shell 元字符或不在允许集内的值
    if editor in ALLOWED_EDITORS:
        return editor
    return None

@app.route("/config/edit", methods=["POST"])
def edit_config():
    # source: editor 来自客户端可控的请求参数
    editor = request.form.get("editor") or os.environ.get("EDITOR", "vim")
    safe_editor = _resolve_editor(editor)
    if safe_editor is None:
        return jsonify({"status": "rejected"}), 400
    cmd = f"{safe_editor} {shlex.quote(CONFIG_PATH)}"
    if os.name == "nt":
        subprocess.run(["cmd", "/c", cmd], shell=False)
    else:
        # sink: 经白名单校验后的固定值，不再拼接未净化输入
        subprocess.run([safe_editor, CONFIG_PATH], shell=False)
    return jsonify({"status": "opened"})

if __name__ == "__main__":
    app.run(port=5000)
```

3. 新增第 11-16 行 `_resolve_editor` 辅助函数实施白名单精确允许集（vim/vi/nano），任何包含 `;`、`|`、反引号等元字符的输入都不在集合内而被拒绝；这是有效防御（白名单精确允许集），非黑名单/正则过滤。位置正确：位于 source 与 sink 之间、sink 拼接之前，覆盖了 form 参数与环境变量两条进入路径。
4. 第 21 行 sink 改为列表形式 `subprocess.run([safe_editor, CONFIG_PATH], shell=False)` 并用 `shlex.quote` 保护常量路径：列表 + shell=False 使参数不经 shell 解释，类型正确地消除了元字符注入面；Windows 分支（第 18 行）原本就是 shell=False 列表调用，保持不变。
