import os
import shutil
import subprocess
import zipfile

from flask import Flask, request, jsonify, send_file, abort

app = Flask(__name__)

PLUGIN_ROOT = os.path.expanduser("~/.trivy/plugins")

ALLOWED_PLUGINS = {"trivy-kubernetes", "trivy-aws", "trivy-azure", "trivy-gcp"}


def _safe_plugin_path(plugin_name):
    # 白名单精确允许集 + 规范化后边界检查，双重防御
    if plugin_name not in ALLOWED_PLUGINS:
        abort(400)
    target_dir = os.path.realpath(os.path.join(PLUGIN_ROOT, plugin_name))
    if os.path.commonpath([os.path.realpath(PLUGIN_ROOT), target_dir]) != os.path.realpath(PLUGIN_ROOT):
        abort(400)
    return target_dir


@app.route("/plugins/install", methods=["POST"])
def install_plugin():
    # source: 插件名与归档 URL 完全来自外部请求，经白名单校验
    plugin_name = request.form.get("plugin_name")
    archive = request.files.get("archive")

    if not plugin_name or not archive:
        return jsonify({"error": "missing fields"}), 400

    # sink: 路径经白名单与 realpath 边界检查后才拼接
    target_dir = _safe_plugin_path(plugin_name)
    zip_path = target_dir + ".zip"

    os.makedirs(target_dir, exist_ok=True)
    archive.save(zip_path)

    with zipfile.ZipFile(zip_path) as zf:
        for member in zf.namelist():
            member_path = os.path.realpath(os.path.join(target_dir, member))
            if os.path.commonpath([target_dir, member_path]) != target_dir:
                abort(400)
        zf.extractall(target_dir)

    manifest_path = os.path.join(target_dir, "plugin.yaml")
    if os.path.exists(manifest_path):
        with open(manifest_path) as f:
            pass  # 读取 manifest 但不校验其中的路径字段

    return jsonify({"status": "installed", "path": target_dir})


@app.route("/plugins/<name>/output", methods=["GET"])
def plugin_output(name):
    # 同样的 sink 模式：路由参数先过白名单与边界检查再发送
    out_file = os.path.join(_safe_plugin_path(name), "output.txt")
    return send_file(out_file)


@app.route("/plugins/uninstall", methods=["POST"])
def uninstall_plugin():
    name = request.form.get("plugin_name")
    target_dir = _safe_plugin_path(name)
    shutil.rmtree(target_dir)  # 经白名单与边界校验后的删除操作
    return jsonify({"status": "removed"})
