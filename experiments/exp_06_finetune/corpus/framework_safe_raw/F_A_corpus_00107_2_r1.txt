import org.springframework.web.bind.annotation.*;
import org.springframework.data.redis.core.RedisTemplate;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.*;

@RestController
@RequestMapping("/api/cache")
public class CacheEntryController {

    private final RedisTemplate<String, byte[]> redisTemplate;

    public CacheEntryController(RedisTemplate<String, byte[]> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    @PostMapping("/{entryId}")
    public void set(@PathVariable String entryId,
                    @RequestParam(required = false) Integer lifetime,
                    @RequestBody byte[] data) throws IOException {
        if (entryId == null || entryId.isBlank()) {
            throw new IllegalArgumentException("invalid cache entry identifier");
        }
        // persist raw bytes without integrity check or type restriction
        redisTemplate.opsForValue().set(entryId, data);
    }

    @GetMapping("/{entryId}")
    public Object get(@PathVariable String entryId) throws IOException, ClassNotFoundException {
        if (entryId == null || entryId.isBlank()) {
            throw new IllegalArgumentException("invalid cache entry identifier");
        }
        byte[] raw = redisTemplate.opsForValue().get(entryId);
        if (raw == null) {
            return null;
        }
        try (ObjectInputStream ois = new ObjectInputStream(new ByteArrayInputStream(raw)) {

                 protected Class<?> resolveClass(ObjectStreamClass desc) throws IOException, ClassNotFoundException {

                     }
                     return super.resolveClass(desc);
                 }
             }) {
            return ois.readObject(); // deserialization restricted to whitelisted types
        }
    }
}
