from fastapi import FastAPI, Request, HTTPException
import yaml

app = FastAPI()

# 全局配置：模拟 YamlParserProperties，可由运维接口设置安全限制
security_config = {"max_aliases": None, "code_point_limit": None}

@app.post("/admin/parser-config")
async def set_parser_config(request: Request):
    body = await request.json()
    security_config["max_aliases"] = body.get("max_aliases")
    security_config["code_point_limit"] = body.get("code_point_limit")
    return {"status": "ok"}

def build_loader():
    """等价于 newYaml()：始终返回框架原生安全的 SafeLoader。"""
    return yaml.SafeLoader

@app.post("/deploy/cloudformation")
async def deploy_cloudformation(request: Request):
    template = (await request.body()).decode("utf-8")
    loader = build_loader()
    # sink：使用 SafeLoader 反序列化用户提交的 YAML 模板
    parsed = yaml.load(template, Loader=loader)
    return {"parsed": str(parsed)}

@app.post("/baking/cloudfoundry")
async def bake_cloudfoundry(request: Request):
    manifest = (await request.body()).decode("utf-8")
    # sink：使用 SafeLoader，禁止任意 Python 对象构造
    parsed = yaml.load(manifest, Loader=yaml.SafeLoader)
    return {"manifest": str(parsed)}

2. 原漏洞数据流：第 21 行 `yaml.load(template, Loader=loader)` 中 `loader` 来自第 16 行 `build_loader()`，原实现无论配置与否都返回不安全的 `yaml.Loader`，允许任意对象构造导致 RCE（CWE-502）。
3. 修复位置正确性：在第 15 行将 `build_loader()` 的返回值改为框架原生安全 API `yaml.SafeLoader`——这是 PyYAML 官方提供的白名单式构造器集合，仅允许基础标量/序列/映射类型，从类型上杜绝了任意类实例化；该修改位于 sink 上游的 Loader 构造点，覆盖所有经由该函数取 loader 的调用方。
4. 攻击面覆盖完整性：第二入口第 28 行 `yaml.load(manifest, Loader=yaml.Loader)` 同样是 CWE-502 sink，最小修复将其替换为 `yaml.SafeLoader`，两个入口的攻击面均被同一防御原语封堵。
5. 控制流与标识符保持不变：函数名、路由路径、变量名、`security_config` 结构均未改动，仅替换两处 Loader 参数并修正 `build_loader` 返回值，符合 minimal pair 纪律。
6. 第二入口/替代通道检查结论：除上述两处 `yaml.load` 外无其他反序列化通道；`set_parser_config` 仅写配置不再影响解析安全性，无残留风险。
