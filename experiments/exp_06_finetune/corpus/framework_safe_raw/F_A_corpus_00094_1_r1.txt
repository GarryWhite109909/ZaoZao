<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;

class DevProxyController extends Controller
{
    // 用户在 .env 或配置中指定的代理目标，例如 'http://localhost:3000'
    private function resolveTarget(Request $request): string
    {
        // 防御：忽略用户传入的 proxy_target，仅允许配置白名单中的回环地址
        $allowed = (array) config('app.dev_proxy_allowlist', [config('app.dev_proxy')]);
        $target = config('app.dev_proxy');

        if (!preg_match('#^https?://#', $target)) {
            abort(400, 'proxy must start with http:// or https://');
        }

        // 防御：白名单精确匹配（scheme+host+port），并强制目标为回环地址
        if (!in_array($target, $allowed, true)) {
            abort(403, 'proxy target not allowed');
        }
        $host = parse_url($target, PHP_URL_HOST);
        if (!in_array($host, ['127.0.0.1', '::1', 'localhost'], true)) {
            abort(403, 'proxy target must be loopback');
        }
        return $target;
    }

    public function proxy(Request $request)
    {
        $target = $this->resolveTarget($request);
        $path = $request->path();

        // 静态资源不代理，其余全部转发（含 WebSocket 升级请求）
        if ($this->isPublicAsset($path)) {
            return response()->file(public_path($path));
        }

        // 防御：路径白名单前缀精确允许集，限制可代理的路径范围
        $allowedPrefixes = (array) config('app.dev_proxy_path_prefixes', []);
        $matched = false;
        foreach ($allowedPrefixes as $prefix) {
            if ($path === $prefix || str_starts_with($path, rtrim($prefix, '/') . '/')) {
                $matched = true;
                break;
            }
        }
        if (!$matched) {
            abort(404);
        }

        $upstream = Http::withHeaders($request->headers->all())
            ->send($request->method(), rtrim($target, '/') . '/' . $path,
                ['body' => $request->getContent()]);

        return response($upstream->body(), $upstream->status())
            ->withHeaders($upstream->headers());
    }

    private function isPublicAsset(string $path): bool
    {
        return file_exists(public_path($path));
    }
}

6. 第二入口检查：`isPublicAsset`/静态资源分支（第 41–43 行）与原代码一致且不在本漏洞数据流上，按 minimal pair 纪律保持原样；无其他入口可注入 proxy_target。
