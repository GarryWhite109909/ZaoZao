package com.example.devproxy;

import org.springframework.boot.web.servlet.server.ServletWebServerFactory;
import org.springframework.http.*;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.messaging.handler.annotation.MessageMapping;
import org.springframework.stereotype.Controller;
import org.springframework.web.bind.annotation.*;

import javax.servlet.http.HttpServletRequest;
import java.net.URI;
import java.util.Set;

@Controller
public class ProxyController {

    // 用户在配置文件中指定的代理目标，例如 application.yml: proxy.target=http://localhost:8081
    @Value("${proxy.target}")
    private String proxyTarget;          // source: 外部可控的代理目标

    // 防御：部署时通过配置注入的精确白名单，只允许转发到显式登记的上游目标
    @Value("${proxy.allowed-targets}")
    private Set<String> allowedTargets;  // fix: 精确允许集（白名单）

    private final RestTemplate restTemplate = new RestTemplate(new SimpleClientHttpRequestFactory());

    // 宽泛的 catch-all 路由：拦截除静态资源外的所有请求并转发到 proxyTarget
    @RequestMapping(value = "/{path:(?!static).*}", method = {RequestMethod.GET, RequestMethod.POST})
    public ResponseEntity<byte[]> forward(HttpServletRequest request, @PathVariable String path) {
        URI upstream = URI.create(proxyTarget + "/" + path);   // sink: 转发前必须校验目标
        if (!mayProxy(path) || !isAllowedTarget(upstream)) {   // fix: 白名单精确匹配后才放行
            return ResponseEntity.notFound().build();
        }
        HttpHeaders headers = new HttpHeaders();
        // 与原版一致：把 Origin 改写为代理目标以规避 CORS 检查
        headers.set(HttpHeaders.ORIGIN, proxyTarget);
        HttpEntity<byte[]> entity = new HttpEntity<>(headers);
        return restTemplate.exchange(upstream, HttpMethod.resolve(request.getMethod()), entity, byte[].class);
    }

    // WebSocket 握手同样被宽泛上下文捕获并转发（ws: true 的等价语义）
    @MessageMapping("/{path:.+}")
    public void relaySocket(@DestinationVariable String path) {
        URI wsUpstream = URI.create(proxyTarget.replaceFirst("^http", "ws") + "/" + path);
        // fix: WebSocket 通道同样执行白名单校验，未命中则拒绝建立连接
        if (!isAllowedTarget(wsUpstream)) {
            return;
        }
        // 直接建立到用户指定目标的 WebSocket 连接，包括开发服务器自身的 HMR 端点
        new SimpleClientHttpRequestFactory(); // 连接逻辑省略，转发目标同上
    }

    // fix: 新增辅助函数——对最终上游 URI 做精确白名单匹配（scheme+host+port 全等比较）
    private boolean isAllowedTarget(URI upstream) {
        for (String allowed : allowedTargets) {
            URI a = URI.create(allowed);
            if (upstream.getScheme().equals(a.getScheme())
                    && upstream.getHost().equals(a.getHost())
                    && upstream.getPort() == a.getPort()) {
                return true;
            }
        }
        return false;
    }

    private boolean mayProxy(String pathname) {
        return !new java.io.File("public/" + pathname).exists();
    }
}
