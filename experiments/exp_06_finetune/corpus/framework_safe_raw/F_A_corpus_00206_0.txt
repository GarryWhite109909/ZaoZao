from fastapi import FastAPI, Request
from fastapi.responses import HTMLResponse
from urllib.parse import urlencode
import html

app = FastAPI()

@app.get("/oauth2/authorize/consent", response_class=HTMLResponse)
async def consent_page(request: Request):
    params = dict(request.query_params)
    client_name = params.get("client_name", "")
    client_desc = params.get("client_description", "")
    user_name = params.get("user_name", "")
    scopes = params.get("scopes", "").split()

    # build redirect target from raw request path + query string (escaped)
    target = request.url.path
    if request.url.query:
        target = target + "?" + request.url.query

    scope_items = "".join(
        f"<li>{html.escape(s)}</li>" for s in scopes
    )

    # FIXED: escape all interpolated values with html.escape before
    # inserting into the HTML template; escape the URL for attribute context
    page = f"""
    <html><body>
      <h1>Authorize {html.escape(client_name)}</h1>
      <p>{html.escape(client_desc)}</p>
      <p>Logged in as: {html.escape(user_name)}</p>
      <ul>{scope_items}</ul>
      <form method="post" action="/oauth2/authorize/decision?target={urlencode({'target': target})}">
        <input type="hidden" name="csrf" value="">
        <button type="submit">Allow</button>
      </form>
    </body></html>
    """
    return HTMLResponse(content=page)
