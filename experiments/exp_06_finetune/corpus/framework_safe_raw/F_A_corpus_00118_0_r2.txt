from fastapi import FastAPI, Request, HTTPException
import json

app = FastAPI()

ALLOWED_TYPES = {"str", "int", "float", "bool", "list", "dict"}

def safe_deserialize(raw: bytes):
    # 用 JSON 替代 pickle，并做类型白名单校验
    obj = json.loads(raw.decode("utf-8"))
    if type(obj).__name__ not in ALLOWED_TYPES:
        raise ValueError("type not allowed")
    return obj

@app.post("/deserialize")
async def deserialize_payload(request: Request):
    # 接收客户端上传的原始二进制序列化数据
    raw = await request.body()
    if not raw:
        raise HTTPException(status_code=400, detail="empty payload")

    try:
        # 反序列化：仅允许 JSON 基本类型白名单
        obj = safe_deserialize(raw)
    except Exception:
        raise HTTPException(status_code=400, detail="invalid payload")

    return {"type": type(obj).__name__, "value": str(obj)[:200]}

@app.post("/session/restore")
async def restore_session(request: Request):
    # 另一个入口：从请求头读取 base64 编码的会话快照并反序列化
    import base64
    encoded = request.headers.get("X-Session-Snapshot", "")
    if not encoded:
        raise HTTPException(status_code=400, detail="missing snapshot")
    data = base64.b64decode(encoded)
    state = json.loads(data.decode("utf-8"))  # JSON 解析 + 白名单校验
    if not isinstance(state, dict) or any(type(v).__name__ not in ALLOWED_TYPES for v in state.values()):
        raise HTTPException(status_code=400, detail="invalid payload")
    return {"restored_keys": list(state.keys())}
