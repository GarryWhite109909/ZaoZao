```go
package main

import (
	"net/http"

	"github.com/gin-gonic/gin"
)

type OAuth2Client struct {
	ClientID     string `json:"client_id"`
	ClientSecret string `json:"client_secret"`
	Name         string `json:"name"`
	RedirectURL  string `json:"redirect_url"`
}

var clients = map[string]OAuth2Client{
	"app1": {ClientID: "app1", ClientSecret: "s3cret", RedirectURL: "https://app.example.com/callback"},
}

func authorizeHandler(c *gin.Context) {
	clientID := c.Query("client_id")
	redirectURI := c.Query("redirect_uri")

	client, ok := clients[clientID]
	if !ok {
		c.JSON(http.StatusBadRequest, gin.H{"error": "unknown client"})
		return
	}

	if redirectURI == client.RedirectURL {
		c.Redirect(http.StatusFound, redirectURI)
		return
	}
	c.JSON(http.StatusForbidden, gin.H{"error": "illegal_redirect_uri"})
}

func main() {
	r := gin.Default()
	r.GET("/oauth2/authorize", authorizeHandler)
	r.Run(":8080")
}
```

1. 第 17-18 行：用户可控输入点为 c.Query("client_id") 与 c.Query("redirect_uri")，二者均直接来自请求查询参数，未经任何校验即进入后续逻辑。
2. 第 20-24 行：clientID 用于 map 查找，属于键查找而非重定向目标，本身不构成 sink；redirect_uri 是本漏洞数据流的 source，沿控制流直达第 29 行。
3. 原代码第 28 行使用 strings.HasPrefix(redirectURI, client.RedirectURL) 做前缀匹配，攻击者可注册或利用如 https://app.example.com/callback.evil.com 或 https://app.example.com/callback/../../attacker 等前缀吻合但主机/路径不同的 URI 绕过校验，随后第 29 行 c.Redirect 将其作为 Location 输出，构成 CWE-601 开放重定向 sink。
4. 防御类型正确性：将前缀匹配替换为 redirectURI == client.RedirectURL 的白名单精确相等比较（RFC 6749 第 3.1.2.3 节要求的精确字符串匹配），仅允许预先在 clients 注册表中登记的完整 URL，任何附加路径、子域、尾缀均被拒绝；该比较不依赖黑名单或正则，属于白名单精确允许集防御。
5. 位置正确性：修复位于第 28 行，即 source（redirect_uri）进入 sink（c.Redirect）之前的唯一必经分支点，覆盖了该 handler 中 redirect_uri 到 c.Redirect 的全部数据流路径；c.Redirect 仅在此分支内可达，无旁路。
6. 第二入口检查：main 函数第 36 行仅注册 /oauth2/authorize 一个路由，无其他 handler 接收 redirect_uri 并执行重定向；clients 注册表为硬编码常量，无动态写入通道。除已修复的主链外不存在替代开放重定向通道，结论为漏洞已消除。
