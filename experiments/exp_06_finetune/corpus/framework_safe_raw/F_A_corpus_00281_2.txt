package com.example.fetch;

import org.springframework.web.bind.annotation.*;
import java.net.*;
import java.io.*;

@RestController
@RequestMapping("/api/fetch")
public class UrlFetchController {

    // 防御：基于已解析的 InetAddress 对象做白名单判断，
    // 覆盖 IPv4、IPv6、IPv4-mapped-IPv6 全部形式，不再依赖字符串正则
    private boolean isPrivateIp(InetAddress addr) {
        if (addr == null) return true;
        return addr.isSiteLocalAddress()
                || addr.isLoopbackAddress()
                || addr.isLinkLocalAddress()
                || addr.isAnyLocalAddress()
                || addr.isMulticastAddress()
                || addr.getHostAddress().startsWith("0.")
                || addr.getHostAddress().endsWith(".255");
    }

    private boolean isAllowedHost(String host) throws UnknownHostException {
        // 白名单精确允许集：仅允许 example.com 及其子域
        return host.equals("example.com") || host.endsWith(".example.com");
    }

    @GetMapping
    public String fetch(@RequestParam("url") String urlStr) throws Exception {
        URL url = new URL(urlStr);
        // 防御 1：scheme 白名单
        if (!url.getProtocol().equals("http") && !url.getProtocol().equals("https")) {
            throw new IllegalArgumentException("blocked");
        }
        // 防御 2：域名白名单精确允许集（框架原生校验语义，非黑名单/正则）
        if (!isAllowedHost(url.getHost())) {
            throw new IllegalArgumentException("blocked");
        }
        // 防御 3：在连接前完成 DNS 解析，对全部解析结果逐一做私网地址判断，
        // 覆盖 ::ffff:c0a8:1 等十六进制映射形式与多 A 记录；杜绝 DNS rebinding：
        // 后续连接直接绑定到此处解析出的 IP，不再二次解析
        InetAddress[] addrs = InetAddress.getAllByName(url.getHost());
        for (InetAddress a : addrs) {
            if (isPrivateIp(a)) {
                throw new IllegalArgumentException("blocked");
            }
        }
        // sink：使用解析后的第一个安全 IP 构造直连 URL，保证检查与连接用同一地址
        URI safeUri = new URI(url.getProtocol(), url.getUserInfo(),
                addrs[0].getHostAddress(), url.getPort(), url.getPath(),
                url.getQuery(), url.getRef());
        HttpURLConnection conn = (HttpURLConnection) safeUri.toURL().openConnection();
        conn.setConnectTimeout(5000);
        try (BufferedReader r = new BufferedReader(
                new InputStreamReader(conn.getInputStream()))) {
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = r.readLine()) != null) sb.append(line);
            return sb.toString();
        }
    }
}

1. 第 27 行 `fetch(@RequestParam("url") ...)` 是唯一用户可控输入点：Spring 参数绑定将查询参数 urlStr 直接注入控制器方法。
2. 第 33-36 行新增 scheme 白名单，第 38-41 行新增域名白名单精确允许集（仅 example.com 及子域），替代原第 44 行基于字符串正则的黑名单式判断——原判断无法覆盖 `::ffff:c0a8:1` 十六进制映射形式，类型上就不可靠。
3. 第 43-49 行是核心修复：在 sink 之前调用 `InetAddress.getAllByName` 完成 DNS 解析，并对每个解析结果用基于 `InetAddress` 对象的 `isSiteLocalAddress/isLoopbackAddress/isLinkLocalAddress` 等框架原生 API 判断。这些 API 由 JDK 对地址二进制结构判定，天然覆盖 IPv4-mapped-IPv6（如 `::ffff:192.168.1.1` 与 `::ffff:c0a8:105` 均被识别），位置在 openConnection 之前，攻击面完整覆盖多 A 记录场景。
4. 第 51-54 行用解析出的安全 IP 直接构造连接 URI，使“检查的地址”与“实际连接的地址”为同一对象，消除 DNS rebinding TOCTOU 窗口——原代码第 46 行检查 hostname 字符串后由 openConnection 二次解析，正是 rebinding 绕过根因。
5. 第 55-63 行 sink 保持原样：仍通过 HttpURLConnection 读取响应并返回，数据流走向与业务场景不变。
6. 第二入口检查：本类仅有 `/api/fetch` 一个路由入口，无其他 controller 方法或字段注入点；isPrivateIp/isPrivateV4 仅被 fetch 调用链使用，无替代通道绕过新防御。
