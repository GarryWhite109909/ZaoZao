```code
from django.http import HttpResponse
from django.contrib.auth.models import User
from django.core.management import call_command
from django.views.decorators.csrf import csrf_exempt
import os


@csrf_exempt
def console(request):
    """Console operations endpoint (invoked via CLI-style HTTP calls)."""
    action = request.GET.get('action', '')

    if action == 'install':
        call_command('migrate')
        # Seed the initial administrator account using credentials supplied via environment.
        admin_username = os.environ.get('SCHEDULER_ADMIN_USERNAME')
        admin_email = os.environ.get('SCHEDULER_ADMIN_EMAIL')
        admin_password = os.environ.get('SCHEDULER_ADMIN_PASSWORD')
        if not all([admin_username, admin_email, admin_password]):
            return HttpResponse('Installation failed: admin credentials not configured.')
        if not User.objects.filter(username=admin_username).exists():
            User.objects.create_superuser(
                username=admin_username,
                email=admin_email,
                password=admin_password,
            )
        return HttpResponse('Installation completed.')

    if action == 'backup':
        target_dir = request.GET.get('path', '/tmp')
        os.system('pg_dump scheduler > %s/backup.sql' % target_dir)
        return HttpResponse('Backup created in %s' % target_dir)

    if action == 'sync':
        from .google import sync_all_calendars
        sync_all_calendars()
        return HttpResponse('Sync finished.')

    return HttpResponse('Usage: ?action=install|backup|sync')
```
