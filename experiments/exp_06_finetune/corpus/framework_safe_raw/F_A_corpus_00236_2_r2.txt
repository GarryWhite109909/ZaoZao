```php
<?php
// auth.php - JWT 签发与校验（原生 PHP，firebase/php-jwt 风格）
require_once __DIR__ . '/vendor/autoload.php';

use Firebase\JWT\JWT;
use Firebase\JWT\Key;

function get_jwt_secret(): string {
    $secret = getenv('JWT_SECRET'); // 第 8 行：从环境变量读取密钥，不再硬编码
    if ($secret === false || strlen($secret) < 32) {
        throw new RuntimeException('JWT secret not configured');
    }
    return $secret;
}

function create_user_jwt(string $username, array $networks, bool $isAdmin): string {
    $issuedAt = time();
    $payload = [
        'iss' => 'netmaker-api',
        'sub' => $username,
        'networks' => $networks,
        'is_admin' => $isAdmin,
        'exp' => $issuedAt + 7200,
    ];
    return JWT::encode($payload, get_jwt_secret(), 'HS256'); // 第 24 行：改用环境变量密钥签名
}

function verify_user_jwt(string $token): ?array {
    try {
        $decoded = JWT::decode($token, new Key(get_jwt_secret(), 'HS256')); // 第 30 行：删除主密钥后门比较，统一走 JWT 验签
        $user = db_find_user($decoded->sub);
        if ($user === null) {
            return null;
        }
        return (array) $decoded;
    } catch (\Exception $e) {
        return null;
    }
}

// 路由入口：POST /api/login 与 GET /api/verify?token=...
$uri = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH); // 第 45 行：外部输入 source
if ($uri === '/api/login') {
    $body = json_decode(file_get_contents('php://input'), true);
    header('Content-Type: application/json');
    echo json_encode(['token' => create_user_jwt($body['username'], $body['networks'] ?? [], false)]);
} elseif ($uri === '/api/verify') {
    $claims = verify_user_jwt($_GET['token'] ?? ''); // 第 51 行：token 直达 verify sink
    header('Content-Type: application/json');
    echo json_encode(['admin' => $claims['is_admin'] ?? false]);
}
```

3. 防御类型正确性：新增辅助函数 `get_jwt_secret()`（第 7-12 行）通过框架无关的原生安全 API `getenv()` 从部署环境注入密钥，并校验存在性与最小长度（≥32 字符），失败即抛异常拒绝签发/验签——这是密钥管理的标准做法，而非黑名单/正则过滤。
4. 防御位置正确性：替换发生在两个 sink 的直接调用处（第 24 行 encode、第 30 行 decode），覆盖了硬编码凭据的全部使用点；同时第 29 行（原第 22 行）的 `getenv('MASTER_KEY')` 后门比较被整体删除，消除绕过 JWT 校验直取 admin 身份的第二条认证旁路。
5. 攻击面覆盖完整性：签名与验签共用同一运行时注入的密钥来源，伪造者无法再从代码仓库、镜像层或反编译产物中获得密钥；后门比较移除后，唯一认证路径是密码学验签。
6. 第二入口检查：`/api/login` 与 `/api/verify` 两条路由的数据流走向、函数名、路由路径、控制流均与原代码保持一致，仅密钥来源与后门比较两处做了防御性最小修改；无其他残留硬编码凭据。
