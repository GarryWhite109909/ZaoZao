package com.example.web.config;

import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;

@Component
public class TemplateFetchInterceptor implements HandlerInterceptor {

    // 模板白名单目录，本应限制在 /var/app/templates 内
    private static final Path TEMPLATE_ROOT = Paths.get("/var/app/templates");

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) throws IOException {
        String templateName = request.getParameter("template");   // source: 外部输入入口
        if (templateName == null) {
            response.sendError(400, "missing template param");
            return false;
        }

        // 防御：白名单精确允许集校验，仅允许合法模板文件名
        if (!isAllowedTemplateName(templateName)) {
            response.sendError(400, "invalid template name");
            return false;
        }

        // 拼接路径并读取文件：sink（已由白名单保证无路径穿越）
        Path target = TEMPLATE_ROOT.resolve(templateName).normalize();
        byte[] content = Files.readAllBytes(target);

        response.setContentType("text/plain");
        response.getOutputStream().write(content);
        return false;
    }

    // 防御辅助函数：精确允许集校验，拒绝任何包含路径分隔符或 .. 的名称
    private static boolean isAllowedTemplateName(String name) {
        if (name.isEmpty() || name.length() > 64) {
            return false;
        }
        for (char c : name.toCharArray()) {
            if (!(Character.isLetterOrDigit(c) || c == '_' || c == '-' || c == '.')) {
                return false;
            }
        }
        if (name.contains("..") || name.startsWith("/") || name.contains("\") || name.contains("/")) {
            return false;
        }
        return Files.exists(TEMPLATE_ROOT.resolve(name).normalize())
                && TEMPLATE_ROOT.equals(TEMPLATE_ROOT.resolve(name).normalize().getParent());
    }
}
