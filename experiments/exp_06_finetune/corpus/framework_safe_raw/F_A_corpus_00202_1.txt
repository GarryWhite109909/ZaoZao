<?php

namespace App\Http\Controllers;

use App\Models\Post;
use App\Models\PostRevision;
use Illuminate\Http\Request;

class PostHistoryController extends Controller
{
    // GET /admin/posts/{post}/history
    public function show(Request $request, Post $post)
    {
        $revisions = $post->revisions()
            ->orderByDesc('created_at')
            ->get()
            ->map(function (PostRevision $revision, int $index) {
                return [
                    'index' => $index,
                    'title' => e($revision->title),
                    'content' => Purifier::clean($revision->content), // 历史版本 HTML 内容：白名单净化
                    'created_at' => $revision->created_at,
                ];
            });

        return view('admin.post-history', [
            'post' => $post,
            'revisions' => $revisions,
        ]);
    }

    // POST /admin/posts/{post}/history/preview
    // 预览某个历史版本的渲染结果
    public function preview(Request $request, Post $post)
    {
        $validated = $request->validate([
            'revision_index' => 'required|integer',
        ]);

        $revision = $post->revisions()
            ->orderByDesc('created_at')
            ->skip($validated['revision_index'])
            ->firstOrFail();

        // 预览响应中输出经过 HTMLPurifier 白名单净化的历史版本 HTML
        return response()->json([
            'title' => e($revision->title),
            'html' => Purifier::clean($revision->content),
        ]);
    }
}
