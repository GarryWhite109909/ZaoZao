import org.springframework.web.bind.annotation.*;
import java.util.*;
import java.util.regex.Pattern;

@RestController
@RequestMapping("/api/alerts")
public class AlertActionController {

    // 模拟管理员在配置中注册的告警命令模板（含 {{mustache}} 占位符）
    private final Map<String, String> alertCommands = new HashMap<>();

    private final Map<String, String> lastStatus = new HashMap<>();

    // 白名单精确允许集：占位符值仅允许字母、数字、空格及有限安全标点，杜绝 shell 元字符
    private static final Pattern SAFE_VALUE = Pattern.compile("^[A-Za-z0-9 ._:/-]{0,128}$");

    public AlertActionController() {
        alertCommands.put("cpu_critical", "logger -t glances 'CPU critical on host {{hostname}}' >> /var/log/glances.log");
    }

    /** 防御辅助函数：白名单校验单个字符串值 */
    private boolean isSafeValue(String s) {
        return s != null && SAFE_VALUE.matcher(s).matches();
    }

    /** 防御辅助函数：递归校验任意结构（含嵌套 List）中的所有字符串值 */
    private boolean isSafeStructure(Object v) {
        if (v instanceof String s) {
            return isSafeValue(s);
        }
        if (v instanceof List<?> list) {
            for (Object item : list) {
                if (!isSafeStructure(item)) {
                    return false;
                }
            }
            return true;
        }
        return true;
    }

    /** 外部输入点1：监控探针上报的指标数据（含进程 cmdline、主机名等攻击者可控字段） */
    @PostMapping("/trigger")
    public Map<String, Object> trigger(@RequestBody Map<String, Object> stats) {
        String statName = (String) stats.get("stat_name");
        String criticality = (String) stats.get("criticality");
        String cmdTemplate = alertCommands.get(statName);
        if (cmdTemplate == null) {
            return Map.of("ran", false);
        }
        if (criticality.equals(lastStatus.get(statName))) {
            return Map.of("ran", false); // 去重，等价 repeat=false
        }
        // 白名单防御：递归校验所有待渲染值（含嵌套列表），任一不合法即拒绝执行
        for (Map.Entry<String, Object> e : stats.entrySet()) {
            if (!isSafeStructure(e.getValue())) {
                return Map.of("ran", false);
            }
        }
        String cmd = cmdTemplate;
        for (Map.Entry<String, Object> e : stats.entrySet()) {
            cmd = cmd.replace("{{" + e.getKey() + "}}", String.valueOf(e.getValue()));
        }
        // sink：经 /bin/sh 解释执行；因所有插值已通过白名单精确允许集校验，shell 元字符无法注入
        try {
            Process p = new ProcessBuilder("/bin/sh", "-c", cmd).start();
            p.waitFor();
        } catch (Exception ignored) {}
        lastStatus.put(statName, criticality);
        return Map.of("ran", true);
    }

    /** 第二入口：同一模板渲染逻辑被 /api/processes 间接复用 */
    @PostMapping("/processes/report")
    public Map<String, Object> report(@RequestBody Map<String, Object> procStats) {
        procStats.put("stat_name", "cpu_critical");
        return trigger(procStats);
    }
}

分析：

1. **第 34 行**：用户可控输入点为 `@PostMapping("/trigger")` 的 `@RequestBody Map<String, Object> stats`，其中 `hostname`、进程 cmdline 列表等字段全部由监控探针上报，攻击者可完全控制其内容。
2. **第 55–60 行**：原代码的黑名单 `replace` 只清洗顶层字符串且可被编码绕过，嵌套 List 直接透传。修复在第 56–59 行改为调用 `isSafeStructure` 对每个待渲染值做**白名单精确允许集**校验（第 20–21 行的 `SAFE_VALUE` 正则仅允许字母数字与有限标点），任何包含 shell 元字符（`;`、`&`、`|`、反引号、`$` 等）的值都会使请求直接返回 `ran=false`，不进入渲染。
3. **第 62–65 行**：模板替换逻辑保持不变；由于所有插值已通过白名单，`{{hostname}}` 等占位符只能被安全字符集内的字符串填充，类型正确性上 `String.valueOf` 输出已被上游约束，位置正确性上校验发生在 sink 之前的唯一汇聚点，覆盖了所有键的值（含嵌套 cmdline 列表的递归遍历，第 26–37 行）。
4. **第 68–71 行**：sink 仍为 `new ProcessBuilder("/bin/sh", "-c", cmd)`，保留原框架习语与业务语义（logger 重定向到日志文件）；但因第 56–59 行的白名单门禁，注入载荷无法到达此 sink，属于有效防御而非黑名单过滤。
5. **第二入口检查**：`/api/processes/report`（第 79–83 行）仅设置 `stat_name` 后直接委托给 `trigger`，因此同样经过第 56–59 行的白名单校验，替代通道无绕过路径。
