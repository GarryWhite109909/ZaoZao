<?php
// share.php — 文件分享访问入口（原生 PHP + PDO）
declare(strict_types=1);

$pdo = new PDO('sqlite:/var/data/app.db');
$pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);

function getShareById(PDO $pdo, string $id): ?array {
    $stmt = $pdo->prepare('SELECT * FROM shares WHERE id = :id');
    $stmt->execute([':id' => $id]);
    $row = $stmt->fetch(PDO::FETCH_ASSOC);
    return $row ?: null;
}

// 防御辅助：将路径规范化为绝对形式，供归属校验使用（新增，仅防御用途）
function resolveUnderRoot(string $root, string $relative): ?string {
    $base = realpath($root);
    if ($base === false) { return null; }
    $target = realpath($base . '/' . ltrim($relative, '/'));
    if ($target === false) { return null; }
    // 前缀校验：目标必须严格位于根目录之内（分隔符感知）
    if ($target !== $base && strpos($target, $base . DIRECTORY_SEPARATOR) !== 0) {
        return null;
    }
    return $target;
}

// source: 用户通过 GET 参数传入分享 ID 与要访问的子路径
$shareId = $_GET['share_id'] ?? '';
$subPath = $_GET['path'] ?? '/';

$share = getShareById($pdo, $shareId);
if ($share === null) {
    http_response_code(404);
    exit('share not found');
}

$files = json_decode($share['files_raw'], true) ?? [];

if (count($files) === 1) {
    // sink: 拼接根目录与用户路径后，先做规范化+前缀归属校验再读取
    $realPath = rtrim($share['root_dir'], '/') . '/' . ltrim($subPath, '/');
} else {
    $child = explode('/', trim($subPath, '/'))[0];
    $rest  = implode('/', array_slice(explode('/', trim($subPath, '/')), 1));
    $mapPath = '';
    foreach ($files as $f) {
        if (basename($f) === $child) { $mapPath = $f; break; }
    }
    if ($mapPath === '') { http_response_code(403); exit('not allowed'); }
    $realPath = $mapPath . '/' . $rest;
}

// sink: 规范化后校验最终路径仍位于分享根目录内，越界则拒绝
$finalPath = resolveUnderRoot('/srv/storage/' . rtrim($share['root_dir'], '/'), substr($realPath, strlen(rtrim($share['root_dir'], '/'))));
if ($finalPath === null || !is_file($finalPath)) {
    http_response_code(403);
    exit('not allowed');
}

header('Content-Type: application/octet-stream');
readfile($finalPath);
