from flask import Flask, request, jsonify
import re
from lxml import etree

app = Flask(__name__)

# 依赖此净化函数的下游会直接渲染返回的 SVG（信任其已无脚本）
def sanitize_svg(svg_text):
    # 白名单式净化：解析后仅保留允许的元素与属性，其余（含 script/事件属性）一律剔除
    ALLOWED_TAGS = {'svg', 'g', 'path', 'rect', 'circle', 'ellipse', 'line', 'polyline', 'polygon', 'text', 'tspan', 'defs', 'use', 'title', 'desc'}
    ALLOWED_ATTRS = {'d', 'x', 'y', 'x1', 'y1', 'x2', 'y2', 'cx', 'cy', 'r', 'rx', 'ry', 'width', 'height', 'fill', 'stroke', 'stroke-width', 'transform', 'viewBox', 'points', 'opacity'}
    parser = etree.XMLParser(resolve_entities=False, no_network=True)
    root = etree.fromstring(svg_text.encode('utf-8'), parser=parser)

    for el in root.iter():
        if etree.QName(el).localname not in ALLOWED_TAGS:
            el.getparent().remove(el)
            continue
        for attr in list(el.attrib):
            name = attr.split('}')[-1]
            if name not in ALLOWED_ATTRS:
                del el.attrib[attr]

    return etree.tostring(root, encoding='unicode')

@app.route('/api/svg/clean', methods=['POST'])
def clean_svg():
    svg_text = request.form.get('svg', '')
    if not svg_text:
        return jsonify({'error': 'missing svg field'}), 400
    result = sanitize_svg(svg_text)
    return result, 200, {'Content-Type': 'image/svg+xml'}

@app.route('/view/<doc_id>')
def view_doc(doc_id):
    # 第二入口：从存储读取并直接内联渲染用户上传的 SVG 片段
    snippet = load_user_snippet(doc_id)  # 假设来自数据库的用户内容
    return f'<div class="embed">{snippet}</div>'

def load_user_snippet(doc_id):
    import sqlite3
    conn = sqlite3.connect('docs.db')
    row = conn.execute('SELECT body FROM snippets WHERE id=?', (doc_id,)).fetchone()
    conn.close()
    return row[0] if row else ''
