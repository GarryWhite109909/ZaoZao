```python
import re
import subprocess
import shlex
from flask import Flask, request, jsonify

app = Flask(__name__)

WORKSPACE = "/srv/workspace"
TIMEOUT = 60

ALLOWED_BINARIES = {"ls", "cat", "grep", "find", "wc", "head", "tail", "echo"}

def guard_command(cmd, cwd):
    try:
        tokens = shlex.split(cmd)
    except ValueError:
        return "Command blocked (malformed command)"
    if not tokens:
        return "Command is required"
    if tokens[0] not in ALLOWED_BINARIES:
        return "Command blocked by safety guard"
    for token in tokens[1:]:
        if token.startswith("-") and token not in {"-l", "-n", "-i", "-r"}:
            return "Command blocked by safety guard"
        if ".." in token or token.startswith("/"):
            return "Command blocked (path traversal detected)"
    return ""

@app.route("/api/exec", methods=["POST"])
def exec_command():
    data = request.get_json(silent=True) or {}
    command = data.get("command")
    if not command:
        return jsonify({"error": "command is required"}), 400

    cwd = data.get("working_dir") or WORKSPACE
    err = guard_command(command, cwd)
    if err:
        return jsonify({"error": err}), 403

    try:
        result = subprocess.run(
            shlex.split(command),
            cwd=cwd, capture_output=True, text=True, timeout=TIMEOUT
        )
        output = result.stdout + result.stderr
        return jsonify({"output": output or "(no output)"})
    except subprocess.TimeoutExpired:
        return jsonify({"error": "Command timed out"}), 504
    except (FileNotFoundError, PermissionError):
        return jsonify({"error": "Command execution failed"}), 400

if __name__ == "__main__":
    app.run()
```
