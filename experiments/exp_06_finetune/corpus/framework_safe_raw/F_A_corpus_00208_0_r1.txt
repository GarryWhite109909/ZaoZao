from flask import Flask, request, jsonify
import re
import xml.etree.ElementTree as ET

app = Flask(__name__)

# SVG 白名单：仅允许结构性/绘图标签与安全属性，事件属性一律剔除
ALLOWED_TAGS = {'svg', 'g', 'path', 'rect', 'circle', 'ellipse', 'line', 'polyline', 'polygon', 'text', 'tspan', 'defs', 'use', 'title', 'desc'}
ALLOWED_ATTRS = {'id', 'class', 'd', 'x', 'y', 'cx', 'cy', 'r', 'rx', 'ry', 'width', 'height', 'fill', 'stroke', 'stroke-width', 'transform', 'viewBox', 'xmlns', 'points', 'x1', 'y1', 'x2', 'y2', 'dx', 'dy', 'font-size', 'text-anchor'}

def _sanitize_element(elem):
    # 白名单精确允许集：非白名单标签整体丢弃（含 script/foreignObject/on* 事件载体）
    if not (elem.tag in ALLOWED_TAGS):
        return None
    for attr in list(elem.attrib.keys()):
        if attr not in ALLOWED_ATTRS:
            del elem.attrib[attr]
    kept = []
    for child in list(elem):
        cleaned_child = _sanitize_element(child)
        if cleaned_child is None:
            elem.remove(child)
        else:
            kept.append(cleaned_child)
    return elem

# 依赖此净化函数的下游会直接渲染返回的 SVG（信任其已无脚本）
def sanitize_svg(svg_text):
    # 白名单式净化：解析后仅保留允许的标签/属性，替代原黑名单正则
    root = ET.fromstring(svg_text)
    _sanitize_element(root)
    return ET.tostring(root, encoding='unicode')

@app.route('/api/svg/clean', methods=['POST'])
def clean_svg():
    svg_text = request.form.get('svg', '')
    if not svg_text:
        return jsonify({'error': 'missing svg field'}), 400
    result = sanitize_svg(svg_text)
    return result, 200, {'Content-Type': 'image/svg+xml'}

@app.route('/view/<doc_id>')
def view_doc(doc_id):
    # 第二入口：从存储读取并直接内联渲染用户上传的 SVG 片段
    snippet = load_user_snippet(doc_id)  # 假设来自数据库的用户内容
    return f'<div class="embed">{snippet}</div>'

def load_user_snippet(doc_id):
    import sqlite3
    conn = sqlite3.connect('docs.db')
    row = conn.execute('SELECT body FROM snippets WHERE id=?', (doc_id,)).fetchone()
    conn.close()
    return row[0] if row else ''
