package com.example.booking.config;

import org.springframework.context.annotation.Configuration;
import org.springframework.web.bind.annotation.*;
import org.springframework.jdbc.core.JdbcTemplate;
import javax.servlet.http.HttpServletRequest;
import java.util.List;
import java.util.Map;
import org.jsoup.Jsoup;
import org.jsoup.safety.Safelist;

@Configuration
@RestController
@RequestMapping("/api/booking-settings")
public class BookingSettingsController {

    private final JdbcTemplate jdbc;

    public BookingSettingsController(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    @GetMapping
    public Map<String, Object> index(HttpServletRequest request) {
        return Map.of("settings", jdbc.queryForList("SELECT name, value FROM booking_settings"));
    }

    @PostMapping("/save")
    public String save(HttpServletRequest request, @RequestBody List<Map<String, Object>> settings) {
        if (!"ADMIN".equals(request.getSession().getAttribute("role"))) {
            return "redirect:/login";
        }
        for (Map<String, Object> setting : settings) {
            String name = (String) setting.get("name");
            String value = (String) setting.get("value");
            if ("disable_booking_message".equals(name)) {
                value = Jsoup.clean(value, Safelist.basic()); // whitelist-based sanitization
            }
            Integer existing = jdbc.queryForObject(
                "SELECT COUNT(*) FROM booking_settings WHERE name = ?", Integer.class, name);
            if (existing > 0) {
                jdbc.update("UPDATE booking_settings SET value = ? WHERE name = ?", value, name);
            } else {
                jdbc.update("INSERT INTO booking_settings(name, value) VALUES (?, ?)", name, value);
            }
        }
        return "ok";
    }
}

2. 第 31-32 行原代码用 `replaceAll("<script>", "")` 黑名单过滤，可被 `<img onerror=...>`、大小写变体、嵌套标签等绕过，属于无效防御。
3. 第 32 行替换为 `Jsoup.clean(value, Safelist.basic())`：基于白名单解析 HTML 并剥离所有脚本/事件属性/危险协议（javascript: 等），类型正确（返回净化后的 String），位置正确（在入库前对唯一目标字段处理）。
4. 攻击面覆盖：Safelist 白名单机制覆盖所有标签与属性注入向量，而非仅匹配特定模式，杜绝黑名单绕过路径；其余字段（name 仅用于参数化 SQL 匹配）不进入 HTML 渲染上下文，无需额外处理。
5. 第 34-40 行 SQL 已使用 `?` 参数化占位符，无 SQL 注入问题，保持原样以维持 minimal pair 局部对照性。
6. 第二入口检查：GET `/api/booking-settings` 第 20 行仅读取数据回显，写入通道只有 POST `/save` 一条；该入口已修复，无未防护替代通道。
