from flask import Flask, request, render_template_string
from markupsafe import escape

app = Flask(__name__)

RAW_CONTENT_TAGS = {"style", "script", "iframe", "noembed", "noscript", "noframes", "plaintext", "xmp"}

def escape_html(s):
    return s.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")

def escape_attr(s):
    return escape_html(s).replace('"', "&quot;")

def serialize_node(tag, attrs, raw_children):
    out = "<" + tag
    for k, v in attrs.items():
        out += ' ' + k + '="' + escape_attr(v) + '"'
    out += ">"
    if tag in RAW_CONTENT_TAGS:
        # 修复：raw-content 元素内容不再原样输出，
        # 统一使用框架原生转义 API markupsafe.escape 处理全部用户内容，
        # 彻底消除闭合标签变体绕过（大小写/空白/嵌套）的可能
        out += str(escape(raw_children))
    else:
        out += escape_html(raw_children)
    out += "</" + tag + ">"
    return out

@app.route("/render")
def render_page():
    # source: 用户可控的 iframe 内容与属性
    user_content = request.args.get("content", "")
    src = request.args.get("src", "")
    html = serialize_node("iframe", {"src": src}, user_content)
    return render_template_string("<html><body>{{ body|safe }}</body></html>", body=html)

if __name__ == "__main__":
    app.run()
