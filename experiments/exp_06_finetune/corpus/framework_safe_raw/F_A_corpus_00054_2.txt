```go
package main

import (
	"net/http"
	"os"
	"path/filepath"
	"strings"
)

// 模型权重文件编辑服务：允许客户端读取/下载服务器上的 .h5/.keras 权重文件
func main() {
	mux := http.NewServeMux()
	mux.HandleFunc("/api/weights", weightsHandler)
	http.ListenAndServe(":8080", mux)
}

// weightsRoot 是权重文件的唯一允许根目录（白名单精确允许集）
const weightsRoot = "/var/lib/model-weights"

// allowedWeightsFiles 是服务端预定义的文件名白名单，客户端只能按名引用，不能传路径
var allowedWeightsFiles = map[string]bool{
	"resnet50.h5":    true,
	"bert-base.h5":   true,
	"vit-l16.keras":  true,
}

func weightsHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, "method not allowed", http.StatusNotImplemented)
		return
	}

	// source：外部输入仅作为白名单键名，不再作为路径
	fileName := r.URL.Query().Get("path")
	if fileName == "" {
		http.Error(w, "missing path param", http.StatusBadRequest)
		return
	}

	// 防御 1：拒绝任何包含路径分隔符的输入，杜绝目录穿越语义
	if strings.ContainsAny(fileName, `/\`) || fileName == "." || fileName == ".." {
		http.Error(w, "invalid file name", http.StatusBadRequest)
		return
	}

	// 防御 2：后缀白名单校验（保留原校验）
	if filepath.Ext(fileName) != ".h5" && filepath.Ext(fileName) != ".keras" {
		http.Error(w, "unsupported file type", http.StatusBadRequest)
		return
	}

	// 防御 3：精确允许集——文件名必须存在于服务端白名单中
	if !allowedWeightsFiles[fileName] {
		http.Error(w, "file not allowed", http.StatusNotFound)
		return
	}

	// sink：路径由受信常量根 + 白名单文件名拼接，外部输入无法控制目录部分
	filePath := filepath.Join(weightsRoot, fileName)

	f, err := os.Open(filePath)
	if err != nil {
		http.Error(w, "file not found", http.StatusNotFound)
		return
	}
	defer f.Close()

	w.Header().Set("Content-Type", "application/octet-stream")
	http.ServeContent(w, r, filepath.Base(filePath), f.ModTime(), f)
}
```

3. 第 45~48 行：保留原有 `.h5`/`.keras` 后缀白名单，维持 minimal pair 中原有的业务校验不变。
4. 第 51~54 行：防御二为精确允许集（map 白名单），文件名必须命中服务端预定义集合，这是硬性要求认可的白名单精确允许集防御，位置在 sink 之前、source 之后，覆盖了符号链接外的所有直接输入攻击面。
5. 第 57 行：sink 处路径由受信常量 `weightsRoot` 与已验证文件名经 `filepath.Join` 拼接，外部输入完全无法控制目录分量，任意本地文件读取链路被切断。
6. 第二入口检查：本 handler 仅注册于 `/api/weights` 一个路由，无其他 query/body/header 数据流进入 `os.Open`；`http.ServeContent` 仅消费已打开的受控文件句柄，无替代通道残留。
