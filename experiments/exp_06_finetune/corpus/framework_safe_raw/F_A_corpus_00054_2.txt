package main

import (
	"net/http"
	"os"
	"path/filepath"
)

// 模型权重文件编辑服务：允许客户端读取/下载服务器上的 .h5/.keras 权重文件
func main() {
	mux := http.NewServeMux()
	mux.HandleFunc("/api/weights", weightsHandler)
	http.ListenAndServe(":8080", mux)
}

// weightsRootDir：权重文件唯一允许的根目录（白名单精确允许集）
const weightsRootDir = "/var/lib/model-weights"

func weightsHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}

	// source：外部输入直接作为文件路径
	filePath := r.URL.Query().Get("path")
	if filePath == "" {
		http.Error(w, "missing path param", http.StatusBadRequest)
		return
	}

	// 仅做后缀白名单校验，未限制目录范围
	if filepath.Ext(filePath) != ".h5" && filepath.Ext(filePath) != ".keras" {
		http.Error(w, "unsupported file type", http.StatusBadRequest)
		return
	}

	// 防御：将路径限制在白名单根目录内，拒绝任何穿越
	cleanPath := filepath.Clean("/" + filePath) // 去除 ../ 等相对成分并锚定到根
	fullPath := filepath.Join(weightsRootDir, cleanPath)
	if !isWithinWeightsRoot(fullPath) {
		http.Error(w, "forbidden path", http.StatusForbidden)
		return
	}

	f, err := os.Open(fullPath)
	if err != nil {
		http.Error(w, "file not found", http.StatusNotFound)
		return
	}
	defer f.Close()

	w.Header().Set("Content-Type", "application/octet-stream")
	http.ServeContent(w, r, filepath.Base(fullPath), f.ModTime(), f)
}

// isWithinWeightsRoot：校验解析后的绝对路径严格位于白名单根目录之下
func isWithinWeightsRoot(p string) bool {
	absRoot, err := filepath.Abs(weightsRootDir)
	if err != nil {
		return false
	}
	absPath, err := filepath.Abs(p)
	if err != nil {
		return false
	}
	rel, err := filepath.Rel(absRoot, absPath)
	if err != nil || rel == ".." || filepath.IsAbs(rel) {
		return false
	}
	return true
}

分析：

1. 第 26 行：用户可控输入点为 `r.URL.Query().Get("path")`，即 `filePath` 直接来自查询参数，是本漏洞数据流的唯一 source。
2. 第 33–36 行：保留原有 `.h5`/`.keras` 后缀白名单——它只约束文件类型，不约束目录范围，攻击者可用 `/etc/passwd.h5` 之类绕过，因此必须补充目录级防御。
3. 第 39–44 行：防御核心。先用 `filepath.Clean("/" + filePath)` 把输入锚定为根下相对路径，剥离所有 `../` 穿越；再通过 `filepath.Join(weightsRootDir, cleanPath)` 把路径强制限定在白名单根目录 `/var/lib/model-weights` 内。这是白名单精确允许集式修复（框架原生惯用做法），而非黑名单或正则过滤。
4. 第 45–48 行：新增 `isWithinWeightsRoot` 校验函数（第 66–79 行），用 `filepath.Rel` 做规范化后的前缀关系判定：若相对结果为 `..`、以 `..` 开头或是绝对路径则拒绝。该检查位于 sink（第 50 行 `os.Open`）之前，位置正确；且基于 Clean 后的路径做 Rel 比较，可抵御符号化穿越与编码变体，类型正确、攻击面覆盖完整。
5. 第 50 行起：sink 改为打开经校验的 `fullPath`，后续 `http.ServeContent` 数据流走向不变；方法检查、错误响应等控制流均保持原样，构成 minimal pair。
6. 第二入口检查：路由仅注册 `/api/weights` 一个 handler，无其他读取文件的入口；`http.ServeContent` 的 name 参数取自已校验的 `fullPath` 的 Base，不构成替代通道。结论：无残留路径穿越风险。
