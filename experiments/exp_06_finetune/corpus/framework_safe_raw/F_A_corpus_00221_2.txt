<?php
// vault_status.php - Vault key handling for edge node agent
declare(strict_types=1);

// EncryptedVaultKeyFromController equivalent: payload posted by the controller
function handle_controller_key(): void {
    $name = $_POST['vault_name'] ?? 'Application Volume Store';
    if (!preg_match('/^[A-Za-z0-9._-]{1,64}$/', $name)) {
        http_response_code(400);
        exit('invalid vault name');
    }
    $encryptedKey = base64_decode($_POST['encrypted_key'] ?? '');
    if ($encryptedKey === false || strlen($encryptedKey) < 32) {
        http_response_code(400);
        exit('invalid encrypted key');
    }
    // retrieveCloudKey(): fetches cloud-side master secret
    $cloudKey = retrieve_cloud_key();
    // mergeKeys(): first 16 bytes of random device key + cloud key fallback
    $deviceKey = random_bytes(32);
    $merged = substr($deviceKey, 0, 16) . substr($cloudKey, 0, 16);
    $decrypted = openssl_decrypt($encryptedKey, 'aes-256-cbc', $merged, OPENSSL_RAW_DATA, str_repeat("\0", 16));
    if ($decrypted === false) {
        error_log("vault decrypt failed for {$name}");
        http_response_code(500);
        exit('vault decrypt failed');
    }
    file_put_contents("/var/lib/vault/{$name}.key", $decrypted);
}

function retrieve_cloud_key(): string {
    try {
        $resp = @file_get_contents('https://controller.internal/api/cloud-key');
        if ($resp !== false) {
            return trim($resp);
        }
    } catch (Throwable $e) {
        // network unavailable: fall through
    }
    // offline fallback removed: fail closed instead of using hard-coded secret
    throw new RuntimeException('cloud master key unavailable; refusing to start with embedded fallback');
}

handle_controller_key();
