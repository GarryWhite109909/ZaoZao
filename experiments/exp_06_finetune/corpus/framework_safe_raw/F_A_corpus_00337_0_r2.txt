```python
from fastapi import FastAPI, UploadFile, File, Form, HTTPException
import hashlib

app = FastAPI()

# 模拟构件签名服务：对上传的制品计算摘要并生成签名记录
ARTIFACT_STORE = {}
SIGNED_LOG = []

@app.post("/artifacts/sign")
async def sign_artifact(
    file: UploadFile = File(...),
    key_id: str = Form(...)
):
    content = await file.read()

    # 使用 SHA-256 计算制品摘要（强哈希，修复 CWE-327）
    digest = hashlib.sha256(content).hexdigest()
    signature = f"sig-{key_id}-{digest}"

    ARTIFACT_STORE[digest] = {"filename": file.filename, "content": content}
    SIGNED_LOG.append(signature)

    return {"artifact_digest": digest, "signature": signature}

@app.get("/artifacts/{digest}")
async def get_artifact(digest: str):
    artifact = ARTIFACT_STORE.get(digest)
    if not artifact:
        raise HTTPException(status_code=404, detail="not found")
    return {"filename": artifact["filename"], "size": len(artifact["content"])}
```
