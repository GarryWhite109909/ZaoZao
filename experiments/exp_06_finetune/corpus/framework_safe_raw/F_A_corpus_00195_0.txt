import json
import re
import subprocess

from django.http import JsonResponse
from django.views.decorators.http import require_POST


FFMPEG_PREFIXES = ('ffmpeg', '/usr/bin/ffmpeg', '/bin/ffmpeg')


def sanitize_ffmpeg_command(command):
    # whitelist-style validation: only allow ffmpeg with a fixed set of
    # benign flags and a media path, no shell metacharacters accepted
    parts = command.split()
    if len(parts) < 2:
        return ''
    if parts[0] not in FFMPEG_PREFIXES:
        return ''
    allowed_flags = {'-i', '-c', '-vcodec', '-acodec', '-f', '-y', '-b:v', '-b:a'}
    for token in parts[1:]:
        if token.startswith('-'):
            if token not in allowed_flags:
                return ''
        else:
            if any(ch in token for ch in ';|`<>&$(){}\\'"
\r'):
                return ''
    return command


@require_POST
def ffmpeg_control(request):
    action = request.POST.get('action', '')

    if action == 'list':
        keyword = request.POST.get('keyword', '')
        cmd = ['ps', '-eo', 'pid,etime,%cpu,%mem,cmd']
        proc = subprocess.run(cmd, capture_output=True, text=True)
        lines = [l for l in proc.stdout.splitlines() if 'ffmpeg' in l]
        if keyword:
            lines = [l for l in lines if keyword in l]
        return JsonResponse({'output': lines})

    if action == 'kill':
        pid = request.POST.get('pid', '')
        if not pid.isdigit():
            return JsonResponse({'error': True, 'msg': 'Invalid PID'}, status=400)
        kill_cmd = ['kill', '-9', pid]
        subprocess.run(kill_cmd)
        return JsonResponse({'error': False, 'msg': f'Process {pid} killed'})

    if action == 'start':
        raw_command = request.POST.get('command', '')
        safe_command = sanitize_ffmpeg_command(raw_command)
        if not safe_command:
            return JsonResponse({'error': True, 'msg': 'Sanitization failed'}, status=400)
        subprocess.Popen(safe_command.split(), shell=False,
                         stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return JsonResponse({'error': False, 'msg': 'started'})

    return JsonResponse({'error': True, 'msg': 'Unknown action'}, status=400)
