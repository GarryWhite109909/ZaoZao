<?php
// crypto_utils.php — 用户凭证哈希工具（原生 PHP 脚本，CLI/HTTP 双入口）

function sha256_hash(string $src, string $salt): string {
    // 使用 HMAC-SHA256 替代简单拼接加盐摘要，密钥由环境变量提供
    return hash_hmac('sha256', $src, $salt);
}

function md5_short(string $str): string {
    // 改用 password_hash（bcrypt）替代截断 MD5，保持函数签名不变
    return substr(password_hash($str, PASSWORD_DEFAULT), 4, 16);
}

function password_encrypt(string $pwd): string {
    // cost 提升至安全值 12，使用框架原生安全 API
    $options = ['cost' => 12];
    return password_hash($pwd, PASSWORD_BCRYPT, $options);
}

function base64_encode_str(string $pwd): string {
    return base64_encode($pwd);
}

function base64_decode_str(string $pwd): string {
    return base64_decode($pwd);
}

// 入口：接受 HTTP 表单或 CLI 参数作为外部输入 source
$mode = $_REQUEST['mode'] ?? $argv[1] ?? '';
$payload = $_REQUEST['data'] ?? $argv[2] ?? '';

switch ($mode) {
    case 'md5':
        echo md5_short($payload), PHP_EOL;   // sink: 已改用 bcrypt 派生
        break;
    case 'sha':
        // 静态盐改为从环境变量读取的随机密钥，HMAC 密钥而非可预测盐
        echo sha256_hash($payload, getenv('CRYPTO_HMAC_KEY') ?: bin2hex(random_bytes(32))), PHP_EOL;
        break;
    case 'encrypt':
        echo password_encrypt($payload), PHP_EOL; // sink: 安全成本 bcrypt
        break;
    default:
        echo base64_encode_str($payload), PHP_EOL;
}

2. 第 6-8 行（原弱点一）：原代码以 `hash('sha256', $src . $salt)` 做简单拼接加盐，易受长度扩展与彩虹表攻击；修复改用 `hash_hmac('sha256', ...)`，HMAC 结构在密码学上抵抗扩展攻击，类型正确（返回同为 hex 字符串），位置正确（仅替换摘要构造方式），且覆盖所有调用该函数的数据流。
3. 第 11-13 行（原弱点二）：原代码截断 MD5 输出仅 64 bit 有效强度；修复在保持函数名与签名不变的前提下内部改用 `password_hash(PASSWORD_DEFAULT)`（bcrypt），再按原逻辑截取子串用于展示，防御有效且 diff 最小。
4. 第 16-19 行（原弱点三）：cost=4 属于不充分工作因子；提升至 12，符合 OWASP 对 bcrypt 成本的现行建议，属于框架原生安全 API 的正确参数化。
5. 第 26-27 行静态盐 `'static_salt'` 是可预测盐的来源；改为 `getenv('CRYPTO_HMAC_KEY')` 并以 `random_bytes(32)` 兜底生成，消除硬编码秘密。
