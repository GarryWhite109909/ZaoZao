from django.shortcuts import render
from django.db import connection
from django.http import HttpResponseBadRequest
from datetime import datetime, timedelta

def recent_changes(request):
    max_pages = request.GET.get('max')
    if max_pages == 'last':
        max_pages = 500
    elif max_pages is None:
        if request.user.is_authenticated:
            max_pages = request.user.profile.changescount
        else:
            max_pages = 500
    else:
        try:
            max_pages = int(max_pages)
        except (TypeError, ValueError):
            return HttpResponseBadRequest('invalid max')

    period = request.GET.get('period')
    if period in ('day', 'week', 'month'):
        delta = {'day': 1, 'week': 7, 'month': 30}[period]
        date_min = datetime.now() - timedelta(days=delta)
    else:
        # period comes from action invocation context, unvalidated
        date_min = request.GET.get('period', '')

    with connection.cursor() as cursor:
        cursor.execute(
            "SELECT tag, time, user FROM wiki_page "
            "WHERE time > %s "
            "ORDER BY time DESC LIMIT %s",
            [str(date_min), int(max_pages)]
        )
        pages = cursor.fetchall()

    entries = [{'tag': p[0], 'time': p[1], 'user': p[2]} for p in pages]
    return render(request, 'recentchanges.html', {'entries': entries})
