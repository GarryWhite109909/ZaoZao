from flask import Flask, request, render_template_string, Markup, escape

app = Flask(__name__)

# 模拟富文本片段存储：每个 piece 含 string 内容与 attributes 属性
PIECES = {}

ALLOWED_TAGS = {"b", "i", "em", "strong", "u", "br", "p", "span"}
ALLOWED_ATTRS = {"class", "data-attrs"}

def sanitize_rich_text(html):
    # 防御：使用框架原生 Markup + 白名单精确允许集清洗富文本片段，
    # 仅保留 ALLOWED_TAGS/ALLOWED_ATTRS，其余一律转义为纯文本
    from html.parser import HTMLParser

    class WhitelistParser(HTMLParser):
        def __init__(self):
            super().__init__(convert_charrefs=True)
            self.out = []

        def handle_starttag(self, tag, attrs):
            if tag in ALLOWED_TAGS:
                safe_attrs = " ".join(
                    '%s="%s"' % (k, escape(v))
                    for k, v in attrs if k in ALLOWED_ATTRS
                )
                self.out.append("<%s%s>" % (tag, (" " + safe_attrs) if safe_attrs else ""))

        def handle_endtag(self, tag):
            if tag in ALLOWED_TAGS:
                self.out.append("</%s>" % tag)

        def handle_data(self, data):
            self.out.append(escape(data))

    parser = WhitelistParser()
    parser.feed(html)
    return "".join(parser.out)

@app.route("/piece/save", methods=["POST"])
def save_piece():
    piece_id = request.form.get("id", "p1")
    content = request.form.get("string", "")
    attributes = request.form.get("attributes", "")
    PIECES[piece_id] = {"string": content, "attributes": attributes}
    return {"saved": True, "length": len(content)}

@app.route("/piece/render/<piece_id>")
def render_piece(piece_id):
    piece = PIECES.get(piece_id)
    if not piece:
        return {"error": "not found"}, 404

    # 拼接属性后渲染，模拟 consolidateWith 的合并行为；
    # sink 前先经白名单清洗，属性值经 escape 转义
    clean_attrs = escape(piece["attributes"])
    clean_string = sanitize_rich_text(piece["string"])
    html = "<span data-attrs='%s'>%s</span>" % (clean_attrs, clean_string)

    # sink: render_template_string 渲染已清洗的 HTML（白名单允许集内的标签安全）
    return render_template_string(Markup(html))

if __name__ == "__main__":
    app.run(debug=True)
