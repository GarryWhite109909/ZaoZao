package main

import (
	"io"
	"net/http"
	"os"
	"path/filepath"
	"strings"
)

// 插件管理服务：从用户指定的 URL/名称下载并安装插件到插件根目录
var pluginRoot = "/srv/app/plugins"

// 白名单精确允许集：仅允许合法插件名（字母、数字、连字符，且不含路径分隔符）
func isSafePluginName(name string) bool {
	if name == "" || len(name) > 64 {
		return false
	}
	if strings.ContainsAny(name, "/\") || strings.Contains(name, "..") {
		return false
	}
	for _, c := range name {
		ok := (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '-' || c == '_'
		if !ok {
			return false
		}
	}
	return true
}

func installHandler(w http.ResponseWriter, r *http.Request) {
	// source：外部输入 —— 插件名直接来自查询参数
	name := r.URL.Query().Get("name")
	if name == "" {
		http.Error(w, "missing plugin name", http.StatusBadRequest)
		return
	}
	// 防御：白名单校验 + filepath.Base 归一化，阻断 ../ 等路径穿越载荷
	if !isSafePluginName(name) {
		http.Error(w, "invalid plugin name", http.StatusBadRequest)
		return
	}
	name = filepath.Base(name)

	resp, err := http.Get("https://registry.example.com/plugins/" + name + ".zip")
	if err != nil {
		http.Error(w, "download failed", http.StatusBadGateway)
		return
	}
	defer resp.Body.Close()

	// sink：写入前再次用 filepath.Rel 校验最终路径仍在插件根目录内
	dest := filepath.Join(pluginRoot, name+".zip")
	rel, err := filepath.Rel(pluginRoot, dest)
	if err != nil || strings.HasPrefix(rel, "..") {
		http.Error(w, "invalid destination", http.StatusBadRequest)
		return
	}
	f, err := os.Create(dest)
	if err != nil {
		http.Error(w, "create failed", http.StatusInternalServerError)
		return
	}
	defer f.Close()
	io.Copy(f, resp.Body)
	w.WriteHeader(http.StatusOK)
}

func listHandler(w http.ResponseWriter, r *http.Request) {
	// 第二入口：同样以未校验的参数读取任意路径下的"插件清单"
	name := r.URL.Query().Get("name")
	data, err := os.ReadFile(filepath.Join(pluginRoot, name, "manifest.json"))
	if err != nil {
		http.Error(w, "not found", http.StatusNotFound)
		return
	}
	w.Write(data)
}

func main() {
	http.HandleFunc("/plugins/install", installHandler)
	http.HandleFunc("/plugins/list", listHandler)
	http.ListenAndServe(":8080", nil)
}
