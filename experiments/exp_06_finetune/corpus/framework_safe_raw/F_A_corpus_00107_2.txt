package com.example.cache;

import org.springframework.web.bind.annotation.*;
import org.springframework.data.redis.core.RedisTemplate;
import java.io.*;

@RestController
@RequestMapping("/api/cache")
public class CacheEntryController {

    private static final long MAX_PAYLOAD_SIZE = 64 * 1024;
    // whitelist: only application-owned DTO types may ever be deserialized
    private static final Set<Class<?>> ALLOWED_TYPES = Set.of(CacheEntryPayload.class);

    private final RedisTemplate<String, byte[]> redisTemplate;

    public CacheEntryController(RedisTemplate<String, byte[]> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    @PostMapping("/{entryId}")
    public void set(@PathVariable String entryId,
                    @RequestParam(required = false) Integer lifetime,
                    @RequestBody byte[] data) throws IOException {
        if (entryId == null || entryId.isBlank()) {
            throw new IllegalArgumentException("invalid cache entry identifier");
        }
        if (data == null || data.length == 0 || data.length > MAX_PAYLOAD_SIZE) {
            throw new IllegalArgumentException("invalid cache payload");
        }
        // persist raw bytes; integrity enforced at read time via whitelist filter
        redisTemplate.opsForValue().set(entryId, data);
    }

    @GetMapping("/{entryId}")
    public Object get(@PathVariable String entryId) throws IOException, ClassNotFoundException {
        if (entryId == null || entryId.isBlank()) {
            throw new IllegalArgumentException("invalid cache entry identifier");
        }
        byte[] raw = redisTemplate.opsForValue().get(entryId);
        if (raw == null) {
            return null;
        }
        try (ObjectInputStream ois = new WhitelistedObjectInputStream(
                new ByteArrayInputStream(raw), ALLOWED_TYPES)) {
            return ois.readObject(); // only whitelisted classes can be resolved
        }
    }

    // framework-idiomatic defensive wrapper: resolveObject whitelist gate
    private static final class WhitelistedObjectInputStream extends ObjectInputStream {
        private final Set<Class<?>> allowedTypes;

        WhitelistedObjectInputStream(InputStream in, Set<Class<?>> allowedTypes) throws IOException {
            super(in);
            this.allowedTypes = allowedTypes;
        }

        @Override
        protected Class<?> resolveClass(ObjectStreamClass desc) throws IOException, ClassNotFoundException {
            Class<?> clazz = super.resolveClass(desc);
            if (!allowedTypes.contains(clazz)) {
                throw new InvalidClassException("class not allowed", desc.getName());
            }
            return clazz;
        }
    }
}
