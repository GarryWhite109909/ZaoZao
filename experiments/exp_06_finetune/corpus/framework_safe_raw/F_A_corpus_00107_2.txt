package com.example.cache;

import org.springframework.web.bind.annotation.*;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.data.redis.serializer.StringRedisSerializer;
import org.springframework.data.redis.serializer.Jackson2JsonRedisSerializer;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.*;

@RestController
@RequestMapping("/api/cache")
public class CacheEntryController {

    private final RedisTemplate<String, byte[]> redisTemplate;
    // whitelist of allowed cacheable types: only safe DTOs may be stored/restored
    private static final java.util.Set<String> ALLOWED_TYPES =
            java.util.Set.of("com.example.cache.CacheEntryDTO");

    public CacheEntryController(RedisTemplate<String, byte[]> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    @PostMapping("/{entryId}")
    public void set(@PathVariable String entryId,
                    @RequestParam(required = false) Integer lifetime,
                    @RequestBody String json) throws IOException {
        if (entryId == null || entryId.isBlank()) {
            throw new IllegalArgumentException("invalid cache entry identifier");
        }
        // validate payload is a whitelisted type before persisting
        ObjectMapper mapper = new ObjectMapper();
        mapper.activateDefaultTyping(
                new com.fasterxml.jackson.databind.jsontype.impl.LaissezFaireSubTypeValidator() {
                    @Override
                    public boolean validateSubType(Class<?> subType, com.fasterxml.jackson.databind.JavaType baseType) {
                        return ALLOWED_TYPES.contains(subType.getName());
                    }
                },
                ObjectMapper.DefaultTyping.NON_FINAL);
        Object value = mapper.readValue(json, Object.class);
        redisTemplate.opsForValue().set(entryId, mapper.writeValueAsBytes(value));
    }

    @GetMapping("/{entryId}")
    public Object get(@PathVariable String entryId) throws IOException, ClassNotFoundException {
        if (entryId == null || entryId.isBlank()) {
            throw new IllegalArgumentException("invalid cache entry identifier");
        }
        byte[] raw = redisTemplate.opsForValue().get(entryData(entryId));
        if (raw == null) {
            return null;
        }
        // replace native Java deserialization with JSON + strict subtype whitelist
        ObjectMapper mapper = new ObjectMapper();
        mapper.activateDefaultTyping(
                new com.fasterxml.jackson.databind.jsontype.impl.LaissezFaireSubTypeValidator() {
                    @Override
                    public boolean validateSubType(Class<?> subType, com.fasterxml.jackson.databind.JavaType baseType) {
                        return ALLOWED_TYPES.contains(subType.getName());
                    }
                },
                ObjectMapper.DefaultTyping.NON_FINAL);
        return mapper.readValue(raw, Object.class);
    }

    private String entryData(String entryId) {
        return entryId; // key unchanged, kept for clarity
    }
}
