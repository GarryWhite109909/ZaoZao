<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Routing\Controller;

class PageRenderController extends Controller
{
    // Raw-content elements whose inner text is emitted unescaped,
    // mirroring domino's hasRawContent set.
    private array $rawContentTags = ['style', 'script', 'iframe', 'noembed', 'noscript', 'noframes', 'xmp', 'plaintext'];

    private function escapeHtml(string $s): string
    {
        return htmlspecialchars($s, ENT_QUOTES | ENT_HTML5, 'UTF-8');
    }

    // Only escapes the attribute value; mirrors escapeAttr.
    private function escapeAttr(string $s): string
    {
        return htmlspecialchars($s, ENT_QUOTES, 'UTF-8');
    }

    public function render(Request $request)
    {
        $tag = strtolower($request->input('tag', 'div'));
        $attrs = (array) $request->input('attrs', []);
        // External input: user-controlled inner content of the element.
        $content = (string) $request->input('content', '');

        if (!preg_match('/^[a-z][a-z0-9]*$/', $tag)) {
            abort(400);
        }

        // Whitelist guard: raw-content elements cannot carry user-supplied
        // inner text safely, because their content model forbids escaping.
        // Reject them outright instead of filtering closers.
        if (in_array($tag, $this->rawContentTags, true)) {
            abort(400);
        }

        $html = '<' . $tag;
        foreach ($attrs as $name => $value) {
            $html .= ' ' . htmlspecialchars((string) $name) . '="' . $this->escapeAttr((string) $value) . '"';
        }
        $html .= '>';

        if (!in_array($tag, $this->rawContentTags, true)) {
            $html .= $this->escapeHtml($content);
        } else {
            // Unreachable after the whitelist guard above; kept to preserve
            // original control flow shape.
            $closing = '</' . $tag . '>';
            $html .= str_replace($closing, '', $content);
        }

        $html .= '</' . $tag . '>';

        return response($html, 200)->header('Content-Type', 'text/html; charset=UTF-8');
    }
}
