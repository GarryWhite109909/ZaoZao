```python
from django.contrib.auth.decorators import login_required, user_passes_test
from django.http import JsonResponse, HttpResponseBadRequest
from django.views.decorators.http import require_POST
import json

from myapp.models import GlobalSet


def _is_staff(user):
    return user.is_staff


@login_required
@user_passes_test(_is_staff)
@require_POST
def reorder_sets(request):
    """Reorders global sets. Fixed: now restricted to staff/admin."""
    try:
        payload = json.loads(request.body)
        set_ids = payload["ids"]
    except (json.JSONDecodeError, KeyError):
        return HttpResponseBadRequest("missing ids")

    for position, set_id in enumerate(set_ids):
        GlobalSet.objects.filter(pk=set_id).update(sort_order=position)

    return JsonResponse({"success": True})


@login_required
def delete_set(request, set_id):
    """Delete action that DOES enforce the admin check."""
    if not request.user.is_staff:
        return JsonResponse({"error": "forbidden"}, status=403)

    GlobalSet.objects.filter(pk=set_id).delete()
    return JsonResponse({"success": True})
```
