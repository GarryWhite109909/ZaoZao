```code
package com.example.fetch;

import org.springframework.web.bind.annotation.*;
import org.springframework.http.*;
import java.net.*;
import java.util.regex.*;

@RestController
@RequestMapping("/api/fetch")
public class UrlFetchController {

    // 检查 IP 是否为内网地址（与原代码相同的私有段判断）
    private boolean isPrivateIp(String addr) {
        if (addr == null || addr.isEmpty()) return true;
        try {
            InetAddress ia = InetAddress.getByName(addr);
            if (ia.isLoopbackAddress() || ia.isSiteLocalAddress()
                || ia.isLinkLocalAddress() || ia.isAnyLocalAddress()) return true;
        } catch (UnknownHostException e) { return true; }
        return false;
    }

    @GetMapping
    public ResponseEntity<String> fetch(@RequestParam String url) throws Exception {
        // 仅做格式校验，未做 scheme/host 白名单
        if (!Pattern.matches("^https?://[\\w.:-]+.*", url)) {
            return ResponseEntity.badRequest().body("invalid url");
        }
        URL target = new URL(url);
        // 防御：只解析一次 DNS，后续连接复用同一 InetAddress，消除 TOCTOU
        InetAddress resolved = InetAddress.getByName(target.getHost());
        if (isPrivateIp(resolved.getHostAddress())) {
            return ResponseEntity.status(403).body("private ip blocked");
        }
        // sink：使用已校验的 IP 构造连接 URL，并保留原 Host 头，防止 rebinding 绕过
        URL pinnedUrl = new URL(target.getProtocol(), resolved.getHostAddress(), target.getPort(), target.getFile());
        HttpURLConnection conn = (HttpURLConnection) pinnedUrl.openConnection();
        conn.setRequestProperty("Host", target.getHost());
        conn.setConnectTimeout(5000);
        try (var in = conn.getInputStream()) {
            return new ResponseEntity<>(new String(in.readAllBytes()), HttpStatus.OK);
        }
    }
}
```

1. 第 22 行 `@RequestParam String url` 是唯一用户可控输入点（框架参数绑定），数据流经第 25 行正则校验、第 28 行 `new URL(url)` 进入请求逻辑。
2. 原漏洞成因：第 31 行 `isPrivateIp(target.getHost())` 内部通过域名解析一次 DNS，而第 36 行 `target.openConnection()` 连接时再次解析 DNS；两次解析结果可不同（DNS rebinding），攻击者首次返回公网 IP 通过检查、第二次返回 127.0.0.1/内网 IP 即绕过。
3. 修复位置正确性：在第 30-31 行将 DNS 解析提前为一次性操作 `InetAddress.getByName(target.getHost())`，并把解析结果（而非域名）传入 `isPrivateIp` 校验，保证被检查的地址就是将要连接的地址——类型上为 `InetAddress` 精确 IP，非可重解析的域名字符串。
4. 攻击面覆盖完整性：第 35 行用已校验的 `resolved.getHostAddress()` 构造 `pinnedUrl` 并据此 `openConnection()`，连接阶段不再触发新的 DNS 解析；第 36 行显式设置 `Host` 头为原始域名，保证基于虚拟主机的目标服务仍能正确路由，功能语义不变。TOCTOU 窗口被完全关闭。
5. 第 25 行原有正则格式校验保持不变，仅作为输入合法性前置过滤，不承担安全职责；业务场景（URL 抓取代理）、路由 `/api/fetch`、类名与方法名均未改动，符合最小 diff 纪律。
6. 第二入口/替代通道检查：本控制器仅此一个 GET 映射，无其他接收 url 参数的端点；`isPrivateIp` 为私有辅助方法无外部入口；结论为不存在其他可达同一 sink 的替代通道。
