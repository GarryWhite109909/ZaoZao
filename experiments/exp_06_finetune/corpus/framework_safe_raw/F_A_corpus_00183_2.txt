package com.example.scanner.controller;

import org.springframework.web.bind.annotation.*;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.File;
import java.util.ArrayList;
import java.util.List;

@RestController
@RequestMapping("/api/projects")
public class ProjectScanController {

    // Scan workspace for serverless config files by shelling out to find
    @GetMapping("/scan")
    public List<String> scanWorkspace(@RequestParam("workspaceDir") String workspaceDir) {
        String rootDir = (workspaceDir == null || workspaceDir.isEmpty())
                ? System.getProperty("user.dir") : workspaceDir;

        List<String> results = new ArrayList<>();
        try {
            // Fixed: no shell interpolation; use ProcessBuilder argument list directly,
            // so user input is passed as a single argv element, never parsed by /bin/sh
            List<String> command = new ArrayList<>();
            command.add("/usr/bin/find");
            command.add(rootDir);
            command.add("-name");
            command.add("serverless.yml");
            command.add("-not");
            command.add("-path");
            command.add("*/node_modules/*");
            ProcessBuilder pb = new ProcessBuilder(command);
            Process proc = pb.start();
            BufferedReader reader = new BufferedReader(new InputStreamReader(proc.getInputStream()));
            String line;
            while ((line = reader.readLine()) != null) {
                results.add(line);
            }
        } catch (Exception e) {
            // swallow errors, return partial results like the original
        }
        return results;
    }
}

2. 第 24-31 行：原漏洞 sink 是 `new ProcessBuilder("/bin/sh", "-c", "find '" + rootDir + "' ...")`，字符串拼接使 `rootDir` 进入 shell 解释器，攻击者可用 `' ; rm -rf / #` 等注入任意命令（CWE-77）。
3. 修复位置正确性：在第 24 行起将命令改为参数列表形式，直接调用 `/usr/bin/find` 并把 `rootDir` 作为独立 argv 元素传入；由于不再经过 `/bin/sh -c`，shell 元字符（分号、反引号、管道、引号）不会被解释，类型上每个元素都是字面量字符串参数。
4. 攻击面覆盖完整性：该修复属于框架/标准库原生安全 API（ProcessBuilder 参数化调用），消除了唯一的 shell 拼接点；业务语义不变——仍按名称查找 serverless.yml 且排除 node_modules 路径，数据流走向与返回结构完全一致。

6. 第二入口检查：本控制器无其他端点，无其他用户可控数据流入进程执行 API；`System.getProperty("user.dir")` 为 JVM 属性非用户输入。结论：无残留命令注入通道。
