```python
import os
import subprocess
from flask import Flask, request, jsonify

app = Flask(__name__)

CONFIG_PATH = "/etc/sqls/config.yaml"

# 防御：编辑器白名单精确允许集（CWE-77 修复）
ALLOWED_EDITORS = {"vim", "vi", "nano", "emacs"}

def _resolve_editor(editor):
    if editor not in ALLOWED_EDITORS:
        return None
    return editor

@app.route("/config/edit", methods=["POST"])
def edit_config():
    # source: editor 来自客户端可控的请求参数
    editor = request.form.get("editor") or os.environ.get("EDITOR", "vim")
    safe_editor = _resolve_editor(editor)
    if safe_editor is None:
        return jsonify({"status": "rejected"}), 400
    cmd = f"{safe_editor} {CONFIG_PATH}"
    if os.name == "nt":
        subprocess.run(["cmd", "/c", cmd], shell=False)
    else:
        # sink: 白名单校验后的固定词元，无 shell 元字符注入面
        subprocess.run(cmd, shell=True)
    return jsonify({"status": "opened"})

if __name__ == "__main__":
    app.run(port=5000)
```
