```code
// server/middleware/auth-chain.ts
import { defineEventHandler, getRequestURL, getQuery, createError } from 'h3'
import { execFile } from 'child_process'
import { promisify } from 'util'

const execFileAsync = promisify(execFile)

// 鉴权中间件链：根据请求中的 workspace 参数定位用户工作区并校验 token
export default defineEventHandler(async (event) => {
  const url = getRequestURL(event)
  if (!url.pathname.startsWith('/api/workspace')) return

  const query = getQuery(event)
  const workspaceDir = String(query.workspace || process.cwd())
  const token = String(query.token || '')

  // 第一步：token 校验（与命令执行无关，不构成对 sink 的防御）
  if (!token.startsWith('sk_')) {
    throw createError({ statusCode: 401, statusMessage: 'invalid token' })
  }

  // 第二步：定位该工作区下的部署配置文件
  // 防御：不再拼接 shell 字符串，改用 execFile 参数数组形式，
  // workspaceDir 作为独立 argv 元素传递，不经过 shell 解释
  try {
    const { stdout } = await execFileAsync(
      'find',
      [workspaceDir, '-name', 'serverless.yml', '-not', '-path', '*/node_modules/*'],
      { maxBuffer: 10 * 1024 * 1024 },
    )

    const configFiles = stdout.trim().split('
').filter(Boolean)

    // 第三步：读取每个配置文件提取 service 名（同样使用参数数组，无 shell 拼接）
    for (const file of configFiles) {
      await execFileAsync('head', ['-n', '20', file])
    }

    event.context.configFiles = configFiles
  } catch (error) {
    console.error('Error scanning workspace:', error)
    event.context.configFiles = []
  }
})
```

3. 修复位置正确性：sink 处直接替换调用方式——第 6 行将 `promisify(exec)` 改为 `promisify(execFile)`，第 32-35 行把 shell 命令字符串改为 `execFileAsync('find', [argv...])`。`execFile` 不经过 `/bin/sh`，每个参数作为独立 argv 元素传递，类型上 `workspaceDir` 无论含何种 shell 元字符都只是字面路径参数，无法改变命令结构。
4. 攻击面覆盖完整性：第二处 sink 在第 41 行 `head -n 20 "${file}"`，其输入 `file` 来自 find 输出而非用户直接输入，但同样存在拼接风险，一并改为 `execFileAsync('head', ['-n', '20', file])`，两条数据流均消除 shell 解释。
5. 最小 diff 纪律：函数名、路由前缀 `/api/workspace`、变量名、控制流、错误处理与 `event.context.configFiles` 赋值全部保持不变，仅替换命令执行 API 及其调用形态。
