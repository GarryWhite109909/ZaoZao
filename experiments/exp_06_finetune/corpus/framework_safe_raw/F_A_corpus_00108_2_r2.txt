package main

import (
	"bytes"
	"encoding/base64"
	"encoding/json"
	"io/ioutil"
	"net/http"

	"github.com/gin-gonic/gin"
)

// 模拟一个可被反序列化的执行器类型
type TaskRunner struct {
	Cmd string
}

func (t *TaskRunner) Run() { /* exec t.Cmd */ }

// 白名单结构体：仅允许反序列化已知的安全字段，禁止任意类型注入
type SafeVariables struct {
	TaskName string `json:"task_name"`
	Payload  string `json:"payload"`
}

var allowedTasks = map[string]bool{
	"report": true,
	"backup": true,
}

func base64Decode(s string) ([]byte, error) {
	return base64.StdEncoding.DecodeString(s)
}

func graphqlHandler(c *gin.Context) {
	body, _ := ioutil.ReadAll(c.Request.Body)
	var req map[string]interface{}
	json.Unmarshal(body, &req)

	// 从请求中取出 variables 字段（base64 编码的 JSON 数据）
	rawVars, ok := req["variables"].(string)
	if !ok {
		c.JSON(http.StatusBadRequest, gin.H{"error": "missing variables"})
		return
	}
	decoded, err := base64Decode(rawVars)
	if err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "bad encoding"})
		return
	}

	// 安全修复：不再使用 gob 反序列化不可信数据（CWE-502），
	// 改用 encoding/json 严格解码到白名单结构体，未知字段直接拒绝
	var vars SafeVariables
	dec := json.NewDecoder(bytes.NewReader(decoded))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&vars); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "invalid variables"})
		return
	}
	if !allowedTasks[vars.TaskName] {
		c.JSON(http.StatusBadRequest, gin.H{"error": "unknown task"})
		return
	}
	runner := &TaskRunner{Cmd: vars.TaskName} // 仅由白名单值构造，不再来自反序列化
	runner.Run()

	c.JSON(http.StatusOK, gin.H{"data": "ok"})
}

func main() {
	r := gin.Default()
	r.POST("/graphql", graphqlHandler)
	r.Run(":8080")
}
