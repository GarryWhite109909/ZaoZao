from fastapi import FastAPI, HTTPException, Depends, Header
from pydantic import BaseModel
import httpx
import os
import secrets

app = FastAPI()

# 修复 CWE-798：凭据不再硬编码，改为从环境变量/密钥管理服务读取，
# 并使用 secrets.compare_digest 做恒定时间比较（框架无关的密码学安全 API）
INTERNAL_ADMIN_USER = os.environ["INTERNAL_ADMIN_USER"]
INTERNAL_ADMIN_PASSWORD = os.environ["INTERNAL_ADMIN_PASSWORD"]

def require_internal_auth(x_internal_user: str = Header(None),
                          x_internal_pass: str = Header(None)):
    if not x_internal_user or not x_internal_pass:
        raise HTTPException(status_code=401, detail="unauthorized")
    # 白名单精确允许集：仅允许内部系统用户名，且用恒定时间比较防时序侧信道
    if not secrets.compare_digest(x_internal_user, INTERNAL_ADMIN_USER) or \
       not secrets.compare_digest(x_internal_pass, INTERNAL_ADMIN_PASSWORD):
        raise HTTPException(status_code=401, detail="unauthorized")
    return {"user": x_internal_user}

class ExportJob(BaseModel):
    dataset: str
    target_url: str

# 白名单精确允许集：仅允许向已注册的内部数据接收端导出
ALLOWED_EXPORT_TARGETS = {
    "https://data-warehouse.internal.example.com/api/ingest",
    "https://analytics.internal.example.com/api/ingest",
}

@app.post("/internal/export")
async def create_export(job: ExportJob, _=Depends(require_internal_auth)):
    # 强防御：对用户可控的 target_url 施加白名单精确允许集校验，阻止 SSRF 外带凭据/内网探测
    if job.target_url not in ALLOWED_EXPORT_TARGETS:
        raise HTTPException(status_code=400, detail="target not allowed")
    async with httpx.AsyncClient() as client:
        resp = await client.post(job.target_url, json={"dataset": job.dataset})
    return {"status": resp.status_code}
