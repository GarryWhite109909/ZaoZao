package com.example.social;

import org.springframework.web.bind.annotation.*;
import org.springframework.http.*;
import org.springframework.web.client.RestTemplate;
import org.springframework.cache.annotation.Cacheable;
import java.util.*;

@RestController
@RequestMapping("/api/social")
public class TwitterFeedController {

    private static final String API_URL = "https://api.twitter.com/1.1/search/tweets.json";

    @GetMapping("/tweets")
    @Cacheable(value = "twitterFeed", key = "#keyword + #count")
    public ResponseEntity<Map<String, Object>> getTweets(
            @RequestParam(required = false) String keyword,
            @RequestParam(defaultValue = "5") int count) {

        // All credentials must come from the environment; fail fast if missing (CWE-798 fix)
        String accessToken = Objects.requireNonNull(
                System.getenv("TWITTER_ACCESS_TOKEN"),
                "Missing required environment variable TWITTER_ACCESS_TOKEN");
        if (accessToken.isEmpty()) {
            throw new IllegalStateException("TWITTER_ACCESS_TOKEN must not be empty");
        }
        String consumerKey = Objects.requireNonNull(
                System.getenv("TWITTER_CONSUMER_KEY"),
                "Missing required environment variable TWITTER_CONSUMER_KEY");
        if (consumerKey.isEmpty()) {
            throw new IllegalStateException("TWITTER_CONSUMER_KEY must not be empty");
        }
        String tokenSecret = Objects.requireNonNull(
                System.getenv("TWITTER_TOKEN_SECRET"),
                "Missing required environment variable TWITTER_TOKEN_SECRET");
        String consumerSecret = Objects.requireNonNull(
                System.getenv("TWITTER_CONSUMER_SECRET"),
                "Missing required environment variable TWITTER_CONSUMER_SECRET");

        RestTemplate rest = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.set("Authorization", "Bearer " + accessToken);
        headers.set("X-Consumer-Key", consumerKey);

        String url = API_URL + "?count=" + count + "&q=" + keyword;
        Map<String, Object> result = rest.getForObject(url, Map.class);
        return ResponseEntity.ok(result);
    }
}
