import re
import subprocess
import shlex
from flask import Flask, request, jsonify

app = Flask(__name__)

WORKSPACE = "/srv/workspace"
TIMEOUT = 60

DENY_PATTERNS = [
    re.compile(r"\brm\s+-[rf]{1,2}\b"),
    re.compile(r"\b(format|mkfs|diskpart)\b\s"),
    re.compile(r"\bdd\s+if="),
    re.compile(r"\b(shutdown|reboot|poweroff)\b"),
]

# 白名单精确允许集：仅允许这些命令以固定参数模板执行
ALLOWED_COMMANDS = {
    "ls": ["ls", "-la"],
    "cat": ["cat"],
    "grep": ["grep"],
    "wc": ["wc", "-l"],
}

def guard_command(cmd, cwd):
    lower = cmd.strip().lower()
    for pattern in DENY_PATTERNS:
        if pattern.search(lower):
            return "Command blocked by safety guard"
    if cwd and ".." in cmd:
        return "Command blocked (path traversal detected)"
    return ""

def build_safe_argv(cmd):
    """白名单校验并构造无 shell 的 argv；非法则返回 None"""
    try:
        tokens = shlex.split(cmd)
    except ValueError:
        return None
    if not tokens:
        return None
    base = tokens[0]
    if base not in ALLOWED_COMMANDS:
        return None
    allowed = ALLOWED_COMMANDS[base]
    extra = tokens[1:]
    # 参数个数与内容限制：禁止额外选项/标志，仅允许普通位置参数
    for tok in extra:
        if tok.startswith("-"):
            return None
    if len(extra) > 4:
        return None
    return allowed + extra

@app.route("/api/exec", methods=["POST"])
def exec_command():
    data = request.get_json(silent=True) or {}
    command = data.get("command")
    if not command:
        return jsonify({"error": "command is required"}), 400

    cwd = data.get("working_dir") or WORKSPACE
    err = guard_command(command, cwd)
    if err:
        return jsonify({"error": err}), 403

    argv = build_safe_argv(command)
    if argv is None:
        return jsonify({"error": "Command not allowed by whitelist"}), 403

    try:
        result = subprocess.run(
            argv,
            cwd=cwd, capture_output=True, text=True, timeout=TIMEOUT,
            shell=False
        )
        output = result.stdout + result.stderr
        return jsonify({"output": output or "(no output)"})
    except subprocess.TimeoutExpired:
        return jsonify({"error": "Command timed out"}), 504

if __name__ == "__main__":
    app.run()
