#### 版本A（vuln，11 行）
```php
<?php
// prefs/import.php
foreach ($_POST as $key => $value) {
    // 动态变量命名空间：键名成为 PHP 变量名并被赋值执行
    $$key = $value;
}
if (!empty(${'is_admin'})) {
    grant_admin();
}
save_prefs($_POST);
echo "imported";
```
#### 版本B（safe，16 行）
```php
<?php
// prefs/import.php
$ALLOWED_PREFS = ['theme', 'pagesize', 'lang'];

$prefs = [];
foreach ($_POST as $key => $value) {
    if (!in_array($key, $ALLOWED_PREFS, true)) {
        continue;
    }
    $prefs[$key] = $value;
}
if (($_SESSION['is_admin'] ?? false) === true) {
    grant_admin();
}
save_prefs($prefs);
echo "imported";
```
