#### 版本A（vuln，28 行）
```go
package audit

import (
	"net/http"
	"sync"
	"time"
)

var mu sync.Mutex
var hits = map[string]int{}

func Middleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		ip := r.Header.Get("X-Forwarded-For")
		if ip == "" {
			ip = r.RemoteAddr
		}
		mu.Lock()
		hits[ip]++
		over := hits[ip] > 600
		mu.Unlock()
		if over {
			http.Error(w, "too many requests", http.StatusTooManyRequests)
			return
		}
		next.ServeHTTP(w, r)
	})
}
```

package audit

import (
	"net/http"
	"sync"
	"time"
)

var mu sync.Mutex
var hits = map[string]int{}

func Middleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		ip := r.RemoteAddr
		auditLog.Printf("xff=%q remote=%s", r.Header.Get("X-Forwarded-For"), ip)
		mu.Lock()
		hits[ip]++
		over := hits[ip] > 600
		mu.Unlock()
		if over {
			http.Error(w, "too many requests", http.StatusTooManyRequests)
			return
		}
		next.ServeHTTP(w, r)
	})
}
