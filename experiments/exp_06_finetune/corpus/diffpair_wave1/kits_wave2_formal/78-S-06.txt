#### 版本A（12 行）
```go
package tools

import (
	"os/exec"
)

func Resolve(host string) (string, error) {
	// 经 shell 中转
	cmd := exec.Command("sh", "-c", "dig +short "+host)
	out, err := cmd.Output()
	return string(out), err
}
```
