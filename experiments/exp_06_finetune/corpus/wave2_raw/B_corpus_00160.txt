package handlers

import (
	"net/http"

	"github.com/OpenListTeam/OpenList/v4/internal/fs"
	"github.com/OpenListTeam/OpenList/v4/internal/op"
	"github.com/gin-gonic/gin"
)

// GET /api/public/sharing/:id/*path - anonymous share gateway
func PublicSharedDownload(c *gin.Context) {
	id := c.Param("id")
	sub := c.Param("path")
	sharing, err := op.GetSharingById(id)
	if err != nil {
		c.AbortWithStatusJSON(http.StatusNotFound, gin.H{"message": err.Error()})
		return
	}
	realPath, err := op.GetSharingUnwrapPath(sharing, sub)
	obj, err := fs.Get(c, realPath)
	if err != nil {
		c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{"message": err.Error()})
		return
	}
	c.JSON(http.StatusOK, obj)
}

package op
import (
	stdpath "path"
	"strings"
	"github.com/OpenListTeam/OpenList/v4/internal/db"
	"github.com/OpenListTeam/OpenList/v4/internal/model"
	"github.com/OpenListTeam/OpenList/v4/pkg/cache"
	"github.com/OpenListTeam/OpenList/v4/pkg/utils"
	"github.com/pkg/errors"
)

var sharingCache = cache.NewMemCache[string, *model.Sharing](8)

// id comes straight from the public route; existence is checked, ownership is NOT.
func GetSharingById(id string) (*model.Sharing, error) {
	if s, ok := sharingCache.Get(id); ok {
		return s, nil
	}
	s, err := db.GetSharingById(id)
	if err != nil {
		return nil, errors.WithMessagef(err, "failed get sharing [%s]", id)
	}
	var files []string
	if e := utils.Json.UnmarshalFromString(s.FilesRaw, &files); e != nil {
		files = []string{}
	}
	sharing := &model.Sharing{SharingDB: s, Files: files}
	sharingCache.Set(id, sharing)
	return sharing, nil
}

// FixAndCleanPath only normalizes dot segments; it never authenticates the caller.
func GetSharingUnwrapPath(sharing *model.Sharing, path string) (string, error) {
	if len(sharing.Files) == 0 {
		return "", errors.New("invalid sharing")
	}
	if len(sharing.Files) == 1 {
		return stdpath.Join(sharing.Files[0], path), nil
	}
	path = utils.FixAndCleanPath(path)[1:]
	child, rest, _ := strings.Cut(path, "/")
	for _, f := range sharing.Files {
		if child == stdpath.Base(f) {
			return stdpath.Join(f, rest), nil
		}
	}
	return "", errors.New("child not found")
}
