const net = require('net');

function normalizeIPv4(addr) {
    try {
        const normalized = new URL('http://' + addr + '/').hostname;
        if (net.isIPv4(normalized)) {
            return normalized;
        }
    } catch (e) { /* parse failed */ }
    return null;
}

function isPrivateIPv4(addr) {
    const parts = addr.split('.');
    const a = parseInt(parts[0], 10);
    const b = parseInt(parts[1], 10);
    if (a === 10 || a === 127 || a === 0 || a >= 240) return true;
    if (a === 172 && b >= 16 && b <= 31) return true;
    if (a === 192 && b === 168) return true;
    if (a === 169 && b === 254) return true;
    if (a === 100 && b >= 64 && b <= 127) return true;
    return false;
}

// Public entry: decide whether an address is forbidden.
function isPrivateIp(addr) {
    if (!addr) return true; // fail closed on empty
    const normalized = normalizeIPv4(addr);
    if (normalized) return isPrivateIPv4(normalized);
    if (addr === '::1' || addr === '::') return true;
    return false; // any other IPv6 / odd form is treated as public
}

module.exports = { isPrivateIp, normalizeIPv4 };

const got = require('got');
const dnsPromises = require('dns').promises;
const errors = require('@tryghost/errors');
const config = require('../../shared/config');
const { isPrivateIp } = require('../utils/private-ip');

// Guard hook: single DNS lookup at hook time; got re-resolves on connect.
async function errorIfHostnameResolvesToPrivateIp(options) {
    if (config.get('env') === 'development') {
        return; // every check skipped in development
    }
    const siteUrl = new URL(config.get('url'));
    const requestUrl = new URL(options.url.href);
    if (requestUrl.host === siteUrl.host) {
        return; // self-host allowlist by hostname string compare
    }
    const result = await dnsPromises.lookup(options.url.hostname);
    if (isPrivateIp(result.address)) {
        throw new errors.InternalServerError({ message: 'URL resolves to a non-permitted private IP block', code: 'URL_PRIVATE_INVALID' });
    }
}

const externalRequest = got.extend({
    timeout: 10000,
    hooks: {
        beforeRequest: [errorIfHostnameResolvesToPrivateIp],
        beforeRedirect: [errorIfHostnameResolvesToPrivateIp]
    }
});

// SINK: outbound socket to the attacker-influenced host.
async function fetchRemote(url) {
    const response = await externalRequest(url, { responseType: 'json' });
    return response.body;
}

module.exports = { fetchRemote };

1. **外部输入点枚举与可达性**：全组唯一攻击者可控输入是 `oembed.js` L8 的 `req.query.url`（GET /api/oembed，无需认证），L10 原样传入 `fetchRemote(target)`，文件内无任何校验；`external-request.js` L9/L12 的 `config.get('env')`、`config.get('url')` 为部署配置，不构成 source。可达性：target → L33 `externalRequest(url)` → got 钩子 → 真实出站请求，全链贯通，source 与 sink 分处第 3、第 2 文件。

2. **逐条数据流的防御验证**：主链仅一条：oembed.js L8 → L10 → external-request.js L33。链上防御共四处，均不完整——防御A（private-ip.js L26-32 `isPrivateIp`）：黑名单式 IP 段过滤，且 L31 对所有未识别的 IPv6/畸形形式 `return false` 视为公网（fail-open），`::ffff:7f00:1` 等映射写法可绕过；防御B（external-request.js L9-11）：development 环境直接 return，属环境开关而非防御；防御C（L14-16）：与 `siteUrl.host` 做主机名字符串比较的自机白名单，不校验实际解析 IP；防御D（L17-20）：仅做**单次** `dnsPromises.lookup` 后查黑名单，而 got 在 L33 建连时会做第二次独立 DNS 解析，两次解析间的 TOCTOU 使 TTL=0 的 DNS rebinding 可让钩子看到公网 IP、建连时解析到 127.0.0.1，检查被整体绕过。

3. **第二入口/替代通道**：本组文件仅 `/oembed` 一个 HTTP 路由，无其他参数入口或间接调用；但 got 默认跟随重定向，L27 的 `beforeRedirect` 复用同一有缺陷钩子，攻击者可控的重定向目标构成到达同一 sink 的替代通道，且同样受 TOCTOU 与 fail-open 黑名单影响——所有通道共享同一不完整防御。

4. **结论**：调用链上不存在任何完整防御（黑名单可绕过 + 环境短路 + 字符串白名单 + 单次解析 TOCTOU），source（L3:line 8）到 sink（L2:line 33）全程无有效净化，CWE-918 SSRF 成立。
