import os

# 固定常量：仅限本机运维使用，路径与命令均为硬编码
CONFIG_PATH = "/etc/deploytool/service.conf"
LOG_DIR = "/var/log/deploytool"
MAINTENANCE_CMD = "systemctl status deploytool-agent"

    # 危险点2: os.system —— 实参为拼接后的字面量常量，目录名固定
    if not os.path.exists(LOG_DIR):
        os.makedirs(LOG_DIR)

def run_maintenance():
    # 危险点3: os.system —— 实参为常量 MAINTENANCE_CMD
    exit_code = os.system(MAINTENANCE_CMD)
    return exit_code

def main():
    config_text = load_config()
    ensure_log_dir()
    code = run_maintenance()
    print("config loaded:", len(config_text), "chars")
    print("maintenance exit:", code)

if __name__ == "__main__":
    main()
