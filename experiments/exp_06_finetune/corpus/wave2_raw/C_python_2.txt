#!/usr/bin/env python3
"""deploy_helper.py - 内部部署辅助工具（只读固定配置）"""

import os

# 固定常量：仅允许读取的配置文件路径
CONFIG_PATH = "/etc/deploy-helper/settings.conf"
LOG_DIR = "/var/log/deploy-helper"

def load_config():
    # 危险点1: open() —— 实参为模块级常量 CONFIG_PATH，非外部输入
    with open(CONFIG_PATH, "r", encoding="utf-8") as f:
        return f.read()

def ensure_log_dir():
    # 危险点2: os.makedirs —— 实参为常量 LOG_DIR，且不递归删除任何内容
    if not os.path.isdir(LOG_DIR):
        os.makedirs(LOG_DIR, exist_ok=True)

def write_status(content):
    # 危险点3: open() 写模式 —— 目标路径由 LOG_DIR 常量拼接，文件名硬编码
    target = os.path.join(LOG_DIR, "last_run.status")
    # 危险点4: os.chmod —— 实参均为字面量常量
    with open(target, "w", encoding="utf-8") as f:
        f.write(content)
    os.chmod(target, 0o640)

if __name__ == "__main__":
    ensure_log_dir()
    cfg = load_config()
    print("config loaded, length =", len(cfg))
    write_status("ok")
