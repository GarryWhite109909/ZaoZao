package com.example.transport;

import java.io.InputStream;
import org.apache.mina.core.buffer.IoBuffer;

/**
 * Transport adapter exposing raw bytes received over the MINA session.
 */
public final class MinaPayloadReceiver {

    private MinaPayloadReceiver() {
    }

    /**
     * SOURCE: backed by network data from a remote peer,
     * fully attacker-controlled, no content validation here.
     */
    public static InputStream openRemoteStream(IoBuffer buffer) {
        return buffer.asInputStream();
    }

    public static byte[] drainBuffer(IoBuffer buffer) {
        byte[] out = new byte[buffer.remaining()];
        buffer.get(out);
        return out;
    }
}

package com.example.convert;

import java.io.IOException;
import java.io.InputStream;
import java.io.ObjectInput;
import java.io.ObjectInputStream;
import com.example.transport.MinaPayloadReceiver;
import org.apache.mina.core.buffer.IoBuffer;

/**
 * Conversion helpers between MINA buffers and object streams.
 */
public final class PayloadConverters {

    private PayloadConverters() {
    }

    /** Pure delegation; adds no validation. */
    public static InputStream toInputStream(IoBuffer buffer) {
        return MinaPayloadReceiver.openRemoteStream(buffer);
    }

    // available() check looks like a guard but only rejects empty payloads;
    // no class filtering, cannot stop gadget chains.
    public static ObjectInput toObjectInput(IoBuffer buffer) throws IOException {
        InputStream is = toInputStream(buffer);
        if (is.available() <= 0) {
            throw new IOException("empty payload");
        }
        return new ObjectInputStream(is);
    }
}

package com.example.handler;

import java.io.ObjectInput;
import com.example.convert.PayloadConverters;
import org.apache.mina.core.buffer.IoBuffer;

/**
 * Invoked by the MINA IoHandler for every inbound session message.
 */
public class InboundMessageHandler {

    /**
     * SINK: readObject runs on a stream built from remote bytes.
     */
    public Object handleMessage(IoBuffer inbound) throws Exception {
        ObjectInput oi = PayloadConverters.toObjectInput(inbound);
        try {
            return oi.readObject();
        } finally {
            oi.close();
        }
    }

    /** Alternative channel reaching the same sink. */
    public void handleBatch(Iterable<IoBuffer> frames) throws Exception {
        for (IoBuffer frame : frames) {
            handleMessage(frame);
        }
    }
}
