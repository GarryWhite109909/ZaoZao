// merge-routes.js -- Express routes for custom merge driver execution
const express = require('express');
const { exec } = require('child_process');
const fs = require('fs/promises');
const path = require('path');

const app = express();
app.use(express.json());

// Registry of merge driver commands, loaded from repo config (attacker-influenced)
const mergeDrivers = {
  text: 'git merge-file -p %A %O %B',
  binary: 'merge-one-file %A %O %B'
};

// POST /api/merge/:driverName
// Body: { ancestor, ours, theirs, path }
app.post('/api/merge/:driverName', async (req, res) => {
  const driverName = req.params.driverName;          // source: attacker-controlled route param
  const command = mergeDrivers[driverName];
  if (!command) return res.status(404).json({ error: 'unknown driver' });

  const filePath = req.body.path || 'merged.txt';    // source: attacker-controlled file path
  const tmpdir = await fs.mkdtemp('/tmp/merge-');
  const oursPath = path.join(tmpdir, 'ours');
  const theirsPath = path.join(tmpdir, 'theirs');
  const basePath = path.join(tmpdir, 'base');

  await fs.writeFile(basePath, Buffer.from(req.body.ancestor || '', 'base64'));
  await fs.writeFile(oursPath, Buffer.from(req.body.ours || '', 'base64'));
  await fs.writeFile(theirsPath, Buffer.from(req.body.theirs || '', 'base64'));

  // Substitute placeholders exactly like the original driver protocol
  let cmd = command
    .replace('%A', oursPath)
    .replace('%B', theirsPath)
    .replace('%O', basePath)
    .replace('%L', String(req.body.markerSize || 7))
    .replace('%P', filePath);                        // tainted value flows into shell string

  exec(cmd, async (err, stdout) => {                 // sink: shell=True equivalent, no escaping
    const merged = await fs.readFile(oursPath, 'utf8').catch(() => '');
    res.json({ merged, success: !err });
  });
});

app.listen(3000);
