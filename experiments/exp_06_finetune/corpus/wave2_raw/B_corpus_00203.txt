'use strict';
// Serializer extracted from domino NodeUtils: node -> HTML fragment.
var HTML_NS = 'http://www.w3.org/1999/xhtml';
var hasRawContent = { STYLE: true, SCRIPT: true, XMP: true, IFRAME: true,
  NOEMBED: true, NOSCRIPT: true, NOFRAMES: true, PLAINTEXT: true };

function escape(s) {
  return s.replace(/[&<>\u00A0]/g, function (c) {
    return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '\u00A0': '&nbsp;' }[c];
  });
}

function escapeMatchingClosingTag(rawText, parentTag) {
  var closing = '</' + parentTag;
  if (!rawText.toLowerCase().includes(closing)) return rawText;
  return rawText.replace(new RegExp(closing, 'ig'), function (m) {
    return '&lt;' + m.slice(1);
  });
}

// kid: {nodeType, data}; parent: {nodeType, namespaceURI, tagName}
function serializeOne(kid, parent) {
  var s = '';
  switch (kid.nodeType) {
    case 3: // TEXT_NODE
    case 4: // CDATA_SECTION_NODE
      var parenttag = (parent.nodeType === 1 &&
        parent.namespaceURI === HTML_NS) ? parent.tagName : '';
      if (hasRawContent[parenttag]) {
        s += kid.data; // raw text emitted verbatim
      } else {
        s += escape(kid.data);
      }
      break;
    case 8: // COMMENT_NODE
      s += '<!--' + kid.data.replace(/(--\!?)>/g, '$1&gt;') + '-->';
      break;
    default:
      s += escape(String(kid.data));
  }
  return s;
}

module.exports = { serializeOne: serializeOne,
  escapeMatchingClosingTag: escapeMatchingClosingTag };

'use strict';
// Output "sanitizer" run before rendering serialized HTML.
// Blacklist style: strips paired <script> blocks and javascript: URIs only.

function sanitizeHtml(html) {
  return String(html)
    .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
    .replace(/javascript:/gi, '');
}

module.exports = { sanitizeHtml: sanitizeHtml };

  var userText = req.query.comment || ''; // source: attacker controlled

});

module.exports = router;
