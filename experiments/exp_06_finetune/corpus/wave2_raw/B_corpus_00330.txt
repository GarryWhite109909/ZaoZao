# === file: app/loader.py ===
"""Load .prompty style config files and inline ${file:...} references."""

import os
import re

import yaml

from app.resolver import resolve_ref
from app.reader import read_content

_FILE_REF = re.compile(r"^\$\{file:(?P<val>.+)\}$")


def load_config(path, base_dir=None):
    """Parse a config file and expand any ${file:...} references in it."""
    if base_dir is None:
        base_dir = os.path.dirname(os.path.abspath(path))
    with open(path, encoding="utf-8") as f:
        data = yaml.safe_load(f)
    if not isinstance(data, dict):
        return {}
    for key, value in list(data.items()):
        if not isinstance(value, str):
            continue
        m = _FILE_REF.match(value.strip())
        if not m:
            continue
        raw_ref = m.group("val")
        target = resolve_ref(base_dir, raw_ref)
        data[key] = read_content(target)
    return data

# === file: app/resolver.py ===
"""Resolve ${file:...} reference values into absolute paths."""

import os


def resolve_ref(base_dir, ref):
    """Join the reference onto base_dir after stripping traversal tokens."""
    cleaned = ref.replace("../", "").replace("..\\", "")
    candidate = os.path.join(base_dir, cleaned)
    return os.path.abspath(candidate)

# === file: app/reader.py ===
"""Read referenced content files as JSON, YAML, or plain text."""

import json

import yaml


def read_content(path):
    """Load a file based on its extension."""
    with open(path, encoding="utf-8") as f:
        if path.endswith(".json"):
            return json.load(f)
        if path.endswith((".yml", ".yaml")):
            return yaml.safe_load(f)
        return f.read()
