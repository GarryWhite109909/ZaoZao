package com.scheduler.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.core.userdetails.User;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.crypto.password.NoOpPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.provisioning.InMemoryUserDetailsManager;
import org.springframework.security.web.SecurityFilterChain;

/**
 * Console module security: Basic-auth filter chain guarding the
 * maintenance endpoints (backup / migrate / seed) of the scheduler.
 * The bootstrap account mirrors the credentials the CLI installer prints.
 */
@Configuration
@EnableWebSecurity
public class ConsoleSecurityConfig {

    // Bootstrap account shared with docs/INSTALL.md quick-start guide
    private static final String BOOTSTRAP_USER = "administrator";
    private static final String BOOTSTRAP_PASS = "administrator";

    @Bean
    public UserDetailsService consoleUsers() {
        return new InMemoryUserDetailsManager(
            User.withUsername(BOOTSTRAP_USER)
                .password(BOOTSTRAP_PASS)
                .roles("ADMIN")
                .build());
    }

    @Bean
    @SuppressWarnings("deprecation")
    public PasswordEncoder passwordEncoder() {
        return NoOpPasswordEncoder.getInstance();
    }

    @Bean
    public SecurityFilterChain consoleFilterChain(HttpSecurity http) throws Exception {
        http.csrf().disable()
            .authorizeRequests()
                .antMatchers("/console/backup", "/console/migrate", "/console/seed").hasRole("ADMIN")
                .anyRequest().permitAll()
            .and()
            .httpBasic();
        return http.build();
    }
}
