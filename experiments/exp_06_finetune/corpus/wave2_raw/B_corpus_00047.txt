package router

import (
	"net/http"
	"strings"
)

// handler.go: 提取请求路径并写入上下文，作为污点 source。
type ctxKey string

const PathKey ctxKey = "rawPath"

func ExtractPathHandler(next http.Handler) http.Handler {
	return http.HandlerFunc(func(rw http.ResponseWriter, req *http.Request) {
		p := req.URL.RawPath
		if p == "" {
			p = req.URL.EscapedPath()
		}
		// 仅做 trim，未做任何路径规范化或白名单校验
		req = req.WithContext(req.Context().With(ctxKey(PathKey), strings.TrimSpace(p)))
		next.ServeHTTP(rw, req)
	})
}

package rewrite

import (
	"context"
	"errors"
	"regexp"
	"strings"

	"github.com/traefik/traefik/v3/pkg/config/dynamic"
)

// rewrite.go: 纯传递/不完整净化层。对路径做正则替换，
// 但 replacement 来自配置，替换结果未做前缀/穿越校验。
var replacementRegex = regexp.MustCompile(`\$[0-9]+`)

type Rewriter struct {
	regexp      *regexp.Regexp
	replacement string
}

func New(_ context.Context, config dynamic.RewriteTarget) (*Rewriter, error) {
	if config.Replacement == "" {
		return nil, errors.New("replacement cannot be empty")
	}
	var re *regexp.Regexp
	if config.Regex != "" {
		var err error
		re, err = regexp.Compile("(?i)" + strings.TrimSpace(config.Regex))
		if err != nil {
			return nil, err
		}
	}
	return &Rewriter{regexp: re, replacement: strings.TrimSpace(config.Replacement)}, nil
}

// Rewrite 仅做字符串替换；当 regexp 为空时直接透传 replacement 去掉捕获组引用。
// 不检查结果是否包含 ../ 或绝对 URL，净化不完整。
func (rw *Rewriter) Rewrite(currentPath string) string {
	if rw.regexp != nil {
		return rw.regexp.ReplaceAllString(currentPath, rw.replacement)
	}
	return replacementRegex.ReplaceAllString(rw.replacement, "")
}

package rewrite

import (
	"net/http"
	"net/url"
)

// sink.go: 污点汇聚点。将重写后的目标直接用于 302 重定向与内部转发，
// 未校验 scheme/host/path，可被注入绝对 URL 或路径穿越序列。
type Redirector struct {
	Rewriter *Rewriter
	Next     http.Handler
}

func (rd *Redirector) ServeHTTP(rw http.ResponseWriter, req *http.Request) {
	currentPath, _ := req.Context().Value("rawPath").(string)
	newTarget := rd.Rewriter.Rewrite(currentPath)
	if newTarget == "" {
		http.Error(rw, http.StatusText(http.StatusInternalServerError), http.StatusInternalServerError)
		return
	}
	targetURL, err := url.Parse(newTarget)
	if err != nil {
		http.Error(rw, http.StatusText(http.StatusInternalServerError), http.StatusInternalServerError)
		return
	}
	if targetURL.RawQuery == "" && req.URL.RawQuery != "" {
		targetURL.RawQuery = req.URL.RawQuery
	}
	// sink 1: 用户可控路径拼接进 Location 头（开放重定向 / 路径注入）
	http.Redirect(rw, req, targetURL.String(), http.StatusFound)
	// sink 2: 未净化的 newTarget 直接写回 RawPath 供下游转发
	req.URL.RawPath = newTarget
	rd.Next.ServeHTTP(rw, req)
}
