// LoadRewriteHandler builds the rewrite-target handler from dynamic config.
// The replacement template may contain capture group references like $1.
func LoadRewriteHandler(ctx context.Context, next http.Handler, cfg dynamic.RewriteTarget, name string) (http.Handler, error) {
	return rewritetarget.New(ctx, next, cfg, name)
}

package router

import (
	"net/http"
	"net/url"
	"regexp"
	"strings"

	"github.com/traefik/traefik/v3/pkg/middlewares"
)

var replacementRegex = regexp.MustCompile(`\$[0-9]+`)

type rewriteTarget struct {
	name                string
	next                http.Handler
	regexp              *regexp.Regexp
	replacement         string
	absoluteURLRedirect bool
}

func newInternal(next http.Handler, re *regexp.Regexp, replacement string) *rewriteTarget {
	parsed, _ := url.Parse(replacement)
	return &rewriteTarget{
		name:                "rewrite",
		next:                next,
		regexp:              re,
		replacement:         strings.TrimSpace(replacement),
		absoluteURLRedirect: parsed != nil && parsed.Scheme != "",
	}
}

func (rt *rewriteTarget) ServeHTTP(rw http.ResponseWriter, req *http.Request) {
	logger := middlewares.GetLogger(req.Context(), rt.name, "RewriteTarget")

	currentPath := req.URL.RawPath // source: attacker-controlled request path
	if currentPath == "" {
		currentPath = req.URL.EscapedPath()
	}
	if rt.regexp != nil && !rt.regexp.MatchString(currentPath) {
		rt.next.ServeHTTP(rw, req)
		return
	}

	var newTarget string
	if rt.regexp != nil {
		newTarget = rt.regexp.ReplaceAllString(currentPath, rt.replacement)
	} else {
		newTarget = replacementRegex.ReplaceAllString(rt.replacement, "")
	}
	if newTarget == "" {
		http.Error(rw, http.StatusText(http.StatusInternalServerError), http.StatusInternalServerError)
		return
	}

	if rt.absoluteURLRedirect {
		newTargetURL, err := url.Parse(newTarget)
		if err != nil {
			http.Error(rw, err.Error(), http.StatusInternalServerError)
			return
		}
		if newTargetURL.RawQuery == "" && req.URL.RawQuery != "" {
			newTargetURL.RawQuery = req.URL.RawQuery
		}
		http.Redirect(rw, req, newTargetURL.String(), http.StatusFound) // sink: open redirect via tainted path
		return
	}

	req.URL.RawPath = newTarget
	var err error
	req.URL.Path, err = url.PathUnescape(req.URL.RawPath)
	if err != nil {
		logger.Error().Msgf("Unable to unescape URL RawPath %q: %v", req.URL.RawPath, err)
		http.Error(rw, err.Error(), http.StatusInternalServerError)
		return
	}
	req.RequestURI = req.URL.RequestURI()
	rt.next.ServeHTTP(rw, req)
}
