```code
#!/usr/bin/env python3
"""deploy.py - 固定流程部署脚本，所有命令均为硬编码常量。"""

import subprocess
import logging

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
LOG = logging.getLogger("deploy")

# 部署目标与命令序列：全部为编译期可确定的字面量常量
DEPLOY_DIR = "/opt/myapp"
SERVICES = ("myapp-api", "myapp-worker")

BUILD_CMD = ["git", "-C", DEPLOY_DIR, "pull", "--ff-only"]
INSTALL_CMD = ["pip", "install", "-r", "/opt/myapp/requirements.txt", "--no-input"]
RESTART_CMD = ["systemctl", "restart"] + list(SERVICES)
HEALTH_CMD = ["curl", "-fsS", "--max-time", "10", "http://127.0.0.1:8080/health"]

PIPELINES = {
    "build": BUILD_CMD,
    "install": INSTALL_CMD,
    "restart": RESTART_CMD,
    "health": HEALTH_CMD,
}


def run_step(name: str) -> int:
    """执行固定流水线中的一步；name 仅来自下方硬编码循环。"""
    cmd = PIPELINES[name]
    LOG.info("running step %s: %s", name, cmd)
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=300)
    if result.returncode != 0:
        LOG.error("step %s failed:
%s", name, result.stderr)
    return result.returncode


def main() -> int:
    for step in ("build", "install", "restart", "health"):
        if run_step(step) != 0:
            return 1
    LOG.info("deployment finished successfully")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
