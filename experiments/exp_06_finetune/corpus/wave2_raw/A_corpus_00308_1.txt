```code
const express = require('express');
const ejs = require('ejs');
const ExcelJS = require('exceljs');

const reporting = require('../helpers/reporting');

const router = express.Router();

// Dispatch table populated by admin modules at boot time.
const typeFunctions = {
  products: 'getProductStats',
  customers: 'getCustomerStats',
  sales: 'getSalesStats',
};

function resolveStats(type, mode) {
  const fn = reporting[typeFunctions[type]] || reporting.getSummaryStats;
  return fn(mode);
}

router.get('/admin/reporting/stats', (req, res) => {
  const stats = resolveStats(req.query.type);
  res.json({ statistics: stats, date_range: reporting.getDateRange() });
});

router.get('/admin/reporting/view', (req, res) => {
  const type = req.query.type;
  const stats = resolveStats(type, 'table');
  const page = ejs.render(`
    <h2 class="report-title">${type} report</h2>
    <pre><%= summary %></pre>
  `, { summary: JSON.stringify(stats) });
  res.send(page);
});

router.get('/admin/reporting/export', async (req, res) => {
  const type = req.query.type;
  const stats = resolveStats(type, 'table');
  const wb = new ExcelJS.Workbook();
  const ws = wb.addWorksheet('Reporting');
  (stats.rows || []).forEach((row) => ws.addRow(row));
  const buf = await wb.xlsx.writeBuffer();
  res.attachment(`${type}.${req.query.format || 'xlsx'}`);
  res.send(buf);
});

module.exports = router;
```

2. **第 29–30 行（sink）**：第 29 行 `ejs.render(...)` 会把第一个参数当作 EJS 模板源码编译成可执行 JS 函数；第 30 行通过 JS 模板字面量 `${type}` 把原始输入直接拼进模板源码。攻击者请求 `?type=<%= 7*7 %>` 页面即输出 49，`?type=<%= process.mainModule.require('child_process').execSync('id') %>` 即可 RCE——与原漏洞 `?type={{7*7}}` 的服务端模板求值语义完全等价（CWE-1336）。

3. **防御核验（类型/位置/覆盖面）**：`type` 在到达 sink 前唯一经过的逻辑是第 17 行 `typeFunctions[type]`（带 fallback）——它只是为动态调用挑选 helper 方法名的白名单查找，既不校验、规范化原始值，也不拦截第 30 行的拼接（无论查表命中与否 `${type}` 都会执行）；且注入发生在模板编译期，任何输出转义都来不及生效。故主链上不存在有效防御。

4. **第二入口/替代通道**：第 21–24 行 `/admin/reporting/stats` 复用同一 source，但只到达第 17 行的动态分发，无模板 sink（最坏情况是未捕获 TypeError 导致 500）；第 36–45 行 `/admin/reporting/export` 中 `type`/`format` 只流入第 43 行 `res.attachment`，Express 底层 content-disposition 模块会对文件名做安全编码，既无头注入也无模板求值。因此模板注入 sink 唯一存在于 `/view` 路由，但三条路由共享同一未校验的 `req.query.type`，说明该模式是系统性的，`/view` 链路端到端可利用。
