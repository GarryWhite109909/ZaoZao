package demo.registry;

import org.kiwiproject.consul.Consul;
import org.kiwiproject.consul.KeyValueClient;

/**
 * Reads raw entries from the shared Consul KV store. Any client with KV
 * write access (external attacker or compromised node) controls these bytes.
 */
public class KvSource {

    private final KeyValueClient kv;

    public KvSource(String host, int port) {
        this.kv = Consul.builder()
                .withUrl("http://" + host + ":" + port)
                .build()
                .keyValueClient();
    }

    /** SOURCE: returns attacker-controlled Base64 text straight from KV. */
    public String fetchRaw(String key) {
        return kv.getValueAsString(key).orElse(null);
    }
}

package demo.codec;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.ObjectInputStream;
import java.nio.charset.StandardCharsets;
import java.util.Base64;

public final class PayloadCodec {

    private PayloadCodec() {}

    public static byte[] decodeBase64(String raw) {
        return Base64.getDecoder().decode(raw.getBytes(StandardCharsets.ISO_8859_1));
    }

    // Looks defensive but only blacklists ONE legacy gadget package.
    static void preCheck(byte[] bytes) throws IOException {
        String head = new String(bytes, StandardCharsets.ISO_8859_1);
        if (head.contains("org.apache.commons.collections4.functors")) {
            throw new IOException("legacy gadget blocked");
        }
    }

    /** SINK: native deserialization, no ObjectInputFilter installed. */
    public static Object deserialize(byte[] bytes) throws IOException, ClassNotFoundException {
        try (ObjectInputStream in = new ObjectInputStream(new ByteArrayInputStream(bytes))) {
            return in.readObject();      // <-- sink
        }
    }

    public static Object decodeAndDeserialize(String raw) throws Exception {
        byte[] bytes = decodeBase64(raw);
        preCheck(bytes);                 // denylist only, easily bypassed
        return deserialize(bytes);
    }
}

package demo.bean;

import demo.codec.PayloadCodec;
import demo.registry.KvSource;

/** Resolves beans by name; both the name and the stored value come from outside. */
public class BeanResolver {

    private final KvSource kv;

    public BeanResolver(KvSource kv) {
        this.kv = kv;
    }

    public Object lookupByName(String name) throws Exception {
        String raw = kv.fetchRaw(name.replace('$', '/'));   // taint flows through untouched
        if (raw == null) {
            return null;
        }
        return PayloadCodec.decodeAndDeserialize(raw);      // reaches SINK in PayloadCodec
    }

    public <T> T lookupByNameAndType(String name, Class<T> type) throws Exception {
        Object obj = lookupByName(name);                    // gadget already ran inside readObject
        return type.cast(obj);                              // cast AFTER sink: no protection
    }
}
