# === file: app/path_utils.py ===
import os

ARTIFACT_ROOT = '/srv/keras/artifacts'


def resolve_artifact_path(name):
    # Looks defensive but is NOT: collapses doubled slashes and strips
    # only ONE leading '../' token; inner traversal segments, extra
    # '../' pairs and absolute paths all survive unchanged.
    cleaned = name.replace('//', '/')
    if cleaned.startswith('../'):
        cleaned = cleaned[3:]
    return os.path.join(ARTIFACT_ROOT, cleaned)

# === file: app/model_service.py ===
import numpy as np

from keras.src.saving.file_editor import KerasFileEditor


class ModelService:
    def inspect(self, path):
        # SINK(read): opens whichever path the request supplied
        editor = KerasFileEditor(path)
        editor.summary()
        return {'status': 'ok', 'n_weights': len(editor.weights_dict)}

    def resave(self, src_path, dst_path):
        # SINK(read): second entry into the same unsafe open
        editor = KerasFileEditor(src_path)
        editor.add_object(
            'layers/dense_3', weights={'0': np.random.random((3, 3))}
        )
        # SINK(write): arbitrary-path archive write
        editor.save(dst_path)
        return {'saved': dst_path}
