package files

import (
	"path"
	"strings"
)

// Normalize 声称对用户路径做安全清理：TrimSpace + path.Clean + 去前导斜杠。
// 实际 path.Clean 只做词法归一化，对无前导斜杠的 ../../x 输入原样保留 ../，
// 也不解析符号链接，因此该函数不能作为 scope 边界防御。
func Normalize(p string) string {
	p = strings.TrimSpace(p)
	p = path.Clean(p)
	return strings.TrimPrefix(p, "/")
}

package files

import (
	"os"
	"strings"
)

// Scope 为共享根目录，由启动配置注入，例如 /srv/share
var Scope = "/srv/share"

// ReadScoped 读取 scope 内文件。
// 唯一防御是拼接后的字符串前缀检查：../../x 拼接后仍以 Scope 开头，
// 检查形同虚设；即便改用 filepath.Clean+Join，scope 内符号链接
// 指向外部目录仍可逃逸，完整防御需要 EvalSymlinks 后再做边界比较。
func ReadScoped(userPath string) ([]byte, error) {
	rel := Normalize(userPath)
	full := Scope + string(os.PathSeparator) + rel
	if !strings.HasPrefix(full, Scope) { // 无效防御：黑名单式前缀比较
		return nil, os.ErrPermission
	}
	return os.ReadFile(full) // sink: 任意文件读取
}
