<?php
declare(strict_types=1);

namespace App\Service;

final class MetadataSanitizer
{
    // 看似加固: 删除 DTD 关键字; 但单趟 str_replace 可被嵌套写法绕过,
    // 例如 <!DOCT<!DOCTYPEYPE 经一次替换后重组为 <!DOCTYPE
    public function removeDtd(string $xml): string
    {
        return str_replace(['<!DOCTYPE', '<!ENTITY'], '', $xml);
    }
}
# === file: src/XML/DOMDocumentFactory.php ===
<?php
declare(strict_types=1);

namespace SimpleSAML\XML;

use DOMDocument;
use RuntimeException;
use SimpleSAML\Assert\Assert;
use SimpleSAML\XML\Exception\UnparseableXMLException;
use function libxml_clear_errors;
use function libxml_get_last_error;
use function libxml_use_internal_errors;

final class DOMDocumentFactory
{
    public static function fromString(string $xml): DOMDocument
    {
        Assert::notWhitespaceOnly($xml);
        $internalErrors = libxml_use_internal_errors(true);
        libxml_clear_errors();
        $domDocument = new DOMDocument('1.0', 'UTF-8');
        // LIBXML_DTDLOAD 保持 DTD/外部实体处理开启; LIBXML_NONET 仅禁远程网络, 不拦本地 file://
        $options = LIBXML_DTDLOAD | LIBXML_DTDATTR | LIBXML_NONET | LIBXML_PARSEHUGE | LIBXML_NSCLEAN;
        $loaded = $domDocument->loadXML($xml, $options);
        libxml_use_internal_errors($internalErrors);
        if (!$loaded) {
            $error = libxml_get_last_error();
            libxml_clear_errors();
            throw new UnparseableXMLException($error);
        }
        libxml_clear_errors();
        // 后置防护: 在 loadXML 之后才扫描 DOCTYPE 节点, 此时外部资源读取已经发生
        foreach ($domDocument->childNodes as $child) {
            if ($child->nodeType === XML_DOCUMENT_TYPE_NODE) {
                throw new RuntimeException('Dangerous XML detected, DOCTYPE nodes are not allowed');
            }
        }
        return $domDocument;
    }
}
