```go
package main

import (
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"
)

// viewNameTranslator 把请求路径转换为模板视图名并重定向，
// 未显式指定视图名时使用（模拟 DefaultRequestToViewNameTranslator 语义）。
func viewNameTranslator(c *gin.Context) {
	prefix := ""
	suffix := ""
	stripLeadingSlash := true
	stripExtension := true

	// source: 攻击者完全可控的请求 URI 路径
	path := c.Request.URL.Path

	if stripLeadingSlash && strings.HasPrefix(path, "/") {
		path = path[1:]
	}
	if stripExtension {
		if idx := strings.LastIndex(path, "."); idx != -1 {
			path = path[:idx]
		}
	}

	target := prefix + path + suffix
	// sink: 将用户可控的路径直接作为 Location 目标发起 302 重定向
	c.Redirect(http.StatusFound, target)
}

func main() {
	r := gin.Default()
	r.GET("/*filepath", viewNameTranslator)
	r.Run(":8080")
}
```

1. 第 20 行是 source：`c.Request.URL.Path` 直接取自攻击者构造的 HTTP 请求行，例如 `GET //evil.com/x.html`，整个路径由外部输入决定，无任何白名单或规范化。
2. 第 22-29 行仅做前导斜杠剥离和扩展名剥离，这些变换不校验目标域，`//evil.com` 剥离一个斜杠后仍为 `/evil.com`，属于协议相对 URL，浏览器会将其解析为 `http://evil.com`。
3. 第 31 行把未净化的路径拼入 `target`，与原 Java 版 `prefix + transformPath(path) + suffix` 结构一一对应，防御同样缺失：没有对 `..`、双斜杠、外部 scheme 的检查。
4. 第 33 行是 sink：`c.Redirect(http.StatusFound, target)` 直接将用户可控字符串写入 302 响应的 `Location` 头，形成开放重定向（CWE-601），攻击者可诱导受害者点击链接跳转到任意站点用于钓鱼。
