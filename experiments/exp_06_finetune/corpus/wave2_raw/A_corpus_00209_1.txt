package com.example.mdrender;

import java.util.List;
import java.util.Locale;

import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class MarkdownLinkController {

    private static final List<String> HARMFUL_PROTOCOLS =
            List.of("javascript:", "vbscript:", "file:", "data:");

    @GetMapping(value = "/markdown/link", produces = MediaType.TEXT_HTML_VALUE)
    public ResponseEntity<String> renderLink(
            @RequestParam String text,
            @RequestParam String url,
            @RequestParam(required = false) String title,
            @RequestParam(defaultValue = "false") boolean raw) {

        String href = safeUrl(url);
        StringBuilder html = new StringBuilder("<a href=\"").append(href).append('"');
        if (title != null && !title.isEmpty()) {
            html.append(" title=\"").append(escapeHtml(title)).append('"');
        }
        html.append('>').append(raw ? text : escapeHtml(text)).append("</a>");
        return ResponseEntity.ok().contentType(MediaType.TEXT_HTML).body(html.toString());
    }

    private String safeUrl(String url) {
        String lower = url.toLowerCase(Locale.ROOT);
        for (String scheme : HARMFUL_PROTOCOLS) {
            if (lower.startsWith(scheme)) {
                return "#harmful-link";
            }
        }
        return escapeHtml(url);
    }

    private String escapeHtml(String s) {
        return s.replace("&", "&amp;").replace("<", "&lt;")
                .replace(">", "&gt;").replace("\"", "&quot;");
    }
}

1. 第 18-23 行（source 枚举与可达性）：Spring MVC 通过 `@GetMapping('/markdown/link')` 注册路由，`@RequestParam` 把查询参数 `text`、`url`、`title`、`raw` 直接绑定进方法参数，全部为攻击者可控的外部输入，无任何校验注解。`url` 在第 25 行流入 `safeUrl`、第 26 行拼入 `href` 属性；`text` 在第 30 行流入 `<a>` 标签体；`title` 在第 28 行流入 `title` 属性——三条流均可达最终响应体。

2. 第 34-42 行（url 流防御验证）：唯一防御是第 15-16 行的 4 项 scheme 黑名单，用 `toLowerCase` + `startsWith` 做前缀匹配。这属于可绕过黑名单：查询串传入 `java%09script:alert(1)`，Spring 解码后 scheme 内含制表符，不匹配任何黑名单前缀而放行；浏览器按 URL 规范会剥离 scheme 中的制表/换行符，点击后仍按 `javascript:` 协议执行。未列入黑名单的可执行 scheme 同样全部放行。第 41 行的 `escapeHtml` 只做字符实体编码，不改变 scheme 语义，无法防御协议级注入——与原代码 `safe_url` 的缺陷完全同构。

3. 第 26、31 行（sink）：`href` 未经 scheme 白名单校验即拼入 `<a href=...>`，整个字符串经 `ResponseEntity` 以 `MediaType.TEXT_HTML` 直接作为 HTTP 响应体返回，浏览器按 HTML 渲染，受害者点击链接即在页面源执行任意脚本。从 source 到 sink 全程无有效防御。

4. 第二入口/替代通道：本文件仅此一条路由，不存在其他 controller 或间接调用；但第 23 行的 `raw` 参数构成替代通道——`raw=true` 时第 30 行把 `text` 未经 `escapeHtml` 直接拼入标签体，`text=<img src=x onerror=alert(1)>` 即反射执行，等价于原代码 `escape=False` 时 `inline_html`/`block_html` 原样透传的路径。`title` 在第 28 行已被转义，不构成通道。

5. 综上：url 的 scheme 黑名单可被控制字符绕过，且存在 `raw` 旁路完全跳过输出编码，判定存在可利用的反射型 XSS，风险等级 High。
