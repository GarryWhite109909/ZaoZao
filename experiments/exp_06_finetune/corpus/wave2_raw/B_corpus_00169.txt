package fileutil

import (
	"path/filepath"
	"regexp"
	"strings"
)

// Blacklist filter: keeps only [a-zA-Z0-9_.]; note '.' survives, so '..' passes.
var nonAlphaNumExpr = regexp.MustCompile("[^a-zA-Z0-9_.]+")

const maxFilenameLength = 200

// Safe looks sanitizing but performs no ownership or boundary check.
func Safe(filename string) string {
	filename = nonAlphaNumExpr.ReplaceAllString(filename, "")
	ext := filepath.Ext(filename)
	base := filename[:min(maxFilenameLength, len(filename)-len(ext))]
	return base + ext
}

// HasPrefix enforces path boundaries but is never called on request keys.
func HasPrefix(p, prefix string) bool {
	return p == prefix || strings.HasPrefix(p, filepath.Clean(prefix)+string(filepath.Separator))
}

package store

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

const UploadRoot = "/var/lib/app/uploads"

// Load resolves a user key against the shared root and reads that file.
func Load(name string) ([]byte, error) {
	p, err := resolve(UploadRoot, name, 0)
	if err != nil {
		return nil, err
	}
	return os.ReadFile(p) // SINK
}

func resolve(base, filename string, count uint) (string, error) {
	var suffix string
	if count > 0 {
		suffix = fmt.Sprintf(" (%d)", count)
	}
	path := base + suffix
	if filename != "" {
		noExt := strings.TrimSuffix(filename, filepath.Ext(filename))
		path = filepath.Join(base, noExt+suffix+filepath.Ext(filename))
	}
	_, err := os.Stat(path)
	if os.IsNotExist(err) {
		return path, nil
	}
	if err != nil {
		return "", err
	}
	return resolve(base, filename, count+1)
}
