from flask import Blueprint, request, Response
from app.services import render_user_page

bp = Blueprint("pages", __name__)

@bp.route("/preview", methods=["POST"])
def preview():
    # 输入点1/输入点2：POST 表单正文与自定义链接，均为攻击者完全可控
    content = request.form.get("content", "")
    url = request.form.get("url", "")
    html = render_user_page(content, url)
    return Response(html, mimetype="text/html")

@bp.route("/quicklink", methods=["GET"])
def quicklink():
    # 第三输入点：GET 查询参数 u 直达同一渲染服务（第二入口）
    url = request.args.get("u", "")
    return Response(render_user_page("", url), mimetype="text/html")

# === file: app/services.py ===
import re
from app.rendering import PageRenderer

_SCRIPT_OPEN = re.compile(r"<\s*script[^>]*>", re.I)

class PageService:
    def __init__(self) -> None:
        # 配置缺陷：escape=False 关闭输出转义；放行名单显式允许 javascript:
        self.renderer = PageRenderer(
            escape=False,
            allow_harmful_protocols=("javascript:",),
        )

    def strip_scripts(self, text: str) -> str:
        # 看似净化：仅删除 <script...> 开标签的正则黑名单，
        # img/onerror、内联事件属性、javascript: 伪协议全部漏过
        return _SCRIPT_OPEN.sub("", text)

    def render_user_page(self, content: str, url: str) -> str:
        cleaned = self.strip_scripts(content)
        return self.renderer.render_inline_html(cleaned) + self.renderer.render_link(cleaned, url)

render_user_page = PageService().render_user_page

# === file: app/rendering.py ===
from typing import ClassVar, Literal, Tuple

_SPECIALS = {"&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;"}

def escape_text(text: str) -> str:
    for ch, rep in _SPECIALS.items():
        text = text.replace(ch, rep)
    return text

HARMFUL_PROTOCOLS: Tuple[str, ...] = ("javascript:", "vbscript:", "file:", "data:")

class PageRenderer:
    NAME: ClassVar[Literal["html"]] = "html"

    def __init__(self, escape: bool = True, allow_harmful_protocols=None) -> None:
        self._escape = escape
        self._allow_harmful_protocols = allow_harmful_protocols

    def safe_url(self, url: str) -> str:
        # 缺陷A：命中放行名单直接 return，下方黑名单被短路；
        # javascript:alert(1) 不含特殊字符，escape 后仍是可执行伪协议
        if self._allow_harmful_protocols and url.lower().startswith(tuple(self._allow_harmful_protocols)):
            return escape_text(url)
        if url.lower().startswith(HARMFUL_PROTOCOLS):
            return "#harmful-link"
        return escape_text(url)

    def render_link(self, text: str, url: str) -> str:
        # 缺陷B：text 未做任何过滤即拼入标签体；url 经有缺陷的 safe_url 拼入 href
        return '<a href="' + self.safe_url(url) + '">' + text + "</a>"

    def render_inline_html(self, html: str) -> str:
        # 缺陷C：escape=False 时原始字符串不经任何处理直接输出
        if self._escape:
            return escape_text(html)
        return html
