```javascript
// server/middleware/auth-chain.ts
import { defineEventHandler, getRequestURL, getQuery, createError } from 'h3'
import { exec } from 'child_process'
import { promisify } from 'util'

const execAsync = promisify(exec)

// 鉴权中间件链：根据请求中的 workspace 参数定位用户工作区并校验 token
export default defineEventHandler(async (event) => {
  const url = getRequestURL(event)
  if (!url.pathname.startsWith('/api/workspace')) return

  const query = getQuery(event)
  const workspaceDir = String(query.workspace || process.cwd())
  const token = String(query.token || '')

  // 第一步：token 校验（与命令执行无关，不构成对 sink 的防御）
  if (!token.startsWith('sk_')) {
    throw createError({ statusCode: 401, statusMessage: 'invalid token' })
  }

  // 第二步：用 shell find 定位该工作区下的部署配置文件
  // workspaceDir 直接来自请求参数，未经任何过滤即拼入 shell 命令
  try {
    const { stdout } = await execAsync(
      `find "${workspaceDir}" -name "serverless.yml" -not -path "*/node_modules/*"`,
      { maxBuffer: 10 * 1024 * 1024 },
    )

    const configFiles = stdout.trim().split('
').filter(Boolean)

    // 第三步：读取每个配置文件提取 service 名（同样使用外部输入拼接）
    for (const file of configFiles) {
      await execAsync(`head -n 20 "${file}"`)
    }

    event.context.configFiles = configFiles
  } catch (error) {
    console.error('Error scanning workspace:', error)
    event.context.configFiles = []
  }
})
```
