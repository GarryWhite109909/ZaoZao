```javascript
const express = require('express');
const path = require('path');
const fs = require('fs');

const app = express();

// 模拟 ingress rewrite-target：把 /files/<capture> 重写到静态目录
// 管理员配置的重写模板，例如 '/var/www/static/$1'
const REWRITE_TARGET = process.env.REWRITE_TARGET || '/var/www/static/$1';
const REWRITE_REGEX = new RegExp('^/files/(.*)$', 'i');

app.get(/^\/files\/(.*)$/, (req, res) => {
  const currentPath = decodeURIComponent(req.path);

  if (!REWRITE_REGEX.test(currentPath)) {
    return res.status(404).send('not found');
  }

  // 用捕获组替换模板中的 $1 —— 与 ingress-nginx rewrite-target 行为一致
  let newTarget = REWRITE_TARGET.replace('$1', currentPath.replace(REWRITE_REGEX, '$1'));

  if (newTarget === '') {
    return res.status(500).send('internal error');
  }

  // 若重写目标是绝对 URL 则 302 跳转（保留原查询串）
  if (/^https?:\/\//.test(newTarget)) {
    const parsed = new URL(newTarget);
    if (!parsed.search && req.url.includes('?')) {
      parsed.search = req.url.split('?')[1];
    }
    return res.redirect(302, parsed.toString());
  }

  // 直接把重写结果当作文件路径读取并返回
  const filePath = path.normalize(newTarget);
  fs.readFile(filePath, (err, data) => {
    if (err) return res.status(500).send('internal error');
    res.type(path.extname(filePath) || 'text/plain').send(data);
  });
});

app.listen(8080);
```
