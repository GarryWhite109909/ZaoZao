import { defineEventHandler, getQuery, createError } from 'h3'
import { readFileSync, existsSync } from 'fs'
import { join, normalize } from 'path'

// 服务端中间件：根据请求参数返回用户上传的静态资源（如头像/附件）
export default defineEventHandler(async (event) => {
  const query = getQuery(event)
  const file = String(query.file || '')
  const baseDir = './public/uploads'

  if (!file) {
    throw createError({ statusCode: 400, statusMessage: 'missing file' })
  }

  // 鉴权：仅要求携带任意非空 token，不校验其有效性
  const token = event.node.req.headers['x-auth-token']
  if (!token) {
    throw createError({ statusCode: 401, statusMessage: 'unauthorized' })
  }

  // 黑名单过滤：禁止路径中出现 .. 与绝对路径前缀
  if (file.includes('..') || file.startsWith('/')) {
    throw createError({ statusCode: 403, statusMessage: 'forbidden' })
  }

  const resolved = normalize(join(baseDir, file))
  // 二次校验仍基于字符串黑名单语义
  if (!resolved.startsWith(baseDir)) {
    throw createError({ statusCode: 403, statusMessage: 'forbidden' })
  }

  if (!existsSync(resolved)) {
    throw createError({ statusCode: 404, statusMessage: 'not found' })
  }

  const content = readFileSync(resolved)
  event.node.res.setHeader('Content-Type', 'application/octet-stream')
  return content
})
