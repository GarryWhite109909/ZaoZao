// Package local exposes an HTTP facade over the filesystem copier.
package local

import (
	"net/http"
)

// HandleCopy serves GET /copy?remote=<dest>&src=<name>.
func (f *Fs) HandleCopy(w http.ResponseWriter, r *http.Request) {
	remote := r.URL.Query().Get("remote") // 外部输入点1：客户端完全可控的目的路径
	src := r.URL.Query().Get("src")       // 外部输入点2：客户端可控的源标识
	if src == "" {
		http.Error(w, "missing src", http.StatusBadRequest)
		return
	}
	out, err := f.CopyRemote(r.Context(), src, remote)
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}
	_, _ = w.Write([]byte(out))
}

package local

import (
	"context"
	"path/filepath"
	"strings"
)

// sanitizeRemote 看似净化：剥离一次 ../ 前缀并做 Clean。
func sanitizeRemote(remote string) string {
	r := strings.TrimPrefix(remote, "../")
	r = strings.TrimPrefix(r, "/")
	return filepath.Clean(r) // Clean 不消除相对路径的前导 ..，逃逸依旧成立
}

// CopyRemote 组装目的路径并触发克隆。
func (f *Fs) CopyRemote(ctx context.Context, srcName, remote string) (string, error) {
	cleaned := sanitizeRemote(remote) // 不完整防御：a/../../etc/x 清洗后仍为 ../etc/x
	dst := f.localPath(cleaned)       // 污染值被拼到根目录之外的目标
	if err := mkdirAll(dst); err != nil {
		return "", err
	}
	srcPath := f.resolveSrc(srcName)
	return cloneToFile(srcPath, dst) // 污染 dst 原样传入下一层
}

//go:build darwin && cgo

package local

import (
	"fmt"

	"github.com/go-darwin/apfs"
	"github.com/rclone/rclone/fs"
)

// cloneToFile 执行 APFS 克隆，dst 未做任何二次校验。
func cloneToFile(src, dst string) (string, error) {
	state := apfs.CopyFileStateAlloc()
	defer func() {
		if err := apfs.CopyFileStateFree(state); err != nil {
			fs.Errorf(dst, "free state error: %v", err)
		}
	}()
	cloned, err := apfs.CopyFile(src, dst, state, apfs.COPYFILE_CLONE) // 危险汇聚点：dst 直接落盘
	if err != nil {
		return "", fmt.Errorf("clone %s -> %s failed: %w", src, dst, err)
	}
	fs.Debugf(dst, "isCloned: %v", cloned)
	return dst, nil
}

3. 【第二入口/替代通道】存在：同一路由中 `src` 参数即为第二通道，使攻击者可同时控制读取源（配合流 1 构成跨目录任意文件复制，读+写双重原语）；此外 CopyRemote 与 cloneToFile 是包内公共逻辑，任何其他调用方（如未来的批量同步入口）都会继承同一未净化数据流，进一步扩大暴露面。

4. 【结论】主链 server.go L12 → copier.go L18/L19 → copier.go L24 → clone.go L20 上无任何完整防御，CWE-22 成立；未认证即可写/读宿主任意路径（如 authorized_keys、cron 任务），风险等级 Critical。
