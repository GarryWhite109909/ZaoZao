# -*- coding: utf-8 -*-
from aiohttp import web

from app.file_service import load_local_file


async def handle_fetch(request):
    # 外部输入点 S1：GET 查询参数 path，攻击者完全可控
    path = request.query.get('path', '')
    result = await load_local_file(request.app['config'], path)
    if not result.get('successful'):
        return web.json_response({'error': 'not_found'}, status=404)
    return web.Response(body=result['buffer'],
                        content_type='application/octet-stream')


async def handle_thumb(request):
    # 第二入口 S2：路由段 {image} 同样进入同一加载链
    path = request.match_info.get('image', '')
    result = await load_local_file(request.app['config'], path)
    if not result.get('successful'):
        raise web.HTTPNotFound
    return web.Response(body=result['buffer'])


def make_app(config):
    app = web.Application()
    app['config'] = config
    app.router.add_get('/fetch', handle_fetch)
    app.router.add_get('/thumb/{image}', handle_thumb)
    return app

from os.path import abspath, join

from app.storage import read_checked_path


async def load_local_file(config, path):
    root = config.FILE_LOADER_ROOT_PATH.rstrip('/')
    file_path = abspath(join(root, path.lstrip('/')))

    # 看似净化：仅用 startswith 做根目录前缀检查（黑名单式弱防御）
    # 缺口1：未追加路径分隔符，前缀可比对到同级目录名
    # 缺口2：此处不解码，下游 unquote 还原出的穿越序列不受本检查约束
    if not file_path.startswith(abspath(root)):
        return {'successful': False, 'error': 'not_found'}

    # 过检后原样透传，链上不再复核
    return await read_checked_path(file_path)

# -*- coding: utf-8 -*-
from datetime import datetime
from os import fstat
from os.path import exists, isfile
from urllib.parse import unquote


async def read_checked_path(file_path):
    # 盲目信任上游已过检的标记，自身零校验
    if not exists(file_path):
        # 检查之后才解码：%2e%2e%2f 在此还原为 ../
        file_path = unquote(file_path)

    if exists(file_path) and isfile(file_path):
        with open(file_path, 'rb') as source_file:  # sink
            stats = fstat(source_file.fileno())
            return {
                'successful': True,
                'buffer': source_file.read(),
                'metadata': {
                    'size': stats.st_size,
                    'updated_at': datetime.utcfromtimestamp(stats.st_mtime),
                },
            }
    return {'successful': False, 'error': 'not_found'}
