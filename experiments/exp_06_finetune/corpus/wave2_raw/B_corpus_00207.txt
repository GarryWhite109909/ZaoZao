// 富文本编辑器字符串片段模型（拆分自 trix StringPiece）
const normalizeNewlines = (s) => (s == null ? "" : String(s).replace(/\r\n/g, "\n"))

export default class StringPiece {
  // 反序列化入口：pieceJSON 来自服务端文档快照 / 协同编辑消息（外部可控）
  static fromJSON(pieceJSON) {
    return new this(pieceJSON.string, pieceJSON.attributes)
  }

  constructor(string) {
    this.string = normalizeNewlines(string) // 仅归一化换行，非安全过滤
    this.length = this.string.length
  }

  getValue() {
    return this.string
  }

  toString() {
    return this.string.toString()
  }

  canBeConsolidatedWith(piece) {
    return !!piece && this.constructor === piece.constructor
  }

  consolidateWith(piece) {
    return new this.constructor(this.toString() + piece.toString(), this.attributes)
  }
}

# === file: app/services/text_pipeline.js ===
import StringPiece from "../models/string_piece"

// 所谓“净化”实为黑名单正则：仅删除成对 <script>…</script>，
// 对 <img onerror>、<svg onload>、未闭合标签等完全不设防 —— 可绕过
const stripScriptTags = (html) => html.replace(/<script[\s\S]*?<\/script>/gi, "")

// 把反序列化出的片段列表合并为一个 HTML 字符串
export function buildDocumentHTML(pieceList) {
  let merged = null
  for (const item of pieceList || []) {
    const piece = StringPiece.fromJSON(item) // 触达 source
    if (merged && merged.canBeConsolidatedWith(piece)) {
      merged = merged.consolidateWith(piece) // 字符串原样拼接
    } else {
      merged = piece
    }
  }
  return stripScriptTags(merged ? merged.getValue() : "") // 全链唯一防御点，覆盖不足
}

// 纯传递包装：不编码、不过滤
export function wrapInline(html) {
  return `<span class="doc">${html}</span>`
}

import { buildDocumentHTML, wrapInline } from "../services/text_pipeline"

export default class DocumentView {
  constructor(rootElement) {
    this.root = rootElement
  }

  // 入口一：打开已保存文档（pieces 为用户曾提交的历史数据）
  loadDocument(docJSON) {
    const html = buildDocumentHTML(docJSON.pieces)
    this.root.innerHTML = wrapInline(html) // SINK
  }

  // 入口二：应用协同编辑远端补丁，复用同一渲染链
  applyRemotePatch(patchJSON) {
    const html = buildDocumentHTML([patchJSON.piece])
    this.root.insertAdjacentHTML("beforeend", wrapInline(html)) // SINK 变体
  }
}
