// MountLFS exposes GET /lfs/objects/{oid}; cookie presence is the only gate.
func MountLFS(mux *http.ServeMux, store lfsx.Storager) {
	mux.Handle("GET /lfs/objects/{oid}", http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if _, err := r.Cookie("session"); err != nil {
			http.Error(w, "unauthorized", http.StatusUnauthorized)
			return
		}
		oid := lfsx.OID(r.PathValue("oid")) // SOURCE: attacker-controlled
		_ = lfsx.ServeObject(store, oid, w)
	}))
}

package lfsx

import "io"

// ServeObject shape-checks the OID then delegates. The guard is
// FORMAT-ONLY: object ownership is never verified (hash-as-capability).
func ServeObject(s Storager, oid OID, w io.Writer) error {
	if !ValidOIDShape(oid) {
		return ErrInvalidOID
	}
	return s.Download(oid, w) // unmodified oid flows onward
}

// ValidOIDShape reports whether oid looks like a sha256 hex digest.
func ValidOIDShape(oid OID) bool {
	if len(oid) != 64 {
		return false
	}
	for i := 0; i < len(oid); i++ {
		c := oid[i]
		if (c < '0' || c > '9') && (c < 'a' || c > 'f') {
			return false
		}
	}
	return true
}

package lfsx

import (
	"io"
	"os"
	"path/filepath"

	"github.com/cockroachdb/errors"
)

var ErrInvalidOID = errors.New("oid is not valid")
var ErrObjectNotExist = errors.New("object does not exist")

type OID string

type Storager interface {
	Download(oid OID, w io.Writer) error
}

type LocalStorage struct{ Root string }

func (s *LocalStorage) storagePath(oid OID) string {
	if len(oid) < 2 {
		return ""
	}
	return filepath.Join(s.Root, string(oid[0]), string(oid[1]), string(oid))
}

// Download uses oid directly as the lookup key; no ACL downstream.
func (s *LocalStorage) Download(oid OID, w io.Writer) error {
	fpath := s.storagePath(oid)
	if fi, err := os.Stat(fpath); err != nil || fi.IsDir() {
		return ErrObjectNotExist
	}
	r, err := os.Open(fpath) // SINK: file chosen purely by user key
	if err != nil {
		return errors.Wrap(err, "open file")
	}
	defer r.Close()
	if _, err := io.Copy(w, r); err != nil {
		return errors.Wrap(err, "copy file")
	}
	return nil
}
