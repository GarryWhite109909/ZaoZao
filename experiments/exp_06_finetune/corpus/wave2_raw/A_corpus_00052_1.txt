```php
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Facades\Artisan;
use Symfony\Component\HttpFoundation\StreamedResponse;

class DbController extends Controller
{
    // GET /admin/db/export?filename=xxx.zip
    public function exportContent(\Illuminate\Http\Request $request)
    {
        $filename = $request->query('filename');

        if ($filename) {
            if (!Storage::disk('backups')->exists($filename)) {
                abort(404);
            }

            return new StreamedResponse(function () use ($filename) {
                echo Storage::disk('backups')->get($filename);
            }, 200, [
                'Content-Type' => 'application/zip',
                'Content-Disposition' => 'attachment; filename="' . basename($filename) . '"'
            ]);
        }

        Artisan::call('db:backup');
        return response()->json(['ok' => true]);
    }
}
```

1. 第 12 行：`$request->query('filename')` 是外部输入入口（source），攻击者可通过 URL 查询参数传入任意字符串，如 `../../.env`。
2. 第 14-16 行：`Storage::disk('backups')->exists($filename)` 仅做存在性检查，Laravel 的 local 磁盘驱动在路径解析时不会阻止 `../` 越出配置的 root 目录，因此该检查不构成路径遍历防御。
3. 第 18-21 行：`Storage::disk('backups')->get($filename)` 直接以用户输入作为相对路径读取文件内容并流式返回（sink），`basename()` 只用于响应头文件名，未作用于实际读取路径，无任何白名单或 realpath 校验。
4. 第二入口分析：本控制器中无其他路由读取该磁盘；但同一 disk 若被其他控制器复用则同样可被利用；`Artisan::call('db:backup')` 分支不含用户输入，不可达 sink。主链 source→sink 完整可达且防御缺失。
