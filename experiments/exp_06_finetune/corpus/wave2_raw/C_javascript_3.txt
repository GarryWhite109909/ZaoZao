```code
// staticRouteValidator.js
// 常量路由表驱动的静态校验模块 —— 无外部输入入口

const fs = require('fs');
const { exec } = require('child_process');

// 静态路由表（编译期常量，非运行时输入）
const ROUTE_TABLE = Object.freeze({
  '/health':   { handler: 'healthCheck',  cmd: 'echo ok' },
  '/version':  { handler: 'readVersion',  file: './config/version.txt' },
  '/metrics':  { handler: 'readMetrics',  file: './config/metrics.json' }
});

const ALLOWED_HANDLERS = Object.freeze(['healthCheck', 'readVersion', 'readMetrics']);

function healthCheck() {
  // 危险 API：exec，实参为字面量常量
  exec('echo service-alive', (err, stdout) => {
    if (!err) console.log(stdout.trim());
  });
}

function readVersion() {
  const content = fs.readFileSync('./config/version.txt', 'utf8');
  console.log(content);
}

function readMetrics() {
  const raw = fs.readFileSync('./config/metrics.json', 'utf8');
  const data = JSON.parse(raw);
  console.log(Object.keys(data).length + ' metrics loaded');
}

function dispatch(route) {
  // route 仅由下方常量循环传入，非外部参数
  const entry = ROUTE_TABLE[route];
  if (!entry || !ALLOWED_HANDLERS.includes(entry.handler)) return false;
  if (entry.handler === 'healthCheck') healthCheck();
  else if (entry.handler === 'readVersion') readVersion();
  else if (entry.handler === 'readMetrics') readMetrics();
  return true;
}

// 模块自检：仅遍历常量表
Object.keys(ROUTE_TABLE).forEach(dispatch);

module.exports = { dispatch, ROUTE_TABLE };
```
