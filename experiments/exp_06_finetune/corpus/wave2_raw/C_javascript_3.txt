```code
// static_route_guard.js — 常量路由表驱动的静态校验模块
'use strict';

const fs = require('fs');
const { exec } = require('child_process');

// 全部为编译期写死的常量，无任何外部输入入口
const ALLOWED_ROUTES = Object.freeze({
  '/health': { handler: 'noop', timeout: 1000 },
  '/version': { handler: 'version', timeout: 500 },
  '/cleanup-tmp': { handler: 'cleanup', timeout: 3000 }
});

const SAFE_COMMANDS = Object.freeze({
  cleanup: 'rm -rf /tmp/app_cache',
  version: 'node --version'
});

function validateRoute(route) {
  // route 仅来自下方 main() 的字面量调用，非外部输入
  return Object.prototype.hasOwnProperty.call(ALLOWED_ROUTES, route);
}

function runHandler(name) {
  const cmd = SAFE_COMMANDS[name];
  if (typeof cmd !== 'string') {
    throw new Error('unknown handler: ' + name);
  }
  // 危险点1: exec —— 实参完全来自冻结常量表 SAFE_COMMANDS
  exec(cmd, { timeout: ALLOWED_ROUTES['/cleanup-tmp'].timeout }, (err) => {
    if (err) console.error('exec failed:', err.message);
  });
}

function readVersionFile() {
  // 危险点2: fs.readFileSync —— 路径为字面量常量
  return fs.readFileSync('/etc/app_version', 'utf8').trim();
}

function main() {
  const routes = ['/health', '/version', '/cleanup-tmp'];
  for (const r of routes) {
    if (!validateRoute(r)) continue;
    if (r === '/cleanup-tmp') runHandler('cleanup');
    if (r === '/version') console.log(readVersionFile());
  }
  // 危险点3: eval —— 仅用于解析内置常量 JSON，实参为字面量
  const meta = eval('({"build": "v1.2.3", "mode": "static"})');
  console.log('meta:', meta.build, meta.mode);
}

main();
```
