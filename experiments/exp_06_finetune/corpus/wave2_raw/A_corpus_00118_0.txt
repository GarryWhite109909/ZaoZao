```python
import pickle
from io import BytesIO

from fastapi import FastAPI, Request, HTTPException

app = FastAPI()

@app.post("/api/objects/decode")
async def decode_object(request: Request):
    # 接收客户端上传的序列化对象字节流（等价于 IoBuffer 外部输入）
    body = await request.body()
    if not body:
        raise HTTPException(status_code=400, detail="empty payload")

    stream = BytesIO(body)
    try:
        # 直接反序列化，未使用任何受限 Unpickler / 全局白名单过滤
        obj = pickle.load(stream)
    except Exception:
        raise HTTPException(status_code=422, detail="deserialization failed")

    return {"decoded_type": type(obj).__name__}
```
