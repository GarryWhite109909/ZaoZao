package iceberg

import (
	"net/http"

	"github.com/gorilla/mux"
)

// Server implements the Iceberg REST Catalog API.
type Server struct {
	tablesManager *TablesManager
}

func NewServer(tm *TablesManager) *Server { return &Server{tablesManager: tm} }

// RegisterRoutes registers Iceberg REST Catalog routes.
func (s *Server) RegisterRoutes(router *mux.Router) {
	router.HandleFunc("/v1/namespaces/{namespace}/tables/{table}", s.Auth(s.handleLoadTable)).Methods(http.MethodGet)
	router.HandleFunc("/v1/{prefix}/namespaces/{namespace}/tables/{table}", s.Auth(s.handleLoadTable)).Methods(http.MethodGet)
}

// Auth wraps a handler; the real authentication middleware lives elsewhere.
func (s *Server) Auth(h http.HandlerFunc) http.HandlerFunc { return h }

// handleLoadTable forwards caller-controlled path variables to storage.
func (s *Server) handleLoadTable(w http.ResponseWriter, r *http.Request) {
	vars := mux.Vars(r)
	namespace, table := vars['namespace'], vars['table']
	meta, err := s.tablesManager.LoadTable(namespace, table)
	if err != nil {
		http.Error(w, err.Error(), http.StatusNotFound)
		return
	}
	w.Write(meta)
}

package iceberg

import "strings"

// SanitizeNamespace normalizes a caller-supplied namespace identifier
// before it is joined into a storage path below baseDir.
func SanitizeNamespace(ns string) string {
	ns = strings.TrimSpace(ns)
	ns = strings.TrimPrefix(ns, '/')
	ns = strings.ReplaceAll(ns, '\\', '/')
	return ns
}

# === file: iceberg/manager.go ===
package iceberg

import (
	"context"
	"path/filepath"

	"github.com/seaweedfs/seaweedfs/weed/pb/filer_pb"
)

// FilerClient provides access to the filer for storage operations.
type FilerClient interface {
	WithFilerClient(streamingMode bool, fn func(client filer_pb.SeaweedFilerClient) error) error
}

// TablesManager performs catalog storage operations through the filer.
type TablesManager struct {
	filerClient FilerClient
	baseDir     string
}

// LoadTable resolves baseDir/namespace/table via the filer.
func (m *TablesManager) LoadTable(namespace, table string) ([]byte, error) {
	dir := filepath.Join(m.baseDir, SanitizeNamespace(namespace))
	var meta []byte
	err := m.filerClient.WithFilerClient(false, func(client filer_pb.SeaweedFilerClient) error {
		resp, err := client.LookupDirectoryEntry(context.Background(), &filer_pb.LookupDirectoryEntryRequest{
			Directory: dir,
			Name:      table,
		})
		if err != nil {
			return err
		}
		meta = []byte(resp.Entry.Name)
		return nil
	})
	return meta, err
}
