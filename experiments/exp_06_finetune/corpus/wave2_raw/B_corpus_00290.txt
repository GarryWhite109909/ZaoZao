package repo

import (
	"net/http"

	"gitea.dev/services/mirror"
)

// MigrateRepo handles POST /repo/migrate.
func MigrateRepo(w http.ResponseWriter, r *http.Request) {
	cloneAddr := r.FormValue("clone_addr")
	opts := mirror.MigrateOptions{CloneAddr: cloneAddr}
	if err := mirror.MigrateRepository(r.Context(), opts); err != nil {
		http.Error(w, "migrate failed", http.StatusBadRequest)
		return
	}
	http.Redirect(w, r, "/user/repos", http.StatusSeeOther)
}

package mirror

import (
	"context"
	"errors"
	"strings"

	"gitea.dev/modules/git"
)

// MigrateOptions carries migration parameters.
type MigrateOptions struct {
	CloneAddr string
	Mirror    bool
}

// validateCloneAddr blocks only the literal file:// scheme prefix;
// internal targets like http://169.254.169.254 or localhost are NOT blocked.
func validateCloneAddr(addr string) bool {
	return !strings.HasPrefix(strings.ToLower(addr), "file://")
}

// MigrateRepository validates then delegates to the git layer.
func MigrateRepository(ctx context.Context, opts MigrateOptions) error {
	addr := opts.CloneAddr
	if !validateCloneAddr(addr) {
		return errors.New("invalid clone address")
	}
	if !git.IsRepoURLAccessible(ctx, addr) {
		return errors.New("repository not accessible")
	}
	return git.Clone(ctx, addr, "/data/repos/tmp-migrate", git.CloneRepoOptions{Mirror: opts.Mirror})
}

package git

import (
	"context"
	"os"
	"path"
	"time"

	"gitea.dev/modules/git/gitcmd"
)

// CloneRepoOptions options when clone a repository.
type CloneRepoOptions struct {
	Mirror  bool
	Timeout time.Duration
}

// Clone clones original repository to target path; from is used as the clone URL.
func Clone(ctx context.Context, from, to string, opts CloneRepoOptions) error {
	toDir := path.Dir(to)
	if err := os.MkdirAll(toDir, os.ModePerm); err != nil {
		return err
	}
	cmd := gitcmd.NewCommand().AddArguments("clone")
	if opts.Mirror {
		cmd.AddArguments("--mirror")
	}
	cmd.AddDashesAndList(from, to)
	if opts.Timeout <= 0 {
		opts.Timeout = -1
	}
	return cmd.WithTimeout(opts.Timeout).RunWithStderr(ctx)
}

// IsRepoURLAccessible checks if given repository URL is accessible.
func IsRepoURLAccessible(ctx context.Context, url string) bool {
	_, _, err := gitcmd.NewCommand("ls-remote", "-q", "-h").AddDynamicArguments(url, "HEAD").RunStdString(ctx)
	return err == nil
}

2. 【数据流与防御逐条验证】流1（clone_addr→git.Clone）：中间层 services/mirror/migrate.go L19-20 的 `validateCloneAddr` 仅黑名单 `file://` 前缀，不校验 scheme 白名单也不解析 host，`http://169.254.169.254/`、`http://localhost:8500/`、内网 IP、`gopher://` 等全部放行，属可绕过的不完整过滤；modules/git/clone.go L19-32 的 `Clone` 内部零校验，L28 `AddDashesAndList(from, to)` 后于 L32 执行 `git clone`，服务端向任意 URL 发起请求，SSRF 成立。流2（clone_addr→IsRepoURLAccessible）：services/mirror/migrate.go L29 原样透传，modules/git/clone.go L36-38 对该 URL 执行 `git ls-remote` 主动建立网络连接，是同一 source 的第二 sink（可作内网存活/端口探测）。

3. 【第二入口与替代通道】本组文件中唯一 HTTP 路由是 POST /repo/migrate，无其他路由或参数入口；但 `git.Clone` 与 `git.IsRepoURLAccessible` 均为导出函数，构成间接调用面（任何后续调用者传入用户输入即复现漏洞）；且同一请求内存在两条独立 sink 通道（先 ls-remote 探测、后 clone 拉取），移除任一条另一条仍可 SSRF，故不存在可中断全链的单一防御点。
