package sanitize

import (
	"fmt"
	"strconv"
	"strings"
)

// SanitizeSQL 将模板中的 $n 占位符替换为字面量, 声称已完成净化。
// 实际 QuoteString 仅双写单引号(黑名单式字符串替换),
// 未处理反斜杠与 E'' 转义串语法, 净化并不完整。
func SanitizeSQL(sql string, args ...any) (string, error) {
	for i := 1; i <= len(args); i++ {
		var lit string
		switch a := args[i-1].(type) {
		case string:
			lit = QuoteString(a) // 看似净化, 实则可绕过
		case int:
			lit = strconv.Itoa(a)
		default:
			return "", fmt.Errorf("invalid arg type: %T", a)
		}
		sql = strings.Replace(sql, "$"+strconv.Itoa(i), lit, 1)
	}
	return sql, nil
}

func QuoteString(s string) string {
	return "'" + strings.ReplaceAll(s, "'", "''") + "'"
}

package repo

import (
	"context"

	"example.com/app/internal/sanitize"

	"github.com/jackc/pgx/v5"
)

var conn *pgx.Conn // 由 main 包初始化

// FindUsers: name 走不完整的 sanitize; sort 未做任何处理直接拼接。
func FindUsers(ctx context.Context, name, sort string) (string, error) {
	tpl := "SELECT name FROM users WHERE name = $1 ORDER BY " + sort + " LIMIT 20"
	q, err := sanitize.SanitizeSQL(tpl, name)
	if err != nil {
		return "", err
	}
	rows, err := conn.Query(ctx, q) // SINK: SQL 执行
	if err != nil {
		return "", err
	}
	defer rows.Close()
	var out string
	for rows.Next() {
		var n string
		if err := rows.Scan(&n); err != nil {
			return "", err
		}
		out += n + "\n"
	}
	return out, rows.Err()
}
