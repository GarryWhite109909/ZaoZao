```code
package main

import (
	"database/sql"
	"fmt"
	"log"

	_ "github.com/mattn/go-sqlite3"
)

// 迁移脚本：初始化审计日志表结构
const (
	migrationDriver = "sqlite3"
	migrationDSN    = "file:audit.db?mode=rwc"
)

var migrationDDL = []string{
	`CREATE TABLE IF NOT EXISTS audit_log (
		id INTEGER PRIMARY KEY AUTOINCREMENT,
		actor TEXT NOT NULL,
		action TEXT NOT NULL,
		payload TEXT,
		created_at DATETIME DEFAULT CURRENT_TIMESTAMP
	)`,
	`CREATE INDEX IF NOT EXISTS idx_audit_actor ON audit_log(actor)`,
}

func runMigrations(db *sql.DB) error {
	for _, ddl := range migrationDDL {
		if _, err := db.Exec(ddl); err != nil {
			return fmt.Errorf("migration failed: %w", err)
		}
	}
	return nil
}

func main() {
	db, err := sql.Open(migrationDriver, migrationDSN)
	if err != nil {
		log.Fatalf("open db: %v", err)
	}
	defer db.Close()
	if err := runMigrations(db); err != nil {
		log.Fatal(err)
	}
	fmt.Println("migrations applied")
}
```

2. 第 38 行 `sql.Open` 的 driver 与 DSN 均为常量 `migrationDriver`/`migrationDSN`，无环境变量、命令行参数或网络读取。
3. 污点分析：文件中不存在 request/argv/environ/stdin/socket 等任何输入入口；`runMigrations` 仅被 `main` 以固定参数调用，不导出、不被外部以可控参数调用，故不存在第二入口或替代通道。
4. 结论：所有危险 sink（Exec/Open）实参均为常量，无可达污点路径，样本不可利用。
