const express = require('express');
const router = express.Router();
const mobiledocLib = require('../../lib/mobiledoc/image-sizes');

// POST /ghost/api/admin/posts/preview - force-rerender editor payload
router.post('/preview', async (req, res) => {
    const mobiledocJson = req.body.mobiledoc;
    if (!mobiledocJson) {
        return res.status(400).json({error: 'mobiledoc required'});
    }

    // fills missing width/height on image cards before rerender
    const enriched = await mobiledocLib.populateImageSizes(mobiledocJson);
    return res.json({mobiledoc: enriched});
});

module.exports = router;
# === file: core/server/lib/mobiledoc/image-sizes.js ===
const logging = require('@tryghost/logging');
const storageUtils = require('../../adapters/storage/utils');
const {imageSize} = require('../image/image-size');

// used when force-rerendering post content to ensure that old image card
// payloads contain width/height values used when generating srcsets
const populateImageSizes = async function (mobiledocJson) {
    const mobiledoc = JSON.parse(mobiledocJson);

    const sizePromises = mobiledoc.cards.map(async (card) => {
        const [cardName, payload] = card;

        const needsFilling = cardName === 'image' && payload && payload.src && (!payload.width || !payload.height);
        if (!needsFilling) {
            return;
        }

        const isUnsplash = payload.src.match(/images\.unsplash\.com/);
        const isRelativeImagePath = /^\/content\/images\//.test(payload.src);
        try {
            let size;
            if (isUnsplash) {
                size = await imageSize.getImageSizeFromUrl(payload.src);
            } else if (isRelativeImagePath || storageUtils.isLocalImage(payload.src)) {
                size = await imageSize.getOriginalImageSizeFromStorageUrl(payload.src);
            } else {
                size = await imageSize.getImageSizeFromUrl(payload.src);
            }

            if (size && size.width && size.height) {
                payload.width = size.width;
                payload.height = size.height;
            }
        } catch (e) {
            logging.error(e);
        }
    });

    await Promise.all(sizePromises);

    return JSON.stringify(mobiledoc);
};

module.exports = {populateImageSizes};
# === file: core/server/lib/image/image-size.js ===
const request = require('@tryghost/request');
const imageSize = require('image-size');

// resolve a /content/images/... url to a local file and measure it
async function getOriginalImageSizeFromStorageUrl(url) {
    const buffer = await require('../../adapters/storage').readLocalFile(url);
    return imageSize(buffer);
}

// fetches a remote URL server-side to read image dimensions
async function getImageSizeFromUrl(url) {
    const res = await request(url, {method: 'GET'});
    const buffer = await res.body;
    return imageSize(buffer);
}

module.exports = {imageSize, getImageSizeFromUrl, getOriginalImageSizeFromStorageUrl};
