import Component from '@glimmer/component';
import {action} from '@ember/object';
import {tracked} from '@glimmer/tracking';
import {inject as service} from '@ember/service';
import {extractRenderedHTML} from '../../utils/revision-html';
import PreviewPanel from '../post-history/preview-panel';

export default class PostHistoryModal extends Component {
    @service modals;
    @tracked selectedHTML = null;
    @tracked selectedRevisionIndex = 0;

    constructor() {
        super(...arguments);
        this.post = this.args.model.post;
        this.editorAPI = this.args.model.editorAPI;
    }

    get revisionList() {
        const revisions = this.post.get('postRevisions').toArray()
            .sort((a, b) => b.get('createdAt') - a.get('createdAt'));
        return revisions.map((revision) => ({
            lexical: revision.get('lexical'),
            createdAt: revision.get('createdAt')
        }));
    }

    get selectedRevision() {
        return this.revisionList[this.selectedRevisionIndex];
    }

    @action
    async refreshPreview() {
        const state = this.editorAPI.editorInstance.parseEditorState(this.selectedRevision.lexical);
        this.editorAPI.editorInstance.setEditorState(state);
        const hidden = document.querySelector('.gh-post-history-hidden-lexical.current');
        this.selectedHTML = extractRenderedHTML(hidden);
        this.modals.open(PreviewPanel, {html: this.selectedHTML});
    }
}

# === file: app/utils/revision-html.js ===
// Shared helpers for post-history preview rendering.
const PLACEHOLDER_RE = /<div\b[^>]*>(\s*Begin writing your post\.\.\.\s*)<\/div>/i;

export function extractRenderedHTML(element) {
    if (!element) {
        return '';
    }
    const raw = element.innerHTML;
    // NOTE: only strips the Koenig placeholder block. This is NOT an HTML
    // sanitizer: script tags, onerror handlers and javascript: URLs survive.
    return raw.replace(PLACEHOLDER_RE, '');
}

export function freezeContenteditable(root) {
    root.querySelectorAll('[contenteditable]').forEach((el) => {
        el.setAttribute('contenteditable', 'false');
    });
}
