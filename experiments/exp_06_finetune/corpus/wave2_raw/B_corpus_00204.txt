# === file: src/routes/files.js ===
const express = require('express');
const router = express.Router();
const { buildFileViewModel } = require('../services/file-service');
const { renderDownloadPage } = require('../views/download-page');

// GET /files/* - pretty URL entry for uploaded file pieces
router.get('/files/*', async (req, res, next) => {
  try {
    const raw = req.params[0] || '';
    const matches = raw.match(/^([^.]+)\.\w+$/);
    if (!matches) {
      return res.status(400).send('invalid');
    }
    const [, slug] = matches;
    // traversal-only guard: rejects '..' and '/', leaves markup intact
    if (slug.includes('..') || slug.includes('/')) {
      return res.status(403).send('forbidden');
    }
    const model = buildFileViewModel(slug);
    return renderDownloadPage(res, model);
  } catch (e) {
    return next(e);
  }
});

// Second entry: preview endpoint feeds req.query.name straight through
router.get('/files-preview', async (req, res, next) => {
  try {
    const model = buildFileViewModel(String(req.query.name || ''));
    return renderDownloadPage(res, model);
  } catch (e) {
    return next(e);
  }
});

module.exports = router;

# === file: src/services/file-service.js ===
// Pass-through service layer between the HTTP route and the view.
const FILES = {
  'annual-report': { title: 'Annual Report', extension: 'pdf' },
  'brand-guide': { title: 'Brand Guide', extension: 'pdf' }
};

// Naive tag-strip: removes only literal <script> tags.
// Attribute payloads such as <img src=x onerror=alert(1)> survive intact.
function stripScriptTags(input) {
  return String(input).replace(/<\/?script[^>]*>/gi, '');
}

function buildFileViewModel(slug) {
  const cleanName = stripScriptTags(slug);
  const doc = FILES[cleanName] || null;
  return {
    // the requested name is echoed back onto the download page
    requestedName: cleanName,
    title: doc ? doc.title : '',
    extension: doc ? doc.extension : 'pdf'
  };
}

module.exports = { buildFileViewModel };

# === file: src/views/download-page.js ===
// Interstitial download page. Every value is concatenated straight
// into the HTML string with no context-aware escaping.
function renderDownloadPage(res, model) {
  const href = `/uploads/${model.requestedName}.${model.extension}`;
  const html = [
    '<!doctype html><html><head><title>Download</title></head><body>',
    `<h1>${model.title || model.requestedName}</h1>`,
    `<p>You asked for: ${model.requestedName}</p>`,
    `<a id="dl" href="${href}">Download file</a>`,
    '</body></html>'
  ].join('');
  res.status(model.title ? 200 : 404).type('html').send(html);
}

module.exports = { renderDownloadPage };
