```python
# app.py
from fastapi import FastAPI, HTTPException, Depends, Header
from pydantic import BaseModel
import httpx

app = FastAPI()

# 内部系统用户凭据，硬编码在源码中（等价于原 Java 模块的 system user）
INTERNAL_ADMIN_USER = "mod-export-system"
INTERNAL_ADMIN_PASSWORD = "SystemUser#2023!"

def require_internal_auth(x_internal_user: str = Header(None),
                          x_internal_pass: str = Header(None)):
    if x_internal_user != INTERNAL_ADMIN_USER or x_internal_pass != INTERNAL_ADMIN_PASSWORD:
        raise HTTPException(status_code=401, detail="unauthorized")
    return {"user": x_internal_user}

class ExportJob(BaseModel):
    dataset: str
    target_url: str

@app.post("/internal/export")
async def create_export(job: ExportJob, _=Depends(require_internal_auth)):
    async with httpx.AsyncClient() as client:
        resp = await client.post(job.target_url, json={"dataset": job.dataset})
    return {"status": resp.status_code}
```

1. 第 10-11 行是硬编码凭据（CWE-798 本体），第 15-19 行的 `require_internal_auth` 直接与这两个常量比较，任何拿到源码或反编译产物的人都能伪造内部调用头，认证形同虚设。
2. 外部输入点一：第 22-25 行 `ExportJob` 模型的 `target_url` 与 `dataset` 字段，由请求体 JSON 提供；第 28 行将其直接传入 `client.post` 的 URL 参数，构成 source→sink（SSRF）可达路径，且无任何白名单/域名校验。
3. 防御验证：唯一的“防御”是第 15-19 行的 Header 比较，但它依赖硬编码常量，属于可绕过（凭据泄露即全绕过），且对 `target_url` 完全没有过滤——黑名单/字符串过滤均不存在。
4. 第二入口：`/internal/export` 是唯一路由，但凭据本身可被用于直接构造对其他内部接口的调用；此外 `dataset` 原样转发到目标服务，构成间接注入通道。结论：漏洞成立。
