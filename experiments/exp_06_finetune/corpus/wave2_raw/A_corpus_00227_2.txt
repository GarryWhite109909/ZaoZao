```java
package com.example.social;

import org.springframework.web.bind.annotation.*;
import org.springframework.http.*;
import org.springframework.web.client.RestTemplate;
import org.springframework.cache.annotation.Cacheable;
import java.util.*;

@RestController
@RequestMapping("/api/social")
public class TwitterFeedController {

    private static final String API_URL = "https://api.twitter.com/1.1/search/tweets.json";

    // Hard-coded fallback credentials (CWE-798)
    private static final String FALLBACK_ACCESS_TOKEN = "220111598-87eLa7MgXZmd7YeRSkenTSVxhZikok61PXMKZFti";
    private static final String FALLBACK_TOKEN_SECRET = "KsDrxrxoGqVVK0ethvcTTrV58RBH3WUjnPeI616fnxIFS";
    private static final String FALLBACK_CONSUMER_KEY = "WgDmyOjMgX1N7RhcLpQqzUrtR";
    private static final String FALLBACK_CONSUMER_SECRET = "0e8PlzIeKlmGGyH1ajS2Ggaw0anPTX23p3gp2WqZ0PCNxkYYX1";

    @GetMapping("/tweets")
    @Cacheable(value = "twitterFeed", key = "#keyword + #count")
    public ResponseEntity<Map<String, Object>> getTweets(
            @RequestParam(required = false) String keyword,
            @RequestParam(defaultValue = "5") int count) {

        String accessToken = System.getenv("TWITTER_ACCESS_TOKEN");
        if (accessToken == null || accessToken.isEmpty()) {
            accessToken = FALLBACK_ACCESS_TOKEN;   // falls back to hard-coded secret
        }
        String consumerKey = System.getenv("TWITTER_CONSUMER_KEY");
        if (consumerKey == null || consumerKey.isEmpty()) {
            consumerKey = FALLBACK_CONSUMER_KEY;   // same pattern for all four credentials
        }

        RestTemplate rest = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.set("Authorization", "Bearer " + accessToken);
        headers.set("X-Consumer-Key", consumerKey);

        String url = API_URL + "?count=" + count + "&q=" + keyword;
        Map<String, Object> result = rest.getForObject(url, Map.class);
        return ResponseEntity.ok(result);
    }
}
```

分析：
1. 第 30 行：`@GetMapping("/tweets")` 是 Spring MVC 的地道路由注册方式，`@RequestParam` 直接从 HTTP 请求获取外部输入 `keyword` 和 `count`，构成 source。
2. 第 38–43 行：当环境变量缺失时，代码回退到第 19–22 行硬编码的 OAuth token / consumer key 常量，与原 PHP 中 `get_option(...)==false` 时赋值硬编码凭据的语义结构完全一致。
3. 第 46 行：外部输入 `keyword` 未经任何校验/编码直接拼入请求 URL，对应原代码 `$getfield = '?count=' . $count . '&q=' . $query;` 的无防御拼接（原代码虽有 urlencode，此处同样缺失有效防御）。
4. 第 48 行：`rest.getForObject(url, ...)` 使用携带硬编码凭据的 header 发起对外 API 请求，是 sink——任何拿到源码/反编译 jar 的人都能提取这些真实凭据并冒充应用调用 Twitter API。
5. 全程没有对凭据做加密、vault 注入或配置外置校验，防御缺失与原始漏洞等价。
