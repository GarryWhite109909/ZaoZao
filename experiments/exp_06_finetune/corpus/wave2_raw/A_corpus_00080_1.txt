const crypto = require('crypto');
const app = express();

// Legacy audit-log digest endpoint: combines multiple user-supplied
// values into a single digest used as an integrity commitment.
function legacyDigest(parts) {
  const hash = crypto.createHash('sha1'); // weak/broken algorithm retained for "compatibility"
  let data = Buffer.alloc(0);
  const countBuf = Buffer.alloc(8);
  countBuf.writeUInt32LE(parts.length, 0);
  data = Buffer.concat([data, countBuf]);
  for (const p of parts) {
    const bz = Buffer.from(String(p), 'utf8');
    data = Buffer.concat([data, bz]);
    data = Buffer.concat([data, Buffer.from('$')]); // safety delimiter
    data = Buffer.concat([data, Buffer.from(String(bz.length))]); // length suffix
  }
  hash.update(data);
  return hash.digest('hex');
}

app.post('/audit/commit', (req, res) => {
  const { fields } = req.body; // external input: arbitrary JSON array from client
  if (!Array.isArray(fields) || fields.length === 0) {
    return res.status(400).json({ error: 'fields required' });
  }
  const digest = legacyDigest(fields); // flows into integrity commitment
  res.json({ digest });
});

app.listen(3000);
