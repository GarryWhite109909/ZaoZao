package server

import (
	"archive/zip"
	"net/http"
)

type Server struct {
	UploadRoot string
	Webroot    string
}

// handlePut stores the request body at a client-chosen URL path.
func (s *Server) handlePut(w http.ResponseWriter, r *http.Request) {
	target := Resolve(s.UploadRoot, r.URL.Path)
	if err := WriteFile(target, r.Body); err != nil {
		http.Error(w, "upload failed", http.StatusInternalServerError)
		return
	}
	w.WriteHeader(http.StatusCreated)
}

// handleBulk streams selected files back as a zip archive.
func (s *Server) handleBulk(w http.ResponseWriter, r *http.Request) {
	w.Header().Set("Content-Type", "application/zip")
	zw := zip.NewWriter(w)
	defer zw.Close()
	for _, name := range r.URL.Query()["file"] {
		path := Resolve(s.Webroot, name)
		if err := AddToArchive(zw, path); err != nil {
			continue
		}
	}
}

package server

import (
	"path/filepath"
	"strings"
)

// Resolve strips traversal sequences and joins onto root.
// This is the single shared guard for every user-supplied path.
func Resolve(root, raw string) string {
	cleaned := strings.ReplaceAll(raw, "../", "")
	return filepath.Join(root, cleaned)
}
# === file: server/store.go ===
package server

import (
	"archive/zip"
	"io"
	"os"
	"path/filepath"
)

// WriteFile creates or truncates path and copies body into it.
func WriteFile(path string, body io.Reader) error {
	f, err := os.OpenFile(path, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0644)
	if err != nil {
		return err
	}
	defer f.Close()
	_, err = io.Copy(f, body)
	return err
}

// AddToArchive appends the file at path to the zip writer.
func AddToArchive(zw *zip.Writer, path string) error {
	f, err := os.Open(path)
	if err != nil {
		return err
	}
	defer f.Close()
	entry, err := zw.Create(filepath.Base(path))
	if err != nil {
		return err
	}
	_, err = io.Copy(entry, f)
	return err
}
