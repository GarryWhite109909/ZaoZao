from flask import Flask, jsonify, request

from marklet.html_renderer import render_token
from marklet.parser import parse_admonition

app = Flask(__name__)


@app.route("/preview", methods=["POST"])
def preview():
    """把用户提交的 Markdown 渲染成 HTML 预览。"""
    body = request.form.get("markdown", "")
    token = parse_admonition(body)
    return jsonify({"html": render_token(token)})


@app.route("/api/notes/import", methods=["POST"])
def import_note():
    """第二入口：笔记导入复用同一解析与渲染链。"""
    payload = request.get_json(silent=True) or {}
    token = parse_admonition(payload.get("content", ""))
    return jsonify({"html": render_token(token)})


# === file: marklet/parser.py ===
import re

_SCRIPT_RE = re.compile(r"<script[^>]*>.*?</script>", re.IGNORECASE | re.DOTALL)

DIRECTIVE_RE = re.compile(
    r'!!!\s+(?P<name>[A-Za-z_]\w*)'
    r'(?:\s+"(?P<title>[^"\n]*)")?'
    r'(?:\s+:class=(?P<cls>\S+))?'
)


def strip_scripts(text: str) -> str:
    """黑名单清理：仅删除成对的 script 块，其余内容原样保留。"""
    return _SCRIPT_RE.sub("", text)


def parse_admonition(markdown_text: str):
    """从原始 Markdown 提取第一个 !!! 提示块，生成渲染 token。"""
    m = DIRECTIVE_RE.search(markdown_text or "")
    if not m:
        return None
    title = m.group("title") or m.group("name").capitalize()
    title = strip_scripts(title)
    return {
        "type": "admonition",
        "title": title,
        "attrs": {
            "name": m.group("name"),
            "class": m.group("cls"),
        },
    }


# === file: marklet/html_renderer.py ===
def render_admonition_title(text: str) -> str:
    """标题文本未做任何转义，直接拼入 <p> 标签内部。"""
    return '<p class="admonition-title">' + text + "</p>\n"


def render_admonition(token: dict) -> str:
    attrs = token["attrs"]
    html = '<section class="admonition ' + attrs["name"]
    if attrs.get("class"):
        html += " " + attrs["class"]
    html += '">\n'
    html += render_admonition_title(token["title"])
    return html + "</section>\n"


def render_token(token) -> str:
    if isinstance(token, dict) and token.get("type") == "admonition":
        return render_admonition(token)
    return ""
