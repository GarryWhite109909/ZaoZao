```code
// middleware.ts —— Next.js 中间件：路由白名单 + 鉴权 + XML 文本抽取（模拟 partition_xml）
import { NextRequest, NextResponse } from 'next/server';
import { parseXmlString } from 'libxmljs';

const ROUTE_WHITELIST = new Set(['/api/xml/partition']);
const TOKEN_WHITELIST = new Set(['svc-internal-2024']);

async function extractLeafTexts(xmlBody: string): Promise<string[]> {
  // 直接使用 libxmljs 默认解析器：未禁用外部实体（noent 未关闭、未设置非 DTD 模式）
  const doc = parseXmlString(xmlBody);
  const nodes = doc.find('//*[text()]');
  return nodes.map((n) => n.text().trim()).filter(Boolean);
}

export async function middleware(req: NextRequest) {
  const pathname = req.nextUrl.pathname;

  if (!ROUTE_WHITELIST.has(pathname)) {
    return NextResponse.json({ error: 'not found' }, { status: 404 });
  }

  const token = req.headers.get('x-api-token') ?? '';
  if (!TOKEN_WHITELIST.has(token)) {
    return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  }

  if (req.method !== 'POST') {
    return NextResponse.json({ error: 'method not allowed' }, { status: 405 });
  }

  const rawXml = await req.text();
  const elements = await extractLeafTexts(rawXml);

  return NextResponse.json({ elements });
}

export const config = {
  matcher: ['/api/xml/:path*'],
};
```

分析：
1. 第 26 行 `req.nextUrl.pathname` 与第 29 行 `x-api-token` 头是仅有的外部输入点；第 37 行 `await req.text()` 将请求体完整读入，是主数据流的 source。
2. 第 38 行把原始请求体直接传入 `extractLeafTexts`，第 11 行 `parseXmlString(xmlBody)` 是 sink——libxmljs 默认解析会处理 DOCTYPE 与外部实体定义，攻击者可在 body 中内嵌 `<!ENTITY xxe SYSTEM 'file:///etc/passwd'>` 并在叶节点文本中引用，实现任意文件读取/SSRF。
3. 白名单与 token 校验（第 27、31 行）只作用于“是否允许进入处理逻辑”，属于鉴权层防御，对请求体内容不做任何类型或结构校验；source 到 sink 之间没有任何针对 DOCTYPE/ENTITY 的过滤、禁用参数或 schema 校验，防御覆盖面为零。
4. 第二入口确认：matcher 配置（第 45 行）只匹配 `/api/xml/*`，白名单内唯一路由即 `/api/xml/partition`；不存在其他路由或间接调用绕过路径，但这也意味着该唯一入口就是完整的可达链路，结论不受影响。
5. 综上：外部输入（POST body）→ 无净化 → 危险 XML 解析，与原代码中 lxml `iterparse` 接收用户文件并默认展开实体的漏洞语义结构完全等价。
