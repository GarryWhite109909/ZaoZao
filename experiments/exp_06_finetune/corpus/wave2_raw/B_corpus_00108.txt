package demo.web;
import java.util.Map;
import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import demo.exec.RequestExecutionService;
import demo.graphql.DefaultGraphQlRequest;

/** Entry point: accepts a raw GraphQL-over-HTTP body at POST /graphql. */
public class GraphQlHttpController {

	private final RequestExecutionService executionService = new RequestExecutionService();
	private final ObjectMapper json = new ObjectMapper();

	public Map<String, Object> handlePost(String httpBody) throws Exception {
		Map<String, Object> payload = json.readValue(httpBody,
				new TypeReference<Map<String, Object>>() {});
		String document = (String) payload.getOrDefault("query", "");
		String operationName = (String) payload.get("operationName");
		@SuppressWarnings("unchecked")
		Map<String, Object> variables =
				(Map<String, Object>) payload.getOrDefault("variables", Map.of());
		@SuppressWarnings("unchecked")
		Map<String, Object> extensions =
				(Map<String, Object>) payload.getOrDefault("extensions", Map.of());
		DefaultGraphQlRequest request = new DefaultGraphQlRequest(
				document, operationName, variables, extensions);
		return executionService.execute(request);
	}
}

package demo.graphql;
import java.util.LinkedHashMap;
import java.util.Map;

// Minimal port of org.springframework.graphql.GraphQlRequest
interface GraphQlRequest {
	String getDocument();
	String getOperationName();
	Map<String, Object> getVariables();
	Map<String, Object> getExtensions();
}

/** Holds the client-supplied parts of a GraphQL request. */
public class DefaultGraphQlRequest implements GraphQlRequest {

	private final String document;
	private final String operationName;
	private final Map<String, Object> variables;
	private final Map<String, Object> extensions;

	public DefaultGraphQlRequest(String doc, String op,
			Map<String, Object> vars, Map<String, Object> ext) {
		this.document = (doc == null) ? "" : doc.trim();
		this.operationName = (op == null || op.isBlank()) ? null : op.trim();
		// shallow copy: nested values pass through untouched, no type check
		this.variables = new LinkedHashMap<>(vars);
		this.extensions = new LinkedHashMap<>(ext);
	}

	@Override public String getDocument() { return document; }
	@Override public String getOperationName() { return operationName; }
	@Override public Map<String, Object> getVariables() { return variables; }
	@Override public Map<String, Object> getExtensions() { return extensions; }
}

package demo.exec;
import java.util.Map;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.jsontype.impl.LaissezFaireSubTypeValidator;
import demo.graphql.GraphQlRequest;

/** Executes a request; resolves polymorphic types found in client JSON. */
public class RequestExecutionService {

	// default typing + laissez-faire validator honors any '@class' subtype
	private final ObjectMapper mapper = new ObjectMapper().activateDefaultTyping(
			LaissezFaireSubTypeValidator.instance, ObjectMapper.DefaultTyping.NON_FINAL);

	public Map<String, Object> execute(GraphQlRequest request) {
		try {
			String rawVars = mapper.writeValueAsString(request.getVariables());
			Object typedVars = mapper.readValue(rawVars, Object.class); // CWE-502 sink
			String rawExt = mapper.writeValueAsString(request.getExtensions());
			mapper.readValue(rawExt, Object.class); // second sink, same flaw
			return Map.of("ok", String.valueOf(typedVars));
		}
		catch (Exception ex) {
			throw new IllegalStateException("execution failed", ex);
		}
	}
}
