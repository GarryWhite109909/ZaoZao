```code
// static-route-validator.js
// 常量路由表驱动的静态校验模块：所有危险 API 实参均为编译期常量

const ROUTE_TABLE = Object.freeze({
  '/health':   { cmd: 'echo ok',        script: 'return 1;' },
  '/version':  { cmd: 'node --version', script: 'return process.version;' },
  '/ping':     { cmd: 'echo pong',      script: 'return true;' }
});

const ALLOWED_MODULES = Object.freeze(['fs', 'path']);

function validateRoute(route) {
  // route 仅来自下方 MAIN_ROUTES 常量数组，非外部输入
  return Object.prototype.hasOwnProperty.call(ROUTE_TABLE, route);
}

function runDiagnostics() {
  const results = [];
  for (const route of ['health', 'version']) {
    const key = '/' + route;
    if (!validateRoute(key)) continue;
    const entry = ROUTE_TABLE[key];
    // 危险点1：eval —— 实参为冻结常量表中的字面量字符串
    const fn = new Function(entry.script);
    // 危险点2：child_process —— 实参为字面量命令，shell 固定为 false
    require('child_process').execSync(entry.cmd, { shell: false });
    results.push({ route: key, ok: fn() });
  }
  return results;
}

// 模块白名单校验：仅允许冻结列表中的字面量模块名
function loadAllowed(name) {
  if (ALLOWED_MODULES.includes(name)) {
    return require(name); // name 只能是 'fs' 或 'path'
  }
  throw new Error('module not allowed');
}

const report = runDiagnostics();
console.log(JSON.stringify(report));
```

1. 第 20 行 `new Function(entry.script)`：`entry` 来自第 3-7 行 `Object.freeze` 的 `ROUTE_TABLE`，其 `script` 字段均为硬编码字面量（如 `'return 1;'`），且循环变量 `route` 来自第 18 行的常量数组 `['health','version']`，无任何外部输入可改变该值。
2. 第 23 行 `execSync(entry.cmd, { shell: false })`：`cmd` 同样取自冻结常量表中的字面量（`'echo ok'` 等），且 `shell: false` 进一步限制执行方式；文件中不存在 request/argv/env/stdin 等任何污染源，故无污点可达该 sink。
3. 第 31 行 `require(name)`：`name` 受第 30 行白名单约束，仅可能为冻结数组中的 `'fs'` 或 `'path'` 字面量，无法注入任意模块路径。
4. 综上，全部危险 API 的实参链路均终止于编译期常量，不存在从外部输入到危险 sink 的可达路径。
