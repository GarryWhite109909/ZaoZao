const fs = require('fs');
const { exec } = require('child_process');

// 全部为硬编码常量，编译期即确定，运行期不可变
const ALLOWED_ROUTES = Object.freeze({
  '/health': 'ok',
  '/version': 'v1.0.0',
});

const AUDIT_LOG_PATH = '/var/log/app/audit.log';
const HEALTHCHECK_CMD = 'node --version';

function validateRoute(route) {
  // 仅查表比较，route 只能来自下方 main() 的字面量调用
  return Object.prototype.hasOwnProperty.call(ALLOWED_ROUTES, route);
}

function writeAuditLog(entry) {
  // entry 由调用方传入字面量字符串
  fs.appendFileSync(AUDIT_LOG_PATH, entry + '
');
}

function runHealthcheck() {
  // 危险 API：实参为模块级常量 HEALTHCHECK_CMD，非拼接、非外部输入
  exec(HEALTHCHECK_CMD, (err, stdout) => {
    if (!err) {
      writeAuditLog('healthcheck: ' + stdout.trim());
    }
  });
}

function main() {
  const testRoutes = ['/health', '/version', '/unknown'];
  for (const r of testRoutes) {
    console.log(r, validateRoute(r));
  }
  runHealthcheck();
}
