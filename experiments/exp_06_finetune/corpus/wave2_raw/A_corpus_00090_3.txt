package com.vms.admin;

import jakarta.servlet.http.*;
import org.springframework.context.annotation.Configuration;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.servlet.HandlerInterceptor;
import org.springframework.web.servlet.config.annotation.InterceptorRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;
import java.util.Map;

@Configuration
public class CategoryAdminConfig implements WebMvcConfigurer {

    static class AdminAuthInterceptor implements HandlerInterceptor {
        @Override
        public boolean preHandle(HttpServletRequest req, HttpServletResponse res, Object handler) throws Exception {
            HttpSession session = req.getSession(false);
            if (session == null || session.getAttribute("userId") == null) {
                res.sendError(401);
                return false;
            }
            if (!Boolean.TRUE.equals(session.getAttribute("canManageCategories"))) {
                res.sendError(403);
                return false;
            }
            return true;
        }
    }

    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        registry.addInterceptor(new AdminAuthInterceptor()).addPathPatterns("/admin/**");
    }
}

@RestController
class CategoryAdminController {

    private final JdbcTemplate jdbc;

    CategoryAdminController(JdbcTemplate jdbc) { this.jdbc = jdbc; }

    @PostMapping("/admin/categories/delete")
    public Map<String, Object> delete(@RequestParam(name = "id", defaultValue = "0") long id) {
        int n = jdbc.update("DELETE FROM categories WHERE id = ?", id);
        jdbc.update("DELETE FROM category_assets WHERE category_id = ?", id);
        if (n > 0) {
            return Map.of("error", false, "msg", "");
        }
        return Map.of("error", true, "msg", "Error on delete");
    }
}
