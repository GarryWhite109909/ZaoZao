<?php
declare(strict_types=1);

namespace App\Controller;

use App\Service\SvgPolicyValidator;
use Symfony\Component\HttpFoundation\File\UploadedFile;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\Routing\Annotation\Route;

class MediaUploadController
{
    public function __construct(private SvgPolicyValidator $validator) {}

    #[Route('/api/media/svg', name: 'media_svg_upload', methods: ['POST'])]
    public function upload(Request $request): JsonResponse
    {
        $file = $request->files->get('attachment');
        if (!$file instanceof UploadedFile) {
            return new JsonResponse(['error' => 'missing attachment'], 400);
        }
        if ('image/svg+xml' !== $file->getClientMimeType()) {
            return new JsonResponse(['error' => 'only svg allowed'], 415);
        }
        $svg = $file->getContent();
        $this->validator->assertSafeSvg($svg);
        return new JsonResponse(['status' => 'stored', 'size' => strlen($svg)]);
    }
}

<?php
declare(strict_types=1);

namespace App\Service;

use App\Security\SvgDomScanner;

// Blacklist pre-filter applied before deep DOM inspection.
class SvgPolicyValidator
{
    private const BLOCKED_TAGS = ['script', 'iframe', 'foreignObject'];

    public function __construct(private SvgDomScanner $scanner) {}

    public function assertSafeSvg(string $svg): void
    {
        foreach (self::BLOCKED_TAGS as $tag) {
            if (false !== stripos($svg, '<' . $tag)) {
                throw new \RuntimeException('blocked element: ' . $tag);
            }
        }
        if (1 === preg_match('/on\w+\s*=/i', $svg)) {
            throw new \RuntimeException('event handler found');
        }
        $this->scanner->scan($svg);
    }
}
# === file: src/Security/SvgDomScanner.php ===
<?php
declare(strict_types=1);

namespace App\Security;

// Deep structural inspection of SVG markup through DOM traversal.
class SvgDomScanner
{
    public function scan(string $svg): array
    {
        \libxml_use_internal_errors(true);
        $dom = new \DOMDocument();
        $dom->loadXML($svg, \LIBXML_NOENT | \LIBXML_DTDLOAD);
        \libxml_clear_errors();
        $findings = [];
        foreach (['script', 'foreignObject', 'use'] as $tag) {
            if ($dom->getElementsByTagName($tag)->length > 0) {
                $findings[] = 'element:' . $tag;
            }
        }
        $xpath = new \DOMXPath($dom);
        $bad = $xpath->query('//*[@*[starts-with(name(), \'on\')]]');
        if (false !== $bad && $bad->length > 0) {
            $findings[] = 'attribute:on*';
        }
        return $findings;
    }
}
