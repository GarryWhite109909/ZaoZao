'use strict';
// Fake defense: the blacklist only covers spaces, backticks and single
// quotes; <, > and double quotes survive, and when rawHeaderId=true is
// set by the user this function degenerates into a pure pass-through.
function sanitizeHeaderId (rawId, opts) {
  let id = String(rawId);
  if (opts && opts.rawHeaderId) {
    return id.replace(/\s/g, '-'); // spaces -> dashes only
  }
  return id.replace(/[`'\s]/g, '-'); // < > and double quotes untouched
}

export { sanitizeHeaderId };

function renderHeading (text, level, opts) {
  const id = sanitizeHeaderId(text, opts);
  const prefix = opts.prefixHeaderId === true ? 'section-' : (opts.prefixHeaderId || '');
  return '<h' + level + ' id="' + prefix + id + '">' + text + '</h' + level + '>';
}

function renderMarkdown (mdText, opts) {
  const lines = String(mdText).split('\n');
  let html = '';
  for (const line of lines) {
    const m = line.match(/^(#{1,6})\s+(.*)$/);
    html += m ? renderHeading(m[2], m[1].length, opts) : '<p>' + line + '</p>';
  }
  return html;
}

function initPreview () {

}
