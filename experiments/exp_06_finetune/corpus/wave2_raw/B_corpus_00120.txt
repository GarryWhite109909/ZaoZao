package org.apache.camel.component.infinispan.remote;

import org.infinispan.notifications.Listener;
import org.infinispan.notifications.cachelistener.annotation.CacheEntryCreated;
import org.infinispan.notifications.cachelistener.annotation.CacheEntryModified;
import org.infinispan.notifications.cachelistener.event.CacheEntryEvent;

/**
 * Listens for replicated exchange payloads arriving over the HotRod wire.
 */
@Listener
public class InfinispanRemoteEventListener {

    private final ExchangePayloadRouter router;

    public InfinispanRemoteEventListener(ExchangePayloadRouter router) {
        this.router = router;
    }

    @CacheEntryCreated
    public void onCreated(CacheEntryEvent<String, byte[]> event) {
        dispatch(event);
    }

    @CacheEntryModified
    public void onModified(CacheEntryEvent<String, byte[]> event) {
        dispatch(event);
    }

    private void dispatch(CacheEntryEvent<String, byte[]> event) {
        byte[] raw = event.getValue();
        if (raw != null && raw.length > 0) {
            router.route(raw);
        }
    }
}

package org.apache.camel.component.infinispan.remote;

import org.apache.camel.component.infinispan.remote.protostream.DefaultExchangeHolderUtils;

/**
 * Triages incoming payloads and forwards them to the marshaller layer.
 */
public class ExchangePayloadRouter {

    private static final int MAX_BYTES = 65536;

    public void route(byte[] payload) {
        if (!accepted(payload)) {
            return;
        }
        DefaultExchangeHolderUtils.deserialize(payload);
    }

    /**
     * Triage inspects only transport framing: serialization magic bytes and
     * total size. Payload contents are never validated or whitelisted, so a
     * crafted ObjectInputStream body still reaches the sink unchanged.
     */
    private boolean accepted(byte[] payload) {
        return payload.length <= MAX_BYTES
                && payload[0] == (byte) 0xac
                && payload[1] == (byte) 0xed;
    }
}

package org.apache.camel.component.infinispan.remote.protostream;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.ObjectInputStream;

import org.apache.camel.support.DefaultExchangeHolder;
import org.apache.camel.util.ClassLoadingAwareObjectInputStream;

/** Marshalling helpers for {@link DefaultExchangeHolder} over Protostream transports. */
final class DefaultExchangeHolderUtils {

    private DefaultExchangeHolderUtils() {
    }

    public static DefaultExchangeHolder deserialize(byte[] bytes) throws IOException {
        try (ByteArrayInputStream bais = new ByteArrayInputStream(bytes);
             ObjectInputStream ois = new ClassLoadingAwareObjectInputStream(bais)) {
            return (DefaultExchangeHolder) ois.readObject();
        } catch (ClassNotFoundException e) {
            throw new IOException(e);
        }
    }
}
