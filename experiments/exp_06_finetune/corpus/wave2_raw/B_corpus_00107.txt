<?php
declare(strict_types=1);

namespace App\Cache;

use App\Cache\Backend\DatabaseBackend;

/** 缓存存取封装：写入时序列化，读取时只回传原始字符串（纯传递层） */
class CacheStorage
{
    public function __construct(private DatabaseBackend $backend) {}

    public function set(string $entryIdentifier, mixed $data): void
    {
        // 仅校验键名字符集，对缓存值本身无任何约束（防御面不完整）
        if (!preg_match('/^[A-Za-z0-9_.-]{1,250}$/', $entryIdentifier)) {
            throw new \InvalidArgumentException('invalid cache identifier');
        }
        $this->backend->store($entryIdentifier, serialize($data));
    }

    public function getRaw(string $entryIdentifier): string|false
    {
        return $this->backend->fetch($entryIdentifier);
    }
}

<?php
declare(strict_types=1);

namespace App\Service;

use App\Cache\CacheStorage;

/** 从缓存读取小部件配置并还原为对象图 */
class WidgetRenderer
{
    private const DEFAULT_KEY = 'widget.default';

    public function __construct(private CacheStorage $storage) {}

    public function render(string $requestedKey): mixed
    {
        $key = $requestedKey !== '' ? $requestedKey : self::DEFAULT_KEY;
        $raw = $this->storage->getRaw($key);
        if ($raw === false || $raw === '') {
            return null;
        }
        // 未传 allowed_classes 白名单，攻击者预置的序列化对象在此被完整还原
        return unserialize($raw);
    }
}
