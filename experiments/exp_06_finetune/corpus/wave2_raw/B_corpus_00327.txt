<?php declare(strict_types=1);

namespace App\DependencyResolver;

use App\Package\LockFileWriter;

/**
 * 重组自 Composer LockTransaction：保留 updateMirrorAndUrls 的
 * URL 白名单正则与黑名单式替换净化。
 */
class LockTransaction
{
    private LockFileWriter $writer;

    public function __construct(LockFileWriter $writer)
    {
        $this->writer = $writer;
    }

    public function updateMirrorAndUrls(string $packageName, string $distUrl): void
    {
        // 看似净化：黑名单单轮替换，....// 删除一次后仍剩 ../，可绕过
        $safeName = str_replace(['../', '..\\'], '', $packageName);

        // 原版白名单正则只约束 distUrl 内容，不约束 $safeName 的路径语义
        if (preg_match('{^https?://(?:(?:www\.)?bitbucket\.org|(api\.)?github\.com|(?:www\.)?gitlab\.com)/}i', $distUrl)) {
            $distUrl = preg_replace('{(?<=/|sha=)[a-f0-9]{40}(?=/|$)}i', 'HEAD', $distUrl);
        }

        // 污点继续向 sink 传播
        $this->writer->writeLock($safeName, json_encode(['dist_url' => $distUrl], JSON_UNESCAPED_SLASHES));
    }
}

# === file: src/Package/LockFileWriter.php ===
<?php declare(strict_types=1);

namespace App\Package;

/**
 * 锁文件落盘：CWE-22 污点汇聚点。
 */
class LockFileWriter
{
    private string $baseDir;

    public function __construct(?string $baseDir = null)
    {
        $this->baseDir = $baseDir ?? sys_get_temp_dir() . '/app-lock';
    }

    public function writeLock(string $packageName, string $content): void
    {
        // SINK：$packageName 直接拼入路径，无 realpath/基于 baseDir 的前缀校验
        $path = $this->baseDir . '/' . $packageName . '.json';
        if (!is_dir($this->baseDir)) {
            mkdir($this->baseDir, 0755, true);
        }
        file_put_contents($path, $content);
    }
}
