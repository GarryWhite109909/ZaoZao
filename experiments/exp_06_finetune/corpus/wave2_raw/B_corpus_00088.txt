package httpapi

import (
	"io"
	"net/http"
	"path/filepath"
	"strings"

	"goshs.de/goshs/v2/internal/storage"
)

func (fs *FileServer) RegisterRoutes(mux *http.ServeMux) {
	mux.Handle("/upload", Guard(http.HandlerFunc(fs.upload))) // POST /upload: guarded (handler in upload.go)
	mux.HandleFunc("/put/", fs.put)                           // PUT: NOT wrapped by Guard (legacy clients)
}

// put stores a raw-body upload at a caller-chosen path under UploadFolder.
func (fs *FileServer) put(w http.ResponseWriter, req *http.Request) {
	if fs.ReadOnly {
		http.Error(w, "read only", http.StatusForbidden)
		return
	}
	rel := strings.TrimPrefix(req.URL.Path, "/put/")
	savepath := filepath.Join(fs.UploadFolder, rel) // attacker-controlled destination
	body, err := io.ReadAll(req.Body)               // SOURCE: attacker-controlled bytes
	if err != nil {
		http.Error(w, "bad body", http.StatusBadRequest)
		return
	}
	defer req.Body.Close()
	if !fs.applyCustomAuth(w, req) { // ACL only; NO CSRF token check on this route
		return
	}
	if err := storage.WriteUpload(savepath, body); err != nil { // tainted data reaches sink
		http.Error(w, "write failed", http.StatusInternalServerError)
		return
	}
	w.WriteHeader(http.StatusCreated)
}

package httpapi

import (
	"crypto/subtle"
	"net/http"
)

// Guard validates a double-submit CSRF token. Sound implementation, but its
// coverage depends on wiring: only POST /upload is wrapped in routes.go,
// so every other state-changing route silently bypasses it.
func Guard(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
		c, err := req.Cookie("goshs_session")
		if err != nil {
			http.Error(w, "forbidden", http.StatusForbidden)
			return
		}
		tok := req.Header.Get("X-CSRF-Token")
		if subtle.ConstantTimeCompare([]byte(tok), []byte(c.Value)) != 1 {
			http.Error(w, "csrf mismatch", http.StatusForbidden)
			return
		}
		next.ServeHTTP(w, req)
	})
}

// applyCustomAuth: per-directory ACL pass-through, not bound to session/origin.
func (fs *FileServer) applyCustomAuth(w http.ResponseWriter, req *http.Request) bool {
	return req.Header.Get("X-ACL") != "deny"
}

package storage

import (
	"bytes"
	"io"
	"os"
)

// WriteUpload persists caller-supplied bytes at the caller-chosen path.
func WriteUpload(savepath string, body []byte) error {
	// #nosec G304
	dst, err := os.OpenFile(savepath, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0644) // SINK
	if err != nil {
		return err
	}
	defer dst.Close()
	_, err = io.Copy(dst, bytes.NewReader(body))
	return err
}
