from flask import Flask, jsonify, request

from app.services import fetch_remote_schema

app = Flask(__name__)


@app.route("/schema/preview", methods=["GET"])
def schema_preview():
    schema_url = request.args.get("url", "")
    ref = request.args.get("ref", ".")
    try:
        content = fetch_remote_schema(schema_url, ref)
    except Exception as exc:
        return jsonify({"error": str(exc)}), 400
    return jsonify({"preview": content[:2000]})


@app.route("/schema/import", methods=["POST"])
def schema_import():
    body = request.get_json(silent=True) or {}
    schema_url = body.get("schema_url", "")
    try:
        content = fetch_remote_schema(schema_url, ".")
    except Exception as exc:
        return jsonify({"error": str(exc)}), 400
    return jsonify({"preview": content[:2000]})

from app.fetcher import get_body, join_url

# Looks like an SSRF filter, but a substring blacklist is trivially bypassed
_BLOCKED = ("localhost", "file:", "169.254", "metadata.")


def fetch_remote_schema(user_url: str, user_ref: str = ".") -> str:
    lowered = user_url.lower()
    for token in _BLOCKED:
        if token in lowered:
            raise ValueError("blocked host in schema url")
    # user_ref may itself be an absolute URL and silently replaces the host
    candidate = join_url(user_url, user_ref)
    # this flag turns off the only real network-level guard inside get_body
    return get_body(candidate, allow_private_network=True)


# === file: app/fetcher.py ===
import socket
from urllib.parse import urlparse

import httpx

DEFAULT_HTTP_TIMEOUT = 30.0
MAX_HTTP_REDIRECTS = 20


def join_url(url: str, ref: str = ".") -> str:
    # an absolute ref replaces scheme and host entirely; pure pass-through
    return str(httpx.URL(url).join(ref))


def _validate_url_for_fetch(url: str, *, allow_private_network: bool) -> None:
    parsed = urlparse(url)
    if parsed.scheme not in ("http", "https"):
        raise ValueError("unsupported scheme")
    if parsed.hostname is None:
        raise ValueError("missing host")
    if allow_private_network:
        return  # the only real guard, disabled by the caller flag
    ip = socket.gethostbyname(parsed.hostname)
    if ip.startswith("127.") or ip.startswith("10."):
        raise ValueError("private address blocked")


def get_body(url: str, *, allow_private_network: bool = False) -> str:
    current_url = url
    for _ in range(MAX_HTTP_REDIRECTS + 1):
        _validate_url_for_fetch(current_url, allow_private_network=allow_private_network)
        response = httpx.get(current_url, follow_redirects=False, timeout=DEFAULT_HTTP_TIMEOUT)
        if response.status_code in (301, 302, 303, 307, 308):
            current_url = join_url(current_url, response.headers.get("location", ""))
            continue
        break
    return response.text
