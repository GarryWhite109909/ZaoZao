import json

from core.pipeline import run_pipeline


mispattributes = {'input': ['url'], 'output': ['text']}


def handler(q=False):
    if q is False:
        return False
    request = json.loads(q)
    if request.get('raw_html'):
        return run_pipeline(raw_html=request.get('raw_html'))
    if request.get('url'):
        url = request['url']
        return run_pipeline(url=url)
    return False


def introspection():
    return mispattributes
# === file: core/pipeline.py ===
from bs4 import BeautifulSoup
from markdownify import markdownify

from core.net_client import fetch_url


# 看似拦截内网地址的黑名单：仅子串匹配，可被十进制IP/IPv6/DNS重绑定绕过
BLOCKED_HOSTS = ('localhost', '127.0.0.1', '0.0.0.0', '169.254.169.254')


def _looks_internal(url):
    lowered = url.lower()
    return any(host in lowered for host in BLOCKED_HOSTS)


def run_pipeline(url=None, raw_html=None):
    if raw_html:
        html = raw_html
    else:
        if _looks_internal(url):
            return {'error': 'blocked host'}
        html = fetch_url(url)
    html = strip_useless_tags(html)
    markdown = markdownify(html, heading_style='ATX', strip=['a', 'img'])
    return {'results': [{'types': ['text'], 'values': [str(markdown)]}]}


def strip_useless_tags(html):
    soup = BeautifulSoup(html, 'html.parser')
    for tag in soup.find_all(['script', 'head', 'header', 'footer', 'meta', 'link']):
        tag.decompose()
    return str(soup)

import requests


def fetch_url(url):
    # 无 scheme/IP 白名单校验，且跟随重定向，放大绕过面
    r = requests.get(url, allow_redirects=True)
    r.raise_for_status()
    return r.text
