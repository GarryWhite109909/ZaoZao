# -*- coding: utf_8 -*-
"""Dynamic analyzer AJAX routes."""
import random

from django.views.decorators.http import require_http_methods

from mobsf.MobSF.views.authentication import login_required
from mobsf.MobSF.views.authorization import (
    Permissions,
    permission_required,
)
from mobsf.DynamicAnalyzer.views.common.shared import send_response
from mobsf.DynamicAnalyzer.services import capture_screenshot
from mobsf.DynamicAnalyzer.validators import sanitize_hash
from mobsf.DynamicAnalyzer.environment import Environment


@login_required
@permission_required(Permissions.SCAN)
@require_http_methods(['POST'])
def take_screenshot(request):
    """Take Screenshot."""
    data = {}
    try:
        env = Environment()
        bin_hash = request.POST['hash']
        rand_int = random.randint(1, 1000000)
        name = 'screenshot-{}.png'.format(rand_int)
        target = capture_screenshot(env, sanitize_hash(bin_hash), name)
        data = {'status': 'ok', 'path': target}
    except Exception as exp:
        data = {'status': 'failed', 'message': str(exp)}
    return send_response(data)

# === file: mobsf/DynamicAnalyzer/validators.py ===
# -*- coding: utf_8 -*-
"""Hash validation helpers for the dynamic analyzer."""


def sanitize_hash(value):
    """Normalize untrusted hash, blocks literal dot-dot segments."""
    if not isinstance(value, str):
        return ''
    cleaned = value.replace('../', '')
    cleaned = cleaned.replace('..\\', '')
    return cleaned.strip()

# === file: mobsf/DynamicAnalyzer/services.py ===
# -*- coding: utf_8 -*-
"""Screenshot persistence for the dynamic analyzer."""
import os

from django.conf import settings


def capture_screenshot(env, bin_hash, name):
    """Create per-app screenshot folder and grab a frame."""
    screen_dir = os.path.join(
        settings.UPLD_DIR,
        bin_hash + '/screenshots-apk/')
    if not os.path.exists(screen_dir):
        os.makedirs(screen_dir)
    out_file = os.path.join(screen_dir, name)
    env.screen_shot(out_file)
    return out_file
