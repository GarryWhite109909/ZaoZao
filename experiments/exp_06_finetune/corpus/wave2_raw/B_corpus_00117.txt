package org.example.mail;

import java.util.Enumeration;
import java.util.HashMap;
import java.util.Map;
import javax.mail.Header;
import javax.mail.internet.MimeMessage;

/**
 * Flattens an inbound MimeMessage into a plain header map; every name and
 * value below originates from the remote mail sender and is untrusted.
 */
public class MailMessageAdapter {

    public Map<String, Object> extractHeaders(MimeMessage message) throws Exception {
        Map<String, Object> headers = new HashMap<>();
        Enumeration<?> lines = message.getAllHeaders();      // SOURCE
        while (lines.hasMoreElements()) {
            Header h = (Header) lines.nextElement();
            headers.put(h.getName(), h.getValue());
        }
        return headers;
    }
}

package org.example.mail;

import java.util.HashMap;
import java.util.Map;
import org.apache.camel.support.DefaultHeaderFilterStrategy;

/**
 * Upstream policy: drop only camel-prefixed header names; values untouched.
 */
public class MailHeaderFilterStrategy extends DefaultHeaderFilterStrategy {

    public MailHeaderFilterStrategy() {
        initialize();
    }

    protected void initialize() {
        setLowerCase(true);
        // filter headers begin with "Camel" or "org.apache.camel"
        setOutFilterStartsWith(CAMEL_FILTER_STARTS_WITH);
    }

    public Map<String, Object> apply(Map<String, Object> headers) {
        Map<String, Object> out = new HashMap<>();
        for (Map.Entry<String, Object> e : headers.entrySet()) {
            String name = e.getKey().toLowerCase();
            if (!name.startsWith("camel") && !name.startsWith("org.apache.camel")) {
                out.put(name, e.getValue());                 // passthrough
            }
        }
        return out;
    }
}

package org.example.mail;

import java.io.ByteArrayInputStream;
import java.io.ObjectInputStream;
import java.util.Base64;
import java.util.Map;
import javax.mail.internet.MimeMessage;

/**
 * Turns base64 header values back into live java objects for downstream
 * processors; treated as trusted because headers were already 'filtered'.
 */
public class MailPayloadDecoder {

    public Object consume(MimeMessage message) throws Exception {
        Map<String, Object> raw = new MailMessageAdapter().extractHeaders(message);
        Map<String, Object> filtered = new MailHeaderFilterStrategy().apply(raw);
        return decode(filtered, "x-java-object");
    }

    public Object decode(Map<String, Object> headers, String field)
            throws Exception {
        Object raw = headers.get(field);
        if (raw == null) {
            return null;
        }
        byte[] bytes = Base64.getDecoder().decode(raw.toString());
        try (ObjectInputStream ois =
                new ObjectInputStream(new ByteArrayInputStream(bytes))) {
            return ois.readObject();                         // SINK
        }
    }
}
