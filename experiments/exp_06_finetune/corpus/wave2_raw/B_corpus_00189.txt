<?php
namespace App\Http;

use App\Support\PathFilter;
use App\Reports\Gitblame;

/**
 * Renders per-file version-control blame reports.
 */
class BlameController
{
    public function show(): void
    {
        $raw = $_GET['file'] ?? '';
        if ($raw === '') {
            http_response_code(400);
            exit('missing file parameter');
        }
        $clean  = PathFilter::clean($raw);
        $report = new Gitblame();
        header('Content-Type: text/plain');
        echo implode(PHP_EOL, $report->generate($clean));
    }
}

# === file: src/Support/PathFilter.php ===
<?php
namespace App\Support;

/**
 * Lightweight guard applied before invoking VCS tools.
 */
final class PathFilter
{
    public static function clean(string $path): string
    {
        $path = str_replace(['../', '..\\'], '', $path);
        $path = preg_replace('/[;&|`]/', '', $path);
        return ltrim($path, '/');
    }
}

# === file: src/Reports/Gitblame.php ===
<?php
namespace App\Reports;

/**
 * Produces git blame output for one tracked file.
 */
class Gitblame
{
    public function generate(string $filename): array
    {
        return $this->getBlameContent($filename);
    }

    protected function getBlameContent(string $filename): array
    {
        $cwd = getcwd();
        chdir(dirname($filename));
        $command = 'git blame --date=short "' . basename($filename) . '" 2>&1';
        $handle  = popen($command, 'r');
        if ($handle === false) {
            chdir($cwd);
            return ['ERROR: blame command failed'];
        }
        $raw = stream_get_contents($handle);
        pclose($handle);
        chdir($cwd);

        return explode("\n", $raw);
    }
}
