```code
#!/usr/bin/env python3
"""deploy.py - 固定流程部署脚本，所有命令均为硬编码常量。"""

import subprocess
import logging

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
logger = logging.getLogger("deploy")

# 部署配置常量（部署目标固定为内网构建机）
DEPLOY_HOST = "build-01.internal.corp"
REMOTE_DIR = "/srv/app/releases"
ARCHIVE_NAME = "app-bundle.tar.gz"

# 硬编码命令序列（全部为字面量常量）
COMMANDS = [
    ["ssh", DEPLOY_HOST, "mkdir", "-p", REMOTE_DIR],
    ["scp", ARCHIVE_NAME, f"{DEPLOY_HOST}:{REMOTE_DIR}/"],
    ["ssh", DEPLOY_HOST, "tar", "-xzf", f"{REMOTE_DIR}/{ARCHIVE_NAME}", "-C", REMOTE_DIR],
    ["ssh", DEPLOY_HOST, "systemctl", "restart", "app.service"],
]


def run_step(cmd):
    """执行单条命令；cmd 来自模块级常量列表 COMMANDS。"""
    logger.info("executing: %s", " ".join(cmd))
    result = subprocess.run(
        cmd,
        shell=False,          # 不经过 shell 解析
        capture_output=True,
        text=True,
        timeout=120,
        check=True,
    )
    return result.stdout


def main():
    for cmd in COMMANDS:
        try:
            out = run_step(cmd)
            if out:
                logger.info("output: %s", out.strip())
        except subprocess.CalledProcessError as exc:
            logger.error("step failed: %s (rc=%s)", cmd[0], exc.returncode)
            raise SystemExit(1)


if __name__ == "__main__":
    main()
```
