import subprocess

# 部署配置常量（写死在文件中）
DEPLOY_DIR = "/opt/myapp"
BACKUP_DIR = "/var/backups/myapp"
SERVICE_NAME = "myapp.service"
GIT_BRANCH = "release/1.4.2"

def stop_service():
    # 危险点1: subprocess.run, 实参为字面量列表常量

    subprocess.run(
        ["rsync", "-a", DEPLOY_DIR + "/", BACKUP_DIR + "/"],
        check=True,
    )

def pull_latest():
    # 危险点3: subprocess.run, 实参为字面量列表 + 常量 GIT_BRANCH
    subprocess.run(
        ["git", "-C", DEPLOY_DIR, "fetch", "origin", GIT_BRANCH],
        check=True,
    )
    subprocess.run(
        ["git", "-C", DEPLOY_DIR, "checkout", GIT_BRANCH],
        check=True,
    )

def start_service():
    # 危险点4: subprocess.run, 实参为字面量列表常量
    subprocess.run(["sudo", "systemctl", "start", SERVICE_NAME], check=True)

if __name__ == "__main__":
    stop_service()
    backup_current()
    pull_latest()
    start_service()
