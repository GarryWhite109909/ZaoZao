CONFIG_PATH = "/etc/deploy/app.conf"
LOG_DIR = "/var/log/deploy"

def load_config():
    # 危险 API：open()，但路径为模块级常量字面量
    with open(CONFIG_PATH, "r", encoding="utf-8") as f:

def check_log_dir():

    if not os.path.isdir(LOG_DIR):

    return LOG_DIR

    log_dir = check_log_dir()
    print("config loaded:", len(config_text), "chars")
    print("log dir ready:", log_dir)

if __name__ == "__main__":
    main()
