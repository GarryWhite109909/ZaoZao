```code
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""内部工具：读取固定路径配置文件并执行初始化 SQL。"""

import os

# 固定配置路径（部署时由运维写入，非用户可控）
CONFIG_PATH = "/etc/myapp/app.conf"
SQL_INIT_PATH = "/opt/myapp/sql/init.sql"

def load_config():
    # 危险点1: open() —— 实参为模块级常量 CONFIG_PATH
    with open(CONFIG_PATH, "r", encoding="utf-8") as f:
        return f.read()

def run_init_sql():
    # 危险点2: os.system() —— 实参为拼接的常量字符串，无任何变量参与
    os.system("psql -f /opt/myapp/sql/init.sql -d myapp")

    # 危险点3: open() 读取固定 SQL 文件，仅打印不执行动态内容
    with open(SQL_INIT_PATH, "r", encoding="utf-8") as f:
        print(f.read())

if __name__ == "__main__":
    cfg = load_config()
    print("config loaded, length =", len(cfg))
    run_init_sql()
```
