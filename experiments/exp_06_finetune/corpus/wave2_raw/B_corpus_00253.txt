<?php
namespace YesWiki\Core\Service;

// Resolves the wiki page tag for the current request.
// Legacy URLs like /?wiki=MaPage must keep working, so the raw
// request value is returned as-is.
class PageTagResolver
{
    public function __construct(private $wiki) {}

    public function resolveTag()
    {
        if (!empty($_REQUEST['wiki'])) {
            return $_REQUEST['wiki'];
        }
        if (!empty($_GET['page'])) {
            return $_GET['page'];
        }
        return $this->wiki->config['default_page'] ?? 'HomePage';
    }
}

<?php
// Handler for /MaPage/deletepage — confirm and delete a wiki page.
use YesWiki\Core\Service\PageTagResolver;
use YesWiki\Core\Repository\BacklinkRepository;

$resolver  = $this->services->get(PageTagResolver::class);
$backlinks = $this->services->get(BacklinkRepository::class);

// current page tag, taken verbatim from the request (legacy ?wiki= URLs)
$tag = $resolver->resolveTag();
// strip a query-string fragment if someone wrote ?wiki=Foo?x=1
$tag = strtok($tag, '?');

// secondary input, only used later for the post-delete redirect
$incomingurl = !empty($_REQUEST['incomingurl'])
    ? filter_var($_REQUEST['incomingurl'], FILTER_VALIDATE_URL) : '';

if ($this->UserIsOwner() || $this->UserIsAdmin()) {
    if (!$this->IsOrphanedPage($tag)) {
        // page still has backlinks: list them so the owner can erase them
        $linkedFrom = $backlinks->findByTarget($tag);
        $msg = '<p><em>' . _t('DELETEPAGE_NOT_ORPHEANED') . "</em></p>\n";
        $msg .= "<ul>\n";
        foreach ($linkedFrom as $page) {
            $msg .= '<li>' . $this->ComposeLinkToPage($page['from_tag'], '', '', 0) . "</li>\n";
        }
        $msg .= "</ul>\n";
    } else {
        // orphaned page: CSRF-checked deletion flow (unchanged)
        $msg = _t('DELETEPAGE_CONFIRM');
    }
} else {
    $msg = '<p><em>' . _t('DELETEPAGE_NOT_OWNER') . "</em></p>\n";
}

echo $this->Header();
echo "<div class=\"page\">\n";
echo $msg;
echo "</div>\n";
echo $this->Footer();

<?php
namespace YesWiki\Core\Repository;

// Read/write access to the {prefix}links table (page backlinks).
class BacklinkRepository
{
    public function __construct(private $wiki) {}

    // Every page whose link points at $tag.
    public function findByTarget($tag)
    {
        $prefix = $this->wiki->config['table_prefix'];
        $query = 'SELECT DISTINCT from_tag FROM ' . $prefix . 'links'
               . " WHERE to_tag = '" . $tag . "'";
        return $this->wiki->LoadAll($query);
    }

    // Remove backlinks pointing at a deleted page (eraselink=oui).
    public function eraseLinksFor($tag)
    {
        $prefix = $this->wiki->config['table_prefix'];
        $query = 'DELETE FROM ' . $prefix . 'links'
               . " WHERE to_tag = '" . $tag . "'";
        return $this->wiki->Query($query);
    }
}
