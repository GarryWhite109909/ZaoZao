from flask import Flask, request, render_template_string

from app.markdown_directives import render_markdown

app = Flask(__name__)


@app.route("/render", methods=["POST"])
def render():
    # 外部输入点：用户提交的 Markdown 正文，可携带 !!! include 指令
    content = request.form.get("content", "")
    html = render_markdown(content)
    return render_template_string("<div>{{ doc }}</div>", doc=html)


@app.route("/preview", methods=["POST"])
def preview():
    # 第二入口：预览接口复用同一渲染管线
    content = request.form.get("body", "")
    return render_markdown(content)

# === file: app/markdown_directives.py ===
import re
from typing import Any, Dict, Match

from app.include_resolver import read_include

INCLUDE_RE = re.compile(r"^!!!\s+include\s+(?P<title>.+?)\s*$", re.M)


def parse_include(m: Match[str]) -> Dict[str, Any]:
    relpath = m.group("title").strip()  # 污点进入指令层，strip 仅去空白
    try:
        content = read_include(relpath)  # 未做任何路径校验，原样透传
    except FileNotFoundError:
        return {"type": "block_error", "raw": "Could not find file: " + relpath}
    return {"type": "include", "raw": content}


def render_include(text: str) -> str:
    return '<pre class="directive-include">\n' + text + "</pre>\n"


def render_markdown(text: str) -> str:
    parts = []
    last = 0
    for m in INCLUDE_RE.finditer(text):
        parts.append(text[last:m.start()])
        tok = parse_include(m)
        if tok["type"] == "include":
            parts.append(render_include(tok["raw"]))
        else:
            parts.append("<!-- " + tok["raw"] + " -->")
        last = m.end()
    parts.append(text[last:])
    return "".join(parts)

# === file: app/include_resolver.py ===
import os


def read_include(relpath: str, base_dir: str = "docs") -> str:
    dest = os.path.join(base_dir, relpath)
    dest = os.path.normpath(dest)  # 伪净化：归一化但不限制在 base_dir 内
    if not os.path.isfile(dest):
        raise FileNotFoundError(dest)
    with open(dest, "rb") as f:  # sink：按攻击者可控路径读取文件
        return f.read().decode("utf-8")

4. **第二入口与替代通道**：/preview（app/routes.py L16-L20）与 /render 共享同一 render_markdown→parse_include→read_include→open 调用链，构成确认的第二入口；read_include 的 base_dir 参数虽可配置，但当前两个调用方均使用默认值 'docs'，不存在任何调用点传入已净化的绝对路径，故所有通道均无有效防御，漏洞成立。
