<?php
declare(strict_types=1);

namespace Plugin\MineAdmin\AppStore\Controller;

use App\Http\Common\Result;
use Hyperf\HttpServer\Annotation\Controller;
use Hyperf\HttpServer\Annotation\PostMapping;
use Plugin\MineAdmin\AppStore\Service\AppService;

#[Controller(prefix: 'admin/plugin/store')]
class AppStoreController
{
    public function __construct(private AppService $service) {}

    #[PostMapping('download')]
    public function download(): Result
    {
        // source: HTTP 请求参数直接进入服务层，无任何校验
        $params = $this->request->all();
        $result = $this->service->downloadPackage($params);
        return $this->success(['result' => $result]);
    }

    private function success(array $data = []): Result
    {
        return new Result($data);
    }
}

# === file: app/Service/AppService.php ===
<?php
declare(strict_types=1);

namespace Plugin\MineAdmin\AppStore\Service;

use Plugin\MineAdmin\AppStore\Support\PathResolver;

class AppService
{
    public function __construct(private PathResolver $resolver) {}

    public function downloadPackage(array $params): array
    {
        $identifier = (string) ($params['identifier'] ?? '');
        // 看似净化：仅做字符串替换过滤 ../ 与 ..\ —— 黑名单替换可被嵌套绕过（....//）
        $cleaned = str_replace(['../', '..\'], '', $identifier);
        // 纯传递：将"净化后"的值交给路径解析器
        $path = $this->resolver->resolveDownloadPath($cleaned);
        return ['file' => $path];
    }
}

# === file: app/Support/PathResolver.php ===
<?php
declare(strict_types=1);

namespace Plugin\MineAdmin\AppStore\Support;

class PathResolver
{
    private string $baseDir = BASE_PATH . '/runtime/appstore/';

    public function resolveDownloadPath(string $name): string
    {
        // sink: 拼接后直接用于文件读取，无 realpath 前缀校验
        $full = $this->baseDir . $name;
        if (!is_file($full)) {
            throw new \RuntimeException('app package not found');
        }
        return file_get_contents($full);
    }
}
