namespace Plugin\MineAdmin\AppStore\Service;

use Plugin\MineAdmin\AppStore\Support\PathResolver;

class DownloadService
{
    public function __construct(private PathResolver $pathResolver)

    }

    public function fetchPackage(string $packageName): array

# === file: app/Support/PathResolver.php ===
<?php
declare(strict_types=1);

namespace Plugin\MineAdmin\AppStore\Support;

class PathResolver
{
    private string $baseDir = BASE_PATH . '/runtime/packages';

    public function resolve(string $name): string
    {
        // 直接拼接后用于文件读取，存在路径穿越（sink）
        $target = $this->baseDir . '/' . $name;
        if (! is_file($target)) {
            throw new \RuntimeException('package not found');
        }
        return file_get_contents($target);
    }
}
