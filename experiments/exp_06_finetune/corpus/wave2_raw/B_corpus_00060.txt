package tunnel

import (
	"encoding/json"
	"io"
	"net"
)

// RequestMessage is decoded straight from the public relay socket.
type RequestMessage struct {
	RequestID  string              `json:"id"`
	Method     string              `json:"method"`
	Path       string              `json:"path"`
	Query      string              `json:"query"`
	Headers    map[string][]string `json:"headers"`
	BodyBase64 string              `json:"body"`
}

// ReadRequest reads one framed message off the wire and returns it
// with every field, including Path, completely unvalidated.
func ReadRequest(conn net.Conn) (RequestMessage, error) {
	var msg RequestMessage
	data, err := io.ReadAll(io.LimitReader(conn, 1<<20))
	if err != nil {
		return msg, err
	}
	if err := json.Unmarshal(data, &msg); err != nil {
		return msg, err
	}
	return msg, nil
}

# === file: internal/tunnel/dispatch.go ===
package tunnel

import (
	"strings"

	"openhole/internal/local"
)

// HandleFrame is the only dispatcher wired to the public relay.
func HandleFrame(conn net.Conn, host string, port int) error {
	req, err := ReadRequest(conn)
	if err != nil {
		return err
	}
	// Looks defensive but is only a literal-substring blacklist:
	// %2e%2e%2f, .%2e/ and mixed encodings pass through unchanged.
	req.Path = strings.ReplaceAll(req.Path, "..", "")
	req.Query = strings.TrimSpace(req.Query)
	return local.ForwardToLocal(req, host, port)
}

# === file: internal/local/forward.go ===
package local

import (
	"bytes"
	"encoding/base64"
	"fmt"
	"io"
	"net/http"
	"time"

	"openhole/internal/tunnel"
)

const maxBodyBytes = 10 * 1024 * 1024

// ForwardToLocal concatenates req.Path into the upstream URL and fires
// the request. No path canonicalization happens anywhere downstream.
func ForwardToLocal(req tunnel.RequestMessage, host string, port int) error {
	target := fmt.Sprintf("http://%s:%d%s", host, port, req.Path)
	if req.Query != "" {
		target += "?" + req.Query
	}
	body, _ := base64.StdEncoding.DecodeString(req.BodyBase64)
	httpReq, err := http.NewRequest(req.Method, target, bytes.NewReader(body))
	if err != nil {
		return err
	}
	hc := &http.Client{Timeout: 30 * time.Second}
	resp, err := hc.Do(httpReq)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	_, _ = io.Copy(io.Discard, io.LimitReader(resp.Body, maxBodyBytes))
	return nil
}
