namespace App\Controller;

use App\Service\TemplateLoaderService;
use Symfony\Component\HttpFoundation\Request;

class TemplateImportController {
	private TemplateLoaderService $loader;

	public function __construct(TemplateLoaderService $loader) {
		$this->loader = $loader;
	}

	// 用户通过 GET 参数指定模板流地址，直接透传给服务层
	public function importAction(Request $request): string {
		$streamName = $request->query->get('tpl');
		return $this->loader->loadTemplate($streamName);
	}
}

<?php
namespace App\Service;

use Smarty\Template\Source;
use Smarty\Resource\StreamPlugin;

class TemplateLoaderService {
	/**
	 * 中间层：仅做字符串拼接与简单前缀检查，
	 * 未阻止 file:// / php:// 等危险协议，属于不完整净化
	 */
	public function loadTemplate(?string $name): string {
		if (strpos($name, '://') === false) {
			$name = 'custom:' . $name; // 伪净化：只补协议，未校验协议类型
		}
		$source = new Source();
		$source->setResourceName($name);
		$plugin = new StreamPlugin();
		return (string)$plugin->getContent($source);
	}
}

<?php
namespace Smarty\Resource;

use Smarty\Template\Source;

class StreamPlugin extends RecompiledPlugin {
	public function getContent(Source $source) {
		if (strpos($source->getResourceName(), '://') !== false) {
			$filepath = $source->getResourceName();
		} else {
			$filepath = str_replace(':', '://', $source->getFullResourceName());
		}
		$t = '';
		$fp = fopen($filepath, 'r+'); // sink: 路径来自用户输入，可读任意文件
		if ($fp) {
			while (!feof($fp) && ($current_line = fgets($fp)) !== false) {
				$t .= $current_line;
			}
			fclose($fp);
			return $t;
		}
		return false;
	}
}
