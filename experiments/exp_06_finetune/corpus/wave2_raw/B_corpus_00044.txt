<?php
namespace App\Controller;

use App\Service\TemplateStreamService;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;

class TemplateImportController {
	private TemplateStreamService $streamService;

	public function __construct(TemplateStreamService $streamService) {
		$this->streamService = $streamService;
	}

	// POST /template/import?resource=php://filter/...&mode=raw
	public function importAction(Request $request): Response {
		$resource = $request->query->get('resource', '');
		$mode = $request->query->get('mode', 'raw');
		if ($mode === 'raw') {
			// 仅做字符串替换，未做协议白名单
			$safe = str_replace('..', '', $resource);
			return new Response($this->streamService->loadTemplate($safe));
		}
		return new Response('unsupported mode', 400);
	}
}

# === file: src/Service/TemplateStreamService.php ===
<?php
namespace App\Service;

use Smarty\Template\Source;
use Smarty\Resource\StreamPlugin;

class TemplateStreamService {

	public function loadTemplate(string $resourceName) {
		// 纯传递层：无任何校验，直接构造 Source 交给插件
		$source = new Source();
		$source->setResourceName($resourceName);
		$plugin = new StreamPlugin();
		return $plugin->getContent($source);
	}
}

# === file: vendor/smarty/smarty/src/Resource/StreamPlugin.php ===
<?php
namespace Smarty\Resource;

use Smarty\Template\Source;

class StreamPlugin extends RecompiledPlugin {

	public function populate(Source $source, ?Template $_template = null) {
		$source->uid = false;
		$source->content = $this->getContent($source);
		$source->timestamp = $source->exists = !!$source->content;
	}

	public function getContent(Source $source) {
		if (strpos($source->getResourceName(), '://') !== false) {
			$filepath = $source->getResourceName();
		} else {
			$filepath = str_replace(':', '://', $source->getFullResourceName());
		}
		$t = '';
		$fp = fopen($filepath, 'r+');
		if ($fp) {
			while (!feof($fp) && ($current_line = fgets($fp)) !== false) {
				$t .= $current_line;
			}
			fclose($fp);
			return $t;
		}
		return false;
	}
}
