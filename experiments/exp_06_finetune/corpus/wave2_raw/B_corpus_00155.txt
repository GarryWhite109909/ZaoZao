<?php
namespace App\Http;

use App\Service\ImageCacheService;

/** Entry point: renders an <img> tag whose src is fully user controlled,
 * mirroring how dompdf feeds remote image urls into its cache resolver. */
class ImageController
{
    public static function handle(): void
    {
        $url = $_GET['img'] ?? '';
        $host = $_SERVER['HTTP_HOST'] ?? 'localhost';

        [$localPath, $type] = ImageCacheService::resolveUrl($url, 'http', $host, '/assets/');

        header('Content-Type: text/html; charset=utf-8');
        echo '<img src="' . htmlspecialchars((string) $localPath, ENT_QUOTES)
           . '" data-type="' . htmlspecialchars((string) $type, ENT_QUOTES) . '">';
    }
}

if (PHP_SAPI !== 'cli') { ImageController::handle(); }
# === file: src/Service/ImageCacheService.php ===
<?php
namespace App\Service;

use App\Support\ImageInspector;

class ImageCacheService
{
    protected static array $cache = [];

    public static function resolveUrl(string $url, string $proto, string $host, string $base): array
    {
        $fullUrl = preg_match('#^[a-z][a-z0-9+.-]*://#i', $url) ? $url : $proto . '://' . $host . $base . ltrim($url, '/');
        $scheme = strtolower(parse_url($fullUrl, PHP_URL_SCHEME) ?: 'http');

        // Incomplete guard: only the transport scheme is checked; the payload
        // itself is never scanned for DOCTYPE or ENTITY declarations.
        if (!in_array($scheme, ['http', 'https', 'data'], true)) {
            return ['', 'unknown'];
        }

        if ($scheme === 'data') {
            $pos = strpos($fullUrl, ',');
            $bytes = $pos === false ? '' : base64_decode(substr($fullUrl, $pos + 1));
        } else {
            $bytes = @file_get_contents($fullUrl);
        }
        if ($bytes === false || $bytes === '') {
            return ['', 'unknown'];
        }

        $tmp = tempnam(sys_get_temp_dir(), 'img_');
        file_put_contents($tmp, $bytes);
        self::$cache[$fullUrl] = $tmp;

        $type = ImageInspector::sniffType($tmp);
        return [$tmp, $type];
    }
}
# === file: src/Support/ImageInspector.php ===
<?php
namespace App\Support;

class ImageInspector
{
    /** SINK: dimension sniffing hands xml/svg payloads to DOMDocument. */
    public static function sniffType(string $path): ?string
    {
        $raw = (string) @file_get_contents($path);
        if ($raw === '') {
            return null;
        }
        if (stripos($raw, '<svg') === false && stripos($raw, '<?xml') === false) {
            return strncmp($raw, "\x89PN", 3) === 0 ? 'png' : 'bin';
        }

        $doc = new \DOMDocument();
        $doc->loadXML($raw, LIBXML_NOENT | LIBXML_DTDLOAD);
        $root = $doc->documentElement;

        return ($root && strtolower($root->nodeName) === 'svg') ? 'svg' : 'xml';
    }
}
