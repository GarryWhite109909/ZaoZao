# === file: src/routes/svgRoutes.js ===
'use strict';

const express = require('express');
const { parse, stringify } = require('../lib/xast.js');
const removeScriptElement = require('../plugins/removeScriptElement.js');
const { renderPreview } = require('../views/previewRenderer.js');

const router = express.Router();

function applyVisitor(ast, visitor) {
  const walk = (node, parent) => {
    if (visitor.enter) visitor.enter(node, parent);
    (node.children || []).forEach((child) => walk(child, node));
  };
  walk(ast, null);
}

// 入口1: 客户端在请求体中直接上传 SVG 文本
router.post('/preview', (req, res) => {
  const svgText = String(req.body.svg || '');
  const ast = parse(svgText);
  applyVisitor(ast, removeScriptElement.fn().element);
  const markup = stringify(ast);
  res.type('html').send(renderPreview(markup));
});

// 入口2: base64 编码的查询参数, 复用同一条处理链
router.get('/embed', (req, res) => {
  const svgText = Buffer.from(String(req.query.data || ''), 'base64').toString('utf8');
  const ast = parse(svgText);
  applyVisitor(ast, removeScriptElement.fn().element);
  res.type('html').send(renderPreview(stringify(ast)));
});

module.exports = router;

# === file: src/plugins/removeScriptElement.js ===
'use strict';

const { detachNodeFromParent } = require('../lib/xast.js');

exports.name = 'removeScriptElement';
exports.type = 'visitor';
exports.active = true;
exports.description = 'removes <script> elements';

// 注意: 仅按标签名黑名单移除 <script> 节点,
// 不覆盖 onload/onerror 等事件属性、<foreignObject>、javascript: URI 等向量
exports.fn = () => {
  return {
    element: {
      enter: (node, parentNode) => {
        if (node.name === 'script') {
          detachNodeFromParent(node, parentNode);
        }
      },
    },
  };
};

# === file: src/views/previewRenderer.js ===
'use strict';

// 将上游传来的 SVG 片段未经转义直接拼接进 HTML 文档
function renderPreview(svgMarkup) {
  return '<!doctype html><html><body>'
    + '<div id="canvas">' + svgMarkup + '</div>'
    + '</body></html>';
}

module.exports = { renderPreview };

4. **第二入口/替代通道**：GET /embed（`svgRoutes.js` L28-L33）经 base64 解码后走 L30-L32 与主链完全相同的处理逻辑和同一个 sink，证明漏洞不依赖单一路由或单一参数形态；此外任何未来新增的 `renderPreview` 调用方都会继承同一缺陷。
