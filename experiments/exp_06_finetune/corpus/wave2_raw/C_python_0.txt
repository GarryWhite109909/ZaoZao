def run_maintenance():
    conn = psycopg2.connect(**DB_CONFIG)
    try:
        with conn.cursor() as cur:

                print("executing:", sql)
                cur.execute(sql)
        conn.commit()
        print("maintenance done, statements =", len(MAINTENANCE_SQL))
    finally:
        conn.close()

if __name__ == "__main__":
    run_maintenance()
