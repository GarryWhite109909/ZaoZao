# === file: app/routes.py ===
from starlette.requests import Request
from starlette.responses import JSONResponse
from starlette.routing import Route

from .services import TokenIntrospectionService


async def introspect_handler(request: Request) -> JSONResponse:
    """RFC-7662 风格内省端点，token 直接来自 POST 表单。"""
    form = await request.form()
    raw_token = form.get("token")
    if not isinstance(raw_token, str) or not raw_token:
        return JSONResponse({"active": False}, status_code=400)
    return await TokenIntrospectionService().introspect(raw_token)


async def revoke_handler(request: Request) -> JSONResponse:
    """第二入口：同一用户可控密钥，换一个动词打到同一存储。"""
    body = await request.json()
    raw_token = str(body.get("token", ""))
    return await TokenIntrospectionService().revoke(raw_token)


routes = [
    Route("/introspect", introspect_handler, methods=["POST"]),
    Route("/revoke", revoke_handler, methods=["POST"]),
]

# === file: app/services.py ===
import re

from .token_store import TOKEN_STORE


class TokenIntrospectionService:
    """中间层：只做格式净化，不做属主或调用方授权校验。"""

    _TOKEN_RE = re.compile(r"^[A-Za-z0-9_\-]{8,64}$")

    async def introspect(self, raw_token: str):
        # 看似防御：strip + 格式白名单，但与调用者身份完全无关
        cleaned = raw_token.strip()
        if not self._TOKEN_RE.match(cleaned):
            return {"active": False}
        record = await TOKEN_STORE.lookup(cleaned)
        if not record:
            return {"active": False}
        return {
            "active": True,
            "client_id": record["client_id"],
            "scope": record["scope"],
            "aud": record["resource"],
        }

    async def revoke(self, raw_token: str):
        cleaned = raw_token.strip()  # 仅字符串清理，无任何鉴权
        await TOKEN_STORE.delete(cleaned)
        return {"revoked": True}

# === file: app/token_store.py ===
"""全局内存令牌库，以原始令牌字符串为主键。"""

TOKENS = {
    "tok-alice-0001": {"client_id": "client-A", "scope": "mcp:read", "resource": "rs://alpha"},
    "tok-bob-0002": {"client_id": "client-B", "scope": "mcp:write", "resource": "rs://beta"},
}


class TokenStore:
    async def lookup(self, token_str: str):
        # SINK: 直接按键取整条记录，不校验请求者与该记录的归属关系
        return TOKENS.get(token_str)

    async def delete(self, token_str: str):
        return TOKENS.pop(token_str, None)


TOKEN_STORE = TokenStore()
