package com.robothy.s3.rest.service;

import com.ctc.wstx.stax.WstxInputFactory;
import com.ctc.wstx.stax.WstxOutputFactory;
import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.dataformat.xml.XmlFactory;
import com.fasterxml.jackson.dataformat.xml.XmlMapper;
import javax.xml.stream.XMLInputFactory;

/** Shared XmlMapper used by all S3 XML request parsers. */
public final class XmlMapperFactory {

  private static XmlMapper mapper;

  public static synchronized XmlMapper getXmlMapper() {
    if (mapper == null) {
      XMLInputFactory input = new WstxInputFactory();
      input.setProperty(XMLInputFactory.IS_NAMESPACE_AWARE, Boolean.FALSE);
      XmlFactory xmlFactory = new XmlFactory(input, new WstxOutputFactory());
      mapper = new XmlMapper(xmlFactory);
      mapper.configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);
    }
    return mapper;
  }

}

package com.robothy.s3.rest.service;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.dataformat.xml.XmlMapper;
import java.io.IOException;

/** Parses S3 XML request bodies (DeleteObjects, CompleteMultipartUpload, etc.). */
public class S3XmlRequestParser {

  public JsonNode parseDeleteRequest(String xmlBody) {
    try {
      XmlMapper mapper = XmlMapperFactory.getXmlMapper();
      return mapper.readValue(xmlBody, JsonNode.class);
    } catch (IOException e) {
      throw new IllegalArgumentException("Malformed XML request body.", e);
    }
  }

}
