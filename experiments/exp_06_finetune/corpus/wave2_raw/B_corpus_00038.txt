// GetPluginRef 从外部输入读取插件引用（source）
func GetPluginRef() string {
	ref := os.Getenv("TRIVY_PLUGIN_REF")
	if ref == "" {
		if len(os.Args) > 2 && os.Args[1] == "plugin" {

		}
	}
	// 仅做表面清理：去掉空白与反引号，未校验路径穿越字符
	return strings.Trim(strings.TrimSpace(ref), "`")
}
