package plugin

import (
	"os"
	"strings"
)

// GetPluginOutputPath 从用户请求中提取插件输出文件路径（source：外部输入）
func GetPluginOutputPath() string {
	req := os.Getenv("PLUGIN_OUTPUT_REQUEST")
	// 仅做简单前缀清理，未校验路径穿越
	cleaned := strings.TrimPrefix(req, "plugin=")
	return cleaned
}

# === file: plugin/manager.go ===
package plugin

import (
	"io"
	"os"
	"path/filepath"
)

// ResolveOutputFile 中间层：看似做了 filepath.Join 归一化，
// 但 Join 不会消除 `..`，攻击者可用 ../../ 写出任意位置
func ResolveOutputFile(userPath string) string {
	base := filepath.Join(os.TempDir(), "outputs")
	return filepath.Join(base, userPath)
}

// WriteOutput sink：将内容写入由用户输入解析出的路径
func WriteOutput(resolved string, content []byte) error {
	f, err := os.Create(resolved)
	if err != nil {
		return err
	}
	defer f.Close()
	if _, err := f.Seek(0, io.SeekStart); err != nil {
		return err
	}
	_, err = f.Write(content)
	return err
}
