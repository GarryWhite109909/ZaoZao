```python
import hashlib
import base64
from flask import Flask, request, jsonify
from werkzeug.security import generate_password_hash

app = Flask(__name__)

@app.route('/api/hash', methods=['POST'])
def make_hash():
    src = request.form.get('src', '')
    salt = request.form.get('salt', '')
    digest = hashlib.sha256((src + salt).encode()).hexdigest()
    return jsonify({'hash': digest})

@app.route('/api/password', methods=['POST'])
def encrypt_password():
    pwd = request.form.get('pwd', '')
    hashed = generate_password_hash(pwd, method='pbkdf2:sha256', salt_length=8)
    return jsonify({'encrypted': hashed})

@app.route('/api/decode', methods=['POST'])
def decode_b64():
    pwd = request.form.get('pwd', '')
    try:
        return jsonify({'decoded': base64.b64decode(pwd).decode('utf-8', 'ignore')})
    except Exception:
        return jsonify({'error': 'bad input'}), 400

@app.route('/api/md5', methods=['POST'])
def make_md5():
    data = request.form.get('data', '')
    return jsonify({'md5': hashlib.md5(data.encode()).hexdigest()[4:12]})

if __name__ == '__main__':
    app.run()
```
