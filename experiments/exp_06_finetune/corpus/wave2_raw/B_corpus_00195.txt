<?php
require_once __DIR__ . '/../lib/sanitizer.php';
require_once __DIR__ . '/../services/stream_manager.php';

function getInput($key, $default = '')
{
    global $argv;
    if (php_sapi_name() === 'cli') {
        foreach ($argv as $arg) {
            if (strpos($arg, "{$key}=") === 0) {
                return substr($arg, strlen("{$key}="));
            }
        }
    } else {
        return isset($_REQUEST[$key]) ? $_REQUEST[$key] : $default;
    }
    return $default;
}

$action = getInput('action', 'status');

if ($action === 'start') {
    $rawCommand = getInput('ffmpegCommand', '');
    $safeCommand = sanitizeFFmpegCommand($rawCommand);
    if ($safeCommand !== '') {
        startStream($safeCommand);
    }
} elseif ($action === 'list') {
    listFFmpegProcesses(getInput('keyword', ''));
}

<?php
// lib/sanitizer.php
// Blacklist sanitizer for ffmpeg commands

function sanitizeFFmpegCommand($command)
{
    $allowedPrefixes = ['ffmpeg', '/usr/bin/ffmpeg', '/bin/ffmpeg'];

    // Remove dangerous characters
    $command = str_replace('&&', '', $command);
    $command = str_replace('rtmp://vlu.me/', 'rtmp://live/', $command);
    $command = preg_replace('/\s*&?>.*(?:2>&1)?/', '', $command);
    $command = preg_replace('/[;|`<>$()\n\r{}]/', '', $command);

    // Ensure it starts with an allowed prefix
    foreach ($allowedPrefixes as $prefix) {
        if (strpos(trim($command), $prefix) === 0) {
            return $command;
        }
    }

    return '';
}

<?php
// services/stream_manager.php

function startStream($ffmpegCommand)
{
    $cmd = $ffmpegCommand . ' > /dev/null 2>&1 &';
    exec($cmd, $output, $status);
    return $status === 0;
}

function listFFmpegProcesses($keyword = '')
{
    $command = "ps -eo pid,etime,%cpu,%mem,cmd | grep '[f]fmpeg'";

    if (!empty($keyword)) {
        $command .= " | grep '$keyword'";
    }

    exec($command, $output, $status);

    $processes = [];
    foreach ($output as $line) {
        preg_match('/^\s*(\d+)\s+([\d:-]+)\s+([\d.]+)\s+([\d.]+)\s+(.+)$/', $line, $matches);
        if (!empty($matches)) {
            $processes[] = [
                'pid' => intval($matches[1]),
                'cpu_usage' => floatval($matches[3]),
                'memory_usage' => floatval($matches[4]),
                'command' => $matches[5]
            ];
        }
    }
    return $processes;
}
