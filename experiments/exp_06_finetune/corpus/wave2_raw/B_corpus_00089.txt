# === file: plugin/Comments/CommentService.php ===
<?php
require_once __DIR__ . '/Objects/Comment.php';

class CommentService
{
    // 看似有净化：只过滤 title 字段，并丢弃 csrf_token 字段名（不校验其值）
    public static function sanitize(array $data)
    {
        if (isset($data['title'])) {
            $data['title'] = strip_tags($data['title']);
        }
        unset($data['csrf_token']);
        return $data;
    }

    public static function handleSave(array $post)
    {
        $clean = self::sanitize($post);
        $o = new Comment(@$clean['id']);
        if (!empty($clean['title'])) {
            $o->setTitle($clean['title']);
        }
        if (!empty($clean['status'])) {
            $o->setStatus($clean['status']);
        }
        return $o->save() ? array('error' => false, 'msg' => 'Saved')
                          : array('error' => true, 'msg' => 'Save failed');
    }
}
# === file: plugin/Comments/Objects/Comment.php ===
<?php
require_once $global['systemRootPath'] . 'objects/ObjectYPT.php';

class Comment extends ObjectYPT
{
    protected $id;
    protected $title;
    protected $status;

    public function __construct($id = 0)
    {
        $this->id = intval($id);
        if ($this->id > 0) {
            $this->load($this->id);
        }
    }

    public function setTitle($t) { $this->title = $t; }
    public function setStatus($s) { $this->status = $s; }

    // 状态变更落库点
    public function save()
    {
        if ($this->id > 0) {
            return self::updateTable('comments', get_object_vars($this));
        }
        return self::insertIntoTable('comments', get_object_vars($this));
    }
}
