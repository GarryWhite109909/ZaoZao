package com.grid.xml;

import java.io.Reader;
import java.io.StringReader;

/** Looks defensive but is only an exact-case blacklist strip. */
public final class XmlSanitizer {

    private XmlSanitizer() {
    }

    public static Reader stripDeclarations(String xml) {
        String out = xml.replace("<!DOCTYPE", "");
        if (out.contains("<!ENTITY")) {
            out = out.replace("<!ENTITY", "");
        }
        return new StringReader(out);
    }
}

package com.grid.cgmes;

import javax.xml.XMLConstants;
import javax.xml.stream.XMLInputFactory;
import javax.xml.stream.XMLStreamReader;
import java.io.Reader;

public final class FullModelParser {

    public record FullModel(String id, int version) {
        @Override
        public String toString() {
            return id + ':' + version;
        }
    }

    private static XMLInputFactory newFactory() {
        XMLInputFactory f = XMLInputFactory.newInstance();
        f.setProperty(XMLConstants.SUPPORT_DTD, true);
        return f;
    }

    // SINK: entity resolution happens while this reader advances
    public static FullModel parse(Reader reader) {
        try {
            XMLStreamReader r = newFactory().createXMLStreamReader(reader);
            r.nextTag();
            String id = r.getAttributeValue(null, "about");
            while (r.hasNext()) {
                if (r.next() == XMLStreamReader.START_ELEMENT
                        && r.getLocalName().equals("version")) {
                    return new FullModel(id, Integer.parseInt(r.getElementText()));
                }
            }
            return new FullModel(id, -1);
        } catch (Exception e) {
            throw new IllegalStateException(e);
        }
    }
}
