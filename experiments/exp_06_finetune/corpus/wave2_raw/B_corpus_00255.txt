<?php
declare(strict_types=1);

namespace Vendor\FormExtension\Controller;

use Psr\Http\Message\ServerRequestInterface;
use TYPO3\CMS\Core\Http\HtmlResponse;
use Vendor\FormExtension\Domain\Dto\SearchCriteria;
use Vendor\FormExtension\Domain\Repository\FormDefinitionRepository;

final class FormManagementController
{
    public function __construct(
        private readonly FormDefinitionRepository $repository,
    ) {}

    /** Backend module list view: GET ?search=...&limit=... */
    public function listAction(ServerRequestInterface $request): HtmlResponse
    {
        $params = $request->getQueryParams();
        $criteria = SearchCriteria::fromQueryParams($params);
        $rows = $this->repository->findAll($criteria);
        $html = '';
        foreach ($rows as $row) {
            $html .= '<li>' . htmlspecialchars((string)$row['label'], ENT_QUOTES) . '</li>';
        }
        return new HtmlResponse('<ul>' . $html . '</ul>');
    }
}
# === file: Classes/Domain/Dto/SearchCriteria.php ===
<?php
declare(strict_types=1);

namespace Vendor\FormExtension\Domain\Dto;

/** Carries the search term from the controller to the repository layer. */
final class SearchCriteria
{
    public function __construct(
        public readonly string $searchTerm = '',
        public readonly int $limit = 0,
    ) {}

    public static function fromQueryParams(array $params): self
    {
        $term = isset($params['search']) ? (string)$params['search'] : '';
        // treat as LIKE-safe: strip SQL wildcard characters only
        $term = str_replace(['%', '_'], '', $term);
        $limit = isset($params['limit']) ? (int)$params['limit'] : 0;
        return new self($term, $limit);
    }

    public function hasLimit(): bool
    {
        return $this->limit > 0;
    }
}
# === file: Classes/Domain/Repository/FormDefinitionRepository.php ===
<?php
declare(strict_types=1);

namespace Vendor\FormExtension\Domain\Repository;

use Doctrine\DBAL\Connection;
use Vendor\FormExtension\Domain\Dto\SearchCriteria;

final class FormDefinitionRepository
{
    public const TABLE_NAME = 'tx_formextension_form_definition';

    public function __construct(
        private readonly Connection $connection,
    ) {}

    /** Fetch all form definitions matching the search term. */
    public function findAll(SearchCriteria $criteria): array
    {
        $sql = 'SELECT * FROM ' . self::TABLE_NAME;
        if ($criteria->searchTerm !== '') {
            $sql .= " WHERE identifier LIKE '%" . $criteria->searchTerm . "%'"
                . " OR label LIKE '%" . $criteria->searchTerm . "%'";
        }
        if ($criteria->hasLimit()) {
            $sql .= ' LIMIT ' . $criteria->limit;
        }
        return $this->connection->fetchAllAssociative($sql);
    }
}
