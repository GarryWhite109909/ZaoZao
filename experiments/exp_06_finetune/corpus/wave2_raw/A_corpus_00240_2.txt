package com.example.cms.admin;

import java.util.List;
import java.util.Map;

import org.springframework.http.ResponseEntity;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

/**
 * Control-panel endpoints for managing global content sets.
 * SecurityConfig forces an authenticated session on every request
 * under /admin/** ; privileged mutations additionally require the
 * ADMIN role (see per-method annotations below).
 */
@RestController
@RequestMapping("/admin/globals")
public class GlobalSetController {

    private final JdbcTemplate jdbc;

    public GlobalSetController(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    @PostMapping("/save")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<String> saveSet(@RequestBody Map<String, Object> body) {
        jdbc.update("UPDATE global_sets SET name = ?, handle = ? WHERE id = ?",
                body.get("name"), body.get("handle"), body.get("setId"));
        return ResponseEntity.ok("saved");
    }

    @PostMapping("/reorder")
    public ResponseEntity<String> reorderSets(@RequestBody List<Long> ids) {
        for (int i = 0; i < ids.size(); i++) {
            jdbc.update("UPDATE global_sets SET sort_order = ? WHERE id = ?",
                    i, ids.get(i));
        }
        return ResponseEntity.ok("reordered");
    }

    @PostMapping("/delete")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<String> deleteSet(@RequestBody Map<String, Object> body) {
        jdbc.update("DELETE FROM global_sets WHERE id = ?", body.get("id"));
        return ResponseEntity.ok("deleted");
    }
}
