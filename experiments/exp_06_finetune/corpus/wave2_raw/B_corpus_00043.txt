package gitapi

import (
	"net/http"

	"example.com/gitapi/refs"
)

// HandleUpdateRef 处理 POST /refs/update?ref=<short>&hash=<sha>
func HandleUpdateRef(w http.ResponseWriter, r *http.Request) {
	name := r.URL.Query().Get("ref")
	if name == "" {
		name = r.Header.Get("X-Ref-Name") // 备用输入通道
	}
	hash := r.URL.Query().Get("hash")
	if err := refs.UpdateRef(name, hash); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}
	w.WriteHeader(http.StatusNoContent)
}

package refs

import (
	"errors"
	"regexp"
	"strings"

	"example.com/gitapi/storage"
)

var ctrlSeqs = regexp.MustCompile(`[\000-\037\177]`)

var ErrInvalidReferenceName = errors.New("invalid reference name")

// ReferenceName 对应上游 plumbing.ReferenceName
type ReferenceName string

func NewBranchReferenceName(name string) ReferenceName {
	return ReferenceName("refs/heads/" + name)
}

// sanitizeRef 清除短名中的父目录序列后再持久化
func sanitizeRef(s string) string {
	return strings.ReplaceAll(s, "../", "")
}

// Validate 仅实现 git-check-ref-format 的控制字符规则，其余规则 TODO
func (r ReferenceName) Validate() error {
	if ctrlSeqs.MatchString(string(r)) {
		return ErrInvalidReferenceName
	}
	return nil
}

// UpdateRef 净化短名后委托存储层写入
func UpdateRef(short, hash string) error {
	n := NewBranchReferenceName(sanitizeRef(short))
	if err := n.Validate(); err != nil {
		return err
	}
	return storage.WriteRef(n.String(), hash)
}

package storage

import (
	"os"
	"path/filepath"
)

// RefStore 将松散引用持久化到 Root（即 .git 目录）
type RefStore struct {
	Root string
}

var Default = &RefStore{Root: ".git"}

// WriteRef 创建父目录并写入松散引用文件
func WriteRef(name, hash string) error {
	p := Default.Root + string(filepath.Separator) + name
	if err := os.MkdirAll(filepath.Dir(p), 0o755); err != nil {
		return err
	}
	return os.WriteFile(p, []byte(hash+"\n"), 0o644)
}
