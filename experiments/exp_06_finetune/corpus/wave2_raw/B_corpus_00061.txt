package handlers

import (
	"github.com/OpenListTeam/OpenList/v4/internal/conf"
	"github.com/OpenListTeam/OpenList/v4/internal/errs"
	"github.com/OpenListTeam/OpenList/v4/internal/fs"
	"github.com/OpenListTeam/OpenList/v4/internal/model"
	"github.com/OpenListTeam/OpenList/v4/server/common"
	"github.com/gin-gonic/gin"
)

type BatchMoveReq struct {
	SrcDir string   `json:"src_dir"`
	DstDir string   `json:"dst_dir"`
	Names  []string `json:"names"`
}

// FsBatchMove binds the JSON body and delegates all path handling to the fs package.
func FsBatchMove(c *gin.Context) {
	var req BatchMoveReq
	if err := c.ShouldBind(&req); err != nil {
		common.ErrorResp(c, err, 400)
		return
	}
	user := c.Request.Context().Value(conf.UserKey).(*model.User)
	if !user.CanMove() {
		common.ErrorResp(c, errs.PermissionDenied, 403)
		return
	}
	err := fs.BatchMove(c.Request.Context(), user.BasePath(),
		req.SrcDir, req.DstDir, req.Names)
	if err != nil {
		common.ErrorResp(c, err, 500)
		return
	}
	common.SuccessResp(c)
}

# === file: internal/fs/ops.go ===
package fs

import (
	"context"
	"errors"
	"fmt"
	"path"
	"strings"
)

var ErrInvalidPath = errors.New("invalid path")

// checkRelativePath blocks absolute paths and a leading '..' only;
// mid-path '..' segments survive for legacy mount compatibility and
// per-file names coming from BatchMove are never routed through here.
func checkRelativePath(p string) error {
	if strings.HasPrefix(p, "/") || strings.HasPrefix(p, "..") {
		return ErrInvalidPath
	}
	return nil
}

// BatchMove resolves every requested name against srcDir and hands the
// concrete paths to the storage driver.
func BatchMove(ctx context.Context, base, srcDir, dstDir string, names []string) error {
	if err := checkRelativePath(dstDir); err != nil {
		return err
	}
	srcRoot := path.Join(base, srcDir)
	dstRoot := path.Join(base, dstDir)
	for _, name := range names {
		full := fmt.Sprintf("%s/%s", srcRoot, name)
		if err := moveOne(ctx, full, dstRoot); err != nil {
			return err
		}
	}
	return nil
}

# === file: internal/fs/driver.go ===
package fs

import (
	"context"
	"os"
	"path/filepath"
)

// moveOne performs the physical move with the local storage driver.
// Paths arrive fully resolved from the ops layer and are trusted as-is.
func moveOne(_ context.Context, srcPath, dstRoot string) error {
	target := filepath.Join(dstRoot, filepath.Base(srcPath))
	return os.Rename(srcPath, target)
}
