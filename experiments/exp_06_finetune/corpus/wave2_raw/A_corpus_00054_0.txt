```javascript
const express = require('express');
const path = require('path');
const h5 = require('h5wasm'); // node 生态常用的 HDF5 读取库

const app = express();
app.use(express.json());

// 模型权重文件编辑器：根据用户传入的模型路径加载 HDF5 权重并返回结构摘要
app.post('/api/model-editor/load', async (req, res) => {
  try {
    // source: 用户完全控制模型文件路径
    const modelPath = req.body.model_path;

    if (!modelPath || typeof modelPath !== 'string') {
      return res.status(400).json({ status: 'error', message: 'missing model_path' });
    }

    // sink: 直接把用户提供的路径交给 HDF5 打开函数，未做目录限制校验
    const f = new h5.File(modelPath, 'r');

    const summary = {};
    for (const key of f.keys()) {
      summary[key] = f.get(key).to_array().length;
    }
    f.close();

    res.json({ status: 'success', weights_summary: summary });
  } catch (err) {
    res.status(500).json({ status: 'error', message: String(err) });
  }
});

app.listen(3000);
```

1. 第 12 行是 source：`req.body.model_path` 来自 HTTP 请求体，攻击者可传入任意字符串，例如 `../../etc/passwd` 或指向任意本地 `.h5`/伪装文件的绝对路径。
2. 第 15 行只做了非空和类型检查，属于存在性校验而非安全防御；没有 `path.resolve` 后与白名单根目录做 `startsWith` 比对，也没有拒绝包含 `..` 的路径。
3. 第 20 行是 sink：`new h5.File(modelPath, 'r')` 直接以用户提供的路径打开本地文件。HDF5 库还会解析 ExternalLink，攻击者可构造一个引用系统内其他文件的恶意 `.h5` 文件，实现任意本地文件内容读取（与原 KerasFileEditor 的 HDF5 ExternalLink 信息泄露语义等价）。
4. 第 22~24 行将文件内容摘要返回给客户端，完成从外部输入到敏感数据外泄的完整链路；全程无沙箱、无路径规范化、无权限隔离。
