const express = require('express');
const axios = require('axios');
const crypto = require('crypto');
const OAuth = require('oauth-1.0a');
const settingsStore = require('./settings-store'); // DB-backed config KV

const app = express();
const feedCache = new Map();
const CACHE_TTL_MS = 90 * 60 * 1000;

async function performApiRequest(url = 'https://api.twitter.com/1.1/search/tweets.json', getField = '') {
  let accessToken = await settingsStore.get('twitter_access_token');
  let accessTokenSecret = await settingsStore.get('twitter_access_token_secret');
  let consumerKey = await settingsStore.get('twitter_consumer_key');
  let consumerSecret = await settingsStore.get('twitter_consumer_secret');

  if (!accessToken) accessToken = '220111598-87eLa7MgXZmd7YeRSkenTSVxhZikok61PXMKZFti';
  if (!accessTokenSecret) accessTokenSecret = 'KsDrxrxoGqVVK0ethvcTTrV58RBH3WUjnPeI616fnxIFS';
  if (!consumerKey) consumerKey = 'WgDmyOjMgX1N7RhcLpQqzUrtR';
  if (!consumerSecret) consumerSecret = '0e8PlzIeKlmGGyH1ajS2Ggaw0anPTX23p3gp2WqZ0PCNxkYYX1';

  const cacheId = crypto.createHash('md5').update(url + getField).digest('hex');
  const hit = feedCache.get(cacheId);
  if (hit && Date.now() - hit.ts < CACHE_TTL_MS) return hit.data;

  const oauth = OAuth({
    consumer: { key: consumerKey, secret: consumerSecret },
    signature_method: 'HMAC-SHA1',
    hash_function: (base, key) => crypto.createHmac('sha1', key).update(base).digest('base64')
  });
  const reqData = { url: url + getField, method: 'GET' };
  const res = await axios.get(reqData.url, {
    headers: oauth.toHeader(oauth.authorize(reqData, { key: accessToken, secret: accessTokenSecret }))
  });

  if (res.data && Object.keys(res.data).length > 0) {
    feedCache.set(cacheId, { data: res.data, ts: Date.now() });
  }
  return res.data;
}

app.get('/api/twitter/search', async (req, res) => {
  const count = parseInt(req.query.count, 10) || 5;
  const q = encodeURIComponent(req.query.q || '');
  const items = await performApiRequest('https://api.twitter.com/1.1/search/tweets.json',
    `?count=${count}&q=${q}`);
  res.json((items && items.statuses) || []);
});

app.get('/api/twitter/user/:handle', async (req, res) => {
  const items = await performApiRequest('https://api.twitter.com/1.1/statuses/user_timeline.json',
    `?include_entities=true&exclude_replies=true&count=5&screen_name=${req.params.handle}`);
  if (items && items.errors) return res.status(502).json({ error: 'upstream error' });
  res.json(items || []);
});

app.listen(3000, () => console.log('twitter feed listening on :3000'));
