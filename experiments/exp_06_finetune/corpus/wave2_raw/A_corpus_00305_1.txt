```javascript
// sandbox-render.js — express 中间件链上的"沙箱模板渲染"服务
const express = require('express');
const app = express();
app.use(express.json());

// 模拟原 JinjavaBeanELResolver 的受限属性黑名单
const RESTRICTED_PROPS = new Set([
  'constructor', 'prototype', '__proto__', 'process',
  'require', 'eval', 'Function', 'module', 'global'
]);

// 路径解析：支持点号与括号两种写法，如 user.name 或 user['name']
function resolvePath(root, path) {
  let cur = root;
  for (const seg of path.split('.')) {
    const bracket = seg.match(/^\[\s*(.+?)\s*\]$/);
    let key;
    if (bracket) {
      key = bracket[1].slice(1, -1); // 去引号；带引号的键被视为数据字面量，跳过黑名单检查
    } else {
      key = seg;
      if (RESTRICTED_PROPS.has(key)) throw new Error('blocked property: ' + key);
    }
    if (cur == null) return '';
    cur = cur[key];
  }
  return cur;
}

// 中间件：把用户提交的 template 渲染后挂到 res.locals
app.use('/api/render', (req, res, next) => {
  const { template } = req.body; // source：外部输入入口
  try {
    res.locals.output = template.replace(/\{\{\s*(.+?)\s*\}\}/g, (_, expr) => {
      const call = expr.match(/^(.*?)\(\)\s*$/); // 支持 {{ expr() }} 零参调用
      let val = resolvePath(req.body.context || {}, call ? call[1] : expr);
      if (call && typeof val === 'function') val = val(); // sink：执行任意函数
      return String(val);
    });
    next();
  } catch (e) {
    res.status(400).send('render error: ' + e.message);
  }
});

app.post('/api/render', (req, res) => {
  res.json({ rendered: res.locals.output });
});

app.listen(3000);
```

2. 第 23 行黑名单仅对**点号段**生效；第 20–21 行对括号写法只做去引号处理并注释为“数据字面量”，完全跳过 `RESTRICTED_PROPS` 检查——这是位置性防御缺口（黑名单本身也属可绕过过滤）。攻击者用 `x['constructor']` 即可取到构造函数。
3. 拿到 `constructor.constructor` 即全局 `Function` 后，第 31 行 `val = val()`（sink）执行零参调用，配合 `Function('return process...')()` 形态完成 RCE，等价于原代码绕过 `DEFAULT_RESTRICTED_METHODS` 后调用任意方法。
4. 第二入口确认：无其他路由或替代通道；但同一中间件对 `context` 内嵌套对象同样适用该解析逻辑，且黑名单是静态集合、不支持配置扩展，覆盖面不足支撑“可被绕过”的最终结论。
