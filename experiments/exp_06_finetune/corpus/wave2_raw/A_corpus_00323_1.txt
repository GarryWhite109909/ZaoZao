package com.example.notify;

import org.springframework.web.bind.annotation.*;
import org.springframework.stereotype.Controller;
import java.util.*;
import java.text.MessageFormat;

@Controller
public class NotifyController {

    // 模板注册表：允许通过 /spec 接口动态注册格式化器（等价于 add_format_spec）
    private final Map<String, java.util.function.Function<Object, String>> formatSpecs = new HashMap<>();

    @PostMapping("/template/register")
    public String registerSpec(@RequestParam String name, @RequestParam String expr) {
        // 注册一个基于 MessageFormat 的自定义格式化函数，expr 为用户可控的格式模式
        formatSpecs.put(name, value -> new MessageFormat(expr).format(new Object[]{value}));
        return "registered";
    }

    // 外部输入入口：用户提交模板字符串与渲染参数
    @GetMapping("/notify/render")
    public String render(@RequestParam String template, @RequestParam Map<String, String> params) {
        StringBuilder out = new StringBuilder();
        int i = 0;
        while (true) {
            int start = template.indexOf('{', i);
            if (start < 0) { out.append(template.substring(i)); break; }
            int end = template.indexOf('}', start);
            if (end < 0) { out.append(template.substring(i)); break; }
            out.append(template, i, start);
            String field = template.substring(start + 1, end);
            Object value = params.get(field);
            // sink：将用户可控字段名/值直接交给 MessageFormat 渲染，
            // 未限制可访问的对象图（如 {env.X}、{systemProperties} 类占位符），
            // 也未对 formatSpecs 中用户注册的 expr 做任何白名单校验
            if (value != null) {
                out.append(new MessageFormat(field).format(new Object[]{value}));
            } else {
                out.append("{" + field + "}");
            }
            i = end + 1;
        }
        return out.toString();
    }
}

2. 第 33-41 行的字段解析循环没有任何黑名单、正则或转义过滤：`field` 直接作为 `MessageFormat` 的 pattern 使用，`params` 的值也未经校验直接进入格式化参数数组；`MessageFormat` 对 `{index}` 之外的元素索引/子格式（如 choice、date 子模式）不做对象访问隔离，配合用户在第 18 行注册的任意 `expr`，攻击者可通过精心构造的占位符探测内部状态并泄露环境信息。
3. 第 15 行的 `formatSpecs` 注册表本身无鉴权与表达式校验，属于第二个 source→sink 通道：注册的 `expr` 在每次渲染时被原样编译执行。
4. 不存在其他替代防御路径：代码中没有对 `template` 或 `field` 做字符集限制（如禁止 `{0,` 等子格式语法），也没有使用只接受纯文本的安全拼接 API。
