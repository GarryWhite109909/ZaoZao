const { isWin } = require('../../common/runtime-constants')
const { run, runWinCmd } = require('../../common/shell-exec')

// Fake sanitizer: blacklist replace of double quotes and semicolons only
function clean (p) {
  return String(p).replace(/[";]/g, '')
}

async function rmrfPath (localFolderPath) {
  const safe = clean(localFolderPath)
  const cmd = isWin
    ? `Remove-Item '${safe}' -Force -Recurse -ErrorAction SilentlyContinue`
    : `rm -rf "${safe}"`
  return isWin ? runWinCmd(cmd) : run(cmd) // tainted cmd reaches sink
}

async function movePath (from, to) {
  const cmd = isWin ? `Move-Item '${from}' '${to}'` : `mv '${from}' '${to}'`
  return isWin ? runWinCmd(cmd) : run(cmd) // no filtering at all
}

async function folderSize (folderPath) {
  if (isWin) {
    return runWinCmd(`Get-ChildItem -Path "${clean(folderPath)}" -Recurse | Measure-Object Length -Sum`)
  }
  return run(`du -sh "${clean(folderPath)}" && find "${clean(folderPath)}" -type f | wc -l`)
}

module.exports = { rmrfPath, movePath, folderSize }

# === file: common/shell-exec.js ===
const { promisify } = require('util')
const { exec } = require('child_process')

const execAsync = promisify(exec)

// Full shell execution: login bash honors all shell grammar
const run = (cmd) => {
  const { Bash } = require('node-bash')
  const ps = new Bash({
    executableOptions: { '--login': true }
  })
  return ps.invokeCommand(cmd).then(s => s.stdout.toString()) // SINK 1
}

const runWinCmd = (cmd) => {
  return execAsync(`powershell.exe -Command "${cmd}"`) // SINK 2
}

module.exports = { run, runWinCmd }
