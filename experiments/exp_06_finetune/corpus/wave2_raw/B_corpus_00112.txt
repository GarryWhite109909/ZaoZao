<?php

namespace App\Controller;

use App\Support\PayloadFilter;
use App\Tool\Serialize;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;

final class ImportController
{
    // POST /admin/import/state   form field: state
    public function importState(Request $request): Response
    {
        $raw = (string) $request->request->get('state', '');
        if ($raw === '') {
            return new Response('missing state', 400);
        }

        $filtered = PayloadFilter::stripDangerousClasses($raw);

        $data = Serialize::unserialize($filtered);

        return new Response('imported keys: ' . implode(',', array_keys((array) $data)));
    }

    // GET /admin/import/preview  query param: blob
    public function previewBlob(Request $request): Response
    {
        $raw = (string) $request->query->get('blob', '');

        return new Response(var_export(Serialize::unserialize(PayloadFilter::stripDangerousClasses($raw)), true));
    }
}
# === file: src/Support/PayloadFilter.php ===
<?php

namespace App\Support;

/**
 * Legacy hardening layer kept from the old Tool\Serialize usage.
 * Strips historically abused gadget class names out of a serialized
 * payload before it is handed to the unserialize helper.
 */
final class PayloadFilter
{
    private const BLOCKED_CLASSES = [
        'Pimcore\Model\Document\PageSnippet',
        'Symfony\Component\Routing\Route',
    ];

    public static function stripDangerousClasses(string $payload): string
    {
        foreach (self::BLOCKED_CLASSES as $needle) {
            $payload = str_replace($needle, '__stripped__', $payload);
        }

        return $payload;
    }
}
# === file: src/Tool/Serialize.php ===
<?php

namespace App\Tool;

final class Serialize
{
    public static function serialize(mixed $data): string
    {
        return serialize($data);
    }

    public static function unserialize(?string $data = null): mixed
    {
        if ($data) {
            $data = unserialize($data);
        }

        return $data;
    }
}
