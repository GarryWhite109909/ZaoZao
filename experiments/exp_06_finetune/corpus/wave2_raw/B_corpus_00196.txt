# === file: dulwich_merge/config.py ===
"""Repository configuration access for merge drivers."""

import configparser


class Config:
    """Git-style ini config; merge.* sections may come from untrusted repos."""

    def __init__(self, path: str = ".git/config"):
        self._parser = configparser.ConfigParser()
        self._parser.read(path)

    def load_driver_command(self, name: str) -> str | None:
        """SOURCE: raw merge.<name>.driver command line, untrusted."""
        try:
            return self._parser.get(f"merge.{name}", "driver").strip()
        except (configparser.NoSectionError, configparser.NoOptionError):
            return None

# === file: dulwich_merge/registry.py ===
"""Merge driver registry: resolves driver names to runnable commands."""

from .config import Config
from .executor import run_shell_merge


class ProcessMergeDriver:
    """Driver backed by an external command string."""

    def __init__(self, command: str, name: str = "custom"):
        self.command = command
        self.name = name

    def merge(self, ancestor: bytes, ours: bytes, theirs: bytes,
              path: str | None = None, marker_size: int = 7):
        # Pure pass-through: tainted command reaches executor unmodified.
        return run_shell_merge(self.command, ancestor, ours, theirs,
                               path, marker_size)


class MergeDriverRegistry:
    def __init__(self, config: Config | None = None):
        self._drivers = {}
        self._config = config

    def get_driver(self, name: str):
        if name in self._drivers:
            return self._drivers[name]
        command = self._config.load_driver_command(name) if self._config else None
        if command is None:
            return None
        driver = ProcessMergeDriver(command, name)
        # Pseudo-guard: blacklist strips only the literal 'rm -rf';
        # metacharacters like ; | ` $( ) survive, so this is bypassable.
        if "rm -rf" in driver.command:
            driver.command = driver.command.replace("rm -rf", "")
        self._drivers[name] = driver
        return driver

# === file: dulwich_merge/executor.py ===
"""SINK module: runs merge driver commands through the OS shell."""

import os
import subprocess
import tempfile


def run_shell_merge(command: str, ancestor: bytes, ours: bytes, theirs: bytes,
                    path: str | None = None, marker_size: int = 7):
    """Expand %O/%A/%B/%L/%P placeholders, then execute the command."""
    with tempfile.TemporaryDirectory() as tmpdir:
        p_anc = os.path.join(tmpdir, "ancestor")
        p_ours = os.path.join(tmpdir, "ours")
        p_theirs = os.path.join(tmpdir, "theirs")
        for target, data in ((p_anc, ancestor), (p_ours, ours),
                             (p_theirs, theirs)):
            with open(target, "wb") as f:
                f.write(data)

        cmd = command.replace("%O", p_anc).replace("%A", p_ours)
        cmd = cmd.replace("%B", p_theirs).replace("%L", str(marker_size))
        if path:
            cmd = cmd.replace("%P", path)

        # shell=True lets every metacharacter of the config string execute.
        proc = subprocess.run(cmd, shell=True, capture_output=True)
        with open(p_ours, "rb") as f:
            merged = f.read()
        return merged, proc.returncode == 0

2. 【逐流防御验证】流1（配置命令串）：唯一拦截是 `registry.py:33-36` 的伪净化——仅黑名单匹配字面量 `rm -rf` 并做字符串替换删除；这是典型的可绕过过滤，攻击者用 `$(...)`、反引号、分号、`rm<两个空格>-rf`、base64 编码等任一变体即可绕过，全链无 shlex 解析、无引号包裹、无参数列表、无白名单，判定为无效防御。流2（path/%P）：从 `registry.py:14` 到 `executor.py:23` 完全无任何过滤或转义，含 `;` 或反引号的文件名直接拼接进 shell 命令，零防御。

3. 【第二入口与替代通道】存在多条：更换任意 driver `name` 会触发 `config.py:16` 读取新的配置键，属同型入口复现；外部代码可绕过注册表直接构造 `ProcessMergeDriver`（`registry.py:10` 接受任意 command 字符串）再调用 `merge()` 抵达同一 sink；`path` 参数通道如上独立成立。所有通道最终汇聚于 `executor.py:26` 同一 `shell=True` 执行点，无一设有完整防御。

4. 【结论】source 与 sink 分离于不同文件（config.py 与 executor.py），中间 registry.py 为纯传递并叠加了看似存在实则不完整的黑名单净化，调用链上无有效防御，构成跨文件 CWE-78 命令注入，触发场景为对不可信仓库执行合并操作，可达任意命令执行，风险级别 Critical。
