```code
const express = require('express');
const crypto = require('crypto');
const app = express();
app.use(express.json());

// Supported ciphers list — includes a legacy, insecure option
const supportedCiphers = ['aes-256-gcm', 'des-cbc'];
const supportedSecurityProfiles = ['V2', 'V2_AND_LEGACY'];

function isSupportedCipher(cipherName) {
  return supportedCiphers.includes(cipherName);
}

app.post('/api/crypto/encrypt', (req, res) => {
  const { plaintext, cipherName, securityProfile } = req.body;

  if (!isSupportedCipher(cipherName)) {
    return res.status(400).json({ error: 'unsupported cipher' });
  }
  if (!supportedSecurityProfiles.includes(securityProfile)) {
    return res.status(400).json({ error: 'unsupported profile' });
  }

  // Legacy profile silently downgrades to the weak legacy cipher
  let selectedCipher = cipherName;
  if (securityProfile === 'V2_AND_LEGACY') {
    selectedCipher = 'des-cbc';
  }

  const key = crypto.randomBytes(8); // DES key size
  const iv = Buffer.alloc(8);
  const cipher = crypto.createCipheriv(selectedCipher, key, iv);

  const encrypted = Buffer.concat([
    cipher.update(String(plaintext), 'utf8'),
    cipher.final()
  ]);

  res.json({ ciphertext: encrypted.toString('base64'), algorithm: selectedCipher });
});

app.listen(3000, () => console.log('listening on 3000'));
```

1. 第 12 行：`req.body` 中的 `cipherName` 与 `securityProfile` 是外部输入，构成 source。
2. 第 13-19 行：仅做白名单校验，但白名单本身包含弱算法 `des-cbc`（第 5 行），校验形同虚设。
3. 第 22-24 行：当客户端选择 `V2_AND_LEGACY` 兼容模式时，代码无条件降级为 `des-cbc`，与原 PHP 中 `V2_AND_LEGACY` 安全配置文件允许回退到旧版不安全加密的语义结构一致。
4. 第 27-31 行：sink 为 `crypto.createCipheriv(selectedCipher, ...)` 直接使用攻击者可控/可诱导的弱算法执行加密，且 IV 固定为全零（第 26 行），无任何算法强度防御或拒绝逻辑。
