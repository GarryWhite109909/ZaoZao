```javascript
const express = require('express');
const crypto = require('crypto');

const app = express();
app.use(express.json());

// 对应原 AbstractCryptoClientV2 的白名单：legacy 安全档仍被放行
const SUPPORTED_CIPHERS = ['gcm'];
const SUPPORTED_SECURITY_PROFILES = ['V2', 'V2_AND_LEGACY'];
const LEGACY_SECURITY_PROFILES = ['V2_AND_LEGACY'];

function isSupportedCipher(cipherName) {
  return SUPPORTED_CIPHERS.includes(cipherName);
}

// 对应原 getCipherOpenSslName/buildCipherMethod：解析出 OpenSSL 算法名
function resolveAlgorithm(cipherName, keySize, securityProfile) {
  if (!SUPPORTED_SECURITY_PROFILES.includes(securityProfile)) {
    throw new Error('Unsupported security profile');
  }
  if (LEGACY_SECURITY_PROFILES.includes(securityProfile)) {
    // legacy 档直接降级为无认证 CBC，cipherName 白名单不参与
    return 'aes-' + keySize + '-cbc';
  }
  if (!isSupportedCipher(cipherName)) {
    throw new Error('Unsupported cipher');
  }
  return 'aes-' + keySize + '-gcm';
}

// 对应原 decrypt()：元数据信封字段由请求体直接提供
app.post('/api/vault/decrypt', (req, res) => {
  const { cipherName, keySize, securityProfile, iv, key, ciphertext, authTag } = req.body;
  const algorithm = resolveAlgorithm(cipherName, Number(keySize) || 256, securityProfile);
  const decipher = crypto.createDecipheriv(algorithm, Buffer.from(key, 'base64'), Buffer.from(iv, 'base64'));
  if (algorithm.endsWith('-gcm')) {
    decipher.setAuthTag(Buffer.from(authTag, 'base64'));
  }
  const plaintext = Buffer.concat([decipher.update(Buffer.from(ciphertext, 'base64')), decipher.final()]);
  res.type('text/plain').send(plaintext.toString('utf8'));
});

app.listen(3000, () => console.log('vault decrypt service listening on :3000'));
```

1. **Source 枚举与可达性**：第 32 行注册唯一路由 POST /api/vault/decrypt，第 5 行 express.json() 中间件把请求体整体解析进 req.body；第 33 行解构出的 cipherName/keySize/securityProfile/iv/key/ciphertext/authTag 七个字段全部为外部输入且无来源校验；第 34 行立即传入 resolveAlgorithm，第 35 行进入解密 sink，source→sink 在同一同步调用链内，可达性成立。

2. **主链防御验证（白名单自身包含风险选项）**：第 18 行对 securityProfile 做白名单检查，但第 9 行 SUPPORTED_SECURITY_PROFILES 自身包含 'V2_AND_LEGACY'——防御把风险选项写进了放行列表，攻击者提交 securityProfile='V2_AND_LEGACY' 即通过；第 21 行命中 LEGACY_SECURITY_PROFILES（第 10 行），第 23 行返回 'aes-<keySize>-cbc'。

3. **第二道防御验证（位置导致覆盖面为零）**：第 25 行 isSupportedCipher（第 12-14 行）仅放行 'gcm'，看似严格，但它位于第 21 行 legacy 分支之后，legacy 路径在第 23 行提前 return，cipherName 校验对主链零覆盖；第 36-38 行 setAuthTag 仅在算法名以 '-gcm' 结尾时执行，CBC 路径完全无完整性校验，且第 33 行 iv 由攻击者提供，CBC 下篡改 IV 可翻转首块明文，密文篡改与 padding oracle 均可行。

4. **Sink 确认**：第 35 行 crypto.createDecipheriv(algorithm, ...) 以第 23 行返回的无认证 CBC 算法初始化解密器；第 33-34 行 keySize 同为外部输入，可进一步降级为 'aes-128-cbc'；无认证 CBC 属于破损/风险密码算法，构成 CWE-327。

5. **第二入口/替代通道**：全文件仅第 32 行一个 HTTP 路由，不存在其他路由或间接调用能到达 createDecipheriv；但同一入口内 keySize（第 33-34 行）与在 legacy 路径被忽略的 authTag 构成替代弱化通道，且第 17 行 resolveAlgorithm 是共享决策点，未来任何复用它的路由都会继承同一降级缺陷；综上主链无有效防御，漏洞成立。
