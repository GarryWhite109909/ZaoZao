```python
from django.http import HttpResponse, JsonResponse
from django.shortcuts import get_object_or_404
from django.template import Template, Context
from django.urls import path

from .models import Form, Submission


def save_form_settings(request, form_id):
    """Any user who can open a form's settings page may store arbitrary text
    into the success message / submission title fields (no permission check)."""
    form = get_object_or_404(Form, pk=form_id)
    if request.method == 'POST':
        form.success_message = request.POST.get('success_message', '')
        form.submission_title = request.POST.get('submission_title', '')
        form.save()
        return JsonResponse({'status': 'ok'})
    return JsonResponse({'status': 'method_not_allowed'}, status=405)


def submission_success(request, submission_id):
    """Success page rendered after a submission: the stored strings are
    compiled and executed by the Django template engine as raw template source."""
    submission = get_object_or_404(
        Submission.objects.select_related('form'), pk=submission_id
    )
    form = submission.form

    ctx = Context({'submission': submission, 'form': form})
    title = Template(form.submission_title).render(ctx)
    message = Template(form.success_message).render(ctx)

    return HttpResponse('<h1>{}</h1><p>{}</p>'.format(title, message))


urlpatterns = [
    path('forms/<int:form_id>/settings/', save_form_settings),
    path('submissions/<int:submission_id>/success/', submission_success),
]
```
