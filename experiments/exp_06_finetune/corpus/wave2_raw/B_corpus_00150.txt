from flask import Flask, request, jsonify
from app.services import TableChatService

app = Flask(__name__)
service = TableChatService()


@app.route("/dataset/chat", methods=["POST"])
def dataset_chat():
    payload = request.get_json(force=True)
    data_ref = payload.get("data", "")      # source: 用户可控的本地路径或远程URL
    result = service.load_dataset(data_ref)
    return jsonify({"summary": result})


@app.route("/dataset/preview", methods=["GET"])
def dataset_preview():
    data_ref = request.args.get("url", "")  # 第二入口: 查询参数同样直达服务层

# === file: app/services.py ===
import re
import urllib.request

from app.parser import load_tabular


_DTD_RE = re.compile(r"<!\s*DOCTYPE", re.IGNORECASE)


def _screen(raw_text: str) -> bool:
    """看似安全检查: 仅扫描前 256 个字符是否出现 DOCTYPE"""
    head = raw_text[:256]
    return _DTD_RE.search(head) is None


class TableChatService:
    def load_dataset(self, data_ref: str) -> str:
        if data_ref.startswith(("http://", "https://")):
            with urllib.request.urlopen(data_ref, timeout=5) as resp:
                raw = resp.read()
        else:
            with open(data_ref, "rb") as fh:
                raw = fh.read()
        text = raw.decode("utf-8", errors="replace")
        if not _screen(text):
            return "blocked: suspicious DTD"
        rows = load_tabular(text)           # 污点原样流入解析器
        cols = sorted({k for row in rows for k in row})
        return f"columns: {', '.join(cols)}"
# === file: app/parser.py ===
import io

from lxml import etree


NS_ROW = ".//row"


def load_tabular(text: str) -> list[dict]:
    # sink: 解析器显式允许实体展开与 DTD 加载, 无任何外部实体限制
    parser = etree.XMLParser(
        resolve_entities=True,
        load_dtd=True,
        no_network=False,
    )
    doc = etree.parse(io.BytesIO(text.encode("utf-8")), parser)
    root = doc.getroot()
    rows = []
    for node in root.findall(NS_ROW):
        rows.append({child.tag: child.text for child in node})
    return rows
