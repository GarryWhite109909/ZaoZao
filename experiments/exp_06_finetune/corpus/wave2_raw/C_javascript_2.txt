```code
// scripts/rebuild-native.js
// 项目维护用辅助脚本：重新构建原生依赖并清理缓存。
// 所有命令均为硬编码常量，不接受任何外部输入。

const { execSync } = require('child_process');

// 配置常量：仅允许的包名与 npm 命令模板
const TARGET_PACKAGES = ['node-sass', 'sharp'];
const NPM_REBUILD_CMD = 'npm rebuild';
const NPM_CACHE_CMD = 'npm cache clean --force';
const LOG_PREFIX = '[rebuild-native]';

function log(message) {
  console.log(`${LOG_PREFIX} ${message}`);
}

function rebuildPackages() {
  for (const pkg of TARGET_PACKAGES) {
    // 危险点 1：execSync 执行拼接后的命令
    const cmd = `${NPM_REBUILD_CMD} ${pkg}`;
    log(`running: ${cmd}`);
    execSync(cmd, { stdio: 'inherit' });
  }
}

function cleanCache() {
  // 危险点 2：execSync 执行写死的缓存清理命令
  log(`running: ${NPM_CACHE_CMD}`);
  execSync(NPM_CACHE_CMD, { stdio: 'inherit' });
}

rebuildPackages();
cleanCache();
log('done');
```

2. 全文未出现 `process.argv`、`process.env`、`require('readline')`、网络请求或文件读取等污染源；`pkg` 仅由常量数组迭代产生，因此第 21 行拼接结果在编译期即可确定为 `npm rebuild node-sass` / `npm rebuild sharp`。
