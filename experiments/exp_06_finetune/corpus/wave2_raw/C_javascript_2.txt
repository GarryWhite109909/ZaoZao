// scripts/rebuild-assets.js
// 构建辅助脚本：重新构建前端静态资源
const { execSync } = require('child_process');

// 全部命令均为写死的构建常量，不接受任何外部输入
const BUILD_COMMANDS = [
  'npm run clean',
  'npm run build:css',
  'npm run build:js',
  'npm run build:docs'
];

const PROJECT_ROOT = '/var/www/app';

function rebuildAssets() {
  console.log('[rebuild] start');
  for (const cmd of BUILD_COMMANDS) {
    // execSync 的实参是常量数组元素，非外部输入
    const output = execSync(cmd, {
      cwd: PROJECT_ROOT,
      stdio: 'pipe',
      encoding: 'utf8'
    });
    console.log(`[rebuild] ${cmd} done`);
    if (output && output.trim().length > 0) {
      console.log(output.trim());
    }
  }
  console.log('[rebuild] all steps finished');
}

rebuildAssets();
