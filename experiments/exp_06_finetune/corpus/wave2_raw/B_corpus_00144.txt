from flask import Blueprint, request, jsonify

from app.services import NoteIngestionService

bp = Blueprint("notes", __name__)
service = NoteIngestionService()


@bp.route("/notes/import", methods=["POST"])
def import_notes():
    # attacker-controlled ENEX export path arrives straight from the request
    enex_path = request.args.get("path") or request.form.get("path")
    if not enex_path:
        return jsonify({"error": "missing path"}), 400
    docs = service.import_export(enex_path)
    return jsonify({"imported": len(docs)}), 200

# === file: app/services.py ===
import os
from typing import List, Dict

from app.enex_parser import EnexNoteParser


class NoteIngestionService:
    """Prepare Evernote ENEX exports before parsing."""

    def __init__(self) -> None:
        self._parser = EnexNoteParser()

    def import_export(self, raw_path: str) -> List[Dict]:
        path = raw_path.strip()
        if not path.lower().endswith(".enex"):
            raise ValueError("only .enex exports accepted")
        # naive scrub of entity markers; plain replace is bypassable
        cleaned = path.replace("&lt;", "").replace("%26", "")
        if not os.path.exists(cleaned):
            raise FileNotFoundError(cleaned)
        return list(self._parser.parse_file(cleaned))

# === file: app/enex_parser.py ===
import logging
from typing import Any, Dict, Iterator, List

logger = logging.getLogger(__name__)


class EnexNoteParser:
    """Stream notes out of an ENEX xml export file."""

    def parse_file(self, xml_file: str) -> Iterator[Dict[str, Any]]:
        from lxml import etree

        # huge_tree + recover stay lenient; entity resolution left enabled
        context = etree.iterparse(
            xml_file,
            encoding="utf-8",
            strip_cdata=False,
            huge_tree=True,
            recover=True,
        )
        for action, elem in context:
            if elem.tag == "note":
                yield self._parse_note(elem)

    def _parse_note(self, note: List) -> Dict[str, Any]:
        note_dict: Dict[str, Any] = {}
        for elem in note:
            if elem.tag == "content":
                note_dict["content"] = elem.text
            else:
                note_dict[elem.tag] = elem.text
        return note_dict

2. **防御逐条核验**：`services.py` L15 的 `.enex` 后缀检查只作用于文件名字符串，完全不校验 XML 内容，DTD/ENTITY 声明原样放行；L18 的 `replace('&lt;','')`/`replace('%26','')` 属黑名单式字符串替换，可用数字字符引用（如 `&#60;`）、大小写变形或直接省略这些标记绕过，且其作用对象是路径字符串而非被解析的文档体，对实际载荷覆盖面为零，不构成有效防御；`enex_parser.py` L14-L20 未设置 `resolve_entities=False`、`no_network=True`，也未注册任何自定义 Resolver，libxml2 默认解析器会展开外部实体。
3. **sink 确认与危害**：`enex_parser.py` L14 启动解析时叠加 `huge_tree=True`（L18，关闭大小/深度保护）与 `recover=True`（L19，容忍畸形 DTD 继续解析），外部实体内容在 L29/L31 被写入 `elem.text` 并装入返回字典向上层回传，实现任意本地文件读取与内网 SSRF。
4. **第二入口与替代通道**：项目仅存在这一个蓝图与一个路由函数，无其他 HTTP 端点，也无其他模块间接调用 `parse_file`（唯一调用点为 `services.py` L21）；但同一路由 L12 的 form 参数构成与 query 参数并列的第二 source 通道，二者汇合后沿完全相同的调用链抵达同一 sink；此外 `services.py` L19 的 `os.path.exists` 还附带提供文件存在性探测的旁路信息泄露通道。多通道汇聚进一步支撑漏洞成立的结论。
5. **结论**：整条调用链上不存在任何有效防御（仅有可绕过的黑名单替换与无关的后缀检查），判定 CWE-611 XXE 成立；因可实现文件读取/SSRF 但无代码执行，风险评级 High。
