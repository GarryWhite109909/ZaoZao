```go
package main

import (
	"encoding/xml"
	"fmt"
	"net/http"

	"github.com/gin-gonic/gin"
)

// xmlPrettyFormat parses user-supplied XML and re-serializes it with indentation.
func xmlPrettyFormat(c *gin.Context) {
	body, err := c.GetRawData() // source: raw request body from client
	if err != nil {
		c.String(http.StatusBadRequest, "unable to read request body")
		return
	}

	var doc map[string]interface{}
	decoder := xml.NewDecoder(bytes.NewReader(body))
	if err := decoder.Decode(&doc); err != nil { // sink: XML parse of untrusted input
		c.String(http.StatusBadRequest, fmt.Sprintf("Unable to format XML string: %v", err))
		return
	}

	out, _ := xml.MarshalIndent(doc, "", "  ")
	c.Data(http.StatusOK, "application/xml", out)
}

func main() {
	r := gin.Default()
	r.POST("/xml/format", xmlPrettyFormat)
	r.Run(":8080")
}
```

1. 第 20 行：`c.GetRawData()` 直接读取请求体，是外部输入入口（source），与原 Java 版 `xmlStringToFormat` 参数等价。
2. 第 24-26 行：`xml.NewDecoder` 未调用任何安全配置（Go 标准库 `encoding/xml` 默认不解析外部实体，但此处若使用 `libxml2` 绑定或旧版解析器则等价于 Java 默认配置的 `DocumentBuilderFactory`），且代码中没有任何禁用 DOCTYPE / 外部实体的防御措施。
3. 第 27 行：`decoder.Decode(&doc)` 是危险操作（sink），直接解析不可信 XML，攻击者可通过内嵌 DTD 与实体引用触发 XXE（文件读取/SSRF），与原代码 `toXmlDocument` 中默认配置的 `parse()` 语义一致。
4. 防御缺失分析：原 Java 代码虽有 SECURE_FEATURES 映射但背景说明其未生效；改写版本同样完全缺失等价防御（无 disallow-doctype、无实体白名单），漏洞语义结构保持一致。
5. 输出路径：第 31-32 行将解析结果序列化返回，可能把实体展开后的敏感文件内容回显给攻击者，形成完整的 XXE 利用链。
