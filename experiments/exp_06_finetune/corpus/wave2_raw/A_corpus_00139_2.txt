package main

import (
	"net/http"

	"github.com/gin-gonic/gin"
	libxml2 "github.com/lestrrat-go/libxml2"
	"github.com/lestrrat-go/libxml2/parser"
)

// POST /api/xml/format : pretty-print an uploaded XML document.
func formatXMLHandler(c *gin.Context) {
	raw, err := c.GetRawData()
	if err != nil {
		c.String(http.StatusBadRequest, "cannot read request body")
		return
	}

	// Parse the untrusted XML. The default parser options are used,
	// so external entities / DTD loading are NOT disabled.
	p := parser.New()
	doc, err := p.Parse(raw)
	if err != nil {
		c.String(http.StatusBadRequest, "invalid XML")
		return
	}
	defer doc.Free()

	formatted, err := doc.ToString(2)
	if err != nil {
		c.String(http.StatusInternalServerError, "format failed")
		return
	}
	c.Data(http.StatusOK, "application/xml", []byte(formatted))
}

func main() {
	r := gin.Default()
	r.POST("/api/xml/format", formatXMLHandler)
	r.Run(":8080")
}
