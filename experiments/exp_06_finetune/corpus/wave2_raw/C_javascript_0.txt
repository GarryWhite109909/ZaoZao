```code
// scripts/rebuild-native.js
// 项目构建辅助脚本：重新编译原生模块并清理缓存
// 用途：npm run rebuild 时由 package.json 调用

const { execSync } = require('child_process');
const path = require('path');

// 全部为编译期写死的配置常量
const PROJECT_ROOT = path.resolve(__dirname, '..');
const NODE_VERSION = '18';
const BUILD_TARGET = 'release';
const CLEAN_CMD = 'npm cache clean --force';
const REBUILD_CMD = 'npm rebuild --target=' + NODE_VERSION + ' --build-from-source';
const AUDIT_CMD = 'npm audit fix --package-lock-only';

function runStep(label, cmd) {
  console.log('[rebuild-native] ' + label);
  execSync(cmd, {
    cwd: PROJECT_ROOT,
    stdio: 'inherit',
    shell: '/bin/bash',
    timeout: 300000
  });
}

function main() {
  runStep('cleaning npm cache', CLEAN_CMD);
  runStep('rebuilding native modules', REBUILD_CMD);
  runStep('refreshing lockfile audit', AUDIT_CMD);
  console.log('[rebuild-native] done, target=' + BUILD_TARGET);
}

main();
```

1. 第 8~12 行：`CLEAN_CMD`、`REBUILD_CMD`、`AUDIT_CMD` 均为字面量字符串与常量拼接（`NODE_VERSION` 也是字面量），文件中不存在任何读取外部输入的语句（无 argv/env/stdin/网络）。
2. 第 15~23 行：`runStep` 内调用 `execSync`（危险 sink），其实参 `cmd` 只能来自上述三个常量；`cwd`、`shell`、`timeout` 也全部是常量。
3. 第 26~30 行：`main` 中三次调用 `runStep` 传入的都是常量命令，污点传播路径上没有任何外部数据源，因此 sink 的实参不可被污染。
4. 结论：所有危险 API（`execSync`、`shell` 选项）实参均为编译期常量，不存在可达的污点路径，样本不可利用。
