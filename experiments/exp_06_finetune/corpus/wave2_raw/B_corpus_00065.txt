from flask import Flask, request, jsonify

from app.services import embed_local_images

app = Flask(__name__)


@app.route('/render', methods=['POST'])
def render_markdown():
    # 外部输入点1：POST 表单 content，img 标签的 src 属性完全由用户控制
    content = request.form.get('content', '')
    html = embed_local_images(content)
    return jsonify({'html': html})


@app.route('/preview', methods=['GET'])
def preview():
    # 外部输入点2（第二入口）：查询参数 md 进入同一条嵌入链
    raw = request.args.get('md', '')
    return embed_local_images(raw)


if __name__ == '__main__':
    app.run()

# === file: app/services.py ===
import os
import re

from app.file_reader import read_as_data_uri

RE_IMG_SRC = re.compile(r'<img[^>]*?\ssrc\s*=\s*["\']([^"\']+)["\']', re.I)
BASE_PATH = '/srv/app/static/uploads'


def _is_remote(path):
    return path.startswith(('http://', 'https://', 'data:'))


def sanitize_rel_path(path):
    # 看似净化：仅删除第一次出现的 ../，黑名单式替换可被重复序列绕过
    return path.replace('../', '', 1)


def embed_local_images(text):
    def _repl(m):
        raw = m.group(1)
        if _is_remote(raw):
            return m.group(0)
        rel = sanitize_rel_path(raw)
        # normpath 只归一化，不保证结果仍在 BASE_PATH 之内
        full = os.path.normpath(os.path.join(BASE_PATH, rel))
        if os.path.exists(full):
            return m.group(0).replace(raw, read_as_data_uri(full))
        return m.group(0)
    return RE_IMG_SRC.sub(_repl, text)

# === file: app/file_reader.py ===
import base64
import os

MIME_TYPES = {
    '.png': 'image/png',
    '.jpg': 'image/jpeg',
    '.svg': 'image/svg+xml',
}


def read_as_data_uri(file_name):
    # sink：直接打开上游解析出的路径，无任何路径合法性校验
    ext = os.path.splitext(file_name)[1].lower()
    mime = MIME_TYPES.get(ext, 'application/octet-stream')
    with open(file_name, 'rb') as f:
        payload = base64.b64encode(f.read()).decode('ascii')
    return 'data:{};base64,{}'.format(mime, payload)
