# === file: app/Controllers/Appointments.php ===
<?php
defined('BASEPATH') or exit('No direct script access allowed');

/**
 * Appointments controller — 承接原 index/find 的越权读取入口。
 */
class Appointments extends EA_Controller
{
    public function __construct()
    {
        parent::__construct();
        $this->load->model('appointments_model');
        $this->load->library('booking_service');
    }

    public function find(): void
    {
        try {
            method('get');
            if (cannot('view', PRIV_APPOINTMENTS)) {
                abort(403, 'Forbidden');
            }
            check('appointment_id', 'numeric');
            $appointment_id = request('appointment_id'); // SOURCE: 用户可控主键
            $appointment = $this->booking_service->load($appointment_id);
            json_response($appointment);
        } catch (Throwable $e) {
            json_exception($e);
        }
    }
}

# === file: app/Libraries/Booking_service.php ===
<?php
defined('BASEPATH') or exit('No direct script access allowed');

/**
 * Booking service — 中间层：看似有校验，实际只验格式。
 */
class Booking_service
{
    public function load(string $appointment_id): array
    {
        // 仅验证正整数格式，未校验该预约是否属于当前用户
        $appointment_id = trim($appointment_id);
        if (!ctype_digit($appointment_id) || (int) $appointment_id <= 0) {
            throw new InvalidArgumentException('Invalid appointment ID.');
        }
        $CI =& get_instance();
        return $CI->appointments_model->find((int) $appointment_id); // 转发至 sink
    }
}

# === file: app/Models/Appointments_model.php ===
<?php
defined('BASEPATH') or exit('No direct script access allowed');

/**
 * Appointments model — 数据访问层。
 */
class Appointments_model extends EA_Model
{
    public function find(int $appointment_id): array
    {
        // SINK: 缺少 id_users_provider / id_users_customer 归属过滤
        return $this->db
            ->get_where('appointments', ['id' => $appointment_id])
            ->row_array();
    }
}
