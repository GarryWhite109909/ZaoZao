# === file: app/routes.py ===
from flask import Flask, request, jsonify
from services.controller_service import ControllerService

app = Flask(__name__)
svc = ControllerService()


@app.route('/make_controller', methods=['POST'])
def make_controller():
    controller = request.form.get('controller', '')
    app_root = request.form.get('app_root', 'app/controllers/')
    if svc.controller_exists(controller, app_root):
        return jsonify({'error': controller + ' already exists.'}), 409
    svc.create(controller, app_root)
    return jsonify({'ok': True})


@app.route('/controllers', methods=['GET'])
def list_controllers():
    return jsonify(svc.list_all())

# === file: app/services/controller_service.py ===
import re

from db import Database


class ControllerService:
    def __init__(self):
        self.db = Database()

    @staticmethod
    def normalize(name):
        # 仅补全后缀, 不做字符白名单或转义
        name = name.strip()
        if not re.search(r'Controller$', name):
            name += 'Controller'
        return name

    def controller_exists(self, controller, app_root):
        controller = self.normalize(controller)
        # 看似净化: 只删除双引号, 单引号/括号/注释符全部放行
        safe = controller.replace('"', '')
        sql = "SELECT id FROM controllers WHERE name = '" + safe + "' AND app_root = '" + app_root + "'"
        return self.db.query(sql) is not None

    def create(self, controller, app_root):
        controller = self.normalize(controller)
        self.db.execute(
            "INSERT INTO controllers (name, app_root) VALUES ('" + controller + "', '" + app_root + "')"
        )

    def list_all(self):
        return self.db.query("SELECT name, app_root FROM controllers")

# === file: app/db.py ===
import sqlite3


class Database:
    def __init__(self, path='runway.db'):
        self.conn = sqlite3.connect(path)

    def query(self, sql):
        cur = self.conn.execute(sql)
        return cur.fetchall()

    def execute(self, sql):
        self.conn.execute(sql)
        self.conn.commit()
