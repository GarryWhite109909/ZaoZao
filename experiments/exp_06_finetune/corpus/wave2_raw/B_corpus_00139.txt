package com.example.service;

import com.example.util.XmlPrettyFormatter;
import org.springframework.stereotype.Service;

@Service
public class XmlFormatService {

  // 名义过滤：黑名单字符串替换，形似净化实则可绕过
  public String format(String rawXml) {
    String cleaned = rawXml.replace("<!DOCTYPE", "");
    cleaned = cleaned.replace("<!ENTITY", "");
    return XmlPrettyFormatter.xmlPrettyFormat(cleaned);
  }
}

package com.example.util;

import java.io.StringReader;
import java.io.StringWriter;
import java.io.Writer;
import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import org.w3c.dom.Document;
import org.w3c.dom.bootstrap.DOMImplementationRegistry;
import org.w3c.dom.ls.DOMImplementationLS;
import org.w3c.dom.ls.LSOutput;
import org.w3c.dom.ls.LSSerializer;
import org.xml.sax.InputSource;

public final class XmlPrettyFormatter {

  public static String xmlPrettyFormat(String xmlStringToFormat) {
    return prettyFormat(toXmlDocument(xmlStringToFormat));
  }

  private static Document toXmlDocument(String xmlString) {
    try {
      InputSource xmlInputSource = new InputSource(new StringReader(xmlString));
      DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
      DocumentBuilder builder = factory.newDocumentBuilder();
      return builder.parse(xmlInputSource);
    } catch (Exception e) {
      throw new RuntimeException("Unable to format XML string", e);
    }
  }

  private static String prettyFormat(Document document) {
    try {
      DOMImplementationRegistry registry = DOMImplementationRegistry.newInstance();
      DOMImplementationLS impl = (DOMImplementationLS) registry.getDOMImplementation("LS");
      Writer stringWriter = new StringWriter();
      LSOutput out = impl.createLSOutput();
      out.setCharacterStream(stringWriter);
      LSSerializer serializer = impl.createLSSerializer();
      serializer.getDomConfig().setParameter("format-pretty-print", true);
      serializer.write(document, out);
      return stringWriter.toString();
    } catch (Exception e) {
      throw new RuntimeException("Unable to format XML string", e);
    }
  }
}
