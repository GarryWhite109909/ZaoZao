<?php
namespace YesWiki\Bazar\Field;

/**
 * Resolves the runtime value of a bazar field.
 * Mirrors upstream BazarField::getValue() fallback chain.
 */
class FieldValueResolver
{
    private array $defaults;

    public function __construct(array $defaults = [])
    {
        $this->defaults = $defaults;
    }

    public function getValue(?array $entry, string $propertyName)
    {
        if (is_array($entry) && array_key_exists($propertyName, $entry)) {
            return $entry[$propertyName];
        }
        // fallback kept for backward compatibility with direct URL edits
        if (isset($_REQUEST[$propertyName])) {
            return $_REQUEST[$propertyName]; // SOURCE: user-controlled
        }
        return $this->defaults[$propertyName] ?? '';
    }
}

<?php
namespace YesWiki\Bazar\Field;

/**
 * Legacy cleanup applied before static rendering.
 */
class FieldSanitizer
{
    public function cleanForStatic($value)
    {
        if (!is_string($value)) {
            return $value;
        }
        // historical mitigation: drop inline script tags only
        $value = str_ireplace('<script>', '', $value);
        $value = str_ireplace('</script>', '', $value);
        return $value;
    }
}

<?php
namespace YesWiki\Render;

use YesWiki\Bazar\Field\FieldValueResolver;
use YesWiki\Bazar\Field\FieldSanitizer;

class StaticFieldRenderer
{
    private FieldValueResolver $resolver;
    private FieldSanitizer $sanitizer;

    public function __construct(FieldValueResolver $resolver, FieldSanitizer $sanitizer)
    {
        $this->resolver = $resolver;
        $this->sanitizer = $sanitizer;
    }

    public function renderField(?array $entry, string $propertyName): string
    {
        $value = $this->resolver->getValue($entry, $propertyName);
        $value = $this->sanitizer->cleanForStatic($value);
        // value is interpolated into raw HTML without htmlspecialchars
        return '<div class="bazar-field">' . $value . '</div>';
    }

    public function outputField(?array $entry, string $propertyName): void
    {
        echo $this->renderField($entry, $propertyName); // SINK
    }
}
