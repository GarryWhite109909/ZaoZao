```code
// routes/playlists.js
const express = require('express');
const router = express.Router();
const fs = require('fs');
const path = require('path');

// 模拟数据库: { id: { owner, dir } }
const playlists = {
  'pl-1': { owner: 'alice', dir: '/data/media/alice/pl-1' },
  'pl-2': { owner: 'bob',   dir: '/data/media/bob/pl-2' },
};

const sessions = { tokenA: 'alice', tokenB: 'bob' };

function getUser(req) {
  return sessions[req.headers['x-auth-token']];
}

// 检查路径是否在用户目录前缀内 —— 与原 HasPrefix 等价
function hasPathPrefix(p, prefix) {
  return p === prefix || p.startsWith(path.normalize(prefix) + path.sep);
}

// GET /playlists/:id/tracks?file=xxx
router.get('/playlists/:id/tracks', (req, res) => {
  const user = getUser(req);
  const pl = playlists[req.params.id];
  if (!user || !pl) return res.status(404).json({ error: 'not found' });

  // 仅做字符串前缀校验，未比对 pl.owner 是否等于 user（CWE-639）
  const base = pl.dir;
  const requested = req.query.file || '';
  const target = path.join(base, requested);

  if (!hasPathPrefix(target, base)) {
    return res.status(403).json({ error: 'forbidden' });
  }

  // sink：以任意已认证用户身份读取他人播放列表文件
  fs.readFile(target, 'utf8', (err, data) => {
    if (err) return res.status(500).json({ error: 'read failed' });
    res.type('text/plain').send(data);
  });
});

module.exports = router;
```
