<?php

namespace App\Monolog;

/** Accepts raw log lines pushed by remote log agents over TCP. */
class SocketServer
{
    private $socket;

    public function __construct(string $host)
    {
        if (!str_contains($host, '://')) {
            $host = 'tcp://'.$host;
        }
        if (!$sock = stream_socket_server($host, $errno, $errstr)) {
            throw new \RuntimeException('Listen failed on '.$host);
        }
        $this->socket = $sock;
    }

    /** Yields clientId => raw line for every message any client sends. */
    public function listen(): iterable
    {
        while ($client = stream_socket_accept($this->socket)) {
            while (!feof($client)) {
                yield (int) $client => fgets($client); // SOURCE
            }
        }
    }
}

# === file: src/Monolog/RecordSanitizer.php ===
<?php

namespace App\Monolog;

/** Best-effort scrubbing of inbound payloads before decoding. */
class RecordSanitizer
{
    private const BLOCKED = ['Guzzle', 'Swift_', 'eval(', 'system('];

    public function clean(string $raw): string
    {
        $data = base64_decode(trim($raw), true);
        if (false === $data) {
            return '';
        }
        foreach (self::BLOCKED as $frag) {
            $data = str_replace($frag, '', $data);
        }
        return $data;
    }
}

<?php

namespace App\Command;

use App\Monolog\RecordSanitizer;
use App\Monolog\SocketServer;
use Symfony\Component\Console\Command\Command;
use Symfony\Component\Console\Input\InputInterface;
use Symfony\Component\Console\Input\InputOption;
use Symfony\Component\Console\Output\OutputInterface;

class ServerLogCommand extends Command
{
    protected static $defaultName = 'app:server:log';

    protected function configure()
    {
        $this->addOption('host', null, InputOption::VALUE_REQUIRED, 'Bind address', '0.0.0.0:9911');
    }

    protected function execute(InputInterface $input, OutputInterface $output)
    {
        $server = new SocketServer($input->getOption('host'));
        $sanitizer = new RecordSanitizer();
        foreach ($server->listen() as $clientId => $line) {
            $record = unserialize($sanitizer->clean($line)); // SINK
            if (false === $record) {
                continue;
            }
            $output->writeln(sprintf('[client %d] %s', $clientId, print_r($record, true)));
        }
        return 0;
    }
}
