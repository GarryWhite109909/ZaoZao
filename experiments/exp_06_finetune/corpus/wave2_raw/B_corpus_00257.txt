namespace App\Controller\Admin;

use App\Service\CompositeIndexService;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\JsonResponse;

class DataObjectController
{
    public function __construct(private CompositeIndexService $indexService) {}

    public function updateIndicesAction(Request $request): JsonResponse
    {
        $table = (string) $request->request->get('table', '');
        $type = (string) $request->request->get('index_type', 'query');
        $indices = $request->request->all('composite_indices');

        // 看似净化：黑名单删除 drop/delete/分号，双写或反引号即可绕过
        foreach (['drop', 'delete', ';'] as $bad) {
            $table = str_ireplace($bad, '', $table);
        }

        $this->indexService->applyCompositeIndices($table, $type, $indices);

        return new JsonResponse(['status' => 'ok']);
    }
}

<?php
declare(strict_types=1);

namespace App\Service;

use App\Model\CompositeIndexRepository;

class CompositeIndexService
{
    public function __construct(private CompositeIndexRepository $repository) {}

    public function applyCompositeIndices(string $table, string $type, array $compositeIndices): void
    {
        // 唯一严格校验只针对 index_type，三个污点变量均未处理
        if (!in_array($type, ['query', 'localized_query'], true)) {
            throw new \InvalidArgumentException('unsupported index type');
        }

        $normalized = [];
        foreach ($compositeIndices as $item) {
            $normalized[] = [
                'index_key' => 'c_' . ($item['index_key'] ?? ''),
                'index_columns' => $item['index_columns'] ?? '',
            ];
        }

        $this->repository->updateCompositeIndices($table, $normalized);
    }
}

<?php
declare(strict_types=1);

namespace App\Model;

use Doctrine\DBAL\Connection;

class CompositeIndexRepository
{
    public function __construct(private Connection $db) {}

    public function updateCompositeIndices(string $table, array $compositeIndices): void
    {
        $existing = $this->db->fetchAllAssociative(
            'SHOW INDEXES FROM ' . $this->db->quoteIdentifier($table) . " WHERE Key_Name LIKE 'c\\_%'"
        );
        $existingMap = [];
        foreach ($existing as $row) {
            $existingMap[$row['Key_name']] = $row['Column_name'];
        }
        foreach ($compositeIndices as $index) {
            $key = $index['index_key'];
            $columns = $index['index_columns'];
            if (isset($existingMap[$key]) && $existingMap[$key] === $columns) {
                continue;
            }
            // sink：表名/索引名/列清单直接拼接，quoteIdentifier 未在此复用
            $this->db->executeQuery('ALTER TABLE `' . $table . '` ADD INDEX `' . $key . '` (' . $columns . ');');
        }
    }
}
