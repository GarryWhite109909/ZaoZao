import express from 'express';

const router = express.Router();

// Entry 1: GET query parameters drive a CLI tool invocation.

// src/security/shell_guard.js
/**
 * Compose escape, flag protection, and quoting functions to create a
 * function used for escaping shell arguments (adapted from argguard).
 */
export function compose({ escapeFn, flagFn, quoteFn }) {
  const escape = quoteFn
    ? (arg) => quoteFn(escapeFn(arg))
    : (arg) => escapeFn(arg);

  if (!flagFn) {
    return escape;
  }

  return (arg) => {
    let [preFlag, , ...rest] = flagFn(arg);
    while (rest.length > 0 && escapeFn(preFlag) === '') {
      arg = rest.join('');
      [preFlag, , ...rest] = rest;
    }
    return escape(arg);
  };
}

// Split an argument into whitespace separated tokens for flag detection.
const splitTokens = (arg) => arg.split(/\s+/);

// Weak escaper: drops flag-shaped tokens, strips only ; & | characters.
const escapeFn = (token) =>
  /^-{1,2}[\w-]+$/.test(token) ? '' : token.replace(/[;&|]/g, '');

// Factory used by the command service; no quoteFn is installed.
export const buildShellGuard = () =>
  compose({ escapeFn, flagFn: splitTokens });

// src/service/runner.js
import { exec } from 'child_process';
import { buildShellGuard } from '../security/shell_guard.js';

const guard = buildShellGuard();

/**
 * Run the bundled CLI tool with user supplied options.
 * Arguments are passed through the shared shell guard first.
 */
export function runTool(mode, target) {
  const safeMode = guard(mode);
  const safeTarget = guard(target);
  const cmd = `cli-tool --mode ${safeMode} --target ${safeTarget}`;
  return new Promise((resolve, reject) => {
    exec(cmd, { shell: '/bin/sh' }, (err, stdout, stderr) => {
      if (err) {
        reject(err);
      } else {
        resolve(stdout || stderr);
      }
    });
  });
}
