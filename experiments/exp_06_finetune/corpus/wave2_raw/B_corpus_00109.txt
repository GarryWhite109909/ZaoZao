package jenkins.web;

import java.io.IOException;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import jenkins.model.PluginStore;
import jenkins.util.XmlGuard;

public class ConfigImportServlet extends HttpServlet {
    private final PluginStore pluginStore = new PluginStore();

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        String configXml = req.getParameter("config");               // [SOURCE]
        if (configXml == null || configXml.isEmpty()) {
            resp.sendError(400);
            return;
        }
        String filtered = XmlGuard.stripDangerousTags(configXml);
        pluginStore.loadFromXml(filtered);                           // [FLOW->SINK FILE]
        resp.setStatus(200);
    }
}

package jenkins.util;

/** Best-effort removal of well-known gadget markers before import. */
public final class XmlGuard {
    private static final String[] BLOCKED = {"<map>", "dynamic-proxy", "ProcessBuilder", "java.lang.Runtime"};

    public static String stripDangerousTags(String xml) {
        String out = xml;
        for (String token : BLOCKED) {
            out = out.replace(token, "");
        }
        return out;
    }
}

package jenkins.model;

import com.thoughtworks.xstream.XStream;
import com.thoughtworks.xstream.converters.MarshallingContext;
import com.thoughtworks.xstream.converters.UnmarshallingContext;
import com.thoughtworks.xstream.converters.collections.AbstractCollectionConverter;
import com.thoughtworks.xstream.io.HierarchicalStreamReader;
import com.thoughtworks.xstream.io.HierarchicalStreamWriter;
import com.thoughtworks.xstream.io.xml.XppDriver;
import com.thoughtworks.xstream.mapper.Mapper;
import java.io.StringReader;
import java.util.ArrayList;
import java.util.List;

public class PluginStore {
    private final XStream xstream = new XStream();

    public void loadFromXml(String xml) {
        HierarchicalStreamReader reader = new XppDriver().createReader(new StringReader(xml));
        xstream.registerConverter(new EntryListConverter(xstream.getMapper()));
        xstream.unmarshal(reader);                               // [SINK]
    }

    public static final class EntryListConverter extends AbstractCollectionConverter {
        public EntryListConverter(Mapper mapper) {
            super(mapper);
        }
        @Override
        public boolean canConvert(Class type) {
            return List.class.isAssignableFrom(type);
        }
        @Override
        public void marshal(Object src, HierarchicalStreamWriter w, MarshallingContext ctx) {
            throw new UnsupportedOperationException();
        }
        @Override
        public Object unmarshal(HierarchicalStreamReader r, UnmarshallingContext ctx) {
            List items = new ArrayList();
            while (r.hasMoreChildren()) {
                r.moveDown();
                items.add(readItem(r, ctx, items));              // gadget resolution
                r.moveUp();
            }
            return items;
        }
    }
}
