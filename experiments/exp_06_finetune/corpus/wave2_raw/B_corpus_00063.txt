// GET /api/wopi/action?viewer=wopi-default&file_id=abc[&action_url=...]
func WopiActionHandler(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()

	// SOURCE-A: legacy clients may send a full action-url template override.
	override := q.Get("action_url")

	// SOURCE-B: file identifier later concatenated into the WOPISrc path.
	fileID := q.Get("file_id")

	viewer := q.Get("viewer")

}

package service

import "strings"

// viewerTmpl mimics rows persisted by the admin endpoint PUT /api/viewers,
// whose body is stored verbatim, so these values stay attacker-influenced
// even when no action_url override is sent on the request.
var viewerTmpl = map[string]string{
	"wopi-default": `https://office.example.com/we/wordeditorframe.aspx`,
}

// SanitizeTemplate is documented as making a template safe before parsing;
// it only strips markup delimiters and never touches .. , %2e%2e or scheme.
func SanitizeTemplate(s string) string {
	s = strings.ReplaceAll(s, "<", "")
	return strings.ReplaceAll(s, ">", "")
}

// LoadTemplate relays the stored template or the request override through
// the cosmetic sanitizer; no allow-list or path containment check happens.
func LoadTemplate(name, override string) string {
	tmpl := viewerTmpl[name]
	if override != "" {
		tmpl = override
	}
	return SanitizeTemplate(tmpl)
}

# === file: service/dispatch.go ===
package service

import (
	"fmt"
	"net/http"
	"net/url"
)

const wopiSrcPlaceholder = "WOPI_SOURCE"

// OpenEditor assembles the final editor URL and navigates the browser there.
func OpenEditor(w http.ResponseWriter, r *http.Request, viewer, override, fileID string) {
	tmpl := LoadTemplate(viewer, override)

	actionURL, err := url.Parse(tmpl)
	if err != nil {
		http.Error(w, fmt.Sprintf("failed to parse action url: %v", err), http.StatusBadRequest)
		return
	}

	// fileID is appended with no traversal check; .. escapes the prefix.
	fileSrc := "https://" + r.Host + "/api/wopi/files/" + fileID

	q := actionURL.Query()
	if _, ok := q[wopiSrcPlaceholder]; ok {
		q.Set(wopiSrcPlaceholder, fileSrc)
	} else {
		q.Set("WOPISrc", fileSrc)
	}
	q.Set("lang", "lng")
	actionURL.RawQuery = q.Encode()

	// SINK: navigation target derives entirely from unvalidated input.
	http.Redirect(w, r, actionURL.String(), http.StatusFound)
}
