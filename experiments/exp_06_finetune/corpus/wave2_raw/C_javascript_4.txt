```code
// scripts/maintenance.js
// 项目维护辅助脚本：清理构建缓存并重新安装依赖。
// 所有命令均为硬编码常量，不接受任何外部输入。

const { execSync } = require('child_process');

const PROJECT_ROOT = '/var/www/my-app';
const CLEAN_CMD = 'rm -rf ./node_modules/.cache';
const INSTALL_CMD = 'npm ci --omit=dev';
const AUDIT_CMD = 'npm audit --omit=dev';

function runMaintenance() {
  try {
    console.log('[1/3] 清理构建缓存...');
    execSync(CLEAN_CMD, { cwd: PROJECT_ROOT, stdio: 'inherit' });

    console.log('[2/3] 重新安装生产依赖...');
    execSync(INSTALL_CMD, { cwd: PROJECT_ROOT, stdio: 'inherit' });

    console.log('[3/3] 执行安全审计...');
    const auditOutput = execSync(AUDIT_CMD, {
      cwd: PROJECT_ROOT,
      encoding: 'utf8',
    });
    console.log(auditOutput);
    return true;
  } catch (err) {
    console.error('维护任务失败:', err.message);
    return false;
  }
}

runMaintenance();
```
