```code
// scripts/rebuild-assets.js
// 构建辅助脚本：重新安装依赖并重建前端资源（CI 中使用）
const { execSync } = require('child_process');

// 全部命令均为硬编码常量，禁止修改为动态拼接
const INSTALL_CMD = 'npm ci --omit=dev';
const BUILD_CMD = 'npm run build';
const CLEAN_CMD = 'rm -rf ./dist';

function logStep(name) {
  console.log(`[build] step: ${name}`);
}

function main() {
  try {
    logStep('clean');
    execSync(CLEAN_CMD, { stdio: 'inherit', cwd: __dirname });

    logStep('install');
    execSync(INSTALL_CMD, { stdio: 'inherit' });

    logStep('build');
    const out = execSync(BUILD_CMD, { encoding: 'utf8' });
    console.log(out.trim().split('
').slice(-3).join('
'));

    console.log('[build] done');
  } catch (err) {
    console.error('[build] failed:', err.message);
    process.exit(1);
  }
}

main();
```
