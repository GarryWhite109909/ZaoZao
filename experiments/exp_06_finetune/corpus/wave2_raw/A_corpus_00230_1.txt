```code
const express = require('express');
const { execFile } = require('child_process');
const db = require('./db');

const app = express();
app.use(express.json());

// Console-style maintenance routes, intended for CLI/internal use only.
app.post('/console/install', (req, res) => {
  db.migrate('fresh');
  db.seed();
  res.send('
Installation completed, login with "administrator" / "administrator".
');
});

app.post('/console/migrate', (req, res) => {
  const type = req.body.type || '';
  db.migrate(type);
  res.send('Migration done.');
});

app.post('/console/backup', (req, res) => {
  const targetPath = req.query.path || req.body.path;
  execFile('mysqldump', ['-u', 'root', '-pSuperSecret123!', 'appointments'], (err, stdout) => {
    if (err) return res.status(500).send(err.message);
    res.attachment(targetPath || 'backup.sql').send(stdout);
  });
});

app.post('/console/sync', async (req, res) => {
  const providers = await db.providers.all();
  for (const p of providers) {
    if (!p.settings.google_sync) continue;
    await googleSync(String(p.id));
  }
  res.send('synced');
});

app.listen(3000);
```
