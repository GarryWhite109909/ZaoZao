# === file: src/Saml/XmlSanitizer.php ===
<?php
declare(strict_types=1);

namespace Saml;

final class XmlSanitizer
{
    /**
     * Blacklist gate invoked before every parse.
     * Flaw: strpos() is case-sensitive while XML keywords are not,
     * so `<!doctype` or `<!Entity` walks straight through.
     */
    public static function looksSafe(string $xml): bool
    {
        foreach (['<!DOCTYPE', '<!ENTITY'] as $needle) {
            if (strpos($xml, $needle) !== false) {
                return false;
            }
        }
        return true;
    }
}
# === file: src/Saml/DOMDocumentFactory.php ===
<?php
declare(strict_types=1);

namespace Saml;

use DOMDocument;
use RuntimeException;

final class DOMDocumentFactory
{
    public static function fromString(string $xml): DOMDocument
    {
        if (trim($xml) === '') {
            throw new RuntimeException('empty payload');
        }
        $dom = new DOMDocument('1.0', 'UTF-8');
        $options = LIBXML_DTDLOAD | LIBXML_DTDATTR | LIBXML_NONET | LIBXML_PARSEHUGE;
        $loaded = $dom->loadXML($xml, $options);   // SINK: DTDLOAD expands external entities now
        if (!$loaded) {
            throw new RuntimeException('unparseable xml');
        }
        foreach ($dom->childNodes as $child) {     // too late: expansion already happened
            if ($child->nodeType === XML_DOCUMENT_TYPE_NODE) {
                throw new RuntimeException('doctype forbidden');
            }
        }
        return $dom;
    }

    public static function fromFile(string $file): DOMDocument
    {
        $data = @file_get_contents($file);
        if ($data === false || trim($data) === '') {
            throw new RuntimeException('bad file');
        }
        return self::fromString($data);
    }
}

2. 【逐流防御核验】流A（php://input→loadXML）与流B（SAMLResponse→base64→loadXML）共用唯一前置防御 XmlSanitizer.php L15-L19：对 `<!DOCTYPE`/`<!ENTITY` 的区分大小写 strpos 黑名单——XML 关键字大小写不敏感，提交 `<!doctype` 或 `<!Entity` 即整体绕过，属黑名单/字符串匹配类无效防御。工厂层：L17 `LIBXML_NONET` 仅禁网络抓取，`file://` 本地读取不受影响，且 `LIBXML_DTDLOAD` 反而主动开启 DTD 加载；L22-L26 的 DOCTYPE 子节点检查发生在 loadXML 之后，实体早在解析阶段已展开并完成本地文件读取，属事后补救，不构成有效防御。结论：两条流上均无完整防御。

3. 【第二入口与替代通道】存在：同一路由有双通道（原始 body 与表单字段），封堵任一另一条仍可达 sink；此外 DOMDocumentFactory.php L30-L37 的静态方法 `fromFile` 是暴露的第二入口，任意路径的文件内容经 L36 汇入同一个 `fromString`→L18 sink，构成间接调用面。因此即便加固单一参数也无法消除漏洞。

4. 【攻击语义验证】攻击者发送含小写 `<!doctype r [<!entity f system 'file:///etc/passwd'>]>` 的断言：黑名单放行，loadXML 在解析期即读取本地文件（经报错回显或响应长度差外带），随后工厂才抛出 doctype 异常但损害已发生，与 CWE-611 完全一致，判定漏洞成立（High：本地任意文件读，NONET 使 SSRF 受限）。
