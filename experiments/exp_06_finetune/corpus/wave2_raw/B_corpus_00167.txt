from __future__ import annotations

from fastapi import HTTPException, status

from app.db.dag_repository import DagSourceRepository


def resolve_dag_source(session, dag_id: str, version_number: int | None):
    # 看似净化的中间层：仅做格式规整与黑名单子串检查，
    # 无法阻止任意 dag_id 枚举，亦无资源级属主校验。
    clean_id = dag_id.strip().lower()
    if not clean_id or '..' in clean_id:
        raise HTTPException(status.HTTP_400_BAD_REQUEST, detail='invalid dag_id')
    record = DagSourceRepository.fetch_source(session, clean_id, version_number)
    if record is None:
        raise HTTPException(
            status.HTTP_404_NOT_FOUND,
            detail=f'code not found. dag_id={clean_id!r} version={version_number}',
        )
    return record

from __future__ import annotations

from airflow.models.dag_version import DagVersion


class DagSourceRepository:
    @staticmethod
    def fetch_source(session, dag_id: str, version_number: int | None):
        # SINK: 以调用者可控的 dag_id 直接作主键查询并回传源码，
        # 查询前后均无 '该用户是否可读此具体 DAG' 的对象级判定。
        dag_version = DagVersion.get_version(dag_id, version_number, session=session)
        if not dag_version or not dag_version.dag_code:
            return None
        return {
            'dag_id': dag_id,
            'content': dag_version.dag_code.source_code,
            'version_number': dag_version.version_number,
            'dag_display_name': dag_version.dag_model.dag_display_name,
        }
