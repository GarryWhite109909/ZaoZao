from flask import Blueprint, jsonify, request

from app.services import validate_peppol_document

bp = Blueprint('peppol', __name__)


@bp.route('/validate', methods=['POST'])
def validate_document():
    # raw XML body straight from the client
    document_content = request.get_data()
    schematron = request.args.get('schematron', 'PEPPOL-EN16931-UBL.xsl')
    errors = validate_peppol_document(document_content, [schematron])
    return jsonify(errors=errors)


@bp.route('/validate/upload', methods=['POST'])
def validate_upload():
    # second entry: multipart invoice uploads
    results = []
    for uploaded in request.files.getlist('files'):
        document_content = uploaded.read()
        results.extend(validate_peppol_document(document_content, ['PEPPOL-EN16931-UBL.xsl']))
    return jsonify(results=results)

# === file: app/services.py ===
import re
from typing import List

from app.saxon_backend import run_schematron


_DOCTYPE_RE = re.compile(r'<!DOCTYPE\s+[^>\[]+>', re.IGNORECASE)


def sanitize_xml_header(document_content: bytes) -> bytes:
    # looks like XXE hardening, but the pattern skips internal subsets
    text = document_content.decode('utf-8', errors='ignore')
    return _DOCTYPE_RE.sub('', text).encode('utf-8')


def validate_peppol_document(document_content: bytes, schematron_xsls: List[str]) -> List[dict]:
    cleaned = sanitize_xml_header(document_content)
    return run_schematron(cleaned)

# === file: app/saxon_backend.py ===
from typing import List

from lxml import etree
from saxonche import PySaxonProcessor


_FAILED_ASSERT = '{http://purl.oclc.org/dsdl/svrl}failed-assert'


def run_schematron(document_content: bytes) -> List[dict]:
    errors = []
    with PySaxonProcessor(license=False) as proc:
        xsltproc = proc.new_xslt30_processor()
        # sink: untrusted XML parsed with DTD/entity resolution left on
        document = proc.parse_xml(xml_text=document_content.decode())
        executable = xsltproc.compile_stylesheet(
            stylesheet_file='data/sendpeppol-schematron/PEPPOL-EN16931-UBL.xsl')
        output = executable.transform_to_string(xdm_node=document)
        parsed_output = etree.fromstring(output.encode())
        for e in parsed_output.findall(_FAILED_ASSERT):
            errors.append({
                'text': e.findtext('{*}text'),
                'code': e.get('id'),
                'severity': e.get('flag'),
                'location': e.get('location'),
            })
    return errors
