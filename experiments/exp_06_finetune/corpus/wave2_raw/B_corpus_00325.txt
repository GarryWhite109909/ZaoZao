from flask import Flask, jsonify, request

from app.services import BackupService

app = Flask(__name__)
service = BackupService(base_dir='/var/backups')


@app.route('/api/backups', methods=['POST'])
def create_backup():
    body = request.get_json(force=True)
    backup_id = body['id']
    filename = body.get('filename') or f'{backup_id}.tar'
    service.register(backup_id, filename)
    return jsonify(status='created')


@app.route('/api/backups/<backup_id>', methods=['DELETE'])
def delete_backup(backup_id):
    service.delete(backup_id)
    return jsonify(status='deleted')

# === file: app/services.py ===
from app.storage import BackupStore


class BackupService:
    """Business layer: keeps backup metadata, delegates file operations."""

    def __init__(self, base_dir: str) -> None:
        self.store = BackupStore(base_dir)
        self._meta: dict[str, str] = {}

    def register(self, backup_id: str, filename: str) -> None:
        # Pseudo-sanitization: strips literal '../' only once.
        safe_name = filename.replace('../', '')
        self._meta[backup_id] = safe_name

    def delete(self, backup_id: str) -> None:
        name = self._meta.get(backup_id, f'{backup_id}.tar')
        path = self.store.build_path(name)
        self.store.remove(path)

# === file: app/storage.py ===
from pathlib import Path


class BackupStore:
    """Storage layer: all filesystem access for backups."""

    def __init__(self, base_dir: str) -> None:
        self.base = Path(base_dir)

    def build_path(self, name: str) -> Path:
        """Naive join: keeps '..' segments and absolute overrides."""
        return self.base / name

    def remove(self, path: Path) -> None:
        """Delete a backup file from disk."""
        path.unlink(missing_ok=True)
