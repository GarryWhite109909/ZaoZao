```code
#!/usr/bin/env python3
"""deploy.py - 固定流程部署脚本，所有命令均为硬编码常量。"""

import subprocess
import logging

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
LOG = logging.getLogger("deploy")

# 部署目标与命令全部为编译期常量，不接受任何外部输入
DEPLOY_DIR = "/opt/app/releases/current"
BACKUP_CMD = ["tar", "-czf", "/var/backups/app-current.tar.gz", DEPLOY_DIR]
BUILD_CMD = ["make", "-C", "/opt/app", "build"]
RESTART_CMD = ["systemctl", "restart", "app.service"]


def run_step(cmd, step_name):
    """执行固定命令列表；cmd 为模块级常量列表。"""
    LOG.info("step: %s", step_name)
    result = subprocess.run(
        cmd,
        capture_output=True,
        text=True,
        timeout=300,
        check=False,
    )
    if result.returncode != 0:
        LOG.error("%s failed: %s", step_name, result.stderr)
    return result.returncode


def main():
    steps = [
        ("backup", BACKUP_CMD),
        ("build", BUILD_CMD),
        ("restart", RESTART_CMD),
    ]
    for name, cmd in steps:
        rc = run_step(cmd, name)
        if rc != 0:
            raise SystemExit(f"deployment aborted at {name}")


if __name__ == "__main__":
    main()
```

2. 文件中不存在 `input()`、`sys.argv`、`os.environ`、网络读取等任何污点源（source），因此 `cmd` 不可能携带攻击者可控内容。
3. `subprocess.run` 未使用 `shell=True`，且列表元素均为常量字符串，即使理论上有注入意图也无拼接点。
