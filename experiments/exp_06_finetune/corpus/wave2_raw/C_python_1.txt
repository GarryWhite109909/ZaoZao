```code
#!/usr/bin/env python3
"""deploy.py - 固定流程部署脚本，所有命令均为硬编码常量。"""

import subprocess

DEPLOY_DIR = "/opt/myapp"
BACKUP_CMD = ["tar", "-czf", "/var/backups/myapp.tar.gz", "/opt/myapp/current"]
STOP_CMD = ["systemctl", "stop", "myapp.service"]
COPY_CMD = ["rsync", "-a", "/build/artifacts/", "/opt/myapp/current/"]
START_CMD = ["systemctl", "start", "myapp.service"]


def run_step(cmd):
    """执行固定命令列表；cmd 仅由模块级常量传入。"""
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
    if result.returncode != 0:
        raise RuntimeError("step failed: " + cmd[0])
    return result.stdout


def deploy():
    run_step(BACKUP_CMD)
    run_step(STOP_CMD)
    run_step(COPY_CMD)
    run_step(START_CMD)
    print("deploy finished")


if __name__ == "__main__":
    deploy()
```
