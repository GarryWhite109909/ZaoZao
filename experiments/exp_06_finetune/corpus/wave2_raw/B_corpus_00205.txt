# === file: application/models/Settings_model.php ===
<?php
defined('BASEPATH') or exit('No direct script access allowed');

class Settings_model extends EA_Model
{
    public array $allowed_fields = ['id', 'name', 'value'];

    public function save_setting(array $setting): void
    {
        // Partial defense: strip_tags only for label_custom_field_* names.
        if (!empty($setting['name']) && str_starts_with($setting['name'], 'label_custom_field_')) {
            $setting['value'] = strip_tags($setting['value'] ?? '');
        }

        $existing = $this->db->where('name', $setting['name'])->get('settings')->row_array();

        if (!empty($existing)) {
            $setting['id'] = $existing['id'];
        }

        $row = array_intersect_key($setting, array_flip($this->allowed_fields));

        if (isset($row['id'])) {
            $this->db->update('settings', $row, ['id' => $row['id']]);
        } else {
            $this->db->insert('settings', $row);
        }
    }

    public function get_all(): array
    {
        return $this->db->get('settings')->result_array();
    }
}
