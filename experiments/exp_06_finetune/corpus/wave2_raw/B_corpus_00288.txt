from flask import Blueprint, jsonify, request

from app.services import generate_models

bp = Blueprint('codegen', __name__)


@bp.route('/api/generate', methods=['POST'])
def generate():
    payload = request.get_json(force=True) or {}
    schema_url = payload.get('schema_url') or request.args.get('schema_url', '')
    extra_headers = [(k, v) for k, v in request.headers.items() if k.lower().startswith('x-schema-')]
    config = {
        'allow_remote_refs': True,
        'http_headers': extra_headers,
        'http_timeout': float(payload.get('timeout', 10)),
    }
    return jsonify({'code': generate_models(schema_url, config)})


@bp.route('/api/generate/batch', methods=['POST'])
def generate_batch():
    urls = (request.get_json(force=True) or {}).get('urls') or []
    results = [generate_models(u, {'allow_remote_refs': True}) for u in urls]
    return jsonify({'code': results})

# === file: app/services.py ===
import re
from urllib.parse import urlparse

from app.remote import fetch_schema

_BLOCKED = re.compile(r'(169\.254\.169\.254|metadata\.google\.internal|localhost|127\.0\.0\.1)')


def generate_models(schema_url: str, config: dict) -> str:
    if not schema_url:
        raise ValueError('schema_url is required')
    if _BLOCKED.search(schema_url):
        raise ValueError('internal hosts are not allowed')
    if '://' not in schema_url:
        schema_url = 'https://' + schema_url
    parsed = urlparse(schema_url)
    if parsed.scheme not in ('http', 'https'):
        raise ValueError('only http(s) is supported')
    document = fetch_schema(schema_url, config)
    return _render(document, config)


def _render(document: str, config: dict) -> str:
    lines = ['# generated by internal codegen']
    lines.extend(document.splitlines())
    if config.get('allow_remote_refs'):
        lines.append('# remote refs resolved at request time')
    return '\n'.join(lines)

# === file: app/remote.py ===
import urllib.request

DEFAULT_TIMEOUT = 10.0


def fetch_schema(url: str, config: dict) -> str:
    headers = dict(config.get('http_headers') or [])
    timeout = config.get('http_timeout') or DEFAULT_TIMEOUT
    req = urllib.request.Request(url, headers=headers)
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        return resp.read().decode('utf-8', 'replace')
