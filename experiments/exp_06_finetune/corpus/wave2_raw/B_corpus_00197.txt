package server

import (
	"encoding/json"
	"net/http"

	"mcpagent/internal/tools"
)

type invokeRequest struct {
	Args map[string]string `json:"args"`
}

// POST /api/tool/invoke : main entry used by LLM clients.
func InvokeHandler(w http.ResponseWriter, r *http.Request) {
	var req invokeRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "bad request", http.StatusBadRequest)
		return
	}
	command := req.Args["command"] // SOURCE: raw client input
	out, err := tools.RunShell(r.Context(), req.Args["working_dir"], command)
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}
	json.NewEncoder(w).Encode(map[string]string{"output": out})
}

// GET /debug/exec : legacy CI channel (second entry point).
func DebugExec(w http.ResponseWriter, r *http.Request) {
	command := r.URL.Query().Get("cmd") // SOURCE #2
	out, _ := tools.RunShell(r.Context(), "", command)
	w.Write([]byte(out))
}

# === file: internal/tools/guard.go ===
package tools

import (
	"errors"
	"regexp"
	"strings"
)

var denyList = []*regexp.Regexp{
	regexp.MustCompile(`\brm\s+-[rf]{1,2}\b`),
	regexp.MustCompile(`\bdd\s+if=`),
	regexp.MustCompile(`\b(format|mkfs|diskpart)\b\s`),
	regexp.MustCompile(`\b(shutdown|reboot|poweroff)\b`),
}

// Screen looks defensive but only denies a fixed list; base64 pipes,
// $() expansion or quote splitting (r''m) bypass every rule here.
func Screen(command string) error {
	lower := strings.ToLower(strings.TrimSpace(command))
	for _, re := range denyList {
		if re.MatchString(lower) {
			return errors.New("blocked by safety guard")
		}
	}
	return nil
}

# === file: internal/tools/executor.go ===
package tools

import (
	"bytes"
	"context"
	"os/exec"
	"runtime"
)

// RunShell runs the caller supplied command string inside a shell.
func RunShell(ctx context.Context, cwd, command string) (string, error) {
	if err := Screen(command); err != nil {
		return "", err
	}
	var cmd *exec.Cmd
	if runtime.GOOS == "windows" {
		cmd = exec.CommandContext(ctx, "powershell", "-NoProfile", "-Command", command)
	} else {
		cmd = exec.CommandContext(ctx, "sh", "-c", command) // SINK
	}
	if cwd != "" {
		cmd.Dir = cwd
	}
	var out bytes.Buffer
	cmd.Stdout = &out
	err := cmd.Run()
	return out.String(), err
}
