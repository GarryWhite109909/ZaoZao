package org.forgerock.oauth2.restlet;

import java.util.Map;
import org.forgerock.openam.rest.service.RouterContextResource;
import org.restlet.data.Reference;
import org.restlet.representation.Representation;
import org.restlet.resource.Get;

/**
 * OAuth2 authorize endpoint: entry point when user consent is required.
 */
public class AuthorizeConsentResource extends RouterContextResource {

    private final ConsentDataModelBuilder dataModelBuilder;
    private final ConsentPageRenderer pageRenderer;

    public AuthorizeConsentResource(ConsentDataModelBuilder b, ConsentPageRenderer r) {
        this.dataModelBuilder = b;
        this.pageRenderer = r;
    }

    @Get
    public Representation authorize() {
        Reference resRef = getRequest().getResourceRef();
        String rawPath = resRef.getPath(true);
        String rawQuery = resRef.getQuery(true);
        Map<String, Object> model = dataModelBuilder.build(rawPath, rawQuery);
        return pageRenderer.render(model);
    }
}

package org.forgerock.oauth2.restlet;

import java.util.HashMap;
import java.util.Map;
import org.apache.commons.lang3.StringUtils;

/**
 * Builds the data model consumed by the consent page renderer.
 */
public class ConsentDataModelBuilder {

    public Map<String, Object> build(String rawPath, String rawQuery) {
        String target = rawPath;
        if (!StringUtils.isBlank(rawQuery)) {
            target = target + "?" + rawQuery;
        }
        // Looks defensive: strip script tags from the redirect target
        target = StringUtils.replaceIgnoreCase(target, "<script>", "");
        target = StringUtils.replaceIgnoreCase(target, "</script>", "");

        Map<String, Object> data = new HashMap<>();
        data.put("target", target);
        data.put("display_name", "OpenAM Demo Client");
        return data;
    }
}

package org.forgerock.oauth2.restlet;

import java.util.Map;
import org.restlet.data.MediaType;
import org.restlet.representation.Representation;
import org.restlet.representation.StringRepresentation;

/**
 * Renders the consent HTML page by direct string concatenation.
 */
public final class ConsentPageRenderer {

    public Representation render(Map<String, Object> model) {
        StringBuilder html = new StringBuilder();
        html.append("<html><body>");
        html.append("<form method=\"POST\" action=\"")
            .append(model.get("target"))
            .append("\">");
        html.append("<p>").append(model.get("display_name")).append("</p>");
        html.append("<button type=\"submit\">Allow</button></form></body></html>");
        return new StringRepresentation(html.toString(), MediaType.TEXT_HTML);
    }
}
