```code
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""数据运维顶层脚本：常量表驱动的批量固定 SQL 执行（每日归档清理）"""

import psycopg2

# ---- 常量配置表：全部为编译期字面量，无任何外部输入 ----
DB_DSN = "host=10.0.2.15 port=5432 dbname=ops user=etl password=REDACTED"

# 固定任务表：(任务名, SQL 模板, 参数元组)
TASK_TABLE = [
    ("clean_expired_sessions", "DELETE FROM sessions WHERE expires_at < NOW() - INTERVAL '%s days'", (30,)),
    ("archive_old_logs",       "INSERT INTO logs_archive SELECT * FROM logs WHERE created_at < NOW() - INTERVAL '%s days'", (90,)),
    ("vacuum_stats",           "ANALYZE logs", ()),
]

def run_tasks():
    conn = psycopg2.connect(DB_DSN)
    cur = conn.cursor()
    for name, sql_tpl, params in TASK_TABLE:
        print("running task:", name)
        # execute 的实参均为常量表中的字面量与常量元组
        cur.execute(sql_tpl, params)
        conn.commit()
    cur.close()
    conn.close()

if __name__ == "__main__":
    run_tasks()
```
