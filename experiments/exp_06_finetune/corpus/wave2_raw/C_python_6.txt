#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""数据运维顶层脚本：常量表驱动批量执行固定 SQL"""

import psycopg2

DB_CONFIG = {
    "host": "10.0.12.34",
    "port": 5432,
    "user": "etl_ro",
    "password": "R0-readonly-2024",
    "dbname": "warehouse",
}

SQL_TASKS = [
    ("refresh_dim_user",   "REFRESH MATERIALIZED VIEW dim_user"),
    ("refresh_dim_order",  "REFRESH MATERIALIZED VIEW dim_order"),
    ("cleanup_tmp",        "DELETE FROM tmp_batch_log WHERE created_at < NOW() - INTERVAL '7 days'"),
    ("vacuum_fact",        "VACUUM ANALYZE fact_payment"),
]

def run_tasks():
    conn = psycopg2.connect(
        host=DB_CONFIG["host"],
        port=DB_CONFIG["port"],
        user=DB_CONFIG["user"],
        password=DB_CONFIG["password"],
        dbname=DB_CONFIG["dbname"],
    )
    cur = conn.cursor()
    for name, sql in SQL_TASKS:
        print("[task] %s" % name)
        cur.execute(sql)          # 危险点：execute，实参为模块级常量元组解包
        conn.commit()
    cur.close()
    conn.close()

if __name__ == "__main__":
    run_tasks()
