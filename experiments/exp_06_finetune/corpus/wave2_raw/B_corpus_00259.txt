<?php
namespace App\Report;

use App\Tool\Adapter\SqlAdapter;

class ReportService
{
    public function __construct(private SqlAdapter $adapter)
    {
    }

    public function getReportData($config, array $filters, string $sort, string $dir, $offset, $limit): array
    {
        // hardening is incomplete: blacklist strip only, dir whitelisted
        $offset = $this->stripDangerous($offset);
        $limit  = $this->stripDangerous($limit);
        $dir    = strtoupper($dir) === 'ASC' ? 'ASC' : 'DESC';

        return $this->adapter->getData($config, $filters, $sort, $dir, $offset, $limit);
    }

    private function stripDangerous($value): string
    {
        return str_replace([';', '--', '/*'], '', (string) $value);
    }
}

<?php
namespace App\Tool\Adapter;

use Pimcore\Db;

class SqlAdapter
{
    public function getData($config, array $filters, string $sort, string $dir, $offset, $limit): array
    {
        $db = Db::get();
        $base = $this->getBaseQuery($config, $filters);
        $order = ' ORDER BY ' . $db->quoteIdentifier($sort) . ' ' . $dir;
        $sql = $base['data'] . $order;
        if ($offset !== null && $limit) {
            $sql .= " LIMIT $offset,$limit";
        }
        return [
            'data'  => $db->fetchAllAssociative($sql),
            'total' => $db->fetchOne($base['count']),
        ];
    }

    private function getBaseQuery($config, array $filters): array
    {
        $db = Db::get();
        $cfg = (array) $config;
        $sql = "\n" . ($cfg['sql'] ?? 'SELECT *');
        if (!empty($cfg['where'])) {
            $sql .= "\n WHERE (" . $cfg['where'] . ')';
        }
        $condition = ['1 = 1'];
        foreach ($filters as $filter) {
            $condition[] = $db->quoteIdentifier($filter['property']) . ' = ' . $db->quote($filter['value'] ?? null);
        }
        $where = implode(' AND ', $condition);
        return [
            'data'  => 'SELECT * FROM (' . $sql . ') AS somerandxyz WHERE ' . $where,
            'count' => 'SELECT COUNT(*) FROM (' . $sql . ') AS somerandxyz WHERE ' . $where,
        ];
    }
}
