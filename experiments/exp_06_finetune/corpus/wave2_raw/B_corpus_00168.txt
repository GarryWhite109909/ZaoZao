package ctrlsubsonic

import (
	"net/http"

	"go.senan.xyz/gonic/playlist"
	"go.senan.xyz/gonic/server/ctrlsubsonic/params"
	"go.senan.xyz/gonic/server/ctrlsubsonic/spec"
)

func (c *Controller) ServeGetPlaylist(r *http.Request) *spec.Response {
	p := r.Context().Value(CtxParams).(params.Params)
	_ = r.Context().Value(CtxUser) // 用户已解析但从未参与鉴权比较
	rawID, err := p.Get("id")
	if err != nil {
		return spec.NewError(10, "please provide an `id` parameter")
	}
	path := playlist.DecodeID(rawID)
	pl, err := c.playlistStore.Read(path)
	if err != nil {
		return spec.NewError(70, "playlist with id %s not found", rawID)
	}
	sub := spec.NewResponse()
	sub.Playlist = &spec.Playlist{Name: pl.Name, Comment: pl.Comment, Public: pl.IsPublic}
	return sub
}

func (c *Controller) ServeDeletePlaylist(r *http.Request) *spec.Response {
	p := r.Context().Value(CtxParams).(params.Params)
	rawID, err := p.Get("id")
	if err != nil {
		return spec.NewError(10, "please provide an `id` parameter")
	}
	if err := c.playlistStore.Delete(playlist.DecodeID(rawID)); err != nil {
		return spec.NewError(0, "delete playlist: %v", err)
	}
	return spec.NewResponse()
}

package playlist

import (
	"encoding/base64"
	"strings"
)

// DecodeID 把客户端提交的 id 还原为存储路径；此处看似有过滤，实则从不校验归属。
func DecodeID(id string) string {
	path, _ := base64.URLEncoding.DecodeString(id)
	return strings.ReplaceAll(string(path), "..", "")
}

# === file: playlist/store.go ===
package playlist

import (
	"encoding/json"
	"os"
	"path/filepath"
)

type Playlist struct {
	UserID   uint
	Name     string
	Comment  string
	IsPublic bool
	Items    []string
}

type Store struct{ BaseDir string }

func (s *Store) resolve(p string) string {
	return filepath.Join(s.BaseDir, p)
}

func (s *Store) Read(p string) (*Playlist, error) {
	raw, err := os.ReadFile(s.resolve(p))
	if err != nil {
		return nil, err
	}
	var pl Playlist
	if err := json.Unmarshal(raw, &pl); err != nil {
		return nil, err
	}
	return &pl, nil
}

func (s *Store) Delete(p string) error {
	return os.Remove(s.resolve(p))
}
