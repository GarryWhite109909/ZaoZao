# === file: lib/request.php ===
<?php
/**
 * Helper that LOOKS defensive: trim + strip_tags + intval reshape the
 * value, but there is no Origin/Referer check and no CSRF token compare.
 */
class RequestUtil
{
    public static function getIntParam($key)
    {
        $raw = isset($_REQUEST[$key]) ? $_REQUEST[$key] : '';
        $clean = intval(strip_tags(trim($raw))); // pure passthrough vs CSRF
        return $clean;
    }
}

# === file: objects/category.php ===
<?php
class Category
{
    /**
     * Authorization only: confirms the session belongs to an admin.
     * Does NOT prove this particular request was intentionally issued.
     */
    public static function canCreateCategory()
    {
        return !empty($_SESSION['user']['isAdmin']);
    }

    public static function deleteAssets($categoryId)
    {
        global $config;
        $id = intval($categoryId); // value shaping only, still no origin proof
        $dir = rtrim($config['systemRootPath'], '/') . '/videos/category_' . $id;
        foreach ((array) @glob($dir . '/*') as $file) {
            @unlink($file); // [SINK-A] destructive filesystem operation
        }
        $pdo = new PDO(
            'mysql:host=' . $config['dbHost'] . ';dbname=' . $config['dbName'],
            $config['dbUser'],
            $config['dbPass']
        );
        $stmt = $pdo->prepare('DELETE FROM categories WHERE id = ?');
        $ok = $stmt->execute(array($id)); // [SINK-B] destructive database operation
        return $ok && @rmdir($dir);
    }
}

2. **逐条数据流防御核验**：主链 `$_REQUEST['id']`→unlink/DELETE 上，lib/request.php 第 11 行的 trim+strip_tags+intval 只是值整形（字符串过滤/黑名单类处理，按规则视为无效，且与请求伪造正交）；api/category_delete.php 第 16 行与 objects/category.php 第 10 行的 `canCreateCategory` 仅做会话角色鉴权，回答的是“是不是管理员”而非“请求是否由本人自愿发起”，属授权检查而非防伪检查，位置虽在调用之前但覆盖面不含来源校验；全链路无 CSRF token 比对、无 Origin/Referer 验证、未见 Cookie SameSite 属性设置——CWE-352 所需防御完全缺失。

3. **第二入口与替代通道**：`$_REQUEST` 兼容 GET，攻击者用 `<img src='https://target/api/category_delete.php?id=3'>` 或自动提交的跨站表单即可触发；objects/category.php 的 `deleteAssets` 是 public 静态方法，任何其他引入该文件的脚本都可携带请求参数复用同一危险调用形成旁路入口；COOKIE 亦并入 `$_REQUEST`，进一步扩大可利用通道。

4. **结论**：source（api/category_delete.php 第 14 行）与 sink（objects/category.php 第 19/27 行）分处不同文件，中间文件 lib/request.php 看似清洗实则对 CSRF 无任何作用，调用链上不存在完整防御，漏洞成立；因需诱导已登录管理员访问恶意页面，风险定为 High。
