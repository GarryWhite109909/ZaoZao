package com.spinnaker.pipeline.service;

import com.spinnaker.pipeline.config.YamlFactory;
import org.springframework.stereotype.Service;

@Service
public class PipelineService {

  /** Import a user supplied pipeline manifest into the running app. */
  public String importFromYaml(String manifest) {
    String cleaned = blockKnownGadgets(manifest);
    Object cfg = YamlFactory.parseManifest(cleaned);
    return "imported:" + String.valueOf(cfg);
  }

  /**
   * Blacklist filter: strips only two well-known gadget tag literals.
   */
  private String blockKnownGadgets(String input) {
    if (input == null) {
      return "";
    }
    return input
        .replace("!!javax.script.ScriptEngineManager", "")
        .replace("!com.netflix.spinnaker.", "!safe.");
  }
}

package com.spinnaker.pipeline.config;

import org.springframework.stereotype.Component;
import org.yaml.snakeyaml.DumperOptions;
import org.yaml.snakeyaml.LoaderOptions;
import org.yaml.snakeyaml.Yaml;
import org.yaml.snakeyaml.constructor.Constructor;
import org.yaml.snakeyaml.representer.Representer;
import org.yaml.snakeyaml.resolver.Resolver;

/**
 * Central factory for SnakeYAML parsers shared across services.
 */
@Component
public class YamlFactory {

  /** Permissive parser kept for legacy template compatibility. */
  public static Object parseManifest(String content) {
    LoaderOptions opts = new LoaderOptions();
    opts.setMaxAliasesForCollections(50);
    opts.setCodePointLimit(3145728);
    Constructor ctor = new Constructor(opts);
    Representer rep = new Representer();
    DumperOptions dump = new DumperOptions();
    Yaml yaml = new Yaml(ctor, rep, dump, opts, new Resolver());
    return yaml.load(content);
  }
}
