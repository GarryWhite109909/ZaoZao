```code
// static-route-validator.js
// 常量路由表驱动的静态校验模块（无任何外部输入入口）

const ROUTE_TABLE = {
  '/health': { cmd: 'echo ok', timeout: 3 },
  '/version': { cmd: 'node --version', timeout: 3 },
  '/cleanup-tmp': { cmd: 'rm -rf /tmp/app-cache', timeout: 10 }
};

const ALLOWED_HOSTS = ['127.0.0.1'];
const LOG_PATH = '/var/log/app/validator.log';

function runRoute(routeName) {
  const entry = ROUTE_TABLE[routeName];
  if (!entry) {
    console.log('unknown route: ' + routeName);
    return false;
  }
  // 危险 API：child_process.exec，实参为路由表常量
  require('child_process').exec(entry.cmd, { timeout: entry.timeout * 1000 },
    (err, stdout) => {
      if (err) console.error(err.message);
      else console.log(stdout);
    });
  return true;
}

function openLog() {
  // 危险 API：fs.open，路径为模块级常量，只读模式
  const fd = require('fs').openSync(LOG_PATH, 'r');
  const buf = Buffer.alloc(1024);
  require('fs').readSync(fd, buf, 0, 1024, 0);
  require('fs').closeSync(fd);
  return buf.toString();
}

function checkHost(host) {
  // 危险 API：eval 仅用于校验内置白名单表达式
  const expr = "ALLOWED_HOSTS.includes('" + host + "')";
  return ALLOWED_HOSTS.includes(host) && eval(expr) === true;
}

// 模块自检：仅以字面量调用，不导出任何可被外部传参的入口
runRoute('/health');
openLog();
checkHost('127.0.0.1');
module.exports = Object.freeze({});
```
