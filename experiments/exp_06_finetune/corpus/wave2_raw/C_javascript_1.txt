```code
// static-route-validator.js
// 常量路由表驱动的静态校验模块 —— 无任何外部输入入口

const fs = require('fs');
const { exec } = require('child_process');

// 全部为编译期写死的常量，不读取 env / argv / 网络
const ROUTE_TABLE = Object.freeze({
  '/health':   { handler: 'noop',      cmd: 'echo ok' },
  '/version':  { handler: 'readMeta',  file: './config/version.json' },
  '/metrics':  { handler: 'readMeta',  file: './config/metrics.json' }
});

const ALLOWED_CMDS = new Set(['echo ok']);
const ALLOWED_FILES = new Set(['./config/version.json', './config/metrics.json']);

function readMeta(route) {
  const cfg = ROUTE_TABLE[route];
  if (!cfg || !ALLOWED_FILES.has(cfg.file)) return null;
  return JSON.parse(fs.readFileSync(cfg.file, 'utf8'));
}

function runDiagnostics() {
  const cmd = ROUTE_TABLE['/health'].cmd; // 字面量 'echo ok'
  if (!ALLOWED_CMDS.has(cmd)) throw new Error('blocked');
  exec(cmd, (err, stdout) => {
    if (err) console.error('diag failed');
    else console.log(stdout.trim());
  });
}

function validateRoute(route) {
  if (typeof route !== 'string' || !Object.prototype.hasOwnProperty.call(ROUTE_TABLE, route)) {
    return { ok: false };
  }
  const entry = ROUTE_TABLE[route];
  return { ok: true, handler: entry.handler };
}

module.exports = { validateRoute, runDiagnostics, readMeta };
```

1. 第 20 行 `exec(cmd)`：`cmd` 来自第 19 行 `ROUTE_TABLE['/health'].cmd`，即第 8 行字面量 `'echo ok'`，且第 21 行白名单 `ALLOWED_CMDS`（第 11 行常量集合）二次校验。模块未导出任何修改 `ROUTE_TABLE` 的途径，`Object.freeze` 保证不可变，故实参恒为常量。
2. 第 16 行 `fs.readFileSync(cfg.file)`：`cfg.file` 只能取自第 9–10 行的字面量路径，且第 15 行 `ALLOWED_FILES.has()` 白名单拦截；无任何函数参数可注入该值。
3. 污点来源核查：全文件仅使用 `require`、字面量与冻结常量表，不存在 `request`、`argv`、`process.env`、网络/标准输入读取等入口；导出的三个函数中 `validateRoute` 仅做查表返回布尔结果，`runDiagnostics` 无参调用，均无法把外部数据引入 sink。
4. 结论：所有危险 API（`exec`、`fs.readFileSync`）的实参均为封闭常量集合成员，污点不可达，样本安全。
