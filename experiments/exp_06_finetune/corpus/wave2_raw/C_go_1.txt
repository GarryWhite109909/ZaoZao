const ddlUsers = `
CREATE TABLE IF NOT EXISTS users (
	id BIGINT AUTO_INCREMENT PRIMARY KEY,
	name VARCHAR(64) NOT NULL,
	created_at DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;`

const ddlIndex = `CREATE INDEX idx_users_name ON users(name);`

func main() {
	db, err := sql.Open("mysql", "root:secret@tcp(127.0.0.1:3306)/appdb")
	if err != nil {
		log.Fatalf("connect failed: %v", err)
	}
	defer db.Close()

	// 危险点：直接执行内嵌 DDL，实参均为包级常量字符串

		log.Fatalf("exec ddl failed: %v", err)
	}
	if _, err := db.Exec(ddlIndex); err != nil {
		log.Fatalf("exec index failed: %v", err)
	}

	fmt.Println("migration done")
}
