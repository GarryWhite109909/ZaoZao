package com.example.report.web;

import com.example.report.core.XDocArchive;
import com.example.report.service.PreprocessingService;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

@RestController
public class DocumentUploadController {

    private final PreprocessingService service = new PreprocessingService();

    // SOURCE: 客户端上传的 OOXML/ODT 模板包, 字节内容与 entry 条目名全部可控
    @PostMapping(value = "/reports/render", consumes = "multipart/form-data")
    public String render(@RequestParam("file") MultipartFile file,
                         @RequestParam(value = "entry", defaultValue = "word/document.xml") String entryName)
            throws Exception {
        XDocArchive archive = XDocArchive.readArchive(file.getInputStream());
        return service.preprocessEntry(archive, entryName);
    }
}

package com.example.report.service;

import com.example.report.core.XDocArchive;
import com.example.report.document.preprocessor.sax.SAXXDocPreprocessor;
import java.io.InputStream;
import java.io.StringWriter;

public class PreprocessingService {

    // 看似净化: 仅拉黑 .svg 后缀, 且只扫描流前 256 字节中的字面量 DOCTYPE
    private boolean looksSafe(String entryName, InputStream in) throws Exception {
        if (entryName.endsWith(".svg")) {
            return false;
        }
        in.mark(4096);
        byte[] head = new byte[256];
        int n = in.read(head);
        in.reset();
        String headText = new String(head, 0, Math.max(n, 0), "UTF-8");
        return !headText.contains("DOCTYPE");
    }

    public String preprocessEntry(XDocArchive archive, String entryName) throws Exception {
        InputStream entryStream = archive.getEntryInputStream(entryName); // 污点透传
        if (!looksSafe(entryName, entryStream)) {
            throw new IllegalArgumentException("blocked entry");
        }
        StringWriter out = new StringWriter();
        new SAXXDocPreprocessor().preprocess(entryName, entryStream, out, null, null, null);
        return out.toString();
    }
}

package com.example.report.document.preprocessor.sax;

import com.example.report.template.formatter.FieldsMetadata;
import com.example.report.template.formatter.IDocumentFormatter;
import java.io.*;
import java.util.Map;
import org.xml.sax.*;
import org.xml.sax.helpers.XMLReaderFactory;

public class SAXXDocPreprocessor {

    public boolean preprocess(String entryName, InputStream reader, Writer writer,
                              FieldsMetadata fieldsMetadata, IDocumentFormatter formatter,
                              Map<String, Object> sharedContext) throws IOException {
        try {
            // SINK: createXMLReader 默认允许 DTD 与外部实体, 未设置任何限制 feature
            XMLReader xmlReader = XMLReaderFactory.createXMLReader();
            BufferedDocumentContentHandler<?> handler =
                    new BufferedDocumentContentHandler<>(entryName, fieldsMetadata, formatter, sharedContext);
            xmlReader.setContentHandler(handler);
            xmlReader.parse(new InputSource(reader));
            BufferedDocument document = handler.getBufferedDocument();
            if (document != null) {
                document.save(writer);
                return true;
            }
            return false;
        } catch (SAXException e) {
            throw new IOException(e);
        }
    }
}
