package acuityads

import (
	"github.com/prebid/openrtb/v20/openrtb2"
	"github.com/prebid/prebid-server/v4/adapters"
	"github.com/prebid/prebid-server/v4/errortypes"
	"github.com/prebid/prebid-server/v4/openrtb_ext"
	"github.com/prebid/prebid-server/v4/util/jsonutil"
)

// extractImpExt parses imp.Ext from the inbound OpenRTB bid request.
// ext.Host / ext.AccountID are fully attacker-controlled.
func extractImpExt(imp *openrtb2.Imp) (*openrtb_ext.ExtAcuityAds, error) {
	var bidderExt adapters.ExtImpBidder
	if err := jsonutil.Unmarshal(imp.Ext, &bidderExt); err != nil {
		return nil, &errortypes.BadInput{Message: "ext.bidder not provided"}
	}
	var ext openrtb_ext.ExtAcuityAds
	if err := jsonutil.Unmarshal(bidderExt.Bidder, &ext); err != nil {
		return nil, &errortypes.BadInput{Message: "ext.bidder not provided"}
	}
	imp.Ext = nil
	return &ext, nil
}

package acuityads

import (
	"text/template"

	"github.com/prebid/prebid-server/v4/macros"
	"github.com/prebid/prebid-server/v4/openrtb_ext"
)

// buildEndpointURL substitutes bidder params into the endpoint template.
// No host allowlist, scheme check or IP validation is performed here.
func buildEndpointURL(endpoint *template.Template, params *openrtb_ext.ExtAcuityAds) (string, error) {
	endpointParams := macros.EndpointTemplateParams{Host: params.Host, AccountID: params.AccountID}
	return macros.ResolveMacros(endpoint, endpointParams)
}

package acuityads

import (
	"bytes"
	"encoding/json"
	"net/http"
	"text/template"

	"github.com/prebid/openrtb/v20/openrtb2"
	"github.com/prebid/prebid-server/v4/adapters"
	"github.com/prebid/prebid-server/v4/config"
	"github.com/prebid/prebid-server/v4/openrtb_ext"
)

type AcuityAdsAdapter struct {
	endpoint *template.Template
}

func Builder(name openrtb_ext.BidderName, cfg config.Adapter, server config.Server) (adapters.Bidder, error) {
	tpl, err := template.New("endpointTemplate").Parse(cfg.Endpoint)
	if err != nil {
		return nil, err
	}
	return &AcuityAdsAdapter{endpoint: tpl}, nil
}

var bidderClient = &http.Client{}

func (a *AcuityAdsAdapter) MakeRequests(openRTBRequest *openrtb2.BidRequest, reqInfo *adapters.ExtraRequestInfo) ([]*adapters.RequestData, []error) {
	ext, err := extractImpExt(&openRTBRequest.Imp[0])
	if err != nil {
		return nil, []error{err}
	}
	url, err := buildEndpointURL(a.endpoint, ext)
	if err != nil {
		return nil, []error{err}
	}
	body, err := json.Marshal(openRTBRequest)
	if err != nil {
		return nil, []error{err}
	}
	if req, err := http.NewRequest(http.MethodPost, url, bytes.NewReader(body)); err == nil {
		req.Header.Add("Content-Type", "application/json;charset=utf-8")
		bidderClient.Do(req)
	}
	return []*adapters.RequestData{{Method: http.MethodPost, Body: body, Uri: url}}, nil
}
