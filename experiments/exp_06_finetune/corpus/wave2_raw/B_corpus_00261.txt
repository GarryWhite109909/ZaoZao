<?php
declare(strict_types=1);

namespace phpMyFAQ\Setup;

/**
 * Reads environment values submitted by the setup wizard form.
 */
final class UrlConfigProvider
{
    /** Source: raw installer POST value, fully attacker-controlled. */
    public static function getDefaultUrlFromRequest(): string
    {
        return isset($_POST['faqDefaultUrl']) ? (string) $_POST['faqDefaultUrl'] : 'http://localhost/faq/';
    }

    /** parse_url only parses; it does not neutralize spaces, '#' or raw newlines. */
    public static function getServerPath(): string
    {
        $path = parse_url(self::getDefaultUrlFromRequest(), PHP_URL_PATH);
        return ($path === null || $path === false || $path === '') ? '/' : $path;
    }
}

<?php
declare(strict_types=1);

namespace phpMyFAQ\Setup;

/**
 * Composes .htaccess directive argument values.
 * rtrim() below only strips trailing slashes: raw newlines, leading
 * blanks and '#' survive, so this 'sanitization' is incomplete.
 */
final class HtaccessRuleComposer
{
    /** Pure pass-through of the tainted path. */
    public function rewriteBaseValue(): string
    {
        return UrlConfigProvider::getServerPath();
    }

    /** Cosmetic transform only; no encoding, no character whitelist. */
    public function errorDocument404Value(): string
    {
        $path = rtrim(UrlConfigProvider::getServerPath(), '/');
        return $path . '/index.php?action=404';
    }
}

<?php
declare(strict_types=1);

namespace phpMyFAQ\Setup;

use phpMyFAQ\Core\Exception;

final class EnvironmentConfigurator
{
    private HtaccessRuleComposer $composer;

    public function __construct(private string $htaccessPath)
    {
        $this->composer = new HtaccessRuleComposer();
    }

    public function adjustRewriteBaseHtaccess(): bool
    {
        if (!file_exists($this->htaccessPath)) {
            throw new Exception('The .htaccess file does not exist!');
        }

        $lines = file($this->htaccessPath, FILE_IGNORE_NEW_LINES) ?: [];
        $out = [];
        foreach ($lines as $line) {
            if (preg_match('/^\s*RewriteBase\b/i', $line)) {
                $out[] = 'RewriteBase ' . $this->composer->rewriteBaseValue();
            } elseif (preg_match('/^\s*ErrorDocument\s+404\b/i', $line)) {
                $out[] = 'ErrorDocument 404 ' . $this->composer->errorDocument404Value();
            } else {
                $out[] = $line;
            }
        }

        // Sink: tainted directive values are written verbatim into .htaccess
        return (bool) file_put_contents($this->htaccessPath, implode(PHP_EOL, $out));
    }
}
