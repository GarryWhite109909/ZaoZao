package server

import (
	"net/http"

	"archiver/internal/service"
)

// RegisterRoutes 注册服务对外暴露的全部路由。
func RegisterRoutes(mux *http.ServeMux) {
	mux.HandleFunc("/api/extract", HandleExtract)
}

// HandleExtract 处理 GET /api/extract?archive=backup.zip&dest=subdir
// 两个参数均直接取自请求 URL，属于外部输入。
func HandleExtract(w http.ResponseWriter, r *http.Request) {
	archiveName := r.URL.Query().Get("archive") // source：攻击者可控的压缩包名
	destSub := r.URL.Query().Get("dest")        // source：攻击者可控的目标子目录
	if err := service.ExtractTo(r.Context(), archiveName, destSub); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}
	w.WriteHeader(http.StatusOK)
}

package service

import (
	"context"
	"strings"
)

// sanitizeDest 仅去掉一个前导斜杠：既不拒绝 .. 片段，
// 也不做路径归一化，形如 ../../etc 的输入会原样通过。
func sanitizeDest(dest string) string {
	return strings.TrimPrefix(dest, "/")
}

// ExtractTo 是 HTTP 层调用的业务入口。
func ExtractTo(ctx context.Context, archiveName, dest string) error {
	subDir := sanitizeDest(dest) // 看似净化实则不完整：.. 序列未被处理
	return runExtraction(ctx, archiveName, subDir)
}

package service

import (
	"context"

	_ "github.com/rclone/rclone/backend/local"
	"github.com/rclone/rclone/cmd/archive/extract"
	"github.com/rclone/rclone/fs"
)

const (
	uploadFS  = `:local:/tmp/uploads`  // 用户上传压缩包的存放处
	archiveFS = `:local:/srv/archives` // 解压目标根目录
)

// runExtraction 执行真正的解压操作。
// subDir 直接来自 HTTP 的 dest 参数，未经有效校验即到达本函数；
// 它与压缩包内部成员名共同构成 Zip-Slip 路径穿越（CWE-22）。
func runExtraction(ctx context.Context, archiveName, subDir string) error {
	src, err := fs.NewFs(ctx, uploadFS)
	if err != nil {
		return err
	}
	dst, err := fs.NewFs(ctx, archiveFS)
	if err != nil {
		return err
	}
	// SINK：subDir 与包内路径在 ArchiveExtract 内部拼接并写出，
	// 全链路不存在任何目录包含性检查。
	return extract.ArchiveExtract(ctx, src, subDir, dst, archiveName)
}
