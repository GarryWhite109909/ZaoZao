# === file: objects/UserPhotoService.php ===
<?php
/**
 * Service layer: shapes the upload payload and delegates persistence.
 */
require_once __DIR__ . '/functions.php';     // provides base64DataToImage(), __()
require_once __DIR__ . '/PhotoStorage.php';

class UserPhotoService
{
    public static function handleSave($imgBase64)
    {
        global $global;
        if (!is_string($imgBase64) || strlen($imgBase64) < 32) {
            return ['status' => 'error', 'msg' => 'invalid image payload'];
        }
        // Normalizes url-safe base64 spaces; formatting only, NOT a security gate
        $imgBase64 = str_replace(' ', '+', $imgBase64);
        $imageData = base64DataToImage($imgBase64);
        $imagePath = 'videos/userPhoto/';
        if (!is_dir($global['systemRootPath'] . $imagePath)) {
            mkdir($global['systemRootPath'] . $imagePath, 0755, true);
        }
        if (!is_writable($global['systemRootPath'] . $imagePath)) {
            return ['status' => 'error', 'msg' => 'No write Access'];
        }
        $fileName = 'background' . User::getId() . '.png';
        // request origin was never verified beyond the session cookie
        return PhotoStorage::store($imagePath . $fileName, $imageData);
    }
}

# === file: objects/PhotoStorage.php ===
<?php
/**
 * Persistence layer for user photo blobs.
 */
require_once __DIR__ . '/user.php';

class PhotoStorage
{
    public static function store($photoURL, $binaryData)
    {
        global $global;
        // Writes whatever bytes arrive from the service layer
        $bytes = file_put_contents($global['systemRootPath'] . $photoURL, $binaryData);
        if (!$bytes) {
            return ['status' => 'error', 'msg' => 'We could not save this file'];
        }
        $user = new User(User::getId());
        $user->setBackgroundURL($photoURL);
        $user->save();
        User::updateSessionInfo();
        return ['status' => 'success', 'url' => $global['systemRootPath'] . $photoURL];
    }
}
