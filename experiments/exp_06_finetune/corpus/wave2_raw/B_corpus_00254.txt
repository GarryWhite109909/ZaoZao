namespace Wiki\Core;

class Request
{
    private array $get;

    public function __construct()
    {
        $this->get = $_GET; // 全量接收外部 GET 输入，无任何过滤
    }

    public function param(string $name, ?string $default = null): ?string
    {
        return isset($this->get[$name]) ? (string)$this->get[$name] : $default;
    }
}
# === file: controllers/RecentChangesController.php ===
<?php
// controllers/RecentChangesController.php —— 中间层：白名单看似存在，回退分支却放行原始输入
namespace Wiki\Controllers;

use Wiki\Core\Request;
use Wiki\Db\PageRepository;

class RecentChangesController
{
    public function __construct(private Request $request, private PageRepository $repo) {}

    private function resolveDateMin(): string
    {
        $period = $this->request->param('period'); // 污点进入业务层
        if (is_string($period) && in_array($period, ['day', 'week', 'month'], true)) {
            $map = ['day' => '-1 day', 'week' => '-1 week', 'month' => '-1 month'];
            return date('Y-m-d H:i:s', strtotime($map[$period])); // 命中白名单：返回安全常量
        }
        return $period ?? date('Y-m-d H:i:s', strtotime('-1 month')); // 回退分支：原始输入原样返回
    }

    public function show(): void
    {
        $max = (int)($this->request->param('max') ?? 500); // 第二输入点 max：(int) 强转，有效防御
        $dateMin = $this->resolveDateMin();                // 污点继续下传
        $rows = $this->repo->loadRecentlyChanged($max, $dateMin); // 污点实参流向 sink
        foreach ($rows as $page) {
            echo '<small>', htmlspecialchars($page['time']), '</small><br>', "\n";
        }
    }
}

namespace Wiki\Db;

use PDO;

class PageRepository
{
    public function __construct(private PDO $pdo) {}

    public function loadRecentlyChanged(int $max, string $dateMin): array
    {
        $sql = 'SELECT id, tag, time, user FROM pages'
             . " WHERE time >= '" . $dateMin . "'" // sink：$dateMin 未参数化直接拼入
             . ' ORDER BY time DESC LIMIT ' . $max;
        return $this->pdo->query($sql)->fetchAll(PDO::FETCH_ASSOC);
    }
}
