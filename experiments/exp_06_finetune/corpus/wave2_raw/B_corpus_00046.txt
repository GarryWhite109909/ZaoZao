package replacepathregex

import (
	"context"
	"fmt"
	"net/http"
	"regexp"
	"strings"

	"github.com/traefik/traefik/v2/pkg/config/dynamic"
)

const typeName = "ReplacePathRegex"

type replacePathRegex struct {
	next        http.Handler
	regexp      *regexp.Regexp
	replacement string
	name        string
}

// New creates a new replace path regex middleware.
func New(ctx context.Context, next http.Handler, config dynamic.ReplacePathRegex, name string) (http.Handler, error) {
	exp, err := regexp.Compile(strings.TrimSpace(config.Regex))
	if err != nil {
		return nil, fmt.Errorf("error compiling regular expression %s: %w", config.Regex, err)
	}
	return &replacePathRegex{
		regexp:      exp,
		replacement: strings.TrimSpace(config.Replacement),
		next:        next,
		name:        name,
	}, nil
}

func (rp *replacePathRegex) ServeHTTP(rw http.ResponseWriter, req *http.Request) {
	currentPath := req.URL.RawPath
	if currentPath == "" {
		currentPath = req.URL.EscapedPath()
	}
	if rp.regexp != nil && rp.regexp.MatchString(currentPath) {
		req.URL.RawPath = rp.regexp.ReplaceAllString(currentPath, rp.replacement)
		req.RequestURI = req.URL.RequestURI()
	}
	rp.next.ServeHTTP(rw, req)
}

package router

import (
	"net/http"
	"net/url"
)

// NormalizeURL 试图规范化 URL，但只处理 Path 字段，
// 未校验 RawPath，也未阻止 `..` 段，属于不完整的净化。
func NormalizeURL(req *http.Request) {
	p := req.URL.Path
	for len(p) > 1 && p[len(p)-1] == '/' {
		p = p[:len(p)-1]
	}
	req.URL.Path = p
	// RawPath 保持原样（可能含 %2e%2e 等编码穿越段）
	_ = url.PathEscape(p)
}

package handler

import (
	"net/http"
	"os"
	"path/filepath"

	"github.com/traefik/traefik/v2/pkg/log"
)

// FileHandler 直接使用请求 URL 的 RawPath 定位文件，
// 中间件改写后的 RawPath 与未净化的 Path 均可携带 `..`。
func (h *FileHandler) ServeFile(rw http.ResponseWriter, req *http.Request) {
	target := filepath.Join(h.RootDir, req.URL.RawPath)
	data, err := os.ReadFile(target)
	if err != nil {
		log.FromContext(req.Context()).Error(err)
		http.Error(rw, err.Error(), http.StatusInternalServerError)
		return
	}
	rw.Write(data)
}
