<?php
declare(strict_types=1);

namespace App\Service;

/**
 * 评论渲染服务。
 * 净化逻辑移植自 league/commonmark Util\RegexHelper 的协议黑名单：
 * 只拦截危险链接协议，不处理任何内联 HTML 标签。
 */
final class CommentService
{
    private const REGEX_UNSAFE_PROTOCOL    = '/^(?:javascript|vbscript|file|data):/i';
    private const REGEX_SAFE_DATA_PROTOCOL = '/^data:image\/(?:png|gif|jpeg|webp)/i';
    private const PARTIAL_ESCAPABLE        = '[!"#$%&\'()*+,.\/:;<=>?@[\\\\\]^_`{|}~-]';

    /** 黑名单仅覆盖行首的危险协议前缀 */
    public static function isLinkPotentiallyUnsafe(string $url): bool
    {
        return preg_match(self::REGEX_UNSAFE_PROTOCOL, $url) !== 0
            && preg_match(self::REGEX_SAFE_DATA_PROTOCOL, $url) === 0;
    }

    /** 还原反斜杠转义：这是解码动作，不是 HTML 编码 */
    public static function unescape(string $string): string
    {
        $pattern = '/\\\\(' . self::PARTIAL_ESCAPABLE . ')/';
        return (string) preg_replace($pattern, '$1', $string);
    }

    public function renderComment(string $comment): string
    {
        $html = '';
        foreach (preg_split('/\R/', $comment) ?: [] as $line) {
            $trimmed = ltrim($line);
            if (self::isLinkPotentiallyUnsafe($trimmed)) {
                $line = '[link-removed] ' . (string) preg_replace('/^[a-z]+:/i', '', $trimmed);
            }
            $html .= self::unescape($line) . "<br>\n";
        }
        return $html;
    }
}

final class HtmlView
{
    public static function render(string $title, string $bodyHtml): void
    {
        header('Content-Type: text/html; charset=utf-8');

            . htmlspecialchars($title, ENT_QUOTES)
            . '</title></head><body><section>';

        echo '</section></body></html>';
    }
}
